## --file-- server.js

```js
import { createServer } from 'node:http';

const books = [{ id: 1, title: 'Babička' }];

const server = createServer(async (req, res) => {
  if (req.method === 'POST' && req.url === '/books') {
    let body = '';
    for await (const chunk of req) body += chunk;
    const book = { id: books.length + 1, ...JSON.parse(body), type: req.headers['content-type'] };
    books.push(book);
    res.writeHead(201, { 'Content-Type': 'application/json; charset=utf-8' });
    res.end(JSON.stringify(book));
    return;
  }
  res.writeHead(200, { 'Content-Type': 'application/json; charset=utf-8', 'X-Pocet': String(books.length) });
  res.end(JSON.stringify(books));
});

server.listen(Number(process.env.PORT ?? 3000), () => {
  console.log('Knihovna běží');
});
```
