# --description--

Kuchyňský tablet ztratí wifi na dvacet sekund. Až se vrátí, server o tom neví
a objednávky, které mezitím přišly, na obrazovce prostě nebudou. Aby se daly
doplnit, musí mít každá událost **svoje číslo** a server si musí posledních pár
událostí pamatovat.

**Úkol:** v `server.js`:

1. vedle registru klientů si založ prázdnou historii událostí a počítadlo
   posledního čísla,
2. ve funkci `rozesli` číslo zvyš o jedna, ulož si událost (číslo, typ, data)
   do historie a rozešli ji i s tímhle číslem,
3. historii ořízni na posledních **50** událostí — paměť serveru není archiv.

Uvítací snapshot číslo nedostane, ten posíláš zvlášť a není to změna.

Padesát událostí je kompromis: víc paměti za delší výpadek, který ještě umíme
dohnat. Delší výpadek se řeší novým snapshotem, ne nekonečnou historií.

const otevriProud = async (adresa, hlavicky = {}) => {
  const stop = new AbortController();
  const res = await fetch(adresa, { headers: hlavicky, signal: stop.signal });
  const ctecka = res.body.getReader();
  const dekoder = new TextDecoder();
  let text = '';
  const dalsi = async ({ limit = 4000, komentare = false } = {}) => {
    for (;;) {
      const konec = text.indexOf('\n\n');
      if (konec !== -1) {
        const kus = text.slice(0, konec);
        text = text.slice(konec + 2);
        if (!komentare && kus.split('\n').every((radek) => radek.startsWith(':'))) continue;
        return kus;
      }
      const kousek = await Promise.race([ctecka.read(), helpers.wait(limit).then(() => 'cas')]);
      if (kousek === 'cas' || kousek.done) return null;
      text += dekoder.decode(kousek.value, { stream: true });
    }
  };
  const rozloz = (ramec) => {
    const pole = { data: [] };
    for (const radek of (ramec ?? '').split('\n')) {
      const dvojtecka = radek.indexOf(':');
      if (dvojtecka <= 0) continue;
      const klic = radek.slice(0, dvojtecka);
      const hodnota = radek.slice(dvojtecka + 1).replace(/^ /, '');
      if (klic === 'data') pole.data.push(hodnota);
      else pole[klic] = hodnota;
    }
    return { ...pole, data: pole.data.join('\n') };
  };
  return { res, dalsi, rozloz, zavri: () => stop.abort() };
};
const server = await helpers.startServer('server.js');
const proud = await otevriProud(new URL('/api/udalosti', server.url));
try {
  const uvitani = proud.rozloz(await proud.dalsi());
  assert.equal(uvitani.id, undefined, 'Uvítací snapshot není změna, kterou by šlo doplnit — řádek id: do něj nepatří');
  const posli = (stul) => fetch(new URL('/api/objednavky', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ stul, polozky: ['Espresso'] }),
  });
  await posli(2);
  const prvni = proud.rozloz(await proud.dalsi());
  await posli(3);
  const druha = proud.rozloz(await proud.dalsi());
  assert.equal(prvni.id, '1', `První rozeslaná událost má mít id: 1, přišlo: ${JSON.stringify(prvni)}`);
  assert.equal(druha.id, '2', `Druhá rozeslaná událost má mít id: 2, přišlo: ${JSON.stringify(druha)}`);
} finally {
  proud.zavri();
}

# --help--

## --tip--

Počítadlo i historie musí přežít mezi požadavky, takže patří mezi konstanty nahoře. Počítadlo se mění, historie jen roste a zkracuje se — podle toho zvol `const`, nebo `let`.

## --tip--

Postav si nejdřív objekt události a teprve ten předej `formatujUdalost` i historii. Jinak se ti číslo v rámci a v historii rozejde.

## --tip--

Nejstarší prvek z pole odebereš jednou metodou, která rovnou zkrátí původní pole. Volej ji, až když je pole delší než limit.

// Objednávky bistra U Modré kotvy. Držíme je v paměti — po restartu serveru
// se seznam vrátí do tohohle stavu, databáze tu není potřeba.
export const objednavky = [
  { id: 1, stul: 3, polozky: ['Svíčková', 'Kofola'], stav: 'nova', prijato: '11:42' },
  { id: 2, stul: 7, polozky: ['Guláš', 'Pivo', 'Štrúdl'], stav: 'hotova', prijato: '11:45' },
  { id: 3, stul: 1, polozky: ['Smažený sýr', 'Limonáda'], stav: 'nova', prijato: '11:51' },
];

// Čas ve tvaru HH:MM, jak ho kuchyň čte z lístku.
export function cas(ted = new Date()) {
  return ted.toTimeString().slice(0, 5);
}

export function dalsiId(seznam) {
  return seznam.reduce((nejvyssi, objednavka) => Math.max(nejvyssi, objednavka.id), 0) + 1;
}

export function pridejObjednavku(seznam, stul, polozky) {
  const objednavka = { id: dalsiId(seznam), stul, polozky, stav: 'nova', prijato: cas() };
  seznam.push(objednavka);
  return objednavka;
}

// Jeden rámec Server-Sent Events: řádky s klíčem a prázdný řádek na konci.
export function formatujUdalost({ id, typ, data }) {
  let ramec = '';
  if (id !== undefined) ramec += `id: ${id}\n`;
  if (typ) ramec += `event: ${typ}\n`;
  ramec += `data: ${JSON.stringify(data)}\n`;
  return ramec + '\n';
}

}

async function prectiJson(req) {
  const kousky = [];
  for await (const kousek of req) kousky.push(kousek);
  if (kousky.length === 0) return {};
  try {
    return JSON.parse(Buffer.concat(kousky).toString('utf8'));
  } catch {
    return null;
  }
}

const server = createServer(async (req, res) => {
  const url = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && url.pathname === '/api/objednavky') {
    posliJson(res, 200, objednavky);
    return;
  }

  if (req.method === 'POST' && url.pathname === '/api/objednavky') {
    const telo = await prectiJson(req);
    if (!telo || typeof telo.stul !== 'number' || !Array.isArray(telo.polozky) || telo.polozky.length === 0) {
      posliJson(res, 400, { chyba: 'Objednávka potřebuje číslo stolu a aspoň jednu položku.' });
      return;

    const objednavka = pridejObjednavku(objednavky, telo.stul, telo.polozky);
    rozesli('objednavka', objednavka);
    posliJson(res, 201, objednavka);
    return;

  if (req.method === 'GET' && url.pathname === '/') {
    posliJson(res, 200, { bistro: 'U Modré kotvy', objednavek: objednavky.length });
    return;
  }

  if (req.method === 'GET' && url.pathname === '/api/metriky') {
    posliJson(res, 200, { klientu: klienti.size });
    return;
  }

  if (req.method === 'GET' && url.pathname === '/api/udalosti') {
    res.writeHead(200, {
      'Content-Type': 'text/event-stream; charset=utf-8',
      'Cache-Control': 'no-cache',
      Connection: 'keep-alive',
    });
    res.flushHeaders();

    res.write(formatujUdalost({ typ: 'snapshot', data: objednavky }));
    klienti.add(res);

    // Komentářový rámec, který drží spojení otevřené i v hodinu ticha.
    const pingMs = Number(process.env.PING_MS ?? 15000);
    const ping = setInterval(() => res.write(': ping\n\n'), pingMs);

    // Když obrazovka spojení zavře, ať v registru nezůstane a časovač po ní taky ne.
    req.on('close', () => {
      clearInterval(ping);
      klienti.delete(res);
    });
    return;

import { createServer } from 'node:http';
import { objednavky, pridejObjednavku } from './objednavky.js';
import { formatujUdalost } from './sse.js';

const port = Number(process.env.PORT ?? 3000);

// Otevřené proudy kuchyňských obrazovek. Do každého z nich server píše události.
const klienti = new Set();

// Posledních 50 rozeslaných událostí a číslo té poslední. Podle nich se dá
// obrazovce po výpadku doplnit, co zmeškala.
const historie = [];
let posledniId = 0;

// Každá JSON odpověď serveru jde přes tuhle funkci.
function posliJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Pošle jednu událost všem obrazovkám, které zrovna poslouchají.
function rozesli(typ, data) {
  posledniId += 1;
  const udalost = { id: posledniId, typ, data };
  historie.push(udalost);
  if (historie.length > 50) historie.shift();
  const ramec = formatujUdalost(udalost);
  for (const klient of klienti) {
    klient.write(ramec);
  }
}

    return null;

}
