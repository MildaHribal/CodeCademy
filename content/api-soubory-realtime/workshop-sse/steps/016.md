# --description--

Poslední krok: na straně prohlížeče. A je překvapivě krátký — o celý protokol,
rozdělení na rámce i znovupřipojení se stará objekt [[EventSource]]. Ty jen
řekneš, kam se připojit a co dělat s jednotlivými typy událostí.

**Úkol:** v `public/kuchyn.js` otevři proud na adrese `/api/udalosti`
a zpracuj tři typy událostí, které server posílá:

| událost | co s ní |
|---|---|
| `snapshot` | data jsou celý seznam — ulož ho do `stav` a vykresli |
| `objednavka` | data jsou jedna nová objednávka — přidej ji do `stav` a vykresli |
| `ceka` | data jsou `{ ceka: <počet> }` — číslo zapiš do `pocitadlo` |

Na každý typ si zaregistruj vlastní posluchače. V obsluze dostaneš objekt
události a data v jeho vlastnosti `data` — jsou to vždycky **text**, takže
je nejdřív rozbal z JSON. Kreslení už máš hotové ve funkci `vykresli`.

Po **Spustit** otevři `http://localhost:3000` a ve druhém okně pošli
`POST /api/objednavky`. Objednávka na obrazovce naskočí sama, bez obnovení —
a když server na chvíli zastavíš a pustíš znovu, obrazovka se připojí sama.

## Co máš hotové

Server, který drží desítky otevřených spojení, posílá do nich změny v okamžiku,
kdy nastanou, uklidí po odpojené obrazovce, drží spojení naživu tlukotem a po
výpadku dožene, co uteklo. To je celé [[Server-Sent Events|SSE]] — a stojí to
na obyčejném HTTP, bez jediné knihovny.

Příště přijde lab, ve kterém si stejný proud postavíš sám, ale s ostrahou:
notifikace uvidí jen ten, komu patří.

## Udělej po svém

- Dej hotovým objednávkám možnost zmizet: přidej `PATCH /api/objednavky/:id`,
  který změní stav na `hotova`, a rozešli o tom událost.
- Ať je na obrazovce vidět, že spojení žije — třeba tečka, která se při každé
  události na chvíli rozsvítí.
- Přebarvi objednávky podle stáří: co čeká přes deset minut, ať je vidět na
  první pohled. Čas přijetí v datech už máš.
- Přehoď to na téma, které tě zajímá: výdejní okénko lékárny, semafor stavu
  serverů, tabule odletů. Protokol zůstane, změní se jen data a vzhled.

```js
// Objednávky bistra U Modré kotvy. Držíme je v paměti — po restartu serveru
// se seznam vrátí do tohohle stavu, databáze tu není potřeba.
export const objednavky = [
  { id: 1, stul: 3, polozky: ['Svíčková', 'Kofola'], stav: 'nova', prijato: '11:42' },
  { id: 2, stul: 7, polozky: ['Guláš', 'Pivo', 'Štrúdl'], stav: 'hotova', prijato: '11:45' },
  { id: 3, stul: 1, polozky: ['Smažený sýr', 'Limonáda'], stav: 'nova', prijato: '11:51' },
];

// Čas ve tvaru HH:MM, jak ho kuchyň čte z lístku.
export function cas(ted = new Date()) {
  return ted.toTimeString().slice(0, 5);
}

export function dalsiId(seznam) {
  return seznam.reduce((nejvyssi, objednavka) => Math.max(nejvyssi, objednavka.id), 0) + 1;
}

export function pridejObjednavku(seznam, stul, polozky) {
  const objednavka = { id: dalsiId(seznam), stul, polozky, stav: 'nova', prijato: cas() };
  seznam.push(objednavka);
  return objednavka;
}

export function pocetCekajicich(seznam) {
  return seznam.filter((objednavka) => objednavka.stav === 'nova').length;
}
```

```js
// Jeden rámec Server-Sent Events: řádky s klíčem a prázdný řádek na konci.
export function formatujUdalost({ id, typ, data }) {
  let ramec = '';
  if (id !== undefined) ramec += `id: ${id}\n`;
  if (typ) ramec += `event: ${typ}\n`;
  ramec += `data: ${JSON.stringify(data)}\n`;
  return ramec + '\n';
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { readFile } from 'node:fs/promises';
import { objednavky, pridejObjednavku, pocetCekajicich } from './objednavky.js';
import { formatujUdalost } from './sse.js';

const port = Number(process.env.PORT ?? 3000);

// Otevřené proudy kuchyňských obrazovek. Do každého z nich server píše události.
const klienti = new Set();

// Posledních 50 rozeslaných událostí a číslo té poslední. Podle nich se dá
// obrazovce po výpadku doplnit, co zmeškala.
const historie = [];
let posledniId = 0;

// Každá JSON odpověď serveru jde přes tuhle funkci.
function posliJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Pošle jednu událost všem obrazovkám, které zrovna poslouchají.
function rozesli(typ, data) {
  posledniId += 1;
  const udalost = { id: posledniId, typ, data };
  historie.push(udalost);
  if (historie.length > 50) historie.shift();
  const ramec = formatujUdalost(udalost);
  for (const klient of klienti) {
    klient.write(ramec);
  }
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí null.
async function prectiJson(req) {
  const kousky = [];
  for await (const kousek of req) kousky.push(kousek);
  if (kousky.length === 0) return {};
  try {
    return JSON.parse(Buffer.concat(kousky).toString('utf8'));
  } catch {
    return null;
  }
}

const server = createServer(async (req, res) => {
  const url = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && url.pathname === '/api/objednavky') {
    posliJson(res, 200, objednavky);
    return;
  }

  if (req.method === 'POST' && url.pathname === '/api/objednavky') {
    const telo = await prectiJson(req);
    if (!telo || typeof telo.stul !== 'number' || !Array.isArray(telo.polozky) || telo.polozky.length === 0) {
      posliJson(res, 400, { chyba: 'Objednávka potřebuje číslo stolu a aspoň jednu položku.' });
      return;
    }
    const objednavka = pridejObjednavku(objednavky, telo.stul, telo.polozky);
    rozesli('objednavka', objednavka);
    rozesli('ceka', { ceka: pocetCekajicich(objednavky) });
    posliJson(res, 201, objednavka);
    return;
  }

  if (req.method === 'GET' && url.pathname === '/') {
    const stranka = await readFile(new URL('./public/kuchyn.html', import.meta.url));
    res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
    res.end(stranka);
    return;
  }

  if (req.method === 'GET' && url.pathname === '/kuchyn.js') {
    const skript = await readFile(new URL('./public/kuchyn.js', import.meta.url));
    res.writeHead(200, { 'Content-Type': 'text/javascript; charset=utf-8' });
    res.end(skript);
    return;
  }

  if (req.method === 'GET' && url.pathname === '/api/metriky') {
    posliJson(res, 200, { klientu: klienti.size });
    return;
  }

  if (req.method === 'GET' && url.pathname === '/api/udalosti') {
    res.writeHead(200, {
      'Content-Type': 'text/event-stream; charset=utf-8',
      'Cache-Control': 'no-cache',
      Connection: 'keep-alive',
    });
    res.flushHeaders();

    // Kdo se vrací po výpadku, pošle číslo poslední události, kterou viděl.
    const odkud = Number(req.headers['last-event-id'] ?? 0);
    if (odkud > 0) {
      for (const zmeskana of historie.filter((udalost) => udalost.id > odkud)) {
        res.write(formatujUdalost(zmeskana));
      }
    } else {
      res.write(formatujUdalost({ typ: 'snapshot', data: objednavky }));
    }
    klienti.add(res);

    // Komentářový rámec, který drží spojení otevřené i v hodinu ticha.
    const pingMs = Number(process.env.PING_MS ?? 15000);
    const ping = setInterval(() => res.write(': ping\n\n'), pingMs);

    // Když obrazovka spojení zavře, ať v registru nezůstane a časovač po ní taky ne.
    req.on('close', () => {
      clearInterval(ping);
      klienti.delete(res);
    });
    return;
  }

  posliJson(res, 404, { chyba: `Cesta ${req.method} ${url.pathname} tady není.` });
});

server.listen(port, () => {
  console.log(`Kuchyň bistra běží na http://localhost:${port}`);
});
```

```js
// Kuchyňská obrazovka. Tenhle soubor běží v prohlížeči.
const seznam = document.querySelector('#objednavky');
const pocitadlo = document.querySelector('#ceka');

// Jeden řádek objednávky. Vzhled řeší kuchyn.html, tady skládáme jen obsah.
function radek(objednavka) {
  return `<li class="objednavka" data-stav="${objednavka.stav}">
      <span class="stul">Stůl ${objednavka.stul}</span>
      <span class="polozky">${objednavka.polozky.join(', ')}</span>
      <time>${objednavka.prijato}</time>
    </li>`;
}

function vykresli(objednavky) {
  seznam.innerHTML = objednavky.map(radek).join('');
}

// Poslední známý stav kuchyně. Události ho mění, vykresli() ho kreslí.
let stav = [];

const proud = new EventSource('/api/udalosti');

proud.addEventListener('snapshot', (udalost) => {
  stav = JSON.parse(udalost.data);
  vykresli(stav);
});

proud.addEventListener('objednavka', (udalost) => {
  stav = [...stav, JSON.parse(udalost.data)];
  vykresli(stav);
});

proud.addEventListener('ceka', (udalost) => {
  pocitadlo.textContent = JSON.parse(udalost.data).ceka;
});
```
