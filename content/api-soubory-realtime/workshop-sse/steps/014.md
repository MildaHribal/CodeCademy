# --description--

Kuchař chce nad seznamem velké číslo: **kolik jídel ještě čeká**. Kdyby si ho
obrazovka počítala sama, každá by ho počítala po svém a po výpadku by byla
vedle. Spočítá ho tedy server a pošle ho jako vlastní událost.

**Úkol:**

1. v `objednavky.js` napiš a vyexportuj funkci `pocetCekajicich(seznam)`,
   která vrátí, kolik objednávek má stav `'nova'`. Prázdný seznam vrátí nulu.
2. v `server.js` ji naimportuj a po rozeslání nové objednávky rozešli ještě
   jednu událost — typ `ceka`, data `{ ceka: <počet> }`.

Jak počet spočítáš, je čistě na tobě: nic ti nepředepisuju a testy nekoukají na
to, jakou cestou jsi k číslu došel. Vyber, co je tady nejčitelnější.

Že má počítadlo vlastní událost, a ne že se přibalí k objednávce, není náhoda:
obrazovka pak může každou z nich zpracovat zvlášť a později přibude třeba
událost o jídle, které už je hotové.

const otevriProud = async (adresa, hlavicky = {}) => {
  const stop = new AbortController();
  const res = await fetch(adresa, { headers: hlavicky, signal: stop.signal });
  const ctecka = res.body.getReader();
  const dekoder = new TextDecoder();
  let text = '';
  const dalsi = async ({ limit = 4000, komentare = false } = {}) => {
    for (;;) {
      const konec = text.indexOf('\n\n');
      if (konec !== -1) {
        const kus = text.slice(0, konec);
        text = text.slice(konec + 2);
        if (!komentare && kus.split('\n').every((radek) => radek.startsWith(':'))) continue;
        return kus;
      }
      const kousek = await Promise.race([ctecka.read(), helpers.wait(limit).then(() => 'cas')]);
      if (kousek === 'cas' || kousek.done) return null;
      text += dekoder.decode(kousek.value, { stream: true });
    }
  };
  const rozloz = (ramec) => {
    const pole = { data: [] };
    for (const radek of (ramec ?? '').split('\n')) {
      const dvojtecka = radek.indexOf(':');
      if (dvojtecka <= 0) continue;
      const klic = radek.slice(0, dvojtecka);
      const hodnota = radek.slice(dvojtecka + 1).replace(/^ /, '');
      if (klic === 'data') pole.data.push(hodnota);
      else pole[klic] = hodnota;
    }
    return { ...pole, data: pole.data.join('\n') };
  };
  return { res, dalsi, rozloz, zavri: () => stop.abort() };
};
const server = await helpers.startServer('server.js');
const proud = await otevriProud(new URL('/api/udalosti', server.url));
try {
  await proud.dalsi();
  await fetch(new URL('/api/objednavky', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ stul: 3, polozky: ['Svíčková'] }),
  });
  const objednavka = proud.rozloz(await proud.dalsi());
  const ceka = proud.rozloz(await proud.dalsi());
  assert.equal(objednavka.event, 'objednavka', 'Nejdřív dorazí událost s novou objednávkou');
  assert.equal(ceka.event, 'ceka', `Hned za ní má dorazit událost ceka, přišlo: ${JSON.stringify(ceka)}`);
  assert.deepEqual(JSON.parse(ceka.data), { ceka: 3 }, 'V seznamu jsou dvě objednávky ve stavu nova, třetí právě přibyla — čeká jich tedy 3');
} finally {
  proud.zavri();
}

# --help--

## --tip--

Ze seznamu potřebuješ **jedno číslo** — kolik prvků splňuje podmínku. Jde to spočítat cyklem, vybráním podmnožiny i sečtením jedniček; všechny tři cesty jsou správně.

## --tip--

V serveru voláš rozeslání dvakrát za sebou, ve dvou řádcích. Počet si nech spočítat až ve chvíli, kdy je nová objednávka v seznamu — jinak ti bude chybět.

// Jeden rámec Server-Sent Events: řádky s klíčem a prázdný řádek na konci.
export function formatujUdalost({ id, typ, data }) {
  let ramec = '';
  if (id !== undefined) ramec += `id: ${id}\n`;
  if (typ) ramec += `event: ${typ}\n`;
  ramec += `data: ${JSON.stringify(data)}\n`;
  return ramec + '\n';
}

import { objednavky, pridejObjednavku, pocetCekajicich } from './objednavky.js';
import { formatujUdalost } from './sse.js';

const port = Number(process.env.PORT ?? 3000);

function posliJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

    return JSON.parse(Buffer.concat(kousky).toString('utf8'));

    return null;
