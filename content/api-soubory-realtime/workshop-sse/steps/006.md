# --description--

Proud je otevřený, ale zatím do něj nic neteče. Než začneš posílat objednávky,
potřebuješ funkci, která z dat udělá **jeden rámec** v přesném tvaru, jakému
prohlížeč rozumí. Ten tvar je celé kouzlo [[Server-Sent Events|SSE]]: obyčejné
řádky `klíč: hodnota` a prázdný řádek, který říká „a dost, tohle byla celá
událost".

| řádek | k čemu je | povinný |
|---|---|---|
| `id: 7` | číslo události, podle kterého se klient po výpadku dohledá | ne |
| `event: objednavka` | jméno události; bez něj prohlížeč hlásí `message` | ne |
| `data: {"stul":3}` | užitečný obsah, prakticky vždycky JSON | ano |

**Úkol:** seřaď řádky funkce `formatujUdalost` v novém souboru `sse.js`.
Funkce dostane objekt s `id`, `typ` a `data` a vrátí hotový text rámce.
`id` a `typ` můžou chybět — pak se ty řádky nepíšou vůbec. Server funkci
zapojíš hned v dalším kroku. Dva řádky jsou navíc a jedna mezera čeká na
doplnění.

> [!PITFALL]
> Hodnota se do rámce vejde jen jako text. Objekt vložený přímo se změní na
> `[object Object]` a klient na druhé straně dostane nesmysl.

# --hints--

`formatujUdalost` složí rámec z řádků `id:`, `event:` a `data:` a ukončí ho prázdným řádkem.

```js
const { formatujUdalost } = await helpers.importFile('sse.js');
assert.equal(typeof formatujUdalost, 'function', 'sse.js má vyexportovat funkci formatujUdalost');
const ramec = formatujUdalost({ id: 4, typ: 'objednavka', data: { id: 7, stul: 3 } });
assert.ok(ramec.endsWith('\n\n'), `Rámec události má končit prázdným řádkem, přišlo: ${JSON.stringify(ramec)}`);
assert.deepEqual(
  ramec.trimEnd().split('\n'),
  ['id: 4', 'event: objednavka', 'data: {"id":7,"stul":3}'],
  'formatujUdalost({ id: 4, typ: "objednavka", data: { id: 7, stul: 3 } }) má dát řádky id:, event: a data: v tomhle pořadí',
);
```

Chybějící `id` nebo `typ` se do rámce nepíšou vůbec.

```js
const { formatujUdalost } = await helpers.importFile('sse.js');
const bezId = formatujUdalost({ typ: 'snapshot', data: [1, 2] });
assert.deepEqual(
  bezId.trimEnd().split('\n'),
  ['event: snapshot', 'data: [1,2]'],
  'Když id chybí, řádek id: se do rámce nepíše',
);
const holy = formatujUdalost({ data: { ok: true } });
assert.deepEqual(holy.trimEnd().split('\n'), ['data: {"ok":true}'], 'Bez id i bez typu zbude jediný řádek data:');
assert.ok(holy.endsWith('\n\n'), 'I ten nejkratší rámec končí prázdným řádkem');
```

Data jdou do rámce jako JSON.

```js
const { formatujUdalost } = await helpers.importFile('sse.js');
const ramec = formatujUdalost({ typ: 'ceka', data: { ceka: 2, stul: null } });
assert.doesNotMatch(ramec, /\[object Object\]/, 'Data patří do rámce jako JSON, ne jako automaticky převedený objekt');
const radek = ramec.split('\n').find((r) => r.startsWith('data: '));
assert.deepEqual(JSON.parse(radek.slice('data: '.length)), { ceka: 2, stul: null }, 'Řádek data: má obsahovat data převedená do JSON');
```

# --help--

## --tip--

Nejdřív si rozmysli, co funkce vrací: jeden kus textu, který postupně roste. Potřebuješ tedy proměnnou, do které se dá přidávat, a až na konci `return`.

## --tip--

Řádky `id:` a `event:` jsou nepovinné — každý z nich přidáš jen za podmínky. Na `id` se ptej na `undefined`, ne na pravdivost: nula je platné číslo události, ale nepravdivá hodnota.

## --tip--

Konec rámce pozná klient podle **prázdného řádku**. Každý řádek už jeden `\n` má, takže na konci musí být ještě jeden navíc.

```js
// Objednávky bistra U Modré kotvy. Držíme je v paměti — po restartu serveru
// se seznam vrátí do tohohle stavu, databáze tu není potřeba.
export const objednavky = [
  { id: 1, stul: 3, polozky: ['Svíčková', 'Kofola'], stav: 'nova', prijato: '11:42' },
  { id: 2, stul: 7, polozky: ['Guláš', 'Pivo', 'Štrúdl'], stav: 'hotova', prijato: '11:45' },
  { id: 3, stul: 1, polozky: ['Smažený sýr', 'Limonáda'], stav: 'nova', prijato: '11:51' },
];

// Čas ve tvaru HH:MM, jak ho kuchyň čte z lístku.
export function cas(ted = new Date()) {
  return ted.toTimeString().slice(0, 5);
}

export function dalsiId(seznam) {
  return seznam.reduce((nejvyssi, objednavka) => Math.max(nejvyssi, objednavka.id), 0) + 1;
}

export function pridejObjednavku(seznam, stul, polozky) {
  const objednavka = { id: dalsiId(seznam), stul, polozky, stav: 'nova', prijato: cas() };
  seznam.push(objednavka);
  return objednavka;
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { objednavky, pridejObjednavku } from './objednavky.js';

const port = Number(process.env.PORT ?? 3000);

// Každá JSON odpověď serveru jde přes tuhle funkci.
function posliJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí null.
async function prectiJson(req) {
  const kousky = [];
  for await (const kousek of req) kousky.push(kousek);
  if (kousky.length === 0) return {};
  try {
    return JSON.parse(Buffer.concat(kousky).toString('utf8'));
  } catch {
    return null;
  }
}

const server = createServer(async (req, res) => {
  const url = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && url.pathname === '/api/objednavky') {
    posliJson(res, 200, objednavky);
    return;
  }

  if (req.method === 'POST' && url.pathname === '/api/objednavky') {
    const telo = await prectiJson(req);
    if (!telo || typeof telo.stul !== 'number' || !Array.isArray(telo.polozky) || telo.polozky.length === 0) {
      posliJson(res, 400, { chyba: 'Objednávka potřebuje číslo stolu a aspoň jednu položku.' });
      return;
    }
    posliJson(res, 201, pridejObjednavku(objednavky, telo.stul, telo.polozky));
    return;
  }

  if (req.method === 'GET' && url.pathname === '/') {
    posliJson(res, 200, { bistro: 'U Modré kotvy', objednavek: objednavky.length });
    return;
  }

  if (req.method === 'GET' && url.pathname === '/api/udalosti') {
    res.writeHead(200, {
      'Content-Type': 'text/event-stream; charset=utf-8',
      'Cache-Control': 'no-cache',
      Connection: 'keep-alive',
    });
    res.flushHeaders();
    return;
  }

  posliJson(res, 404, { chyba: `Cesta ${req.method} ${url.pathname} tady není.` });
});

server.listen(port, () => {
  console.log(`Kuchyň bistra běží na http://localhost:${port}`);
});
```

```js
// Kuchyňská obrazovka zatím nic živého nemá. Tenhle skript dělá to, co by musela:
// ptá se pořád dokola, jestli nepřibyla objednávka.
const api = process.env.API_URL ?? 'http://localhost:3000';
const pokusu = 8;
const pauza = 50;

let predchozi = null;
let zbytecnych = 0;

for (let pokus = 0; pokus < pokusu; pokus += 1) {
  const odpoved = await fetch(`${api}/api/objednavky`);
  const telo = await odpoved.text();
  if (telo === predchozi) zbytecnych += 1;
  predchozi = telo;
  await new Promise((hotovo) => setTimeout(hotovo, pauza));
}

console.log(JSON.stringify({ pozadavku: pokusu, zbytecnych }));
```

```js
export function formatujUdalost({ id, typ, data }) {
  let ramec = '';
  if (id !== undefined) ramec += `id: ${id}\n`;
  if (typ) ramec += `event: ${typ}\n`;
  ramec += `data: ${__1__}\n`;
  return ramec + '\n';
}
```
