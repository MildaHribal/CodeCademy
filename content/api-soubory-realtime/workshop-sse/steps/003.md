# --description--

Aby měla kuchyň co zobrazovat, musí objednávka nejdřív do systému přijít. Tuhle
adresu bude volat pokladna číšníka.

**Úkol:** v `server.js` obsluž `POST /api/objednavky`:

1. tělo požadavku přečti funkcí `prectiJson`, která už v souboru je,
2. když tělo chybí, není platný JSON, nemá `stul` jako číslo nebo nemá
   neprázdné pole `polozky`, odpověz stavem `400` a tělem
   `{ chyba: 'Objednávka potřebuje číslo stolu a aspoň jednu položku.' }`,
3. jinak objednávku ulož funkcí `pridejObjednavku` a vrať stav `201`
   s uloženou objednávkou.

`pridejObjednavku` musíš doimportovat z `./objednavky.js` — vedle
`objednavky` na prvních řádcích souboru.

Stav `201` znamená „vytvořil jsem nový záznam"; `200` by tvrdilo, že se jen
něco vrátilo. Klientovi to není jedno: podle stavu pozná, jestli má přidat řádek
do tabulky, nebo jen překreslit.

> [!PITFALL]
> Klient si `id` neposílá. Kdyby sis ho vzal z těla, dva číšníci by si během
> oběda přepsali objednávky navzájem.

# --hints--

`POST /api/objednavky` s platným tělem vrátí stav `201` a uloženou objednávku.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(new URL('/api/objednavky', server.url), {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ stul: 4, polozky: ['Utopenec', 'Chleba'] }),
});
assert.equal(res.status, 201, `POST /api/objednavky má vrátit stav 201, přišlo ${res.status}. Výstup serveru:\n${server.output()}`);
const nova = await res.json();
assert.equal(nova.stul, 4, 'Odpověď má obsahovat uloženou objednávku se stolem 4');
assert.deepEqual(nova.polozky, ['Utopenec', 'Chleba'], 'Odpověď má obsahovat položky tak, jak přišly');
assert.equal(nova.stav, 'nova', "Uložená objednávka má stav 'nova'");
assert.equal(nova.id, 4, 'Nová objednávka má dostat id 4 — v seznamu už jsou tři');
```

Uložená objednávka je hned vidět v `GET /api/objednavky`.

```js
const server = await helpers.startServer('server.js');
await fetch(new URL('/api/objednavky', server.url), {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ stul: 9, polozky: ['Svíčková'] }),
});
const seznam = await (await fetch(new URL('/api/objednavky', server.url))).json();
assert.equal(seznam.length, 4, `Po jednom POST má GET /api/objednavky vracet čtyři objednávky, vrátil ${seznam.length}`);
assert.equal(seznam.at(-1).stul, 9, 'Poslední objednávka v seznamu má být ta právě uložená');
```

Objednávka bez stolu, bez položek, s prázdnými položkami nebo s tělem, které není JSON, skončí stavem `400` a nic neuloží.

```js
const server = await helpers.startServer('server.js');
const posli = (body) => fetch(new URL('/api/objednavky', server.url), {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body,
});
const spatna = [
  JSON.stringify({ polozky: ['Guláš'] }),
  JSON.stringify({ stul: '4', polozky: ['Guláš'] }),
  JSON.stringify({ stul: 4 }),
  JSON.stringify({ stul: 4, polozky: [] }),
  'tohle není json',
];
for (const telo of spatna) {
  const res = await posli(telo);
  assert.equal(res.status, 400, `Tělo ${telo} má skončit stavem 400, přišlo ${res.status}`);
  assert.equal(typeof (await res.json()).chyba, 'string', `Odpověď na tělo ${telo} má mít v těle klíč chyba`);
}
const seznam = await (await fetch(new URL('/api/objednavky', server.url))).json();
assert.equal(seznam.length, 3, 'Po odmítnutých požadavcích nesmí v seznamu přibýt objednávka');
```

# --help--

## --tip--

Tělo požadavku je proud a čte se asynchronně — obsluha je už `async`, takže na výsledek `prectiJson` stačí počkat. Proč se tělo nedá přečíst jako vlastnost, připomene [Tělo požadavku je proud](see:node-zaklady/http-v-node#telo-pozadavku-je-proud).

## --tip--

Ověř všechno, co potřebuješ, **jednou podmínkou** hned na začátku a hned odpověz `400`. Na pole se ptej přes `Array.isArray`, na jeho délku hned za tím — viz [Nikdy nevěř klientovi](see:api-http-rest/validace-a-chyby-api#nikdy-never-klientovi).

## --tip--

Vzor nad jinými daty: `if (!telo || typeof telo.email !== 'string') { posliJson(res, 400, { chyba: 'Chybí e-mail.' }); return; }`

```js
// Objednávky bistra U Modré kotvy. Držíme je v paměti — po restartu serveru
// se seznam vrátí do tohohle stavu, databáze tu není potřeba.
export const objednavky = [
  { id: 1, stul: 3, polozky: ['Svíčková', 'Kofola'], stav: 'nova', prijato: '11:42' },
  { id: 2, stul: 7, polozky: ['Guláš', 'Pivo', 'Štrúdl'], stav: 'hotova', prijato: '11:45' },
  { id: 3, stul: 1, polozky: ['Smažený sýr', 'Limonáda'], stav: 'nova', prijato: '11:51' },
];

// Čas ve tvaru HH:MM, jak ho kuchyň čte z lístku.
export function cas(ted = new Date()) {
  return ted.toTimeString().slice(0, 5);
}

export function dalsiId(seznam) {
  return seznam.reduce((nejvyssi, objednavka) => Math.max(nejvyssi, objednavka.id), 0) + 1;
}

export function pridejObjednavku(seznam, stul, polozky) {
  const objednavka = { id: dalsiId(seznam), stul, polozky, stav: 'nova', prijato: cas() };
  seznam.push(objednavka);
  return objednavka;
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
```
```js
import { objednavky, pridejObjednavku } from './objednavky.js';

const port = Number(process.env.PORT ?? 3000);

// Každá JSON odpověď serveru jde přes tuhle funkci.
function posliJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí null.
async function prectiJson(req) {
  const kousky = [];
  for await (const kousek of req) kousky.push(kousek);
  if (kousky.length === 0) return {};
  try {
    return JSON.parse(Buffer.concat(kousky).toString('utf8'));
  } catch {
    return null;
  }
}

const server = createServer(async (req, res) => {
  const url = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && url.pathname === '/api/objednavky') {
    posliJson(res, 200, objednavky);
    return;
  }

  if (req.method === 'POST' && url.pathname === '/api/objednavky') {
    const telo = await prectiJson(req);
    if (!telo || typeof telo.stul !== 'number' || !Array.isArray(telo.polozky) || telo.polozky.length === 0) {
      posliJson(res, 400, { chyba: 'Objednávka potřebuje číslo stolu a aspoň jednu položku.' });
      return;
    }
    posliJson(res, 201, pridejObjednavku(objednavky, telo.stul, telo.polozky));
    return;
  }

  if (req.method === 'GET' && url.pathname === '/') {
    posliJson(res, 200, { bistro: 'U Modré kotvy', objednavek: objednavky.length });
    return;
  }

  posliJson(res, 404, { chyba: `Cesta ${req.method} ${url.pathname} tady není.` });
});

server.listen(port, () => {
  console.log(`Kuchyň bistra běží na http://localhost:${port}`);
});
```
