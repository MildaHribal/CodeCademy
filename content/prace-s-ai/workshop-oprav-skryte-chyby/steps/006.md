# --description--

Chybu už máš doloženou dvěma protipříklady: neznámý kód shodí výpočet a sleva `VIP500`
umí na malé objednávce stáhnout cenu pod nulu (objednávka za 300 Kč vyjde na −200 Kč).
Obojí je jeden a týž problém: výpočet nepočítá s tím, že vstup nemusí dávat smysl.

**Úkol:** seřaď řádky tak, aby vznikla funkce `cenaPoSleve(soucet, kod)`, která:

- u neznámého kódu vrátí částku beze změny,
- u procentní slevy odečte daný podíl,
- u částkové slevy odečte částku, ale nikdy nespadne pod nulu.

Odsazení nastavíš klávesou Tab, dva řádky jsou navíc. Do mezery doplň, co tam patří.
Starou `pouzijSlevu` zatím nech být, vyřídíme ji v dalším kroku.

# --hints--

`cenaPoSleve` zvládne oba druhy slevy.

```js
assert.equal(cenaPoSleve(1000, 'JARO10'), 900, "cenaPoSleve(1000, 'JARO10') má vrátit 900 (sleva 10 %)");
assert.equal(cenaPoSleve(1000, 'VIP500'), 500, "cenaPoSleve(1000, 'VIP500') má vrátit 500");
assert.equal(cenaPoSleve(2000, 'DOPRAVA'), 2000, "cenaPoSleve(2000, 'DOPRAVA') nemá cenu zboží měnit");
```

Neznámý kód cenu nezmění a nic neshodí.

```js
assert.equal(cenaPoSleve(1000, protipriklad3()), 1000, 'neznámý kód má vrátit původní částku');
assert.equal(cenaPoSleve(1000, ''), 1000, 'prázdný kód má vrátit původní částku');
```

Sleva nikdy nestáhne cenu pod nulu.

```js
assert.equal(cenaPoSleve(300, 'VIP500'), 0, "cenaPoSleve(300, 'VIP500') má vrátit 0, ne záporné číslo");
assert.equal(cenaPoSleve(0, 'VIP500'), 0, "cenaPoSleve(0, 'VIP500') má vrátit 0");
```

```js
// kosik.js — tenhle soubor napsal model ze zadání:
// "Modul pro košík e-shopu Cyklo Ráže: součet položek, slevový kód,
//  doprava zdarma od 1500 Kč, popis objednávky do e-mailu."
// Nikdo ho pak nečetl. Prošel, protože na ukázkovém košíku vrací správná čísla.

const SLEVOVE_KODY = [
  { kod: 'JARO10', typ: 'procenta', hodnota: 10 },
  { kod: 'VIP500', typ: 'castka', hodnota: 500 },
  { kod: 'DOPRAVA', typ: 'doprava', hodnota: 0 },
];

const DOPRAVA_ZDARMA_OD = 1500;
const CENA_DOPRAVY = 99;

const UKAZKOVY_KOSIK = [
  { nazev: 'Helma Ráže Air', cena: 1290, pocet: 1 },
  { nazev: 'Blikačka zadní', cena: 250, pocet: 2 },
];

function soucetPolozek(polozky) {
  return polozky.reduce((soucet, polozka) => soucet + polozka.cena * polozka.pocet, 0);
}

function cenaDopravy(soucet) {
  return soucet >= DOPRAVA_ZDARMA_OD ? 0 : CENA_DOPRAVY;
}

function najdiKod(kod) {
  return SLEVOVE_KODY.find((zaznam) => zaznam.kod === kod);
}

function maKod(kod) {
  const poradi = SLEVOVE_KODY.findIndex((zaznam) => zaznam.kod === kod);
  if (poradi) {
    return true;
  }
  return false;
}

function seradPodleCeny(polozky) {
  return polozky.sort((a, b) => a.cena - b.cena);
}

function popisObjednavky(jmeno, poznamka) {
  return '<p>Objednávka pro ' + jmeno + ': ' + poznamka + '</p>';
}

function pouzijSlevu(soucet, kod) {
  const zaznam = najdiKod(kod);
  if (zaznam.typ === 'procenta') {
    return soucet - (soucet * zaznam.hodnota) / 100;
  }
  return soucet - zaznam.hodnota;
}

function celkovaCena(polozky, kod) {
  const poSleve = pouzijSlevu(soucetPolozek(polozky), kod);
  return poSleve + cenaDopravy(poSleve);
}
```

```js
function cenaPoSleve(soucet, kod) {
  const zaznam = najdiKod(kod);
  if (!zaznam) {
    return soucet;
  }
  if (zaznam.typ === 'procenta') {
    return soucet - (soucet * zaznam.hodnota) / 100;
  }
  return __1__(0, soucet - zaznam.hodnota);
}
```
