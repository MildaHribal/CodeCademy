# --description--

Poslední tichá chyba v modulu není chyba v kódu, ale v tom, jak se kód chová, když mu
někdo pošle nesmysl. Když do košíku propadne položka bez ceny nebo bez počtu, součet
vyjde jako `NaN`, projde všemi dalšími výpočty a na faktuře skončí jako „NaN Kč".
Nikde po cestě se nic neozve.

**Úkol:** uprav `soucetPolozek` tak, aby položku bez číselné ceny nebo bez číselného
počtu nespočítala, ale vyhodila `Error`, jehož zpráva obsahuje název té položky.
Platné košíky včetně prázdného musí vycházet pořád stejně.

Dobrá chybová zpráva říká, která položka je vadná — jinak budeš hledat jehlu v košíku
o padesáti řádcích.

# --hints--

Platné košíky se počítají pořád stejně.

```js
assert.equal(soucetPolozek(UKAZKOVY_KOSIK), 1790, 'soucetPolozek(UKAZKOVY_KOSIK) má pořád vrátit 1790');
assert.equal(soucetPolozek([]), 0, 'soucetPolozek([]) má pořád vrátit 0');
```

Položka bez počtu skončí chybou, která ji pojmenuje.

```js
let chyba = null;
try {
  soucetPolozek([{ nazev: 'Bidon 0,7 l', cena: 99 }]);
} catch (zachycena) {
  chyba = zachycena;
}
assert.ok(chyba instanceof Error, 'položka bez počtu má vyhodit Error, ne vrátit NaN');
assert.ok(String(chyba.message).includes('Bidon 0,7 l'), 'zpráva chyby má obsahovat název vadné položky');
```

Položka bez ceny skončí chybou taky.

```js
let chyba = null;
try {
  soucetPolozek([{ nazev: 'Duše 28"', cena: null, pocet: 1 }]);
} catch (zachycena) {
  chyba = zachycena;
}
assert.ok(chyba instanceof Error, 'položka bez ceny má vyhodit Error');
assert.ok(String(chyba.message).includes('Duše 28'), 'zpráva chyby má obsahovat název vadné položky');
```

# --help--

## --tip--

Proč se vyplatí selhat nahlas místo tiše, je v [Testech, které nemůžou selhat](see:prace-s-ai/review-kodu-z-ai#4-testy-ktere-nemuzou-selhat) a v pravidle o tichých chybách.

## --tip--

Do funkce, kterou `reduce` volá na každou položku, přidej kontrolu obou hodnot ještě předtím, než se cokoli přičte. Kontroluj typ, ne pravdivost — nula je platná cena.

## --tip--

Na jiných datech: funkce pro průměrnou známku takhle odmítne žáka, u kterého je známka zapsaná jako prázdný řetězec, a řekne jeho jméno.

# --explain--

Vysvětli vlastními slovy, proč je lepší vadnou položku odmítnout výjimkou než ji jen přeskočit a spočítat zbytek.

## --model--

Přeskočení vypadá vstřícně, ale vyrobí tiché špatné číslo: objednávka projde, zákazník
zaplatí míň, než měl, a nikdo se nic nedozví. Výjimka naopak zastaví výpočet v místě, kde je
problém ještě vidět, a zpráva s názvem položky rovnou ukazuje, kde se data pokazila. Rozhodující
je, že špatná data znamenají chybu někde výš — v importu, ve formuláři, v API — a tu chci opravit
tam, ne ji potichu obcházet v součtu.

## --checklist--

- Přeskočení vyrobí tiše špatné číslo, které nikdo nezachytí.
- Výjimka zastaví výpočet tam, kde je příčina ještě vidět.
- Zpráva s názvem položky ukazuje, kde se data pokazila.
- Vadná data mají příčinu výš a tu je potřeba opravit tam.

// kosik.js — tenhle soubor napsal model ze zadání:
// "Modul pro košík e-shopu Cyklo Ráže: součet položek, slevový kód,

const SLEVOVE_KODY = [
  { kod: 'JARO10', typ: 'procenta', hodnota: 10 },
  { kod: 'VIP500', typ: 'castka', hodnota: 500 },
  { kod: 'DOPRAVA', typ: 'doprava', hodnota: 0 },
];

const DOPRAVA_ZDARMA_OD = 1500;
const CENA_DOPRAVY = 99;

const UKAZKOVY_KOSIK = [
  { nazev: 'Helma Ráže Air', cena: 1290, pocet: 1 },
  { nazev: 'Blikačka zadní', cena: 250, pocet: 2 },
];

--edit--
function soucetPolozek(polozky) {
  return polozky.reduce((soucet, polozka) => soucet + polozka.cena * polozka.pocet, 0);
}
--edit--

function cenaDopravy(soucet) {
  return soucet >= DOPRAVA_ZDARMA_OD ? 0 : CENA_DOPRAVY;
}

function najdiKod(kod) {
  return SLEVOVE_KODY.find((zaznam) => zaznam.kod === kod);
}

function maKod(kod) {
  const poradi = SLEVOVE_KODY.findIndex((zaznam) => zaznam.kod === kod);
  if (poradi !== -1) {
    return true;
  }
  return false;
}

function seradPodleCeny(polozky) {
  return [...polozky].sort((a, b) => a.cena - b.cena);
}

function escapujHtml(text) {
  return String(text)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;');
}

function popisObjednavky(jmeno, poznamka) {
  return '<p>Objednávka pro ' + escapujHtml(jmeno) + ': ' + escapujHtml(poznamka) + '</p>';
}

function celkovaCena(polozky, kod) {
  const poSleve = cenaPoSleve(soucetPolozek(polozky), kod);
  return poSleve + cenaDopravy(poSleve);
}

function cenaPoSleve(soucet, kod) {
  const zaznam = najdiKod(kod);
  if (!zaznam) {
    return soucet;
  }
  if (zaznam.typ === 'procenta') {
    return soucet - (soucet * zaznam.hodnota) / 100;
  }
  return Math.max(0, soucet - zaznam.hodnota);
}

function shrnutiKosiku(polozky, kod) {
  const soucet = soucetPolozek(polozky);
  const poSleve = cenaPoSleve(soucet, kod);
  const doprava = cenaDopravy(poSleve);
  return { soucet, sleva: soucet - poSleve, doprava, celkem: poSleve + doprava };
}

// "Modul pro košík e-shopu Cyklo Ráže: součet položek, slevový kód,
//  doprava zdarma od 1500 Kč, popis objednávky do e-mailu."

  { kod: 'JARO10', typ: 'procenta', hodnota: 10 },

function soucetPolozek(polozky) {
  return polozky.reduce((soucet, polozka) => {
    if (typeof polozka.cena !== 'number' || typeof polozka.pocet !== 'number') {
      throw new Error('Položka "' + polozka.nazev + '" nemá platnou cenu nebo počet.');

}

  return soucet >= DOPRAVA_ZDARMA_OD ? 0 : CENA_DOPRAVY;
}

function najdiKod(kod) {
  return SLEVOVE_KODY.find((zaznam) => zaznam.kod === kod);
}

function maKod(kod) {
  const poradi = SLEVOVE_KODY.findIndex((zaznam) => zaznam.kod === kod);
  if (poradi !== -1) {
    return true;

  return false;
}

function seradPodleCeny(polozky) {
