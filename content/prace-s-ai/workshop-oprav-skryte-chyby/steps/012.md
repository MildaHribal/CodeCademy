# --description--

Sedm chyb je za tebou a modul je teď takový, jaký měl být od začátku. Tohle je dobré místo
na pauzu — do košíku se dá vrátit kdykoli.

Stránka objednávky potřebuje víc než jedno číslo: chce vypsat součet zboží, kolik z něj
ubrala sleva, kolik stojí doprava a kolik se zaplatí celkem.

**Úkol:** napiš funkci `shrnutiKosiku(polozky, kod)`, která vrátí objekt s vlastnostmi
`soucet`, `sleva`, `doprava` a `celkem`. Všechny čtyři hodnoty jsou částky v korunách.

Tenhle krok je opakování bez návodu: všechno, co k němu potřebuješ, už v modulu je
a testy ti řeknou přesná čísla. Doprava se počítá z částky po slevě, stejně jako
v `celkovaCena`. Nic nového nevymýšlej, jen to poskládej.

# --hints--

Shrnutí běžné objednávky se slevovým kódem sedí.

```js
assert.deepEqual(shrnutiKosiku(UKAZKOVY_KOSIK, 'JARO10'), { soucet: 1790, sleva: 179, doprava: 0, celkem: 1611 }, "shrnutiKosiku(UKAZKOVY_KOSIK, 'JARO10') má vrátit { soucet: 1790, sleva: 179, doprava: 0, celkem: 1611 }");
```

Shrnutí prázdného košíku i neznámého kódu projde bez pádu.

```js
assert.deepEqual(shrnutiKosiku([], protipriklad3()), { soucet: 0, sleva: 0, doprava: 99, celkem: 99 }, 'prázdný košík s neznámým kódem má vrátit { soucet: 0, sleva: 0, doprava: 99, celkem: 99 }');
```

Sleva větší než zboží nevyrobí záporné číslo.

```js
assert.deepEqual(shrnutiKosiku([{ nazev: 'Bidon 0,7 l', cena: 200, pocet: 1 }], 'VIP500'), { soucet: 200, sleva: 200, doprava: 99, celkem: 99 }, 'objednávka za 200 Kč s kódem VIP500 má mít slevu 200 a celkem 99');
```

# --help--

## --tip--

Funkce se skládá ze tří kroků, které už v modulu máš hotové: spočítat zboží, uplatnit slevu a dopočítat dopravu. Pořadí kroků je stejné jako v `celkovaCena`.

## --tip--

Sleva není hodnota z tabulky kódů — je to rozdíl mezi součtem zboží a částkou po slevě. Díky tomu vyjde správně i tam, kde sleva narazí na nulu.

## --tip--

Na jiných datech by stejně vypadalo shrnutí faktury: základ, odpočet, poplatek a celkem — každá položka spočítaná z té předchozí.

// kosik.js — tenhle soubor napsal model ze zadání:
// "Modul pro košík e-shopu Cyklo Ráže: součet položek, slevový kód,
//  doprava zdarma od 1500 Kč, popis objednávky do e-mailu."
// Nikdo ho pak nečetl. Prošel, protože na ukázkovém košíku vrací správná čísla.

const SLEVOVE_KODY = [
  { kod: 'JARO10', typ: 'procenta', hodnota: 10 },
  { kod: 'VIP500', typ: 'castka', hodnota: 500 },
  { kod: 'DOPRAVA', typ: 'doprava', hodnota: 0 },
];

const DOPRAVA_ZDARMA_OD = 1500;
const CENA_DOPRAVY = 99;

const UKAZKOVY_KOSIK = [
  { nazev: 'Helma Ráže Air', cena: 1290, pocet: 1 },
  { nazev: 'Blikačka zadní', cena: 250, pocet: 2 },
];

function soucetPolozek(polozky) {
  return polozky.reduce((soucet, polozka) => soucet + polozka.cena * polozka.pocet, 0);
}

function cenaDopravy(soucet) {
  return soucet >= DOPRAVA_ZDARMA_OD ? 0 : CENA_DOPRAVY;
}

function najdiKod(kod) {
  return SLEVOVE_KODY.find((zaznam) => zaznam.kod === kod);
}

function maKod(kod) {
  const poradi = SLEVOVE_KODY.findIndex((zaznam) => zaznam.kod === kod);
  if (poradi !== -1) {
    return true;
  }
  return false;
}

function seradPodleCeny(polozky) {
  return [...polozky].sort((a, b) => a.cena - b.cena);
}

function escapujHtml(text) {
  return String(text)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;');
}

function popisObjednavky(jmeno, poznamka) {
  return '<p>Objednávka pro ' + escapujHtml(jmeno) + ': ' + escapujHtml(poznamka) + '</p>';
}

function celkovaCena(polozky, kod) {
  const poSleve = cenaPoSleve(soucetPolozek(polozky), kod);
  return poSleve + cenaDopravy(poSleve);
}

function cenaPoSleve(soucet, kod) {
  const zaznam = najdiKod(kod);
  if (!zaznam) {
    return soucet;
  }
  if (zaznam.typ === 'procenta') {
    return soucet - (soucet * zaznam.hodnota) / 100;
  }
  return Math.max(0, soucet - zaznam.hodnota);
}

function shrnutiKosiku(polozky, kod) {
  const soucet = soucetPolozek(polozky);
  const poSleve = cenaPoSleve(soucet, kod);
  const doprava = cenaDopravy(poSleve);
  return { soucet, sleva: soucet - poSleve, doprava, celkem: poSleve + doprava };
}
