# --description--

Až výprodej skončí, interval by dál každou sekundu přepisoval `00:00:00` na `00:00:00`. Zbytečná práce, a hlavně chceš uživateli ukázat, že je konec.

**Úkol:** Uprav `updateCountdown` tak, aby po konci výprodeje (zbývající čas je nula nebo méně) zastavila interval `countdownTimer` a hlavičce `.sale` přidala třídu `is-ended`. Styly pak za odpočet připíšou „výprodej skončil".

Interval zastavíš přes `clearInterval` s číslem, které vrátil `setInterval`. Na jiných datech:

```js
if (secondsToDeparture <= 0) {
  clearInterval(departureTimer);
  board.classList.add('is-departed');
}
```

Funkce smí používat `countdownTimer`, i když je konstanta v souboru až pod ní: když se funkce spustí po konci výprodeje, konstanta už existuje.

# --hints--

Po konci výprodeje ukazuje `#countdown` čas `00:00:00` a hlavička `.sale` má třídu `is-ended`.

```js
const setClock = (time) => {
  const RealDate = Date;
  window.Date = class extends RealDate {
    constructor(...args) { super(...(args.length ? args : [time])); }
    static now() { return time; }
  };
};
setClock(saleEndsAt + 5000);
updateCountdown();
assert.equal(document.querySelector('#countdown').textContent.trim(), '00:00:00', 'po konci výprodeje má #countdown ukazovat 00:00:00');
assert.ok(document.querySelector('.sale').classList.contains('is-ended'), 'po konci výprodeje má mít .sale třídu is-ended');
```

// Horská výbava v bleskovém výprodeji. Ceny jsou v korunách.
const products = [
  { id: 1, name: 'Stan Hory 2', price: 3490, oldPrice: 4990 },
  { id: 2, name: 'Spacák Sněžka −5 °C', price: 1890, oldPrice: 2690 },
  { id: 3, name: 'Čelovka 400 lm', price: 490, oldPrice: 790 },
  { id: 4, name: 'Batoh Šumava 35 l', price: 1990, oldPrice: 2890 },
  { id: 5, name: 'Trekové hole Jizera', price: 890, oldPrice: 1290 },
  { id: 6, name: 'Karimatka Krkonoše', price: 690, oldPrice: 990 },
  { id: 7, name: 'Vařič Beskydy', price: 1190, oldPrice: 1590 },

];

// V ukázce končí výprodej 2 h 15 min po načtení stránky
// a uzávěrka doručení na zítřek je za 47 minut.
const saleEndsAt = Date.now() + (2 * 60 + 15) * 60 * 1000;
const deliveryDeadline = Date.now() + 47 * 60 * 1000;

const priceFormat = new Intl.NumberFormat('cs-CZ', { style: 'currency', currency: 'CZK', maximumFractionDigits: 0 });

// Nejnižší cena za posledních 30 dní, kterou musí e-shop u slevy ukázat.
// Prochází celou historii cen, a proto u jednoho produktu trvá pár milisekund.
function lowestPrice30Days(product) {
  let lowest = product.oldPrice;
  for (let record = 0; record < 60000; record++) {
    const price = product.oldPrice - ((record * 7919) % 11) * 10;
    if (price < lowest && price >= product.price) lowest = price;
  }
  return lowest;
}

// Karta jednoho produktu (prvek li, zatím mimo stránku).
function createProductCard(product) {
  const card = document.createElement('li');
  card.className = 'product';
  card.dataset.id = product.id;

  const badge = document.createElement('span');
  badge.className = 'product__badge';
  badge.textContent = `−${Math.round((1 - product.price / product.oldPrice) * 100)} %`;

  const name = document.createElement('h2');
  name.className = 'product__name';
  name.textContent = product.name;

  const price = document.createElement('p');
  price.className = 'product__price';
  price.textContent = priceFormat.format(product.price);

  const note = document.createElement('p');
  note.className = 'product__note';
  note.textContent = `Nejnižší cena za 30 dní: ${priceFormat.format(lowestPrice30Days(product))}`;

  const button = document.createElement('button');
  button.type = 'button';
  button.className = 'button product__add';
  button.textContent = 'Do košíku';

  card.append(badge, name, price, note, button);
  return card;
}

// Celý katalog: 600 variant velikostí a barev.
const fullCatalog = Array.from({ length: 600 }, (_, index) => {
  const base = products[index % products.length];
  return { ...base, id: 100 + index, name: `${base.name} · varianta ${index + 1}` };
});

// Vykreslí karty produktů a nahradí předchozí obsah seznamu.
function renderProducts(items) {
  const list = document.querySelector('#products');
  list.replaceChildren(...items.map(createProductCard));
}

renderProducts(products);

}
