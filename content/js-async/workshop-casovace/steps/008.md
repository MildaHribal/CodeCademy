# --description--

Server, který neodpoví vůbec, je horší než server, který vrátí chybu: tabule čeká a nic se neděje. Potřebuješ časový limit — když odpověď nepřijde do dané doby, považuj to za chybu.

**Úkol:** Pod `loadAvailable` napiš funkci `withTimeout(promise, ms)`. Vrátí Promise, která:

- se splní hodnotou `promise`, když ta doběhne do `ms` milisekund,
- se zamítne chybou z `promise`, když ta do té doby selže,
- se zamítne chybou se zprávou `Vypršel čas 150 ms` (s číslem z `ms`), když `promise` do limitu nedoběhne.

Hodí se jedna z kombinačních metod Promise, `sleep` a `throw`. Která z nich bere výsledek podle toho, **kdo doběhne první**?

Tahle funkce zatím nic nevypíše, v konzoli se nic nezmění. Budeš ji potřebovat na opakování pokusů.

# --hints--

Když Promise doběhne do limitu, `withTimeout` se splní její hodnotou.

```js
const value = await withTimeout(sleep(20).then(() => 'odjezdy'), 200);
assert.equal(value, 'odjezdy', "withTimeout(Promise splněná za 20 ms hodnotou 'odjezdy', 200) se má splnit hodnotou 'odjezdy'");
```

Když Promise do limitu nedoběhne, `withTimeout` se zamítne chybou `Vypršel čas 50 ms`.

```js
let caught = null;
const started = performance.now();
await withTimeout(sleep(400).then(() => 'pozdě'), 50).catch((error) => {
  caught = error;
});
assert.ok(caught instanceof Error, 'withTimeout(Promise za 400 ms, 50) se má zamítnout chybou (Error)');
assert.equal(caught.message, 'Vypršel čas 50 ms', "withTimeout(…, 50) se má zamítnout se zprávou 'Vypršel čas 50 ms'");
assert.ok(performance.now() - started < 300, 'withTimeout(…, 50) má skončit po zhruba 50 ms, nečekat na pomalou Promise');
```

Když Promise selže dřív, `withTimeout` se zamítne její chybou.

```js
let caught = null;
await withTimeout(fetchDepartures('trams', 'Náměstí Míru'), 500).catch((error) => {
  caught = error;
});
assert.equal(caught?.message, 'Zastávka Náměstí Míru neexistuje', 'withTimeout má předat chybu Promise, která selhala dřív než limit');
```

# --help--

## --tip--

Hledáš „kdo dřív přijde, ten vyhraje" — viz tabulka v části [Víc Promise najednou](see:js-async/promise#vic-promise-najednou-all-allsettled-race-a-any) a tip pod ní.

## --tip--

Závodí dvě Promise: ta zadaná a druhá, která po `ms` milisekundách vyhodí chybu. Druhou vyrobíš ze `sleep(ms)` a `then`, ve kterém je `throw`.

## --tip--

Na jiných datech: `Promise.race([loadVideo(), sleep(3000).then(() => { throw new Error('Video se nenačetlo'); })])`.

  'Mendlovo náměstí': [
    { provider: 'buses', line: '67', destination: 'Líšeň', time: '14:12' },
    { provider: 'trams', line: '1', destination: 'Bystrc', time: '14:06' },
    { provider: 'buses', line: '44', destination: 'Bystrc', time: '14:03' },
    { provider: 'trams', line: '6', destination: 'Starý Lískovec', time: '14:04' },
  ],
};

// Servery dopravců. Testy si je přepínají: delay = odezva v ms,
// failures = kolikrát za sebou server selže, než odpoví, down = neodpovídá vůbec.
const servers = {
  trains: { label: 'vlaky', delay: 60, failures: 0, down: false },
  buses: { label: 'autobusy', delay: 40, failures: 0, down: false },
  trams: { label: 'tramvaje', delay: 20, failures: 0, down: false },
};

// Statistika sítě: kolik požadavků proběhlo a kolik jich běželo naráz.
const network = { calls: 0, active: 0, maxActive: 0 };

// Odjezdy jednoho dopravce ze zastávky. Vrací Promise a umí se zrušit přes signal.
function fetchDepartures(provider, stop, { signal } = {}) {
  const server = servers[provider];
  network.calls++;
  network.active++;
  network.maxActive = Math.max(network.maxActive, network.active);
  return new Promise((resolve, reject) => {
    const finish = () => {
      network.active--;
      signal?.removeEventListener('abort', onAbort);
    };
    const onAbort = () => {
      clearTimeout(timer);
      finish();
      reject(signal.reason);
    };
    const timer = setTimeout(() => {
      finish();
      if (server.down) return reject(new Error(`Server ${server.label} neodpovídá`));
      if (server.failures > 0) {
        server.failures--;
        return reject(new Error(`Server ${server.label} je přetížený`));
      }
      if (!timetable[stop]) return reject(new Error(`Zastávka ${stop} neexistuje`));
      resolve(timetable[stop].filter((item) => item.provider === provider).map((item) => ({ ...item })));
    }, server.delay);
    if (signal?.aborted) return onAbort();
    signal?.addEventListener('abort', onAbort);
  });
}

// Počká, až doběhnou všechny požadavky (používají testy).
function networkIdle() {
  return new Promise((resolve) => {
    const check = () => (network.active === 0 ? resolve() : setTimeout(check, 5));
    check();
  });
}

// Jeden řádek tabule, třeba „14:05  Os 4908  Břeclav".
function formatDeparture(departure) {
  return `${departure.time}  ${departure.line.padEnd(7)} ${departure.destination}`;
}

// Vypíše tabule do konzole.
function showBoard(stop, departures) {
  console.log(`Odjezdy — ${stop}`);
  for (const departure of departures) {
    console.log(formatDeparture(departure));
  }
}

// ===== Tvůj kód =====

// Nový seznam odjezdů seřazený od nejbližšího.
function sortByTime(departures) {
  return departures.toSorted((a, b) => a.time.localeCompare(b.time));
}

// Počká zadaný počet milisekund.
function sleep(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

// Odjezdy jednoho dopravce seřazené od nejbližšího.
async function loadProvider(provider, stop) {
  const departures = await fetchDepartures(provider, stop);
  return sortByTime(departures);
}

// Odjezdy všech tří dopravců, načtené jeden po druhém.
async function loadAllSequential(stop) {
  const trains = await fetchDepartures('trains', stop);
  const buses = await fetchDepartures('buses', stop);
  const trams = await fetchDepartures('trams', stop);
  return sortByTime([...trains, ...buses, ...trams]);
}

// Odjezdy všech tří dopravců, načtené souběžně.
async function loadAll(stop) {
  const [trains, buses, trams] = await Promise.all([
    fetchDepartures('trains', stop),
    fetchDepartures('buses', stop),
    fetchDepartures('trams', stop),
  ]);
  return sortByTime([...trains, ...buses, ...trams]);
}

// Od kolegy: nadpis tabule s počtem odjezdů.
async function countDepartures(stop) {
  const departures = await loadAll(stop);
  return departures.length;
}

async function headline(stop) {
  const count = await countDepartures(stop);
  return `${stop} · odjezdů: ${count}`;
}

// Odjezdy dopravců, kteří odpověděli, a seznam těch, kteří ne.
async function loadAvailable(stop) {
  const providers = ['trains', 'buses', 'trams'];
  const results = await Promise.allSettled(providers.map((provider) => fetchDepartures(provider, stop)));
  const departures = [];
  const failed = [];
  results.forEach((result, index) => {
    if (result.status === 'fulfilled') {
      departures.push(...result.value);
    } else {
      failed.push(providers[index]);
    }
  });
  return { departures: sortByTime(departures), failed };
}
