# --description--

Pod tabulí má být malá statistika: kolik odjezdů je vlakem, kolik autobusem a kolik tramvají. Tenhle krok je bez návodu.

**Úkol:** Pod `loadBoard` napiš funkci `countByProvider(departures)`, která vrátí objekt, kde klíčem je `provider` a hodnotou počet jeho odjezdů. Dopravci bez odjezdů ve výsledku nejsou, prázdné pole dá prázdný objekt.

Statistiku pod tabulí vypíšeš v ukázce v posledním kroku.

# --hints--

`countByProvider` spočítá odjezdy každého dopravce.

```js
const departures = [
  { provider: 'trams', line: '4' },
  { provider: 'trains', line: 'R 11' },
  { provider: 'trams', line: '9' },
  { provider: 'trams', line: '12' },
];
assert.deepEqual(countByProvider(departures), { trams: 3, trains: 1 }, 'countByProvider pro tři tramvaje a jeden vlak má vrátit { trams: 3, trains: 1 }');
```

Pro prázdné pole vrátí prázdný objekt.

```js
const result = countByProvider([]);
assert.equal(typeof result, 'object', 'countByProvider([]) má vrátit objekt');
assert.deepEqual(Object.keys(result ?? {}), [], 'countByProvider([]) má vrátit objekt bez klíčů');
```

# --help--

## --tip--

Počítání podle klíče znáš ze sekce o poli: [Akumulátor nemusí být číslo](see:js-pole/metody-pole-do-hloubky#akumulator-nemusi-byt-cislo).

## --tip--

Začni prázdným objektem a pro každý odjezd zvyš počet u jeho `provider`. Když tam klíč ještě není, počítej od nuly.

## --tip--

Rozepsané po podcílech:

```text
// 1. prázdný objekt počtů
// 2. pro každý odjezd: dosavadní počet (nebo 0) + 1
// 3. vrať objekt
```

// failures = kolikrát za sebou server selže, než odpoví, down = neodpovídá vůbec.
const servers = {
  trains: { label: 'vlaky', delay: 60, failures: 0, down: false },
  buses: { label: 'autobusy', delay: 40, failures: 0, down: false },
  trams: { label: 'tramvaje', delay: 20, failures: 0, down: false },
};

// Statistika sítě: kolik požadavků proběhlo a kolik jich běželo naráz.
const network = { calls: 0, active: 0, maxActive: 0 };

// Odjezdy jednoho dopravce ze zastávky. Vrací Promise a umí se zrušit přes signal.
function fetchDepartures(provider, stop, { signal } = {}) {
  const server = servers[provider];
  network.calls++;
  network.active++;
  network.maxActive = Math.max(network.maxActive, network.active);
  return new Promise((resolve, reject) => {
    const finish = () => {
      network.active--;
      signal?.removeEventListener('abort', onAbort);
    };
    const onAbort = () => {
      clearTimeout(timer);
      finish();
      reject(signal.reason);
    };
    const timer = setTimeout(() => {
      finish();
      if (server.down) return reject(new Error(`Server ${server.label} neodpovídá`));
      if (server.failures > 0) {
        server.failures--;
        return reject(new Error(`Server ${server.label} je přetížený`));
      }
      if (!timetable[stop]) return reject(new Error(`Zastávka ${stop} neexistuje`));
      resolve(timetable[stop].filter((item) => item.provider === provider).map((item) => ({ ...item })));
    }, server.delay);
    if (signal?.aborted) return onAbort();
    signal?.addEventListener('abort', onAbort);
  });
}

// Počká, až doběhnou všechny požadavky (používají testy).
function networkIdle() {
  return new Promise((resolve) => {
    const check = () => (network.active === 0 ? resolve() : setTimeout(check, 5));
    check();
  });
}

// Jeden řádek tabule, třeba „14:05  Os 4908  Břeclav".
function formatDeparture(departure) {
  return `${departure.time}  ${departure.line.padEnd(7)} ${departure.destination}`;
}

// Vypíše tabule do konzole.
function showBoard(stop, departures) {

    console.log(formatDeparture(departure));
  }
}

function sortByTime(departures) {
  return departures.toSorted((a, b) => a.time.localeCompare(b.time));
}

// Počká zadaný počet milisekund.
function sleep(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));

async function loadAll(stop, { signal } = {}) {
  const [trains, buses, trams] = await Promise.all([
    fetchDepartures('trains', stop, { signal }),
    fetchDepartures('buses', stop, { signal }),
    fetchDepartures('trams', stop, { signal }),
  ]);
  return sortByTime([...trains, ...buses, ...trams]);
}

// Od kolegy: nadpis tabule s počtem odjezdů.
async function countDepartures(stop) {
  const departures = await loadAll(stop);
  return departures.length;
}

async function headline(stop) {
  const count = await countDepartures(stop);
  return `${stop} · odjezdů: ${count}`;
}

// Odjezdy dopravců, kteří odpověděli, a seznam těch, kteří ne.
async function loadAvailable(stop) {
  const providers = ['trains', 'buses', 'trams'];
  const results = await Promise.allSettled(providers.map((provider) => fetchDepartures(provider, stop)));
  const departures = [];
  const failed = [];
  results.forEach((result, index) => {
    if (result.status === 'fulfilled') {
      departures.push(...result.value);
    } else {
      failed.push(providers[index]);
    }
  });
  return { departures: sortByTime(departures), failed };
}

// Zamítne se, když promise nedoběhne do ms milisekund.
function withTimeout(promise, ms) {
  const timeout = sleep(ms).then(() => {
    throw new Error(`Vypršel čas ${ms} ms`);
  });
  return Promise.race([promise, timeout]);
}

// Nejbližší odjezdy od zadaného času, nejvýš count.
function nextDepartures(departures, fromTime, count) {
  return sortByTime(departures.filter((departure) => departure.time >= fromTime)).slice(0, count);
}

async function retry(fn, attempts) {
  let lastError;
  for (let attempt = 1; attempt <= attempts; attempt++) {
    try {
      return await fn();
    } catch (error) {
      lastError = error;
    }
  }
  throw lastError;
}

// Opakuje fn po chybě a před každým dalším pokusem čeká dvakrát déle.
async function retryWithBackoff(fn, { attempts = 3, baseDelay = 100 } = {}) {
  for (let attempt = 1; ; attempt++) {
    try {
      return await fn();
    } catch (error) {
      if (attempt >= attempts) throw error;
      await sleep(baseDelay * 2 ** (attempt - 1));
    }
  }
}

// Od kolegy: odjezdy dopravce, a když server selže, poslední známé.
const lastKnown = {};

async function loadOrCache(provider, stop) {
  const key = `${provider}|${stop}`;
  try {
    return await fetchDepartures(provider, stop).then((departures) => {
      lastKnown[key] = departures;
      return departures;
    });
  } catch (error) {
    return lastKnown[key] ?? [];
  }
}

// Tabule ukazuje jen poslední vybranou zastávku; starší načítání zruší.
let currentLoad = null;

async function showStop(stop) {
  currentLoad?.abort();
  const controller = new AbortController();
  currentLoad = controller;
  try {
    return await loadAll(stop, { signal: controller.signal });
  } catch (error) {
    if (error.name === 'AbortError') return null;
    throw error;
  }
}

// Odjezdy dopravce s časovým limitem; zrušit je jde i zvenku přes signal.
function fetchWithDeadline(provider, stop, { signal, timeoutMs = 1000 } = {}) {
  const deadline = AbortSignal.timeout(timeoutMs);
  const combined = signal ? AbortSignal.any([signal, deadline]) : deadline;
  return fetchDepartures(provider, stop, { signal: combined });
}

// Celá tabule: každý dopravce s limitem a opakováním, výpadek jednoho nevadí.
async function loadBoard(stop, { signal } = {}) {
  const providers = ['trains', 'buses', 'trams'];
  const results = await Promise.allSettled(
    providers.map((provider) =>
      retryWithBackoff(() => {
        signal?.throwIfAborted();
        return fetchWithDeadline(provider, stop, { signal, timeoutMs: 150 });
      }, { attempts: 3, baseDelay: 50 }),
    ),
  );
  signal?.throwIfAborted();
  const departures = [];
  const failed = [];
  results.forEach((result, index) => {
    if (result.status === 'fulfilled') {
      departures.push(...result.value);
    } else {
      failed.push(providers[index]);
    }
  });
  return { departures: sortByTime(departures), failed };
}
