# --description--

Seznam knih zatím vrací všechno najednou. Katalog potřebuje filtr, řazení a stránkování
přesně podle pravidel z minulého workshopu.

Jediná novinka: Express ti query podá už rozebranou v objektu `req.query`. Chybějící
parametr v něm je `undefined` (ne `null` jako u `searchParams.get`) a u opakovaného
parametru (`?page=1&page=2`) je hodnotou pole. Funkce v `catalog.js` s tím počítají.

**Úkol:** uprav routu `GET /api/books` tak, aby se chovala jako minule:

- `genre` filtruje, `sort` řadí jen podle `title`, `author` a `year`, jiná hodnota vrátí `400`,
- `page` (výchozí 1) a `limit` (výchozí 5, nejvýš 50), neplatné hodnoty vrátí `400`,
- odpověď má tvar `{ data, meta }` s `page`, `limit`, `total` a `totalPages`, knihy
  v `data` jsou převedené přes `toBookResponse`.

Zkus to bez nápovědy.

`GET /api/books` vrátí prvních pět knih a metadata `{ page: 1, limit: 5, total: 12, totalPages: 3 }`.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(`${server.url}/api/books`, { signal: AbortSignal.timeout(3000) })
```

```js
const get = (path) => fetch(`${server.url}${path}`, { signal: AbortSignal.timeout(3000) })
  .then((res) => res.json())
  .catch(() => assert.fail(`GET ${path}: server do 3 sekund neodpověděl`));
const sorted = await get('/api/books?sort=title&page=2&limit=3');
assert.deepEqual(sorted.data?.map((book) => book.title), ['Hordubal', 'Chata v Jezerní kotlině', 'Klapzubova jedenáctka'],

const humor = await get('/api/books?genre=humor&limit=3&page=2');
assert.deepEqual(humor.data?.map((book) => book.id), [11], 'GET /api/books?genre=humor&limit=3&page=2 má vrátit jen knihu 11');
assert.deepEqual(humor.meta, { page: 2, limit: 3, total: 4, totalPages: 2 }, 'GET /api/books?genre=humor&limit=3&page=2 má mít total 4 a totalPages 2');

assert.equal(years.length, 12, 'GET /api/books?sort=year&limit=50 má v data vrátit všech 12 knih');
```

Neplatné řazení a stránkování vrátí `400` s klíčem `error`.

```js
const server = await helpers.startServer('server.js');
for (const query of ['sort=heslo', 'sort=constructor', 'page=0', 'limit=51', 'page=abc', 'limit=', 'page=1&page=2']) {
  const res = await fetch(`${server.url}/api/books?${query}`, { signal: AbortSignal.timeout(3000) })
    .catch(() => assert.fail(`GET /api/books?${query}: server do 3 sekund neodpověděl`));
  assert.equal(res.status, 400, `GET /api/books?${query} má vrátit 400, přišlo ${res.status}`);
  const body = await res.json().catch(() => assert.fail(`odpověď na GET /api/books?${query} má být JSON`));
  assert.equal(typeof body.error, 'string', `odpověď 400 na ?${query} má mít klíč error`);
}
```

## --file-- catalog.js

```js
import { loans } from './data.js';

// Povolená řazení. Klient smí poslat jen jméno, které tu je.
export const sorters = {
  title: (a, b) => a.title.localeCompare(b.title, 'cs'),
  author: (a, b) => a.author.localeCompare(b.author, 'cs'),
  year: (a, b) => a.year - b.year,
};

// Převede hodnotu z query na celé číslo v rozsahu. Když to nejde, vrátí null.
// Express dá u opakovaného parametru (?page=1&page=2) pole, proto kontrola typu.
export function parseIntInRange(value, min, max) {
  if (typeof value !== 'string' || value.trim() === '') {
    return null;
  }
  const number = Number(value);
  if (!Number.isInteger(number) || number < min || number > max) {
    return null;
  }
  return number;
}

// Kniha tak, jak ji vidí klient: data z katalogu a dostupnost spočítaná z výpůjček.
export function toBookResponse(book) {
  return { ...book, available: !loans.some((loan) => loan.bookId === book.id) };
}
```

```js
import { books, loans } from './data.js';

// Kontrola těla nové výpůjčky. Vrátí { status, error }, nebo null, když je vstup v pořádku.
export function checkLoanInput(input) {
  if (typeof input !== 'object' || input === null || Array.isArray(input)) {
    return { status: 400, error: 'Tělo požadavku musí být JSON objekt.' };
  }
  if (!Number.isInteger(input.bookId) || input.bookId < 1) {
    return { status: 422, error: 'bookId musí být kladné celé číslo.' };
  }
  if (typeof input.reader !== 'string' || input.reader.trim() === '') {
    return { status: 422, error: 'reader musí být jméno čtenáře.' };
  }
  return null;
}

// Kontrola, jestli jde knihu půjčit. Vrátí { status, error }, nebo null.
export function checkAvailability(bookId) {
  if (!books.some((book) => book.id === bookId)) {
    return { status: 422, error: 'Kniha s tímhle id neexistuje.' };
  }
  if (loans.some((loan) => loan.bookId === bookId)) {
    return { status: 409, error: 'Kniha je právě vypůjčená.' };
  }
  return null;
}

// Datum vrácení: dnes + 28 dní, ve tvaru RRRR-MM-DD.
export function dueDateFromToday() {
  const date = new Date();
  date.setDate(date.getDate() + 28);
  return date.toISOString().slice(0, 10);
}
```

## --file-- app.js

```js
import express from 'express';
import { books } from './data.js';
import { parseIntInRange, sorters, toBookResponse } from './catalog.js';

export const app = express();

app.get('/api/books', (req, res) => {
  const { genre, sort } = req.query;
  let result = genre === undefined ? books : books.filter((book) => book.genre === genre);
  if (sort !== undefined) {
    if (!Object.hasOwn(sorters, sort)) {
      return res.status(400).json({ error: 'Řadit jde jen podle title, author nebo year.' });
    }
    result = result.toSorted(sorters[sort]);
  }

  const page = parseIntInRange(req.query.page ?? '1', 1, Infinity);
  const limit = parseIntInRange(req.query.limit ?? '5', 1, 50);
  if (page === null || limit === null) {
    return res.status(400).json({ error: 'page musí být celé číslo od 1 a limit celé číslo od 1 do 50.' });
  }
  const start = (page - 1) * limit;
  res.json({
    data: result.slice(start, start + limit).map(toBookResponse),
    meta: { page, limit, total: result.length, totalPages: Math.ceil(result.length / limit) },
  });
});

app.get('/api/books/:id', (req, res) => {
  const book = books.find((item) => item.id === Number(req.params.id));
  if (!book) {
    return res.status(404).json({ error: 'Kniha nenalezena.' });
  }
  res.json(toBookResponse(book));
});
```
