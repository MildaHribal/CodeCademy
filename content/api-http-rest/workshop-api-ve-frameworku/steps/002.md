# --description--

Pro detail knihy jsi minule psal celou funkci `matchPath`. V Expressu napíšeš vzor přímo
do cesty routy a hodnoty parametrů najdeš v objektu `req.params`, vždy jako **text**.

Stav jiný než `200` nastavíš přes `res.status(kód)`. Metoda vrací `res`, takže se dá
řetězit s `json`. Na jiných datech:

```js
app.get('/api/cinemas/:id', (req, res) => {
  const cinema = cinemas.find((item) => item.id === Number(req.params.id));
  if (!cinema) {
    return res.status(404).json({ error: 'Kino nenalezeno.' });
  }
  res.json(toCinemaResponse(cinema));
});
```

**Úkol:** přidej do `app.js` routu `GET /api/books/:id`. Vrátí knihu přes `toBookResponse`,
nebo `404` s klíčem `error`, když kniha s takovým id není.

> [!PITFALL]
> `return` před `res.status(404).json(…)` není ozdoba. Bez něj kód po odeslání `404` pokračuje,
> zavolá `toBookResponse(undefined)` a v terminálu serveru se objeví
> `TypeError: Cannot read properties of undefined (reading 'id')`. Kdyby další řádek jen znovu
> odpovídal, dostal bys `ERR_HTTP_HEADERS_SENT` stejně jako v čistém Node.

`GET /api/books/3` vrátí Babičku s `available: true`.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(`${server.url}/api/books/3`, { signal: AbortSignal.timeout(3000) })
  .catch(() => assert.fail('GET /api/books/3: server do 3 sekund neodpověděl'));
assert.equal(res.status, 200, `GET /api/books/3 má vrátit 200, přišlo ${res.status}`);
assert.deepEqual(await res.json(), { id: 3, title: 'Babička', author: 'Božena Němcová', genre: 'klasika', year: 1855, available: true },
  'GET /api/books/3 má vrátit Babičku s available: true');
```

`GET /api/books/99` a `GET /api/books/abc` vrátí `404` s klíčem `error` v JSON.

```js
const server = await helpers.startServer('server.js');
for (const id of ['99', 'abc']) {
  const res = await fetch(`${server.url}/api/books/${id}`, { signal: AbortSignal.timeout(3000) })
    .catch(() => assert.fail(`GET /api/books/${id}: server do 3 sekund neodpověděl`));
  assert.equal(res.status, 404, `GET /api/books/${id} má vrátit 404, přišlo ${res.status}`);
  const body = await res.json().catch(() => assert.fail(`odpověď na GET /api/books/${id} má být JSON, ne HTML stránka Expressu — existuje routa /api/books/:id?`));
  assert.equal(typeof body.error, 'string', `odpověď 404 na GET /api/books/${id} má mít klíč error`);
}
await helpers.wait(200);
assert.doesNotMatch(server.output(), /TypeError|ERR_HTTP_HEADERS_SENT/, `po odpovědi 404 nemá handler pokračovat — v terminálu serveru je chyba:\n${server.output().slice(-400)}`);
```

Seznam knih funguje dál.

```js
const server = await helpers.startServer('server.js');

assert.equal((await res.json()).length, 12, 'GET /api/books má dál vracet 12 knih');
```

import { loans } from './data.js';

// Povolená řazení. Klient smí poslat jen jméno, které tu je.
export const sorters = {
  title: (a, b) => a.title.localeCompare(b.title, 'cs'),
  author: (a, b) => a.author.localeCompare(b.author, 'cs'),
  year: (a, b) => a.year - b.year,
};

// Převede hodnotu z query na celé číslo v rozsahu. Když to nejde, vrátí null.
// Express dá u opakovaného parametru (?page=1&page=2) pole, proto kontrola typu.
export function parseIntInRange(value, min, max) {
  if (typeof value !== 'string' || value.trim() === '') {
    return null;
  }
  const number = Number(value);
  if (!Number.isInteger(number) || number < min || number > max) {
    return null;
  }
  return number;
}

// Kniha tak, jak ji vidí klient: data z katalogu a dostupnost spočítaná z výpůjček.
export function toBookResponse(book) {
  return { ...book, available: !loans.some((loan) => loan.bookId === book.id) };
}

import { books, loans } from './data.js';

// Kontrola těla nové výpůjčky. Vrátí { status, error }, nebo null, když je vstup v pořádku.
export function checkLoanInput(input) {
  if (typeof input !== 'object' || input === null || Array.isArray(input)) {
    return { status: 400, error: 'Tělo požadavku musí být JSON objekt.' };
  }
  if (!Number.isInteger(input.bookId) || input.bookId < 1) {
    return { status: 422, error: 'bookId musí být kladné celé číslo.' };
  }
  if (typeof input.reader !== 'string' || input.reader.trim() === '') {
    return { status: 422, error: 'reader musí být jméno čtenáře.' };
  }
  return null;
}

// Kontrola, jestli jde knihu půjčit. Vrátí { status, error }, nebo null.
export function checkAvailability(bookId) {
  if (!books.some((book) => book.id === bookId)) {
    return { status: 422, error: 'Kniha s tímhle id neexistuje.' };
  }
  if (loans.some((loan) => loan.bookId === bookId)) {
    return { status: 409, error: 'Kniha je právě vypůjčená.' };
  }
  return null;
}

// Datum vrácení: dnes + 28 dní, ve tvaru RRRR-MM-DD.
export function dueDateFromToday() {
  const date = new Date();
  date.setDate(date.getDate() + 28);
  return date.toISOString().slice(0, 10);
}

## --file-- app.js

```js
import express from 'express';
import { books } from './data.js';
import { toBookResponse } from './catalog.js';

export const app = express();

app.get('/api/books', (req, res) => {
  res.json(books.map(toBookResponse));
});

app.get('/api/books/:id', (req, res) => {
  const book = books.find((item) => item.id === Number(req.params.id));
  if (!book) {
    return res.status(404).json({ error: 'Kniha nenalezena.' });
  }
  res.json(toBookResponse(book));
});
```
