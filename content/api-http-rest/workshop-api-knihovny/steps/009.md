# --description--

Zkus v HTTP klientovi `GET /api/books?page=abc`. Server vrátí `200`, prázdná `data`
a v metadatech `"page": null`. `Number('abc')` je `NaN` a `JSON.stringify` z `NaN`
udělá `null`. Frontend pak vykreslí „Strana null z null". A `?limit=1000000` by
u velké knihovny poslal všechno najednou, přesně to, čemu stránkování mělo zabránit.

Všechno z query přichází jako **text**, a to i `page=2`. Než hodnotě uvěříš, musíš ji
převést a zkontrolovat.

**Úkol:** v `listBooks` odmítni neplatné stránkování stavem `400` a vysvětlením v `error`:

- `page` musí být celé číslo od 1,
- `limit` musí být celé číslo od 1 do 50,
- chybějící parametr dál znamená výchozí hodnotu (`page` 1, `limit` 5).

Stránka za koncem seznamu (`?page=9`) chyba není: vrať `200` a prázdná `data`, klient se
z `meta` dozví, kolik stránek je.

Hodí se `Number.isInteger()`, `parseInt()` nebo regulární výraz. Pozor na rozdíly:
`parseInt('2abc')` vrátí `2` a `Number('')` vrátí `0`.

# --hints--

Neplatné `page` (`abc`, `0`, `-1`, `1.5`, prázdné) vrátí `400` s klíčem `error`.

```js
const server = await helpers.startServer('server.js');
for (const value of ['abc', '0', '-1', '1.5', '', '2abc']) {
  const res = await fetch(`${server.url}/api/books?page=${value}`, { signal: AbortSignal.timeout(3000) })
    .catch(() => assert.fail(`GET /api/books?page=${value}: server do 3 sekund neodpověděl`));
  assert.equal(res.status, 400, `GET /api/books?page=${value} má vrátit stav 400`);
  const body = await res.json();
  assert.equal(typeof body.error, 'string', `odpověď 400 na ?page=${value} má mít klíč error`);
}
```

Neplatné `limit` (`0`, `51`, `abc`, `2.5`) vrátí `400`, hraniční `50` projde.

```js
const server = await helpers.startServer('server.js');

    .catch(() => assert.fail(`GET /api/books?limit=${value}: server do 3 sekund neodpověděl`));
  assert.equal(res.status, 400, `GET /api/books?limit=${value} má vrátit stav 400 — limit smí být 1 až 50`);

assert.equal(max.status, 200, 'GET /api/books?limit=50 má vrátit 200 — 50 je ještě povolený limit');
assert.equal((await max.json()).data.length, 12, 'GET /api/books?limit=50 má vrátit všech 12 knih');
```

Stránka za koncem vrátí `200` a prázdná `data`.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(`${server.url}/api/books?page=9`, { signal: AbortSignal.timeout(3000) })
  .catch(() => assert.fail('GET /api/books?page=9: server do 3 sekund neodpověděl'));
assert.equal(res.status, 200, 'GET /api/books?page=9 má vrátit 200 — prázdná stránka není chyba');
const body = await res.json();
assert.deepEqual(body.data, [], 'GET /api/books?page=9 má vrátit prázdné pole data');
assert.deepEqual(body.meta, { page: 9, limit: 5, total: 12, totalPages: 3 }, 'GET /api/books?page=9 má vrátit meta { page: 9, limit: 5, total: 12, totalPages: 3 }');
```

Platné stránkování funguje dál.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(`${server.url}/api/books?page=2&limit=4`, { signal: AbortSignal.timeout(3000) })
  .catch(() => assert.fail('GET /api/books?page=2&limit=4: server do 3 sekund neodpověděl'));
const body = await res.json();
assert.deepEqual(body.data.map((book) => book.id), [5, 6, 7, 8], 'GET /api/books?page=2&limit=4 má dál vracet knihy 5 až 8');
assert.deepEqual(body.meta, { page: 2, limit: 4, total: 12, totalPages: 3 }, 'GET /api/books?page=2&limit=4 má dál vracet správná meta');
```

# --help--

## --tip--

Proč hodnoty z query nejsou čísla a co s nimi, najdeš v
[Stránkování: offset a kurzor](see:api-http-rest/navrh-rest#strankovani-offset-a-kurzor).

## --tip--

Kontrolu dej hned za přečtení `page` a `limit`, před výpočet stránky. Obě hodnoty
kontroluješ stejně, liší se jen rozsahem — vyplatí se malá funkce s parametry pro
minimum a maximum, která vrátí číslo, nebo `null`.

## --tip--

Stejná kontrola pro počet vstupenek (1 až 10):

```text
count = Number(text)
platné, když Number.isInteger(count) a 1 <= count <= 10
```

# --explain--

Vysvětli, proč server musí kontrolovat `page` a `limit`, i když frontend posílá jen čísla z tlačítek.

## --model--

Z query přichází vždycky text a API může zavolat kdokoli, nejen tvůj frontend:
`curl`, jiný web nebo útočník. Bez kontroly se `abc` změní v `NaN`, který v JSON
skončí jako `null`, a obří `limit` pošle celou databázi najednou. `400` řekne klientovi
hned, co poslal špatně, místo tiché nesmyslné odpovědi.

## --checklist--

- Hodnoty z query jsou vždycky text, i když vypadají jako číslo.
- API může volat kdokoli, nejen vlastní frontend.
- `NaN` se v JSON změní na `null` a odpověď tiše nedává smysl.
- Horní hranice `limit` chrání server před obřími odpověďmi.

```js
// Pomocné funkce pro odpovědi. Každá odpověď serveru prochází přes sendJson.

export function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

export function sendError(res, status, message) {
  sendJson(res, status, { error: message });
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí undefined.
export async function readJson(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    return undefined;
  }
}
```

```js
// Porovná cestu se vzorem routy. Vrátí objekt s parametry, nebo null.
export function matchPath(pattern, pathname) {
  const patternParts = pattern.split('/');
  const pathParts = pathname.split('/');
  if (patternParts.length !== pathParts.length) return null;
  const params = {};
  for (let i = 0; i < patternParts.length; i++) {
    if (patternParts[i].startsWith(':')) {
      params[patternParts[i].slice(1)] = pathParts[i];
    } else if (patternParts[i] !== pathParts[i]) {
      return null;
    }
  }
  return params;
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { sendError } from './http.js';
import { matchPath } from './router.js';
import { routes } from './routes.js';

const port = Number(process.env.PORT ?? 3000);

async function handleRequest(req, res) {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  const allowedMethods = [];
  for (const route of routes) {
    const params = matchPath(route.path, pathname);
    if (params === null) {
      continue;
    }
    if (route.method === req.method) {
      return route.handler(req, res, { params, query: searchParams });
    }
    allowedMethods.push(route.method);
  }

  if (allowedMethods.length > 0) {
    res.setHeader('Allow', allowedMethods.join(', '));
    return sendError(res, 405, 'Tahle metoda tu není povolená.');
  }
  sendError(res, 404, 'Tahle adresa neexistuje.');
}

createServer(async (req, res) => {
  try {
    await handleRequest(req, res);
  } catch (error) {
    console.error(error);
    sendError(res, 500, 'Na serveru se něco pokazilo.');
  }
}).listen(port, () => {
  console.log(`API knihovny běží na http://localhost:${port}`);
});
```

## --file-- books.js

```js
import { books } from './data.js';
import { sendError, sendJson } from './http.js';

// Povolená řazení. Klient smí poslat jen jméno, které tu je.
const sorters = {
  title: (a, b) => a.title.localeCompare(b.title, 'cs'),
  author: (a, b) => a.author.localeCompare(b.author, 'cs'),
  year: (a, b) => a.year - b.year,
};

// Převede text z query na celé číslo v rozsahu. Když to nejde, vrátí null.
function parseIntInRange(text, min, max) {
  const number = Number(text);
  if (text.trim() === '' || !Number.isInteger(number) || number < min || number > max) {
    return null;
  }
  return number;
}

export function listBooks(req, res, { query }) {
  const genre = query.get('genre');
  let result = genre === null ? books : books.filter((book) => book.genre === genre);
  const sort = query.get('sort');
  if (sort !== null) {
    if (!Object.hasOwn(sorters, sort)) {
      return sendError(res, 400, 'Řadit jde jen podle title, author nebo year.');
    }
    result = result.toSorted(sorters[sort]);
  }

  const page = parseIntInRange(query.get('page') ?? '1', 1, Infinity);
  const limit = parseIntInRange(query.get('limit') ?? '5', 1, 50);
  if (page === null || limit === null) {
    return sendError(res, 400, 'page musí být celé číslo od 1 a limit celé číslo od 1 do 50.');
  }
  const start = (page - 1) * limit;
  sendJson(res, 200, {
    data: result.slice(start, start + limit),
    meta: { page, limit, total: result.length, totalPages: Math.ceil(result.length / limit) },
  });
}

export function getBook(req, res, { params }) {
  const book = books.find((item) => item.id === Number(params.id));
  if (!book) {
    return sendError(res, 404, 'Kniha nenalezena.');
  }
  sendJson(res, 200, book);
}
```
