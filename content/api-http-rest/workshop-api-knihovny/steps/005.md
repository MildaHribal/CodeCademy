# --description--

Katalog má přepínač žánrů. Frontend pošle `GET /api/books?genre=sci-fi` a čeká jen
sci-fi. Filtr patří do [[query string|query stringu]], ne do cesty: pořád chceš
stejný zdroj, seznam knih, jen jeho výběr.

Server předává handleru parametry z query jako `query` — to je objekt
`URLSearchParams` z `new URL(req.url, …).searchParams`. Hodnotu přečteš přes
`query.get('jméno')`. Když parametr v adrese není, `get` vrátí `null`.

**Úkol:** v `listBooks` v `books.js`:

- když adresa obsahuje `genre`, vrať jen knihy s tímto žánrem,
- když `genre` chybí, vrať všechny knihy jako dosud.

Stejný vzor na jiných datech:

```js
const city = query.get('city');
const result = city === null ? cafes : cafes.filter((cafe) => cafe.city === city);
```

> [!PITFALL]
> Když filtru nic neodpovídá (`?genre=poezie`), odpověď je `200` a prázdné pole, ne
> `404`. Seznam existuje, jen je prázdný. `404` by frontend pochopil jako špatnou adresu.

# --hints--

`GET /api/books?genre=sci-fi` vrátí jen knihy 1, 4 a 12.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(`${server.url}/api/books?genre=sci-fi`, { signal: AbortSignal.timeout(3000) })
  .catch(() => assert.fail('GET /api/books?genre=sci-fi: server do 3 sekund neodpověděl'));
assert.equal(res.status, 200, 'GET /api/books?genre=sci-fi má vrátit stav 200');
const ids = (await res.json()).map((book) => book.id);
assert.deepEqual(ids, [1, 4, 12], 'GET /api/books?genre=sci-fi má vrátit knihy s id 1, 4 a 12');
```

```js
  .catch(() => assert.fail('GET /api/books?genre=dobrodružství: server do 3 sekund neodpověděl'));

assert.deepEqual(ids, [7], 'GET /api/books?genre=dobrodružství má vrátit jen knihu 7 — čteš hodnotu přes query.get?');
```

Žánr, který v knihovně není, vrátí `200` a prázdné pole.

```js

```

```js
export function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

export function sendError(res, status, message) {
  sendJson(res, status, { error: message });
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí undefined.
export async function readJson(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    return undefined;
  }
}
```

```js
// Porovná cestu se vzorem routy. Vrátí objekt s parametry, nebo null.
export function matchPath(pattern, pathname) {
  const patternParts = pattern.split('/');
  const pathParts = pathname.split('/');
  if (patternParts.length !== pathParts.length) return null;
  const params = {};
  for (let i = 0; i < patternParts.length; i++) {
    if (patternParts[i].startsWith(':')) {
      params[patternParts[i].slice(1)] = pathParts[i];
    } else if (patternParts[i] !== pathParts[i]) {
      return null;
    }
  }
  return params;
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { sendError } from './http.js';
import { matchPath } from './router.js';
import { routes } from './routes.js';

const port = Number(process.env.PORT ?? 3000);

async function handleRequest(req, res) {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  const allowedMethods = [];
  for (const route of routes) {
    const params = matchPath(route.path, pathname);
    if (params === null) {
      continue;
    }
    if (route.method === req.method) {
      return route.handler(req, res, { params, query: searchParams });
    }
    allowedMethods.push(route.method);
  }

  if (allowedMethods.length > 0) {
    res.setHeader('Allow', allowedMethods.join(', '));
    return sendError(res, 405, 'Tahle metoda tu není povolená.');
  }
  sendError(res, 404, 'Tahle adresa neexistuje.');
}

createServer(async (req, res) => {
  try {
    await handleRequest(req, res);
  } catch (error) {
    console.error(error);
    sendError(res, 500, 'Na serveru se něco pokazilo.');
  }
}).listen(port, () => {
  console.log(`API knihovny běží na http://localhost:${port}`);
});
```

## --file-- books.js

```js
import { books } from './data.js';
import { sendError, sendJson } from './http.js';

export function listBooks(req, res, { query }) {
  const genre = query.get('genre');
  const result = genre === null ? books : books.filter((book) => book.genre === genre);
  sendJson(res, 200, result);
}

export function getBook(req, res, { params }) {
  const book = books.find((item) => item.id === Number(params.id));
  if (!book) {
    return sendError(res, 404, 'Kniha nenalezena.');
  }
  sendJson(res, 200, book);
}
```
