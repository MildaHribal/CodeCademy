# --description--

Katalog u každé knihy ukazuje štítek „Volná" nebo „Vypůjčená". Frontend by to mohl
spočítat sám, jenže by musel stáhnout všechny výpůjčky a projít je u každé knihy. To je
práce pro server.

Kniha v `data.js` pole `available` nemá a mít nemá: dostupnost se mění s každou výpůjčkou
a uložená hodnota by se rozcházela se skutečností. JSON, který API posílá, je
**reprezentace** zdroje a smí obsahovat pole spočítané v okamžiku požadavku.

**Úkol:** každá kniha v odpovědi `GET /api/books` i `GET /api/books/:id` má navíc pole
`available`:

- `false`, když ke knize existuje výpůjčka, jinak `true`,
- hodnota odpovídá stavu v okamžiku požadavku, takže po vrácení nebo nové výpůjčce se změní,
- ostatní pole knihy zůstávají, jak jsou, a filtr, řazení i stránkování fungují dál.

Pole `books` v `data.js` neměň, dostupnost přidej až do odpovědi.

# --hints--

`GET /api/books/4` vrátí knihu s `available: false` a `GET /api/books/3` s `available: true`.

```js
const server = await helpers.startServer('server.js');
const get = (path) => fetch(`${server.url}${path}`, { signal: AbortSignal.timeout(3000) })
  .then((res) => res.json())
  .catch(() => assert.fail(`GET ${path}: server do 3 sekund neodpověděl`));
assert.deepEqual(await get('/api/books/4'), { id: 4, title: 'Válka s mloky', author: 'Karel Čapek', genre: 'sci-fi', year: 1936, available: false },
  'GET /api/books/4 má vrátit Válku s mloky s available: false — kniha 4 je vypůjčená');
assert.deepEqual(await get('/api/books/3'), { id: 3, title: 'Babička', author: 'Božena Němcová', genre: 'klasika', year: 1855, available: true },
  'GET /api/books/3 má vrátit Babičku s available: true');
```

Seznam knih má `available` u každé knihy: vypůjčené jsou jen knihy 4 a 9.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(`${server.url}/api/books?limit=50`, { signal: AbortSignal.timeout(3000) })
  .catch(() => assert.fail('GET /api/books?limit=50: server do 3 sekund neodpověděl'));
const { data, meta } = await res.json();
assert.deepEqual(data.filter((book) => book.available === false).map((book) => book.id), [4, 9], 'v GET /api/books?limit=50 mají mít available: false jen knihy 4 a 9');
assert.equal(data.filter((book) => book.available === true).length, 10, 'v GET /api/books?limit=50 má mít available: true deset knih');
assert.deepEqual(meta, { page: 1, limit: 50, total: 12, totalPages: 1 }, 'GET /api/books?limit=50 má dál vracet správná meta');
```

Dostupnost se mění s výpůjčkami: po vrácení je kniha 4 volná, po nové výpůjčce je kniha 3 vypůjčená.

```js
const server = await helpers.startServer('server.js');

  .then((res) => res.json())

await fetch(`${server.url}/api/loans`, {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ bookId: 3, reader: 'Eva Malá' }),
  signal: AbortSignal.timeout(3000),
});
const list = await get('/api/books?genre=klasika&sort=title');
assert.deepEqual(list.data.map((book) => [book.title, book.available]),
  [['Babička', false], ['Cirkus Humberto', true], ['Hordubal', true], ['Povídky malostranské', true]],
  'po výpůjčce knihy 3 má GET /api/books?genre=klasika&sort=title ukázat Babičku jako nedostupnou');

// Pomocné funkce pro odpovědi. Každá odpověď serveru prochází přes sendJson.

export function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

export function sendError(res, status, message) {
  sendJson(res, status, { error: message });
}

// Převede text z query na celé číslo v rozsahu. Když to nejde, vrátí null.
export function parseIntInRange(text, min, max) {
  const number = Number(text);
  if (text.trim() === '' || !Number.isInteger(number) || number < min || number > max) {
    return null;
  }
  return number;
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí undefined.
export async function readJson(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    return undefined;
  }
}
```

// Porovná cestu se vzorem routy. Vrátí objekt s parametry, nebo null.
export function matchPath(pattern, pathname) {
  const patternParts = pattern.split('/');
  const pathParts = pathname.split('/');
  if (patternParts.length !== pathParts.length) return null;
  const params = {};
  for (let i = 0; i < patternParts.length; i++) {
    if (patternParts[i].startsWith(':')) {
      params[patternParts[i].slice(1)] = pathParts[i];
    } else if (patternParts[i] !== pathParts[i]) {
      return null;
    }
  }
  return params;
}

## --file-- books.js

```js
import { books } from './data.js';
import { parseIntInRange, sendError, sendJson } from './http.js';

// Povolená řazení. Klient smí poslat jen jméno, které tu je.
const sorters = {
  title: (a, b) => a.title.localeCompare(b.title, 'cs'),
  author: (a, b) => a.author.localeCompare(b.author, 'cs'),
  year: (a, b) => a.year - b.year,
};

export function listBooks(req, res, { query }) {
  const genre = query.get('genre');
  let result = genre === null ? books : books.filter((book) => book.genre === genre);
  const sort = query.get('sort');
  if (sort !== null) {
    if (!Object.hasOwn(sorters, sort)) {
      return sendError(res, 400, 'Řadit jde jen podle title, author nebo year.');
    }
    result = result.toSorted(sorters[sort]);
  }

  const page = parseIntInRange(query.get('page') ?? '1', 1, Infinity);
  const limit = parseIntInRange(query.get('limit') ?? '5', 1, 50);
  if (page === null || limit === null) {
    return sendError(res, 400, 'page musí být celé číslo od 1 a limit celé číslo od 1 do 50.');
  }
  const start = (page - 1) * limit;
  sendJson(res, 200, {
    data: result.slice(start, start + limit),
    meta: { page, limit, total: result.length, totalPages: Math.ceil(result.length / limit) },
  });
}

export function getBook(req, res, { params }) {
  const book = books.find((item) => item.id === Number(params.id));
  if (!book) {
    return sendError(res, 404, 'Kniha nenalezena.');
  }
  sendJson(res, 200, book);
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { sendError } from './http.js';
import { matchPath } from './router.js';
import { routes } from './routes.js';

const port = Number(process.env.PORT ?? 3000);

async function handleRequest(req, res) {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  const allowedMethods = [];
  for (const route of routes) {
    const params = matchPath(route.path, pathname);
    if (params === null) {
      continue;
    }
    if (route.method === req.method) {
      return route.handler(req, res, { params, query: searchParams });
    }
    allowedMethods.push(route.method);
  }

  if (allowedMethods.length > 0) {
    res.setHeader('Allow', allowedMethods.join(', '));
    return sendError(res, 405, 'Tahle metoda tu není povolená.');
  }
  sendError(res, 404, 'Tahle adresa neexistuje.');
}

createServer(async (req, res) => {
  try {
    await handleRequest(req, res);
  } catch (error) {
    console.error(error);
    sendError(res, 500, 'Na serveru se něco pokazilo.');
  }
}).listen(port, () => {
  console.log(`API knihovny běží na http://localhost:${port}`);
});
```

## --file-- loans.js

```js
import { books, loans } from './data.js';
import { parseIntInRange, readJson, sendError, sendJson } from './http.js';

// Datum vrácení: dnes + 28 dní, ve tvaru RRRR-MM-DD.
function dueDateFromToday() {
  const date = new Date();
  date.setDate(date.getDate() + 28);
  return date.toISOString().slice(0, 10);
}

export function listLoans(req, res, { query }) {
  const page = parseIntInRange(query.get('page') ?? '1', 1, Infinity);
  const limit = parseIntInRange(query.get('limit') ?? '5', 1, 50);
  if (page === null || limit === null) {
    return sendError(res, 400, 'page musí být celé číslo od 1 a limit celé číslo od 1 do 50.');
  }
  const start = (page - 1) * limit;
  sendJson(res, 200, {
    data: loans.slice(start, start + limit),
    meta: { page, limit, total: loans.length, totalPages: Math.ceil(loans.length / limit) },
  });
}

export function getLoan(req, res, { params }) {
  const loan = loans.find((item) => item.id === Number(params.id));
  if (!loan) {
    return sendError(res, 404, 'Výpůjčka nenalezena.');
  }
  sendJson(res, 200, loan);
}

export async function createLoan(req, res) {
  const input = await readJson(req);
  if (typeof input !== 'object' || input === null) {
    return sendError(res, 400, 'Tělo požadavku musí být JSON objekt.');
  }
  if (!Number.isInteger(input.bookId) || !books.some((book) => book.id === input.bookId)) {
    return sendError(res, 422, 'bookId musí být id existující knihy.');
  }
  if (typeof input.reader !== 'string' || input.reader.trim() === '') {
    return sendError(res, 422, 'reader musí být jméno čtenáře.');
  }
  if (loans.some((loan) => loan.bookId === input.bookId)) {
    return sendError(res, 409, 'Kniha je právě vypůjčená.');
  }

  const loan = {
    id: Math.max(0, ...loans.map((item) => item.id)) + 1,
    bookId: input.bookId,
    reader: input.reader,
    dueDate: dueDateFromToday(),
  };
  loans.push(loan);
  res.setHeader('Location', `/api/loans/${loan.id}`);
  sendJson(res, 201, loan);
}

export async function updateLoan(req, res, { params }) {
  const loan = loans.find((item) => item.id === Number(params.id));
  if (!loan) {
    return sendError(res, 404, 'Výpůjčka nenalezena.');
  }
  const input = await readJson(req);
  if (typeof input !== 'object' || input === null) {
    return sendError(res, 400, 'Tělo požadavku musí být JSON objekt.');
  }
  const { dueDate } = input;
  if (typeof dueDate !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(dueDate) || dueDate <= loan.dueDate) {
    return sendError(res, 422, 'dueDate musí být datum RRRR-MM-DD pozdější než současné datum vrácení.');
  }

  loan.dueDate = dueDate;
  sendJson(res, 200, loan);
}

export function deleteLoan(req, res, { params }) {
  const index = loans.findIndex((item) => item.id === Number(params.id));
  if (index === -1) {
    return sendError(res, 404, 'Výpůjčka nenalezena.');
  }
  loans.splice(index, 1);
  res.writeHead(204);
  res.end();
}
```

## --file-- books.js

```js
import { books, loans } from './data.js';
import { parseIntInRange, sendError, sendJson } from './http.js';

// Kniha tak, jak ji vidí klient: data z katalogu a dostupnost spočítaná z výpůjček.
function toBookResponse(book) {
  return { ...book, available: !loans.some((loan) => loan.bookId === book.id) };
}

// Povolená řazení. Klient smí poslat jen jméno, které tu je.
const sorters = {
  title: (a, b) => a.title.localeCompare(b.title, 'cs'),
  author: (a, b) => a.author.localeCompare(b.author, 'cs'),
  year: (a, b) => a.year - b.year,
};

export function listBooks(req, res, { query }) {
  const genre = query.get('genre');
  let result = genre === null ? books : books.filter((book) => book.genre === genre);
  const sort = query.get('sort');
  if (sort !== null) {
    if (!Object.hasOwn(sorters, sort)) {
      return sendError(res, 400, 'Řadit jde jen podle title, author nebo year.');
    }
    result = result.toSorted(sorters[sort]);
  }

  const page = parseIntInRange(query.get('page') ?? '1', 1, Infinity);
  const limit = parseIntInRange(query.get('limit') ?? '5', 1, 50);
  if (page === null || limit === null) {
    return sendError(res, 400, 'page musí být celé číslo od 1 a limit celé číslo od 1 do 50.');
  }
  const start = (page - 1) * limit;
  sendJson(res, 200, {
    data: result.slice(start, start + limit).map(toBookResponse),
    meta: { page, limit, total: result.length, totalPages: Math.ceil(result.length / limit) },
  });
}

export function getBook(req, res, { params }) {
  const book = books.find((item) => item.id === Number(params.id));
  if (!book) {
    return sendError(res, 404, 'Kniha nenalezena.');
  }
  sendJson(res, 200, toBookResponse(book));
}
```
