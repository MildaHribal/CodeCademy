# --description--

Routa `GET /api/books/:id` v tabulce je, ale server ji nikdy nenajde: porovnává cestu
přes `===`, a `/api/books/7` se nikdy nerovná `/api/books/:id`. Potřebuješ funkci,
která pozná, že cesta **odpovídá vzoru**, a vytáhne z ní [[parametr cesty|parametry]].

Tohle je jádro každého routeru, i toho v Expressu. Vzor i cestu rozdělíš podle `/`
na části a porovnáš je po jedné:

| vzor | cesta | výsledek |
|---|---|---|
| `/api/books/:id` | `/api/books/7` | `{ id: '7' }` |
| `/api/readers/:readerId/loans/:loanId` | `/api/readers/5/loans/2` | `{ readerId: '5', loanId: '2' }` |
| `/api/books/:id` | `/api/loans/7` | `null` (liší se pevná část) |
| `/api/books/:id` | `/api/books` | `null` (jiný počet částí) |
| `/api/books` | `/api/books` | `{}` (sedí, parametry nemá) |

Část vzoru, která začíná dvojtečkou, sedí na cokoli a její hodnota se uloží pod
jménem bez dvojtečky. Ostatní části se musí shodovat přesně.

**Úkol:** seřaď řádky funkce `matchPath(pattern, pathname)` v novém souboru
`router.js`. Server ji zatím nepoužívá, zapojíš ji v dalším kroku. Dva řádky jsou
navíc.

# --hints--

`matchPath` vrátí objekt s parametry, když cesta vzoru odpovídá.

```js
const { matchPath } = await helpers.importFile('router.js');
assert.deepEqual(matchPath('/api/books/:id', '/api/books/7'), { id: '7' }, "matchPath('/api/books/:id', '/api/books/7') má vrátit { id: '7' }");
assert.deepEqual(
  matchPath('/api/readers/:readerId/loans/:loanId', '/api/readers/5/loans/2'),
  { readerId: '5', loanId: '2' },
  "matchPath('/api/readers/:readerId/loans/:loanId', '/api/readers/5/loans/2') má vrátit { readerId: '5', loanId: '2' }",
);
assert.deepEqual(matchPath('/api/books', '/api/books'), {}, "matchPath('/api/books', '/api/books') má vrátit prázdný objekt {}");
```

`matchPath` vrátí `null`, když se liší pevná část nebo počet částí.

```js
const { matchPath } = await helpers.importFile('router.js');
assert.equal(matchPath('/api/books/:id', '/api/loans/7'), null, "matchPath('/api/books/:id', '/api/loans/7') má vrátit null — books a loans se liší");
assert.equal(matchPath('/api/books/:id', '/api/books'), null, "matchPath('/api/books/:id', '/api/books') má vrátit null — cesta má méně částí");
assert.equal(matchPath('/api/books', '/api/books/7'), null, "matchPath('/api/books', '/api/books/7') má vrátit null — cesta má víc částí");
```

// Pomocné funkce pro odpovědi. Každá odpověď serveru prochází přes sendJson.

export function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

export function sendError(res, status, message) {
  sendJson(res, status, { error: message });
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí undefined.
export async function readJson(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    return undefined;
  }
}
```

import { books } from './data.js';
import { sendError, sendJson } from './http.js';

export function listBooks(req, res, { query }) {
  sendJson(res, 200, books);
}

export function getBook(req, res, { params }) {
  const book = books.find((item) => item.id === Number(params.id));
  if (!book) {
    return sendError(res, 404, 'Kniha nenalezena.');
  }
  sendJson(res, 200, book);
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { sendError } from './http.js';
import { routes } from './routes.js';

const port = Number(process.env.PORT ?? 3000);

async function handleRequest(req, res) {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  const route = routes.find((item) => item.method === req.method && item.path === pathname);
  if (!route) {
    return sendError(res, 404, 'Tahle adresa neexistuje.');
  }
  await route.handler(req, res, { params: {}, query: searchParams });
}

createServer(async (req, res) => {
  try {
    await handleRequest(req, res);
  } catch (error) {
    console.error(error);
    sendError(res, 500, 'Na serveru se něco pokazilo.');
  }
}).listen(port, () => {
  console.log(`API knihovny běží na http://localhost:${port}`);
});
```

# --parsons--

```js
export function matchPath(pattern, pathname) {
  const patternParts = pattern.split('/');
  const pathParts = pathname.split('/');
  if (patternParts.length !== pathParts.length) return null;
  const params = {};
  for (let i = 0; i < patternParts.length; i++) {
    if (patternParts[i].startsWith(':')) {
      params[patternParts[i].__1__] = pathParts[i];
    } else if (patternParts[i] !== pathParts[i]) {
      return null;
    }
  }
  return params;
}
```
