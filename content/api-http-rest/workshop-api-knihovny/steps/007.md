# --description--

## Hlášení

Kolega Tomáš rozšířil řazení, aby frontend nemusel čekat na každé nové pole:
„Teď stačí `?sort=author` nebo `?sort=year`, řadí se podle čehokoli." Podle autora
to opravdu funguje. Jenže:

- `GET /api/books?sort=year` vrátí `500` a v terminálu je
  `TypeError: a[sort].localeCompare is not a function`,
- `GET /api/books?sort=heslo` vrátí `500` a v terminálu
  `TypeError: Cannot read properties of undefined (reading 'localeCompare')`.

`500` znamená „chyba serveru", přitom chybu udělal klient: poslal hodnotu, kterou API
neumí. A server tím navíc prozrazuje, podle jakých polí nejde řadit.

## Úkol

Oprav `listBooks` v `books.js`:

- řadit jde jen podle `title`, `author` (obojí podle české abecedy) a `year`
  (vzestupně, jako čísla),
- jakákoli jiná hodnota `sort` vrátí `400` s vysvětlením v `error`,
- bez `sort` a s `sort=title` se nic nemění.

# --hints--

`?sort=year` vrátí `200` a knihy od nejstarší po nejnovější.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(`${server.url}/api/books?sort=year`, { signal: AbortSignal.timeout(3000) })
  .catch(() => assert.fail('GET /api/books?sort=year: server do 3 sekund neodpověděl'));
assert.equal(res.status, 200, 'GET /api/books?sort=year má vrátit stav 200');
const years = (await res.json()).map((book) => book.year);
assert.equal(years.length, 12, 'GET /api/books?sort=year má vrátit všech 12 knih');
assert.deepEqual(years, years.toSorted((a, b) => a - b), `GET /api/books?sort=year má vrátit roky vzestupně, přišlo: ${years.join(', ')}`);
```

Neznámá pole v `sort` vrátí `400` s klíčem `error`, i když se jmenují jako vlastnost objektu.

```js
const server = await helpers.startServer('server.js');
for (const value of ['heslo', 'id', 'constructor']) {
  const res = await fetch(`${server.url}/api/books?sort=${value}`, { signal: AbortSignal.timeout(3000) })
    .catch(() => assert.fail(`GET /api/books?sort=${value}: server do 3 sekund neodpověděl`));
  const hintText = value === 'constructor'
    ? ' — obyčejný objekt má zděděné vlastnosti jako constructor; povolené hodnoty ověř přes Object.hasOwn nebo seznam jmen'
    : '';
  assert.equal(res.status, 400, `GET /api/books?sort=${value} má vrátit stav 400${hintText}`);
  const body = await res.json();
  assert.equal(typeof body.error, 'string', `odpověď 400 na ?sort=${value} má mít klíč error s vysvětlením`);
}
```

# --help--

## --tip--

Hodnotu z query nesmíš použít jako jméno pole bez kontroly — viz
[Typické chyby a pasti](see:api-http-rest/navrh-rest#typicke-chyby-a-pasti) v lekci Návrh REST API.

## --tip--

Nejdřív ověř, jestli je hodnota `sort` mezi povolenými, a teprve pak řaď. Pro každé
povolené pole potřebuješ vlastní porovnání: texty přes `localeCompare`, čísla odečtením.

## --tip--

Stejný princip na jiných datech: objekt, kde klíč je povolená hodnota a hodnota je
porovnávací funkce. `Object.hasOwn(priceSorters, key)` odmítne i zděděné klíče
jako `toString`.

```text
priceSorters = { cheapest: (a, b) => a.price - b.price, name: (a, b) => a.name.localeCompare(b.name, 'cs') }
```

// Pomocné funkce pro odpovědi. Každá odpověď serveru prochází přes sendJson.

export function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

export function sendError(res, status, message) {
  sendJson(res, status, { error: message });
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí undefined.
export async function readJson(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    return undefined;
  }
}

// Porovná cestu se vzorem routy. Vrátí objekt s parametry, nebo null.
export function matchPath(pattern, pathname) {
  const patternParts = pattern.split('/');
  const pathParts = pathname.split('/');
  if (patternParts.length !== pathParts.length) return null;
  const params = {};
  for (let i = 0; i < patternParts.length; i++) {
    if (patternParts[i].startsWith(':')) {
      params[patternParts[i].slice(1)] = pathParts[i];
    } else if (patternParts[i] !== pathParts[i]) {
      return null;
    }
  }
  return params;
}

## --file-- server.js

```js
import { createServer } from 'node:http';
import { sendError } from './http.js';
import { matchPath } from './router.js';
import { routes } from './routes.js';

const port = Number(process.env.PORT ?? 3000);

async function handleRequest(req, res) {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  const allowedMethods = [];
  for (const route of routes) {
    const params = matchPath(route.path, pathname);
    if (params === null) {
      continue;
    }
    if (route.method === req.method) {
      return route.handler(req, res, { params, query: searchParams });
    }
    allowedMethods.push(route.method);
  }

  if (allowedMethods.length > 0) {
    res.setHeader('Allow', allowedMethods.join(', '));
    return sendError(res, 405, 'Tahle metoda tu není povolená.');
  }
  sendError(res, 404, 'Tahle adresa neexistuje.');
}

createServer(async (req, res) => {
  try {
    await handleRequest(req, res);
  } catch (error) {
    console.error(error);
    sendError(res, 500, 'Na serveru se něco pokazilo.');
  }
}).listen(port, () => {
  console.log(`API knihovny běží na http://localhost:${port}`);
});
```

## --file-- books.js

```js
import { books } from './data.js';
import { sendError, sendJson } from './http.js';

// Povolená řazení. Klient smí poslat jen jméno, které tu je.
const sorters = {
  title: (a, b) => a.title.localeCompare(b.title, 'cs'),
  author: (a, b) => a.author.localeCompare(b.author, 'cs'),
  year: (a, b) => a.year - b.year,
};

export function listBooks(req, res, { query }) {
  const genre = query.get('genre');
  let result = genre === null ? books : books.filter((book) => book.genre === genre);
  const sort = query.get('sort');
  if (sort !== null) {
    if (!Object.hasOwn(sorters, sort)) {
      return sendError(res, 400, 'Řadit jde jen podle title, author nebo year.');
    }
    result = result.toSorted(sorters[sort]);
  }
  sendJson(res, 200, result);
}

export function getBook(req, res, { params }) {
  const book = books.find((item) => item.id === Number(params.id));
  if (!book) {
    return sendError(res, 404, 'Kniha nenalezena.');
  }
  sendJson(res, 200, book);
}
```
