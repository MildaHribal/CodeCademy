# --description--

Katalog je hotový: filtr, řazení i stránkování. Tady je dobré místo na pauzu.

Druhá polovina API jsou výpůjčky. V `data.js` je pole `loans` a přibyl soubor
`loans.js` s importy, které budeš potřebovat. Výpůjčka je samostatný zdroj se
svou adresou, stejně jako kniha.

**Úkol:** přidej routu `GET /api/loans/:id`. Vrátí výpůjčku s daným id se stavem `200`,
nebo `404` s klíčem `error`, když taková není. Handler napiš do `loans.js` a routu
přidej do tabulky v `routes.js`. Všechno potřebné už v projektu máš, zkus to bez nápovědy.

# --help--

## --tip--

Jde o stejný vzor jako detail knihy: routa s [parametrem v cestě](see:api-http-rest/navrh-rest#zdroje-a-adresy).

## --tip--

Podívej se, jak jsou napsané `getBook` v `books.js` a řádek routy pro knihu v `routes.js`.
Handler musíš z `loans.js` exportovat a v `routes.js` naimportovat.

## --tip--

Postup bez kódu:

```text
1. v loans.js: najdi výpůjčku podle Number(params.id)
2. nenalezeno → sendError s 404, jinak sendJson s 200
3. v routes.js: import a nový řádek tabulky s metodou, vzorem cesty a handlerem
```

// Pomocné funkce pro odpovědi. Každá odpověď serveru prochází přes sendJson.

export function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

export function sendError(res, status, message) {
  sendJson(res, status, { error: message });
}

// Přečte tělo požadavku a převede ho z JSON. Když tělo není platný JSON, vrátí undefined.
export async function readJson(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    return undefined;
  }
}

// Porovná cestu se vzorem routy. Vrátí objekt s parametry, nebo null.
export function matchPath(pattern, pathname) {
  const patternParts = pattern.split('/');
  const pathParts = pathname.split('/');
  if (patternParts.length !== pathParts.length) return null;
  const params = {};
  for (let i = 0; i < patternParts.length; i++) {
    if (patternParts[i].startsWith(':')) {
      params[patternParts[i].slice(1)] = pathParts[i];
    } else if (patternParts[i] !== pathParts[i]) {
      return null;
    }
  }
  return params;
}

## --file-- books.js

```js
import { books } from './data.js';
import { sendError, sendJson } from './http.js';

// Povolená řazení. Klient smí poslat jen jméno, které tu je.
const sorters = {
  title: (a, b) => a.title.localeCompare(b.title, 'cs'),
  author: (a, b) => a.author.localeCompare(b.author, 'cs'),
  year: (a, b) => a.year - b.year,
};

// Převede text z query na celé číslo v rozsahu. Když to nejde, vrátí null.
function parseIntInRange(text, min, max) {
  const number = Number(text);
  if (text.trim() === '' || !Number.isInteger(number) || number < min || number > max) {
    return null;
  }
  return number;
}

export function listBooks(req, res, { query }) {
  const genre = query.get('genre');
  let result = genre === null ? books : books.filter((book) => book.genre === genre);
  const sort = query.get('sort');
  if (sort !== null) {
    if (!Object.hasOwn(sorters, sort)) {
      return sendError(res, 400, 'Řadit jde jen podle title, author nebo year.');
    }
    result = result.toSorted(sorters[sort]);
  }

  const page = parseIntInRange(query.get('page') ?? '1', 1, Infinity);
  const limit = parseIntInRange(query.get('limit') ?? '5', 1, 50);
  if (page === null || limit === null) {
    return sendError(res, 400, 'page musí být celé číslo od 1 a limit celé číslo od 1 do 50.');
  }
  const start = (page - 1) * limit;
  sendJson(res, 200, {
    data: result.slice(start, start + limit),
    meta: { page, limit, total: result.length, totalPages: Math.ceil(result.length / limit) },
  });
}

export function getBook(req, res, { params }) {
  const book = books.find((item) => item.id === Number(params.id));
  if (!book) {
    return sendError(res, 404, 'Kniha nenalezena.');
  }
  sendJson(res, 200, book);
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { sendError } from './http.js';
import { matchPath } from './router.js';
import { routes } from './routes.js';

const port = Number(process.env.PORT ?? 3000);

async function handleRequest(req, res) {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  const allowedMethods = [];
  for (const route of routes) {
    const params = matchPath(route.path, pathname);
    if (params === null) {
      continue;
    }
    if (route.method === req.method) {
      return route.handler(req, res, { params, query: searchParams });
    }
    allowedMethods.push(route.method);
  }

  if (allowedMethods.length > 0) {
    res.setHeader('Allow', allowedMethods.join(', '));
    return sendError(res, 405, 'Tahle metoda tu není povolená.');
  }
  sendError(res, 404, 'Tahle adresa neexistuje.');
}

createServer(async (req, res) => {
  try {
    await handleRequest(req, res);
  } catch (error) {
    console.error(error);
    sendError(res, 500, 'Na serveru se něco pokazilo.');
  }
}).listen(port, () => {
  console.log(`API knihovny běží na http://localhost:${port}`);
});
```
