# --description--

Shrnutí ukazuje částky, ale chybí mu kontext, který zákazníkovi pomáhá rozhodnout se.

**Úkol:** Rozšiř `renderSummary()`:

- do `#items-count` počet všech kusů ve správném tvaru (`5 kusů`, `3 kusy`) — počet ti vrátí pomocná `pageItemCount()`,
- do `#vat` DPH 21 % obsaženou v celkové ceně (se započtenou dopravou), česky,
- do `#free-shipping` text o dopravě zdarma podle mezisoučtu.

# --hints--

V `#items-count` je počet kusů ve správném tvaru.

```js
const plain = (selector) => document.querySelector(selector).textContent.replace(/\s/g, ' ').trim();
const qty = (index) => document.querySelectorAll('.cart-row input')[index];

assert.equal(plain('#items-count'), '5 kusů', "#items-count má po načtení ukazovat '5 kusů' (1 + 1 + 3)");
await helpers.type(qty(2), '1');
assert.equal(plain('#items-count'), '3 kusy', "po změně obalů na 1 kus má #items-count ukazovat '3 kusy'");
```

V `#vat` je DPH obsažená v celkové ceně včetně dopravy.

```js
const plain = (selector) => document.querySelector(selector).textContent.replace(/\s/g, ' ').trim();
const qty = (index) => document.querySelectorAll('.cart-row input')[index];

assert.equal(plain('#vat'), '258,02 Kč', "#vat má po načtení ukazovat '258,02 Kč' — 21 % DPH obsažená v 1 486,70 Kč");
```

V `#free-shipping` je, kolik chybí do dopravy zdarma, nebo `Doprava zdarma`.

```js
const plain = (selector) => document.querySelector(selector).textContent.replace(/\s/g, ' ').trim();
const qty = (index) => document.querySelectorAll('.cart-row input')[index];

assert.equal(plain('#free-shipping'), 'Do dopravy zdarma zbývá 102,30 Kč', "#free-shipping má po načtení ukazovat 'Do dopravy zdarma zbývá 102,30 Kč'");
await helpers.type(qty(1), '3');
assert.equal(plain('#free-shipping'), 'Doprava zdarma', "po změně Krycích jmen na 3 kusy má #free-shipping ukazovat 'Doprava zdarma'");
```

# --help--

## --tip--

Tvar slova podle počtu je v části [Množné číslo: Intl.PluralRules](see:js-retezce-cisla/intl-a-datum#mnozne-cislo-intl-pluralrules), DPH a doprava zdarma jsou tvoje funkce z kroků 6 a 19.

## --tip--

Počet kusů si ulož do proměnné. DPH počítej z mezisoučtu plus dopravy v haléřích a zformátuj až výsledek.

## --tip--

Na jiných datech: `document.querySelector('#guests').textContent = guestsLabel(countGuests());`

// Hrací doupě: ceny v košíku e-shopu s deskovými hrami.
// Peníze se počítají v celých haléřích (79900 = 799,00 Kč), na koruny se převádí až při výpisu.

// Popisek dopravy v korunách: 0 → 'Doprava zdarma'.
function shippingLabel(shippingKc) {
  if (shippingKc === 0) {
    return 'Doprava zdarma';
  }
  return `Doprava ${shippingKc} Kč`;
}

// Cena v korunách → celé haléře (4.35 → 435).
function toHalere(koruny) {
  return Math.round(koruny * 100);
}

// Počet kusů z textu formuláře; nesmysl, prázdné pole a méně než 1 → 1.
function parseQuantity(input) {
  const quantity = Number(input);
  if (Number.isNaN(quantity) || quantity < 1) {
    return 1;
  }
  return Math.floor(quantity);
}

// Cena řádku v haléřích: cena za kus × počet kusů.
function lineTotal(unitPriceHalere, quantity) {
  return unitPriceHalere * quantity;
}

// Mezisoučet košíku v haléřích z cen jednotlivých řádků.
function cartSubtotal(lineTotals) {
  let subtotal = 0;
  for (const total of lineTotals) {
    subtotal += total;
  }
  return subtotal;
}

// DPH obsažená v ceně s DPH, v celých haléřích.
function vatIncluded(grossHalere, ratePercent) {
  return Math.round(grossHalere * ratePercent / (100 + ratePercent));
}

// Od kolegy: cena řádku pro starý export do účetnictví, které chce koruny.
function lineTotalKc(priceKc, quantity) {
  return Math.round(priceKc * quantity * 100) / 100;
}

// Cena v haléřích jako český text: 139770 → '1 397,70 Kč'.
function formatCzk(halere) {
  const czk = new Intl.NumberFormat('cs-CZ', { style: 'currency', currency: 'CZK' });
  return czk.format(halere / 100);
}

// Řádek košíku: 'Carcassonne 2× · 1 598,00 Kč'.
function priceLine(name, unitPriceHalere, quantity) {
  return `${name} ${quantity}× · ${formatCzk(lineTotal(unitPriceHalere, quantity))}`;
}

// Cena po slevě v celých haléřích, zaokrouhlená na nejbližší haléř.
function applyDiscount(halere, percent) {
  return Math.round(halere * (100 - percent) / 100);
}

function gamesLabel(count) {
  const category = new Intl.PluralRules('cs').select(count);
  if (category === 'one') {
    return `${count} hra`;
  }
  if (category === 'few') {
    return `${count} hry`;
  }
  return `${count} her`;
}

// Počet kusů ve správném tvaru: 1 kus, 3 kusy, 7 kusů.
function piecesLabel(count) {
  const category = new Intl.PluralRules('cs').select(count);
  if (category === 'one') {
    return `${count} kus`;
  }
  if (category === 'few') {
    return `${count} kusy`;
  }
  return `${count} kusů`;
}

// Cena dopravy v haléřích: zdarma od 1 500 Kč a pro prázdný košík.
function shippingPrice(subtotalHalere) {
  if (subtotalHalere === 0 || subtotalHalere >= 150000) {
    return 0;
  }
  return 8900;
}

// Od kolegy: text do potvrzovacího e-mailu, e-mailová šablona chce tečku a dvě desetinná místa.
function orderTotalText(subtotalHalere, shippingHalere) {
  const subtotal = subtotalHalere / 100;
  const shipping = shippingHalere / 100;
  return `Celkem ${(subtotal + shipping).toFixed(2)} Kč`;
}

// Datum doručení: za 2 dny, víkend se posune na pondělí. Vstup i výstup '2026-09-14'.
function deliveryDate(orderDate) {
  let delivery = Temporal.PlainDate.from(orderDate).add({ days: 2 });
  if (delivery.dayOfWeek === 6) {
    delivery = delivery.add({ days: 2 });
  } else if (delivery.dayOfWeek === 7) {
    delivery = delivery.add({ days: 1 });
  }
  return delivery.toString();
}

// Datum doručení česky: '2026-09-16' → 'středa 16. září'.
function formatDeliveryDate(isoDate) {
  return Temporal.PlainDate.from(isoDate).toLocaleString('cs-CZ', { weekday: 'long', day: 'numeric', month: 'long' });
}

// Počet kalendářních dnů mezi dvěma daty ve tvaru '2026-09-14'.
function daysUntil(fromDate, toDate) {
  return Temporal.PlainDate.from(fromDate).until(toDate).days;
}

// Relativní text doručení: dnes, zítra, pozítří, za 3 dny.
function deliveryText(today, delivery) {
  const relative = new Intl.RelativeTimeFormat('cs', { numeric: 'auto' });
  return relative.format(daysUntil(today, delivery), 'day');
}

// Kolik chybí do dopravy zdarma, nebo 'Doprava zdarma'.
function freeShippingText(subtotalHalere) {
  if (subtotalHalere >= 150000) {
    return 'Doprava zdarma';
  }
  return `Do dopravy zdarma zbývá ${formatCzk(150000 - subtotalHalere)}`;
}

// První splátka v haléřích: ostatní jsou podíl zaokrouhlený dolů, první dorovná zbytek.
function firstInstallment(totalHalere, count) {
  const otherInstallment = Math.floor(totalHalere / count);
  return totalHalere - otherInstallment * (count - 1);
}

## --file-- cart.js

```js
// Živé shrnutí košíku. Funkce ze script.js jsou tu k dispozici.
const rows = document.querySelectorAll('.cart-row');
const today = Temporal.Now.plainDateISO().toString();

// Mezisoučet košíku v haléřích podle počtu kusů na stránce.
function pageSubtotal() {
  let subtotal = 0;
  for (const row of rows) {
    subtotal += lineTotal(Number(row.dataset.price), parseQuantity(row.querySelector('input').value));
  }
  return subtotal;
}

// Počet všech kusů v košíku.
function pageItemCount() {
  let count = 0;
  for (const row of rows) {
    count += parseQuantity(row.querySelector('input').value);
  }
  return count;
}

// Cena každého řádku česky.
function renderRows() {
  for (const row of rows) {
    const total = lineTotal(Number(row.dataset.price), parseQuantity(row.querySelector('input').value));
    row.querySelector('.cart-row__total').textContent = formatCzk(total);
  }
}

// Vypíše shrnutí košíku.
function renderSummary() {
  const subtotal = pageSubtotal();
  const shipping = shippingPrice(subtotal);
  const itemCount = pageItemCount();

  document.querySelector('#subtotal').textContent = formatCzk(subtotal);
  document.querySelector('#shipping').textContent = shipping === 0 ? 'Zdarma' : formatCzk(shipping);
  document.querySelector('#total').textContent = formatCzk(subtotal + shipping);
  document.querySelector('#items-count').textContent = piecesLabel(itemCount);
  document.querySelector('#vat').textContent = formatCzk(vatIncluded(subtotal + shipping, 21));
  document.querySelector('#free-shipping').textContent = freeShippingText(subtotal);
}

// Propojení se stránkou: po změně počtu kusů se všechno přepočítá.
// Posluchače událostí (addEventListener) podrobně probere sekce DOM, události a prohlížeč.
function renderCart() {
  renderRows();
  renderSummary();
}
document.querySelector('.cart').addEventListener('input', renderCart);
renderCart();
```
