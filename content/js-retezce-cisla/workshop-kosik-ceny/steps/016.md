# --description--

`2026-09-16` je tvar pro počítač. Zákazník chce číst „středa 16. září".

**Úkol:** Napiš funkci `formatDeliveryDate(isoDate)`, která text `'2026-09-16'` vrátí jako den v týdnu celým slovem, číslo dne s tečkou a měsíc celým slovem: `středa 16. září`. Rok se nevypisuje.

Kandidáti: `Intl.DateTimeFormat` s volbami `weekday`, `day` a `month`, nebo metoda `toLocaleString` přímo na objektu `Temporal.PlainDate` se stejnými volbami. Jazyk je `'cs-CZ'`.

> [!TIP]
> Tady máš hotové ceny i datum doručení, klidně si dej pauzu. Zbytek workshopu přidá odpočet, splátky a živé shrnutí košíku.

# --hints--

`formatDeliveryDate` vrátí den v týdnu, den a měsíc česky.

```js
assert.equal(formatDeliveryDate('2026-09-16'), 'středa 16. září', "formatDeliveryDate('2026-09-16') má vrátit 'středa 16. září'");
assert.equal(formatDeliveryDate('2026-12-21'), 'pondělí 21. prosince', "formatDeliveryDate('2026-12-21') má vrátit 'pondělí 21. prosince'");
```

Den se píše bez nuly na začátku.

```js
assert.equal(formatDeliveryDate('2027-01-01'), 'pátek 1. ledna', "formatDeliveryDate('2027-01-01') má vrátit 'pátek 1. ledna'");
```

Datum je česky i v prohlížeči nastaveném na jiný jazyk.

```js
// Prohlížeč bez češtiny: formátování data bez zadaného jazyka dostane angličtinu.
const english = (locales) => (locales === undefined || (Array.isArray(locales) && locales.length === 0) ? 'en' : locales);
const OriginalDateTimeFormat = Intl.DateTimeFormat;
const originalPlainDate = Temporal.PlainDate.prototype.toLocaleString;
const originalDateString = Date.prototype.toLocaleString;
const originalDateOnly = Date.prototype.toLocaleDateString;
Intl.DateTimeFormat = function (locales, options) {
  return new OriginalDateTimeFormat(english(locales), options);
};
Temporal.PlainDate.prototype.toLocaleString = function (locales, options) {
  return originalPlainDate.call(this, english(locales), options);
};
Date.prototype.toLocaleString = function (locales, options) {
  return originalDateString.call(this, english(locales), options);
};
Date.prototype.toLocaleDateString = function (locales, options) {
  return originalDateOnly.call(this, english(locales), options);
};
try {
  assert.equal(formatDeliveryDate('2026-09-16'), 'středa 16. září', "formatDeliveryDate('2026-09-16') má vrátit 'středa 16. září' i v prohlížeči nastaveném na angličtinu — jazyk 'cs-CZ' patří do kódu");
} finally {
  Intl.DateTimeFormat = OriginalDateTimeFormat;
  Temporal.PlainDate.prototype.toLocaleString = originalPlainDate;
  Date.prototype.toLocaleString = originalDateString;
  Date.prototype.toLocaleDateString = originalDateOnly;
}
```

# --help--

## --tip--

Volby pro části data vysvětluje [Datum a čas: Intl.DateTimeFormat](see:js-retezce-cisla/intl-a-datum#datum-a-cas-intl-datetimeformat).

## --tip--

Z textu vytvoř `PlainDate` a zavolej na něm `toLocaleString('cs-CZ', …)`. Den v týdnu i měsíc chceš celým slovem (`'long'`), den jako číslo (`'numeric'`).

## --tip--

Na jiných datech: `date.toLocaleString('cs-CZ', { month: 'long', year: 'numeric' })` dá „září 2026".

// Hrací doupě: ceny v košíku e-shopu s deskovými hrami.
// Peníze se počítají v celých haléřích (79900 = 799,00 Kč), na koruny se převádí až při výpisu.

// Popisek dopravy v korunách: 0 → 'Doprava zdarma'.
function shippingLabel(shippingKc) {
  if (shippingKc === 0) {
    return 'Doprava zdarma';
  }
  return `Doprava ${shippingKc} Kč`;
}

// Cena v korunách → celé haléře (4.35 → 435).
function toHalere(koruny) {
  return Math.round(koruny * 100);
}

// Počet kusů z textu formuláře; nesmysl, prázdné pole a méně než 1 → 1.
function parseQuantity(input) {
  const quantity = Number(input);
  if (Number.isNaN(quantity) || quantity < 1) {
    return 1;
  }
  return Math.floor(quantity);
}

// Cena řádku v haléřích: cena za kus × počet kusů.
function lineTotal(unitPriceHalere, quantity) {
  return unitPriceHalere * quantity;
}

// Mezisoučet košíku v haléřích z cen jednotlivých řádků.
function cartSubtotal(lineTotals) {
  let subtotal = 0;
  for (const total of lineTotals) {
    subtotal += total;
  }
  return subtotal;
}

// DPH obsažená v ceně s DPH, v celých haléřích.
function vatIncluded(grossHalere, ratePercent) {
  return Math.round(grossHalere * ratePercent / (100 + ratePercent));
}

// Od kolegy: cena řádku pro starý export do účetnictví, které chce koruny.
function lineTotalKc(priceKc, quantity) {
  return Math.round(priceKc * quantity * 100) / 100;
}

// Cena v haléřích jako český text: 139770 → '1 397,70 Kč'.
function formatCzk(halere) {
  const czk = new Intl.NumberFormat('cs-CZ', { style: 'currency', currency: 'CZK' });
  return czk.format(halere / 100);
}

// Řádek košíku: 'Carcassonne 2× · 1 598,00 Kč'.
function priceLine(name, unitPriceHalere, quantity) {
  return `${name} ${quantity}× · ${formatCzk(lineTotal(unitPriceHalere, quantity))}`;
}

// Cena po slevě v celých haléřích, zaokrouhlená na nejbližší haléř.
function applyDiscount(halere, percent) {
  return Math.round(halere * (100 - percent) / 100);
}

function gamesLabel(count) {
  const category = new Intl.PluralRules('cs').select(count);
  if (category === 'one') {
    return `${count} hra`;
  }
  if (category === 'few') {
    return `${count} hry`;
  }
  return `${count} her`;
}

// Počet kusů ve správném tvaru: 1 kus, 3 kusy, 7 kusů.
function piecesLabel(count) {
  const category = new Intl.PluralRules('cs').select(count);
  if (category === 'one') {
    return `${count} kus`;
  }
  if (category === 'few') {
    return `${count} kusy`;
  }
  return `${count} kusů`;
}

// Cena dopravy v haléřích: zdarma od 1 500 Kč a pro prázdný košík.
function shippingPrice(subtotalHalere) {
  if (subtotalHalere === 0 || subtotalHalere >= 150000) {
    return 0;
  }
  return 8900;
}

// Od kolegy: text do potvrzovacího e-mailu, e-mailová šablona chce tečku a dvě desetinná místa.
function orderTotalText(subtotalHalere, shippingHalere) {
  const subtotal = subtotalHalere / 100;
  const shipping = shippingHalere / 100;
  return `Celkem ${(subtotal + shipping).toFixed(2)} Kč`;
}

// Datum doručení: za 2 dny, víkend se posune na pondělí. Vstup i výstup '2026-09-14'.
function deliveryDate(orderDate) {
  let delivery = Temporal.PlainDate.from(orderDate).add({ days: 2 });
  if (delivery.dayOfWeek === 6) {
    delivery = delivery.add({ days: 2 });
  } else if (delivery.dayOfWeek === 7) {
    delivery = delivery.add({ days: 1 });
  }
  return delivery.toString();
}
