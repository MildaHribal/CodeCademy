# --description--

Hrací doupě má víkendovou akci: vybrané hry o 15 % levněji.

**Úkol:** Napiš funkci `applyDiscount(halere, percent)`, která vrátí cenu po slevě v **celých haléřích**, zaokrouhlenou na nejbližší haléř (přesná polovina nahoru). Sleva `0` cenu nemění, sleva `100` z ní udělá nulu.

Kandidáti na zaokrouhlení jsou `Math.round`, `Math.floor` a `toFixed` — jen jeden vrátí číslo zaokrouhlené na nejbližší celé.

# --hints--

`applyDiscount` vrátí cenu po slevě v celých haléřích.

```js
assert.equal(applyDiscount(79900, 15), 67915, 'applyDiscount(79900, 15) má vrátit 67915');
assert.equal(applyDiscount(4990, 10), 4491, 'applyDiscount(4990, 10) má vrátit 4491');
```

Cena po slevě se zaokrouhlí na nejbližší haléř.

```js
assert.equal(applyDiscount(999, 50), 500, 'applyDiscount(999, 50) má vrátit 500 (499,5 zaokrouhleno nahoru)');
assert.equal(applyDiscount(1001, 30), 701, 'applyDiscount(1001, 30) má vrátit 701 (700,7 zaokrouhleno na nejbližší)');
assert.equal(typeof applyDiscount(1001, 30), 'number', 'applyDiscount má vrátit číslo, ne text');
```

Sleva `0` cenu nemění a sleva `100` dá nulu.

```js
assert.equal(applyDiscount(44900, 0), 44900, 'applyDiscount(44900, 0) má vrátit 44900');
assert.equal(applyDiscount(44900, 100), 0, 'applyDiscount(44900, 100) má vrátit 0');
```

# --help--

## --tip--

Zaokrouhlovací funkce porovnává tabulka v části [Zaokrouhlení: round, floor, ceil a trunc](see:js-retezce-cisla/cisla#zaokrouhleni-round-floor-ceil-a-trunc).

## --tip--

Cena po slevě je cena krát `(100 - percent)` a vydělená stovkou. Celý výpočet zaokrouhli jednou, až na konci.

## --tip--

Na jiných datech: navýšení o 8 % je `Math.round(amount * 108 / 100)`.

// Peníze se počítají v celých haléřích (79900 = 799,00 Kč), na koruny se převádí až při výpisu.

function toHalere(koruny) {
  return Math.round(koruny * 100);
}

function parseQuantity(input) {
  const quantity = Number(input);
  if (Number.isNaN(quantity) || quantity < 1) {
    return 1;
  }
  return Math.floor(quantity);
}

}

// Mezisoučet košíku v haléřích z cen jednotlivých řádků.
function cartSubtotal(lineTotals) {
  let subtotal = 0;
  for (const total of lineTotals) {
    subtotal += total;
  }
  return subtotal;
}

// DPH obsažená v ceně s DPH, v celých haléřích.
function vatIncluded(grossHalere, ratePercent) {
  return Math.round(grossHalere * ratePercent / (100 + ratePercent));
}

// Od kolegy: cena řádku pro starý export do účetnictví, které chce koruny.
function lineTotalKc(priceKc, quantity) {
  return Math.round(priceKc * quantity * 100) / 100;
}

// Cena v haléřích jako český text: 139770 → '1 397,70 Kč'.
function formatCzk(halere) {
  const czk = new Intl.NumberFormat('cs-CZ', { style: 'currency', currency: 'CZK' });
  return czk.format(halere / 100);
}

// Řádek košíku: 'Carcassonne 2× · 1 598,00 Kč'.

  return `${name} ${quantity}× · ${formatCzk(lineTotal(unitPriceHalere, quantity))}`;
