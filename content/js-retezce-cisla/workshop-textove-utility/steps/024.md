# --description--

Karta má titulek, teď popis. V náhledu se ukazuje jen začátek popisu a pod ním počítadlo, kolik znaků z limitu 300 prodejce napsal.

**Úkol:** Rozšiř `renderPreview()`:

- do `#preview-description` vypiš popis z pole `description` bez mezer na krajích, zkrácený na nejvýš 120 znaků **u celého slova**,
- do `#description-count` vypiš počet znaků popisu tak, jak je vidí člověk, ve tvaru `57 / 300`.

Emoji v popisu musí počítadlo spočítat jako jeden znak.

# --hints--

V `#preview-description` je popis zkrácený u celého slova na nejvýš 120 znaků.

```js
const text = (selector) => document.querySelector(selector).textContent.trim();
const field = (name) => document.querySelector(`#listing-form [name="${name}"]`);

const expected = 'Prodám horské kolo Author Traction, velikost rámu L, najeto asi 2 000 km. Kotoučové brzdy Shimano po servisu, nové…';
assert.equal(text('#preview-description'), expected, `#preview-description má po načtení ukazovat popis zkrácený u slova: '${expected}'`);
await helpers.type(field('description'), '  Krátký popis  ');
assert.equal(text('#preview-description'), 'Krátký popis', "po napsání '  Krátký popis  ' má #preview-description ukazovat 'Krátký popis' bez výpustky");
```

V `#description-count` je počet viditelných znaků popisu a limit `300`.

```js
const text = (selector) => document.querySelector(selector).textContent.trim();
const field = (name) => document.querySelector(`#listing-form [name="${name}"]`);

await helpers.type(field('description'), 'Top stav 👍🏽');
assert.equal(text('#description-count'), '10 / 300', "po napsání 'Top stav 👍🏽' má #description-count ukazovat '10 / 300' — palec s odstínem pleti je jeden znak");
await helpers.type(field('description'), 'Kolo');
assert.equal(text('#description-count'), '4 / 300', "po napsání 'Kolo' má #description-count ukazovat '4 / 300'");
```

# --help--

## --tip--

Počítání viditelných znaků je v části [Unicode: diakritika a emoji](see:js-retezce-cisla/retezce#unicode-diakritika-a-emoji), zkrácení u slova máš z kroku 17.

## --tip--

Hodnotu popisu si ulož na začátku funkce, stejně jako titulek. Oříznutý popis pošli do funkce, která zkracuje u slova; počet znaků vlož do šablonového řetězce.

## --tip--

Na jiných datech (počítadlo u krátkého profilu):

```js
document.querySelector('#bio-count').textContent = `${countCharacters(bio)} / 160`;
```

// Kolobazar: textové utility pro inzeráty v bazaru kol.
// Každá funkce dostane text a vrátí nový text, číslo nebo true/false.

// Popisek prodejce pod inzerátem.
function sellerLabel(name) {
  if (name === '') {
    return 'Prodává: anonymní prodejce';
  }
  return `Prodává: ${name}`;
}

// Titulek bez bílých znaků na krajích.
function cleanTitle(title) {
  return title.trim();
}

// Text s velkým prvním písmenem, zbytek beze změny.
function capitalize(text) {
  return text.slice(0, 1).toUpperCase() + text.slice(1);
}

// Obsahuje titulek hledané slovo? Velikost písmen nerozhoduje.
function hasKeyword(title, keyword) {
  return title.toLowerCase().includes(keyword.toLowerCase());
}

// Zkrácený text s výpustkou, když je delší než maxLength.
function truncate(text, maxLength) {
  if (text.length <= maxLength) {
    return text;
  }
  return text.slice(0, maxLength) + '…';
}

// Kód inzerátu s pevnou délkou, např. INZ-000042.
function listingCode(id) {
  return `INZ-${String(id).padStart(6, '0')}`;
}

// Kolega upravil: snese i telefon zadaný s mezerami, třeba '777 123 456'.
function formatPhone(phone) {
  const digits = phone.replaceAll(' ', '');
  return `${digits.slice(0, 3)} ${digits.slice(3, 6)} ${digits.slice(6)}`;
}

// E-mail s hvězdičkami místo jména, doména zůstane.
function maskEmail(email) {
  const at = email.indexOf('@');
  if (at === -1) {
    return email;
  }
  return email.slice(0, 1) + '*'.repeat(at - 1) + email.slice(at);
}

// Platný telefon: po odstranění mezer přesně 9 číslic.
function isValidPhone(phone) {
  const digits = phone.replaceAll(' ', '');
  if (digits.length !== 9) {
    return false;
  }
  for (const char of digits) {
    if (!'0123456789'.includes(char)) {
      return false;
    }
  }
  return true;
}

// Iniciály ze jména: 'Jana Nováková' → 'JN'.
function initials(fullName) {
  let result = '';
  for (const word of fullName.split(' ')) {
    if (word !== '') {
      result += word.slice(0, 1).toUpperCase();
    }
  }
  return result;
}

// Od kolegy: text bez háčků a čárek, pro hledání a adresy inzerátů.
function removeDiacritics(text) {
  return text.normalize('NFD').replace(/\p{M}/gu, '');
}

// Obsahuje titulek hledaný text? Rozhoduje jen písmeno, ne velikost ani diakritika.
function matchesSearch(title, query) {
  const normalizedTitle = removeDiacritics(title).toLowerCase();
  const normalizedQuery = removeDiacritics(query.trim()).toLowerCase();
  return normalizedTitle.includes(normalizedQuery);
}

function slugify(title) {
  const allowed = 'abcdefghijklmnopqrstuvwxyz0123456789';
  let slug = '';
  for (const char of removeDiacritics(title).toLowerCase()) {
    if (allowed.includes(char)) {
      slug += char;
    } else if (slug !== '' && !slug.endsWith('-')) {
      slug += '-';
    }
  }
  return slug.endsWith('-') ? slug.slice(0, -1) : slug;
}

// Hashtag z textu: jen písmena a číslice bez diakritiky, např. #horskekolo29.
function hashtag(text) {
  const allowed = 'abcdefghijklmnopqrstuvwxyz0123456789';
  let tag = '';
  for (const char of removeDiacritics(text).toLowerCase()) {
    if (allowed.includes(char)) {
      tag += char;
    }
  }
  return tag === '' ? '' : `#${tag}`;
}

// Počet znaků tak, jak je vidí člověk (grafémy).
function countCharacters(text) {
  const segmenter = new Intl.Segmenter('cs', { granularity: 'grapheme' });
  return [...segmenter.segment(text)].length;
}

// Zkrácený text, který nekončí uprostřed slova.
function truncateAtWord(text, maxLength) {
  if (text.length <= maxLength) {
    return text;
  }
  let cut = text.slice(0, maxLength);
  const lastSpace = cut.lastIndexOf(' ');
  if (lastSpace !== -1) {
    cut = cut.slice(0, lastSpace);
  }
  return `${cut}…`;
}

// Porovnání titulků podle české abecedy: záporné, 0, nebo kladné číslo.
function compareTitles(a, b) {
  return a.localeCompare(b, 'cs');
}

// Řádek s prodejcem: 'JN · Jana Nováková · 777 123 456'.
function sellerLine(name, phone) {
  const start = `${initials(name)} · ${name.trim()}`;
  if (!isValidPhone(phone)) {
    return start;
  }
  return `${start} · ${formatPhone(phone)}`;
}

// Titulek s prvním výskytem hledaného textu v «uvozovkách».
function highlight(title, query) {
  const cleanQuery = query.trim();
  const position = title.toLowerCase().indexOf(cleanQuery.toLowerCase());
  if (cleanQuery === '' || position === -1) {
    return title;
  }
  const end = position + cleanQuery.length;
  return `${title.slice(0, position)}«${title.slice(position, end)}»${title.slice(end)}`;
}

// Počet slov oddělených libovolným počtem mezer.
function wordCount(text) {
  let count = 0;
  for (const word of text.split(' ')) {
    if (word !== '') {
      count += 1;
    }
  }
  return count;
}

// Adresa inzerátu: /inzerat/42-horske-kolo.
function listingUrl(id, title) {
  const slug = slugify(title);
  if (slug === '') {
    return `/inzerat/${id}`;
  }
  return `/inzerat/${id}-${slug}`;
}

## --file-- preview.js

```js
// Živý náhled inzerátu. Funkce ze script.js jsou tu k dispozici.
const LISTING_ID = 42;
const form = document.querySelector('#listing-form');
const fields = form.elements;

// Vypíše hodnoty z formuláře do karty náhledu.
function renderPreview() {
  const title = fields.title.value;
  const description = fields.description.value;

  document.querySelector('#preview-code').textContent = listingCode(LISTING_ID);
  document.querySelector('#preview-title').textContent = capitalize(cleanTitle(title));
  document.querySelector('#preview-url').textContent = listingUrl(LISTING_ID, title);
  document.querySelector('#preview-description').textContent = truncateAtWord(description.trim(), 120);
  document.querySelector('#description-count').textContent = `${countCharacters(description)} / 300`;
}

// Propojení s formulářem a hledáním: po každé změně se náhled překreslí.
// Posluchače událostí (addEventListener) podrobně probere sekce DOM, události a prohlížeč.
form.addEventListener('input', renderPreview);
document.querySelector('#search').addEventListener('input', renderPreview);
renderPreview();
```
