# --description--

## Hlášení

„Ahoj, přerovnal jsem formulář a teď nejde přidat nic. Vyplním název, vyplním cenu
2200, dám Přidat inzerát — a u ceny vyskočí *Cena musí být kladné celé číslo*,
i když je tam normální číslo. Na název si akce nestěžuje, jen na cenu. Co jsem
rozbil?"

## Úkol

Najdi příčinu a oprav ji. Změň co nejmenší kus kódu — akce v `akce.js` je
v pořádku, chyba je ve formuláři.

# --hints--

Vyplněná cena projde: po odeslání se objeví hlášení o uložení.

```js
await helpers.type(document.querySelector('[name="nazev"]'), 'Stan Husky Bizon 3');
const cena = document.querySelector('form input[type="number"]');
assert.ok(cena, 'Ve formuláři má zůstat číselné políčko na cenu');
await helpers.type(cena, '2200');
await helpers.submit(document.querySelector('form'));
await helpers.waitFor(() => /je na bazárku/.test(document.body.textContent));
assert.match(helpers.normalize(document.body.textContent), /Inzerát je na bazárku\./, 'S vyplněnou cenou má odeslání projít');
```

Akce dostane cenu pod klíčem `cena`.

```js
const formular = document.querySelector('form');
const cena = formular.querySelector('input[type="number"]');
assert.equal(cena.name, 'cena', 'Políčko ceny musí mít name="cena" — pod tím klíčem ho akce hledá');
const data = new FormData(formular);
assert.ok(data.has('cena'), 'V datech formuláře má být klíč cena');
```

Zbytek formuláře funguje dál: krátký název pořád hlásí chybu a popisek u ceny drží.

```js
await helpers.type(document.querySelector('[name="nazev"]'), 'ab');
await helpers.type(document.querySelector('[name="cena"]'), '2200');
await helpers.submit(document.querySelector('form'));
const hlaska = await helpers.waitFor(() => document.querySelector('[role="alert"]'));
assert.match(helpers.normalize(hlaska.textContent), /3 znaky/, 'Krátký název má pořád hlásit chybu');
const cena = document.querySelector('[name="cena"]');
const popisek = document.querySelector('label[for="' + cena.id + '"]');
assert.ok(popisek, 'Popisek u ceny má zůstat propojený s políčkem');
```

# --help--

## --tip-- 2

Akce si data bere z `formData.get(…)`. Pod jakým jménem tam hodnota políčka
skončí, určuje jediný atribut — projdi si ho u všech tří políček a porovnej
je s klíči v `akce.js`.

## --tip-- 2

Atribut `id` slouží popisku, atribut `name` datům formuláře. Nemusí být stejné,
ale tady mají být obojí.

```jsx
import { useActionState } from 'react';

import { vytvorInzerat } from './akce';
import { pocatecniInzeraty } from './data';

const meny = new Intl.NumberFormat('cs-CZ', {
  style: 'currency',
  currency: 'CZK',
  maximumFractionDigits: 0,
});

export function formatujCenu(castka) {
  return meny.format(castka);
}

export default function Bazarek() {
  const inzeraty = pocatecniInzeraty;
  const [stav, akce, ceka] = useActionState(vytvorInzerat, {});

  return (
    <main className="bazarek">
      <header className="hlavicka">
        <h1>Bazárek</h1>
        <p className="podtitulek">Věci, které někomu doma překážejí.</p>
      </header>

      <form className="formular" action={akce}>
        <h2>Přidat inzerát</h2>

        <p className="pole">
          <label htmlFor="nazev">Název</label>
          <input id="nazev" name="nazev" defaultValue={stav.hodnoty?.nazev ?? ''} aria-invalid={Boolean(stav.chyby?.nazev)} />
          {stav.chyby?.nazev && <span className="chyba" role="alert">{stav.chyby.nazev}</span>}
        </p>

        <p className="pole">
          <label htmlFor="cena">Cena v Kč</label>
          <input id="cena" name="castka" type="number" defaultValue={stav.hodnoty?.cena ?? ''} aria-invalid={Boolean(stav.chyby?.cena)} />
          {stav.chyby?.cena && <span className="chyba" role="alert">{stav.chyby.cena}</span>}
        </p>

        <p className="pole">
          <label htmlFor="popis">Popis</label>
          <textarea id="popis" name="popis" defaultValue={stav.hodnoty?.popis ?? ''} />
        </p>

        <button className="tlacitko" type="submit" disabled={ceka}>
          {ceka ? 'Ukládám…' : 'Přidat inzerát'}
        </button>

        <p className="hlaseni">{stav.ok ? 'Inzerát je na bazárku.' : ''}</p>
      </form>

      <section className="nabidka">
        <h2>Na prodej</h2>
        <ul className="seznam" data-testid="seznam">
          {inzeraty.map((inzerat) => (
            <li className="karta" key={inzerat.id}>
              <h3>{inzerat.nazev}</h3>
              <p className="karta__cena">{formatujCenu(inzerat.cena)}</p>
              <p className="karta__popis">{inzerat.popis}</p>
            </li>
          ))}
        </ul>
      </section>
    </main>
  );
}
```

## --file-- akce.js

```js
// Tahle funkce zastupuje serverovou akci. V Next.js by měla na prvním řádku
// 'use server', běžela by na serveru a zapisovala do databáze. Tady jen chvíli
// počká a vrátí stejný tvar výsledku, jaký by vrátila doopravdy.
const pauza = (ms) => new Promise((splnit) => setTimeout(splnit, ms));
export async function vytvorInzerat(predchoziStav, formData) {
  const nazev = String(formData.get('nazev') ?? '').trim();
  const cena = String(formData.get('cena') ?? '').trim();
  const popis = String(formData.get('popis') ?? '').trim();
  await pauza(400);
  const chyby = {};
  if (nazev.length < 3) chyby.nazev = 'Název musí mít aspoň 3 znaky.';
  if (!Number.isInteger(Number(cena)) || Number(cena) <= 0) chyby.cena = 'Cena musí být kladné celé číslo.';
  if (Object.keys(chyby).length > 0) return { chyby, hodnoty: { nazev, cena, popis } };
  return { ok: true, inzerat: { id: 'inzerat-' + Date.now(), nazev, cena: Number(cena), popis } };
}
```
