# --description--

Tři inzeráty se prohlédnou snadno. Třicet už ne.

**Úkol:** nad seznam přidej vyhledávací pole a vypisuj jen ty inzeráty, jejichž
název obsahuje zadaný text — bez ohledu na velikost písmen a bez ohledu na mezery
na okrajích. Filtrovat se musí při psaní.

Pole zabal do odstavce s třídou `hledani`, popisek má text `Hledat v názvech`
a samotný vstup má `id` i `name` `hledani` (podle toho ho hledá kontrola).

Tohle je opakování bez návodu: hledání v textu bez ohledu na velikost písmen
i výběr části pole už znáš z JavaScriptu.

# --hints--

Nad seznamem je vyhledávací pole s popiskem.

```js
const pole = document.querySelector('#hledani');
assert.ok(pole, 'Na stránce má být vyhledávací pole s id="hledani"');
const popisek = document.querySelector('label[for="hledani"]');
assert.ok(popisek, 'K vyhledávacímu poli patří popisek <label htmlFor="hledani">');
```

Po napsání „stan" zůstane v seznamu jediný inzerát.

```js
await helpers.type(document.querySelector('#hledani'), 'stan');
await helpers.waitFor(() => document.querySelectorAll('[data-testid="seznam"] li').length === 1);
const text = helpers.normalize(document.querySelector('[data-testid="seznam"]').textContent);
assert.match(text, /Stan Husky Bizon/, 'V seznamu má zůstat Stan Husky Bizon 3, je tam: ' + text);
```

Filtr si nevšímá velikosti písmen ani mezer na okrajích.

```js
await helpers.type(document.querySelector('#hledani'), '  LYŽE ');
await helpers.waitFor(() => document.querySelectorAll('[data-testid="seznam"] li').length === 1);
assert.match(helpers.normalize(document.querySelector('[data-testid="seznam"]').textContent), /Atomic Redster/, 'Hledání „ LYŽE " má najít Lyže Atomic Redster');
```

Po vymazání hledaného textu jsou zase vidět všechny inzeráty.

```js
await helpers.type(document.querySelector('#hledani'), 'kolo');
await helpers.waitFor(() => document.querySelectorAll('[data-testid="seznam"] li').length === 1);
await helpers.type(document.querySelector('#hledani'), '');
await helpers.waitFor(() => document.querySelectorAll('[data-testid="seznam"] li').length === 3);
assert.equal(document.querySelectorAll('[data-testid="seznam"] li').length, 3, 'S prázdným hledáním mají být vidět všechny inzeráty');
```

# --help--

## --tip-- 2

Hledaný text se mění při psaní, takže patří do stavu komponenty — stejně jako
počet zbývajících znaků v předchozím kroku.

## --tip-- 2

Vykreslovat nebudeš celé pole, ale jen jeho část. Vyber ji těsně před vykreslením;
původní pole ve stavu nech být.

## --tip-- 3

Porovnání bez ohledu na velikost písmen se dělá tak, že se obě strany srovnají
na stejnou velikost. Mezery na okrajích umí uříznout `trim`.

```jsx
import { useActionState, useState } from 'react';
import { useFormStatus } from 'react-dom';

import { vytvorInzerat } from './akce';
import { pocatecniInzeraty } from './data';

const meny = new Intl.NumberFormat('cs-CZ', {
  style: 'currency',
  currency: 'CZK',
  maximumFractionDigits: 0,
});

export function formatujCenu(castka) {
  return meny.format(castka);
}

function TlacitkoOdeslat() {
  const { pending } = useFormStatus();

  return (
    <button className="tlacitko" type="submit" disabled={pending}>
      {pending ? 'Ukládám…' : 'Přidat inzerát'}
    </button>
  );
}

export default function Bazarek() {
  const [inzeraty, setInzeraty] = useState(pocatecniInzeraty);
  const [zbyvaZnaku, setZbyvaZnaku] = useState(200);

  const [stav, akce] = useActionState(async (predchozi, formData) => {
    const vysledek = await vytvorInzerat(predchozi, formData);

    if (vysledek.ok) {
      setInzeraty((soucasne) => [vysledek.inzerat, ...soucasne]);
    }

    return vysledek;
  }, {});

  return (
    <main className="bazarek">
      <header className="hlavicka">
        <h1>Bazárek</h1>
        <p className="podtitulek">Věci, které někomu doma překážejí.</p>
      </header>

      <form className="formular" action={akce}>
        <h2>Přidat inzerát</h2>

        <p className="pole">
          <label htmlFor="nazev">Název</label>
          <input id="nazev" name="nazev" defaultValue={stav.hodnoty?.nazev ?? ''} aria-invalid={Boolean(stav.chyby?.nazev)} />
          {stav.chyby?.nazev && <span className="chyba" role="alert">{stav.chyby.nazev}</span>}
        </p>

        <p className="pole">
          <label htmlFor="cena">Cena v Kč</label>
          <input id="cena" name="cena" type="number" defaultValue={stav.hodnoty?.cena ?? ''} aria-invalid={Boolean(stav.chyby?.cena)} />
          {stav.chyby?.cena && <span className="chyba" role="alert">{stav.chyby.cena}</span>}
        </p>

        <p className="pole">
          <label htmlFor="popis">Popis</label>
          <textarea
            id="popis"
            name="popis"
            defaultValue={stav.hodnoty?.popis ?? ''}
            onChange={(udalost) => setZbyvaZnaku(200 - udalost.target.value.length)}
          />
          <span className="napoveda">Zbývá {zbyvaZnaku} znaků.</span>
        </p>

        <TlacitkoOdeslat />

        <p className="hlaseni">{stav.ok ? 'Inzerát je na bazárku.' : ''}</p>
      </form>

      <section className="nabidka">
        <h2>Na prodej</h2>
        <ul className="seznam" data-testid="seznam">
          {inzeraty.map((inzerat) => (
            <li className="karta" key={inzerat.id}>
              <h3>{inzerat.nazev}</h3>
              <p className="karta__cena">{formatujCenu(inzerat.cena)}</p>
              <p className="karta__popis">{inzerat.popis}</p>
            </li>
          ))}
        </ul>
      </section>
    </main>
  );
}
```

```js
// Tahle funkce zastupuje serverovou akci. V Next.js by měla na prvním řádku
// 'use server', běžela by na serveru a zapisovala do databáze. Tady jen chvíli
// počká a vrátí stejný tvar výsledku, jaký by vrátila doopravdy.
const pauza = (ms) => new Promise((splnit) => setTimeout(splnit, ms));
export async function vytvorInzerat(predchoziStav, formData) {
  const nazev = String(formData.get('nazev') ?? '').trim();
  const cena = String(formData.get('cena') ?? '').trim();
  const popis = String(formData.get('popis') ?? '').trim();
  await pauza(400);
  const chyby = {};
  if (nazev.length < 3) chyby.nazev = 'Název musí mít aspoň 3 znaky.';
  if (!Number.isInteger(Number(cena)) || Number(cena) <= 0) chyby.cena = 'Cena musí být kladné celé číslo.';
  if (Object.keys(chyby).length > 0) return { chyby, hodnoty: { nazev, cena, popis } };
  return { ok: true, inzerat: { id: 'inzerat-' + Date.now(), nazev, cena: Number(cena), popis } };
}
```
