# --description--

Když akce vrátí chyby, uživatel o nich zatím neví — formulář se tváří, že se nic
nestalo. Chyba přitom patří k políčku, kterého se týká.

**Úkol:** u políček `nazev` a `cena`:

1. přidej `aria-invalid={Boolean(stav.chyby?.nazev)}` (u ceny obdobně), aby
   políčko vypadalo i pro čtečku obrazovky jako chybné,
2. hned za políčko dej hlášku
   `<span className="chyba" role="alert">{stav.chyby.nazev}</span>`,
   ale **jen tehdy**, když ta chyba existuje.

Podmíněné vykreslení se v JSX píše přes `&&`: `{maChybu && <span>…</span>}`.
Role `alert` říká čtečce „tohle přečti hned" — proto se hodí na chyby, ne na
běžný text.

# --hints--

Po odeslání krátkého názvu se u políčka objeví hláška o třech znacích.

```js
await helpers.type(document.querySelector('[name="nazev"]'), 'ab');
await helpers.type(document.querySelector('[name="cena"]'), '2200');
await helpers.submit(document.querySelector('form'));
const hlaska = await helpers.waitFor(() => document.querySelector('[role="alert"]'));
assert.match(helpers.normalize(hlaska.textContent), /3 znaky/, 'U názvu má být vidět hláška o třech znacích');
```

Chybné políčko má `aria-invalid="true"`.

```js
await helpers.type(document.querySelector('[name="nazev"]'), 'ab');
await helpers.type(document.querySelector('[name="cena"]'), '2200');
await helpers.submit(document.querySelector('form'));
await helpers.waitFor(() => document.querySelector('[role="alert"]'));
assert.equal(document.querySelector('[name="nazev"]').getAttribute('aria-invalid'), 'true', 'Políčko s chybou má mít aria-invalid="true"');
assert.notEqual(document.querySelector('[name="cena"]').getAttribute('aria-invalid'), 'true', 'Políčko cena je vyplněné správně, nemá být označené jako chybné');
```

U ceny se chyba ukáže taky — a po opravě zmizí.

```js
await helpers.type(document.querySelector('[name="nazev"]'), 'Lyže Atomic Redster');
await helpers.type(document.querySelector('[name="cena"]'), '0');
await helpers.submit(document.querySelector('form'));
await helpers.waitFor(() => document.querySelector('[name="cena"]').getAttribute('aria-invalid') === 'true');
const hlasky = [...document.querySelectorAll('[role="alert"]')].map((prvek) => helpers.normalize(prvek.textContent));
assert.ok(hlasky.some((text) => /celé číslo/.test(text)), 'U nulové ceny má být hláška o kladném celém čísle, hlášky: ' + hlasky.join(' | '));

await helpers.type(document.querySelector('[name="nazev"]'), 'Lyže Atomic Redster');
await helpers.type(document.querySelector('[name="cena"]'), '3900');
await helpers.submit(document.querySelector('form'));
await helpers.waitFor(() => document.querySelectorAll('[role="alert"]').length === 0);
assert.equal(document.querySelectorAll('[role="alert"]').length, 0, 'Po opravě už žádná chybová hláška zůstat nemá');
```

# --help--

## --tip-- 1

Výsledek akce máš ve `stav`. Když akce skončila chybou, je v něm objekt
`chyby` s klíčem podle políčka — přesně tak, jak ho vrací `vytvorInzerat`
v `akce.js`.

## --tip-- 1

Před přístupem ke klíči se hodí volitelné zřetězení (`?.`), protože při prvním
vykreslení je `stav` prázdný objekt a žádné `chyby` v něm nejsou.

## --tip-- 2

Atribut `aria-invalid` čeká text `"true"` nebo `"false"`. React za tebe převede
i boolean — stačí mu dát `Boolean(…)`, ne `undefined`.

```jsx
import { useActionState } from 'react';

import { vytvorInzerat } from './akce';
import { pocatecniInzeraty } from './data';

const meny = new Intl.NumberFormat('cs-CZ', {
  style: 'currency',
  currency: 'CZK',
  maximumFractionDigits: 0,
});

export function formatujCenu(castka) {
  return meny.format(castka);
}

export default function Bazarek() {
  const inzeraty = pocatecniInzeraty;
  const [stav, akce] = useActionState(vytvorInzerat, {});

  return (
    <main className="bazarek">
      <header className="hlavicka">
        <h1>Bazárek</h1>
        <p className="podtitulek">Věci, které někomu doma překážejí.</p>
      </header>

      <form className="formular" action={akce}>
        <h2>Přidat inzerát</h2>

        <p className="pole">
          <label htmlFor="nazev">Název</label>
          <input id="nazev" name="nazev" />
        </p>

        <p className="pole">
          <label htmlFor="cena">Cena v Kč</label>
          <input id="cena" name="cena" type="number" />
        </p>

        <p className="pole">
          <label htmlFor="popis">Popis</label>
          <textarea id="popis" name="popis" />
        </p>

        <button className="tlacitko" type="submit">Přidat inzerát</button>

        <p className="hlaseni">{stav.ok ? 'Inzerát je na bazárku.' : ''}</p>
      </form>

      <section className="nabidka">
        <h2>Na prodej</h2>
        <ul className="seznam" data-testid="seznam">
          {inzeraty.map((inzerat) => (
            <li className="karta" key={inzerat.id}>
              <h3>{inzerat.nazev}</h3>
              <p className="karta__cena">{formatujCenu(inzerat.cena)}</p>
              <p className="karta__popis">{inzerat.popis}</p>
            </li>
          ))}
        </ul>
      </section>
    </main>
  );
}
```

## --file-- akce.js

```js
// Tahle funkce zastupuje serverovou akci. V Next.js by měla na prvním řádku
// 'use server', běžela by na serveru a zapisovala do databáze. Tady jen chvíli
// počká a vrátí stejný tvar výsledku, jaký by vrátila doopravdy.
const pauza = (ms) => new Promise((splnit) => setTimeout(splnit, ms));
export async function vytvorInzerat(predchoziStav, formData) {
  const nazev = String(formData.get('nazev') ?? '').trim();
  const cena = String(formData.get('cena') ?? '').trim();
  const popis = String(formData.get('popis') ?? '').trim();
  await pauza(400);
  const chyby = {};
  if (nazev.length < 3) chyby.nazev = 'Název musí mít aspoň 3 znaky.';
  if (!Number.isInteger(Number(cena)) || Number(cena) <= 0) chyby.cena = 'Cena musí být kladné celé číslo.';
  if (Object.keys(chyby).length > 0) return { chyby, hodnoty: { nazev, cena, popis } };
  return { ok: true, inzerat: { id: 'inzerat-' + Date.now(), nazev, cena: Number(cena), popis } };
}
```
