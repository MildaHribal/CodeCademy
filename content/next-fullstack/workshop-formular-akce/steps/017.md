# --description--

Nová položka se objeví okamžitě a vypadá úplně stejně jako uložené inzeráty.
To je nepoctivé — uživatel netuší, že se ještě čeká na server, a při chybě mu
položka bez vysvětlení zmizí.

**Úkol:** u položky, která má příznak `ceka`:

- přidej k třídě `karta` ještě `karta--ceka` (čárkovaný rámeček už je v CSS
  připravený),
- dej jí `aria-busy="true"`, ať to ví i čtečka obrazovky,
- přidej dovnitř odstavec `<p className="karta__stav">` s textem `Ukládám…`.

Ostatní položky musí zůstat beze změny — žádné `aria-busy` a žádný odstavec navíc.

# --hints--

Během odesílání má nová položka `aria-busy="true"` a čárkovaný rámeček.

```js
await helpers.type(document.querySelector('[name="nazev"]'), 'Snowboard Burton 152');
await helpers.type(document.querySelector('[name="cena"]'), '4300');
helpers.submit(document.querySelector('form'));
const ceka = await helpers.waitFor(() => document.querySelector('[data-testid="seznam"] li[aria-busy="true"]'));
assert.match(helpers.normalize(ceka.textContent), /Snowboard Burton 152/, 'Označená má být ta nová položka');
assert.ok(ceka.classList.contains('karta--ceka'), 'Čekající položka má mít třídu karta--ceka, má: ' + ceka.className);
assert.ok(ceka.classList.contains('karta'), 'Čekající položka má mít i třídu karta');
assert.equal(getComputedStyle(ceka).borderStyle, 'dashed', 'Třída karta--ceka má dát položce čárkovaný rámeček');
```

V čekající položce je vidět text „Ukládám…".

```js
await helpers.type(document.querySelector('[name="nazev"]'), 'Snowboard Burton 152');
await helpers.type(document.querySelector('[name="cena"]'), '4300');
helpers.submit(document.querySelector('form'));
const ceka = await helpers.waitFor(() => document.querySelector('[data-testid="seznam"] li[aria-busy="true"]'));
const stav = ceka.querySelector('.karta__stav');
assert.ok(stav, 'V čekající položce má být odstavec <p className="karta__stav">');
assert.equal(helpers.normalize(stav.textContent), 'Ukládám…', 'V odstavci má být text „Ukládám…"');
```

Po uložení označení zmizí a uložené položky ho nikdy neměly.

```js
assert.equal(document.querySelectorAll('[data-testid="seznam"] li[aria-busy="true"]').length, 0, 'Uložené inzeráty nemají být označené jako čekající');
await helpers.type(document.querySelector('[name="nazev"]'), 'Snowboard Burton 152');
await helpers.type(document.querySelector('[name="cena"]'), '4300');
await helpers.submit(document.querySelector('form'));
await helpers.waitFor(() => /je na bazárku/.test(document.body.textContent));
await helpers.flush();
assert.equal(document.querySelectorAll('[data-testid="seznam"] li[aria-busy="true"]').length, 0, 'Po uložení už položka čekat nemá');
assert.equal(document.querySelectorAll('.karta__stav').length, 0, 'Po uložení má odstavec „Ukládám…" zmizet');
```

# --help--

## --tip-- 1

Příznak `ceka` má jen ta položka, kterou jsi přidal do dočasného seznamu.
U ostatních je `undefined`, tedy nepravdivý.

## --tip-- 1

Atribut `className` je obyčejný řetězec, takže ho jde poskládat ternárním
operátorem. Atribut, který u ostatních položek být nemá, nastav na `undefined` —
React ho pak do HTML vůbec nedá.

## --tip-- 2

Odstavec navíc se vykreslí podmíněně, stejně jako chybová hláška u políčka.

import { useActionState, useOptimistic, useState } from 'react';
import { useFormStatus } from 'react-dom';

import { vytvorInzerat } from './akce';
import { pocatecniInzeraty } from './data';

const meny = new Intl.NumberFormat('cs-CZ', {
  style: 'currency',
  currency: 'CZK',
  maximumFractionDigits: 0,
});

export function formatujCenu(castka) {
  return meny.format(castka);
}

function TlacitkoOdeslat() {
  const { pending } = useFormStatus();

  return (
    <button className="tlacitko" type="submit" disabled={pending}>
      {pending ? 'Ukládám…' : 'Přidat inzerát'}
    </button>
  );
}

export default function Bazarek() {
  const [inzeraty, setInzeraty] = useState(pocatecniInzeraty);
  const [zbyvaZnaku, setZbyvaZnaku] = useState(200);
  const [hledani, setHledani] = useState('');

  const hledanyText = hledani.trim().toLowerCase();
  const nalezene = inzeraty.filter((inzerat) => inzerat.nazev.toLowerCase().includes(hledanyText));

  const [seznam, pridejOptimisticky] = useOptimistic(nalezene, (soucasne, novy) => [novy, ...soucasne]);

  const [stav, akce] = useActionState(async (predchozi, formData) => {
    pridejOptimisticky({
      id: 'ceka',
      nazev: String(formData.get('nazev') ?? ''),
      cena: Number(formData.get('cena')),
      popis: String(formData.get('popis') ?? ''),
      ceka: true,
    });

    const vysledek = await vytvorInzerat(predchozi, formData);

    if (vysledek.ok) {
      setInzeraty((soucasne) => [vysledek.inzerat, ...soucasne]);
    }

    return vysledek;
  }, {});

  return (
    <main className="bazarek">
      <header className="hlavicka">
        <h1>Bazárek</h1>
        <p className="podtitulek">Věci, které někomu doma překážejí.</p>
      </header>

      <form className="formular" action={akce}>
        <h2>Přidat inzerát</h2>

        <p className="pole">
          <label htmlFor="nazev">Název</label>
          <input id="nazev" name="nazev" defaultValue={stav.hodnoty?.nazev ?? ''} aria-invalid={Boolean(stav.chyby?.nazev)} />
          {stav.chyby?.nazev && <span className="chyba" role="alert">{stav.chyby.nazev}</span>}
        </p>

        <p className="pole">
          <label htmlFor="cena">Cena v Kč</label>
          <input id="cena" name="cena" type="number" defaultValue={stav.hodnoty?.cena ?? ''} aria-invalid={Boolean(stav.chyby?.cena)} />
          {stav.chyby?.cena && <span className="chyba" role="alert">{stav.chyby.cena}</span>}
        </p>

        <p className="pole">
          <label htmlFor="popis">Popis</label>
          <textarea
            id="popis"
            name="popis"
            defaultValue={stav.hodnoty?.popis ?? ''}
            onChange={(udalost) => setZbyvaZnaku(200 - udalost.target.value.length)}
          />
          <span className="napoveda">Zbývá {zbyvaZnaku} znaků.</span>
        </p>

        <TlacitkoOdeslat />

        <p className="hlaseni" role="status">
          {stav.ok ? 'Inzerát „' + stav.inzerat.nazev + '" je na bazárku.' : ''}
        </p>
      </form>

      <section className="nabidka">
        <h2>Na prodej</h2>

        <p className="hledani">
          <label htmlFor="hledani">Hledat v názvech</label>
          <input
            id="hledani"
            name="hledani"
            value={hledani}
            onChange={(udalost) => setHledani(udalost.target.value)}
          />
        </p>
        <ul className="seznam" data-testid="seznam">
          {seznam.map((inzerat) => (
            <li className="karta" key={inzerat.id}>
              <h3>{inzerat.nazev}</h3>
              <p className="karta__cena">{formatujCenu(inzerat.cena)}</p>
              <p className="karta__popis">{inzerat.popis}</p>
            </li>
          ))}
        </ul>
      </section>
    </main>
  );
}

// Tahle funkce zastupuje serverovou akci. V Next.js by měla na prvním řádku
// 'use server', běžela by na serveru a zapisovala do databáze. Tady jen chvíli
// počká a vrátí stejný tvar výsledku, jaký by vrátila doopravdy.
const pauza = (ms) => new Promise((splnit) => setTimeout(splnit, ms));
export async function vytvorInzerat(predchoziStav, formData) {
  const nazev = String(formData.get('nazev') ?? '').trim();
  const cena = String(formData.get('cena') ?? '').trim();
  const popis = String(formData.get('popis') ?? '').trim();
  await pauza(400);
  const chyby = {};
  if (nazev.length < 3) chyby.nazev = 'Název musí mít aspoň 3 znaky.';
  if (!Number.isInteger(Number(cena)) || Number(cena) <= 0) chyby.cena = 'Cena musí být kladné celé číslo.';
  if (Object.keys(chyby).length > 0) return { chyby, hodnoty: { nazev, cena, popis } };
  return { ok: true, inzerat: { id: 'inzerat-' + Date.now(), nazev, cena: Number(cena), popis } };
}
