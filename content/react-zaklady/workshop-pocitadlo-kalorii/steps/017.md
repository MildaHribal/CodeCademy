# --description--

Tlačítko **Vynulovat den** jde zmáčknout i ráno, kdy není co nulovat. Uživatel klikne a nic se nestane — a on neví, jestli aplikace funguje.

**Úkol:** Zařiď, aby bylo tlačítko neaktivní, dokud je zápis prázdný. Prohlížeč takové tlačítko sám zešedne a nepustí na něj ani kliknutí, ani klávesnici.

Jakmile je v zápisu aspoň jeden záznam, tlačítko zase funguje.

# --hints--

Na prázdný den je tlačítko neaktivní.

```js
const vynulovat = [...document.querySelectorAll('button')].find((el) => /Vynulovat/.test(el.textContent));
assert.ok(vynulovat, 'tlačítko „Vynulovat den" má zůstat');
assert.equal(vynulovat.disabled, true, 'na prázdný den má být tlačítko neaktivní');
```

Po zapsání jídla je tlačítko aktivní a funguje.

```js
const najdi = (text) => [...document.querySelectorAll('.volba')].find((el) => el.textContent.includes(text));
await helpers.click(najdi('Jablko'));
await helpers.flush();
const vynulovat = [...document.querySelectorAll('button')].find((el) => /Vynulovat/.test(el.textContent));
assert.equal(vynulovat.disabled, false, 'se zapsaným jídlem má být tlačítko aktivní');
await helpers.click(vynulovat);
await helpers.flush();
assert.equal(document.querySelectorAll('.zaznam').length, 0, 'tlačítko má dál vynulovat den');
assert.equal([...document.querySelectorAll('button')].find((el) => /Vynulovat/.test(el.textContent)).disabled, true, 'po vynulování je zase neaktivní');
```

O aktivitě rozhoduje stav, ne pevná hodnota.

```js
const kod = helpers.stripComments(files['App.jsx'], 'js');
assert.match(kod, /disabled=\{/, 'vlastnost disabled nastav výrazem podle zápisu');
assert.doesNotMatch(kod, /disabled=\{true\}|disabled=\{false\}|disabled="/, 'pevná hodnota by tlačítko vypnula nebo zapnula napořád');
```

## --file-- utils.js

```js
// Kolik kalorií mají všechny zapsané porce dohromady.
export function soucetKalorii(zaznamy) {
  return zaznamy.reduce((celkem, zaznam) => celkem + zaznam.kcal * zaznam.porce, 0);
}
```

## --file-- App.jsx

```jsx
import './styles.css';
import { useState } from 'react';
import { DENNI_CIL, potraviny } from './data.js';
import { soucetKalorii } from './utils.js';
import Souhrn from './Souhrn.jsx';

const TYPY = ['Vše', 'Snídaně', 'Oběd', 'Večeře', 'Svačina'];

export default function App() {
  const [zaznamy, setZaznamy] = useState([]);
  const [filtr, setFiltr] = useState('Vše');

  const snedeno = soucetKalorii(zaznamy);
  const zbyva = DENNI_CIL - snedeno;
  const procenta = Math.min(100, Math.round((snedeno / DENNI_CIL) * 100));
  const prekroceno = snedeno > DENNI_CIL;
  const videt = filtr === 'Vše' ? zaznamy : zaznamy.filter((zaznam) => zaznam.typ === filtr);

  function pridej(potravina) {
    setZaznamy([...zaznamy, { ...potravina, id: crypto.randomUUID(), porce: 1 }]);
  }

  function zmenPorci(id, zmena) {
    setZaznamy(
      zaznamy.map((zaznam) =>
        zaznam.id === id ? { ...zaznam, porce: Math.max(1, zaznam.porce + zmena) } : zaznam,
      ),
    );
  }

  function smaz(id) {
    setZaznamy(zaznamy.filter((zaznam) => zaznam.id !== id));
  }

  function vynulujDen() {
    setZaznamy([]);
    setFiltr('Vše');
  }

  return (
    <div className="stranka">
      <Souhrn snedeno={snedeno} zbyva={zbyva} procenta={procenta} prekroceno={prekroceno} />

      <section className="sekce">
        <h2 className="sekce__nadpis">Co jsem snědl</h2>
        <div className="volby">
          {potraviny.map((potravina) => (
            <button className="volba" key={potravina.id} onClick={() => pridej(potravina)}>
              {potravina.nazev} <span className="volba__kcal">{potravina.kcal} kcal</span>
            </button>
          ))}
        </div>
      </section>

      <section className="sekce">
        <div className="sekce__hlavicka">
          <h2 className="sekce__nadpis">Dnešní zápis</h2>
          <div className="filtr">
            {TYPY.map((typ) => (
              <button
                className={typ === filtr ? 'filtr__tlacitko filtr__tlacitko--aktivni' : 'filtr__tlacitko'}
                key={typ}
                onClick={() => setFiltr(typ)}
              >
                {typ}
              </button>
            ))}
          </div>
        </div>
        {videt.length === 0 && (
          <p className="prazdno">{zaznamy.length === 0 ? 'Zatím nic. Klikni na jídlo nahoře.' : 'V téhle kategorii dnes nic nemáš.'}</p>
        )}
        <ul className="zaznamy">
          {videt.map((zaznam) => (
            <li className="zaznam" key={zaznam.id}>
              <p className="zaznam__nazev">
                {zaznam.nazev} <span className="zaznam__typ">{zaznam.typ}</span>
              </p>
              <span className="zaznam__porce">
                <button className="kulate" onClick={() => zmenPorci(zaznam.id, -1)} aria-label="Ubrat porci">−</button>
                {zaznam.porce}×
                <button className="kulate" onClick={() => zmenPorci(zaznam.id, 1)} aria-label="Přidat porci">+</button>
              </span>
              <span className="zaznam__kcal">{zaznam.kcal * zaznam.porce} kcal</span>
              <button className="smazat" onClick={() => smaz(zaznam.id)} aria-label="Smazat záznam">×</button>
            </li>
          ))}
        </ul>
        <button className="tise" onClick={vynulujDen}>
          Vynulovat den
        </button>
      </section>
    </div>
  );
}
```

```jsx
export default function Souhrn({ snedeno, zbyva, procenta, prekroceno }) {
  return (
    <section className="souhrn">
      <h1 className="souhrn__nadpis">Dnes máš snědeno</h1>
      <p className={prekroceno ? 'souhrn__cislo souhrn__cislo--prekroceno' : 'souhrn__cislo'}>{snedeno} kcal</p>
      <p className="souhrn__zbyva">
        {prekroceno ? `Cíl máš překročený o ${-zbyva} kcal.` : `Do cíle zbývá ${zbyva} kcal.`}
      </p>
      <div className="ukazatel" aria-hidden="true">
        <div className={prekroceno ? 'ukazatel__cara ukazatel__cara--prekroceno' : 'ukazatel__cara'} style={{ inlineSize: `${procenta}%` }} />
      </div>
    </section>
  );
}
```
