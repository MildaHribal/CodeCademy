# --description--

Číslo je jedna věc, ale na mobilu si člověk chce jen rychle zkontrolovat, jak daleko je. Ve stylech je připravený pruh: obal `.ukazatel` a čára `.ukazatel__cara`.

**Úkol:** Spočítej, na kolik procent cíle je dnešní součet (zaokrouhleně na celé, nejvýš sto), a pod odstavec se zbývajícími kaloriemi přidej pruh s čárou, která má tuhle šířku. Obalu dej `aria-hidden="true"` — je to jen obrázek čísla, které je hned nad ním.

Na jiných datech vypadá nastavení šířky takhle:

```jsx
<div className="teplomer" style={{ inlineSize: \`${podil}%\` }} />
```

# --hints--

Na prázdný den je čára nulová.

```js
const cara = document.querySelector('.souhrn .ukazatel > .ukazatel__cara');
assert.ok(cara, 'v hlavičce má být .ukazatel a v něm .ukazatel__cara');
assert.ok(cara.getBoundingClientRect().width < 2, 'na prázdný den má být čára nulová');
```

Po zapsání guláše je čára na 35 % šířky.

```js
const najdi = (text) => [...document.querySelectorAll('.volba')].find((el) => el.textContent.includes(text));
await helpers.click(najdi('Guláš'));
await helpers.flush();
const cara = document.querySelector('.ukazatel__cara');
const pomer = cara.getBoundingClientRect().width / cara.parentElement.getBoundingClientRect().width;
assert.ok(Math.abs(pomer - 0.35) < 0.03, `780 z 2200 kcal je 35 % — čára je na ${Math.round(pomer * 100)} %`);
```

Přes sto procent čára nepřeteče.

```js
const najdi = (text) => [...document.querySelectorAll('.volba')].find((el) => el.textContent.includes(text));
for (let i = 0; i < 4; i += 1) await helpers.click(najdi('Guláš'));
await helpers.flush();
const cara = document.querySelector('.ukazatel__cara');
const pomer = cara.getBoundingClientRect().width / cara.parentElement.getBoundingClientRect().width;
assert.ok(pomer <= 1.01, 'při překročení cíle má čára zůstat nejvýš na sto procentech');
assert.ok(pomer > 0.98, 'při překročení cíle má být čára plná');
```

# --help--

## --tip--

Šířka z proměnné se nastavuje objektem ve vlastnosti `style` — viz [Atributy se jmenují jako v JavaScriptu](see:react-zaklady/proc-react#atributy-se-jmenuji-jako-v-javascriptu).

## --tip--

Procenta si spočítej do proměnné vedle ostatních výpočtů. Horní hranici uhlídá `Math.min`, zaokrouhlení `Math.round`.

## --tip--

Na jiných datech: `Math.min(100, Math.round((hotovo / celkem) * 100))` u ukazatele stahování.

## --file-- utils.js

```js
// Kolik kalorií mají všechny zapsané porce dohromady.
export function soucetKalorii(zaznamy) {
  return zaznamy.reduce((celkem, zaznam) => celkem + zaznam.kcal * zaznam.porce, 0);
}
```

## --file-- App.jsx

```jsx
import './styles.css';
import { useState } from 'react';
import { DENNI_CIL, potraviny } from './data.js';
import { soucetKalorii } from './utils.js';

export default function App() {
  const [zaznamy, setZaznamy] = useState([]);

  const snedeno = soucetKalorii(zaznamy);
  const zbyva = DENNI_CIL - snedeno;

  function pridej(potravina) {
    setZaznamy([...zaznamy, { ...potravina, id: crypto.randomUUID(), porce: 1 }]);
  }

  return (
    <div className="stranka">
      <section className="souhrn">
        <h1 className="souhrn__nadpis">Dnes máš snědeno</h1>
        <p className="souhrn__cislo">{snedeno} kcal</p>
        <p className="souhrn__zbyva">{`Do cíle ${DENNI_CIL} kcal zbývá ${zbyva} kcal.`}</p>
      </section>

      <section className="sekce">
        <h2 className="sekce__nadpis">Co jsem snědl</h2>
        <div className="volby">
          {potraviny.map((potravina) => (
            <button className="volba" key={potravina.id} onClick={() => pridej(potravina)}>
              {potravina.nazev} <span className="volba__kcal">{potravina.kcal} kcal</span>
            </button>
          ))}
        </div>
      </section>

      <section className="sekce">
        <div className="sekce__hlavicka">
          <h2 className="sekce__nadpis">Dnešní zápis</h2>
        </div>
        {zaznamy.length === 0 && <p className="prazdno">Zatím nic. Klikni na jídlo nahoře.</p>}
        <ul className="zaznamy">
          {zaznamy.map((zaznam) => (
            <li className="zaznam" key={zaznam.id}>
              <p className="zaznam__nazev">
                {zaznam.nazev} <span className="zaznam__typ">{zaznam.typ}</span>
              </p>
              <span className="zaznam__porce">{zaznam.porce}×</span>
              <span className="zaznam__kcal">{zaznam.kcal * zaznam.porce} kcal</span>
            </li>
          ))}
        </ul>
      </section>
    </div>
  );
}
```

```jsx
import './styles.css';
import { useState } from 'react';
import { DENNI_CIL, potraviny } from './data.js';
import { soucetKalorii } from './utils.js';

export default function App() {
  const [zaznamy, setZaznamy] = useState([]);

  const snedeno = soucetKalorii(zaznamy);
  const zbyva = DENNI_CIL - snedeno;
  const procenta = Math.min(100, Math.round((snedeno / DENNI_CIL) * 100));

  function pridej(potravina) {
    setZaznamy([...zaznamy, { ...potravina, id: crypto.randomUUID(), porce: 1 }]);
  }

  return (
    <div className="stranka">
      <section className="souhrn">
        <h1 className="souhrn__nadpis">Dnes máš snědeno</h1>

        <p className="souhrn__zbyva">{`Do cíle ${DENNI_CIL} kcal zbývá ${zbyva} kcal.`}</p>
        <div className="ukazatel" aria-hidden="true">
          <div className="ukazatel__cara" style={{ inlineSize: `${procenta}%` }} />
        </div>
      </section>

      <section className="sekce">
        <h2 className="sekce__nadpis">Co jsem snědl</h2>
        <div className="volby">
          {potraviny.map((potravina) => (

              {potravina.nazev} <span className="volba__kcal">{potravina.kcal} kcal</span>
            </button>
          ))}
        </div>
      </section>

}
```
