# --description--

Aby šlo něco zapsat, potřebuješ nabídku. V `data.js` je pole `potraviny` — osm jídel s názvem, typem a kaloriemi za porci.

**Úkol:** Mezi hlavičku a zápis přidej sekci `sekce` s nadpisem druhé úrovně **Co jsem snědl** a pod ním `<div className="volby">`, ve kterém z pole vykreslíš tlačítko pro každé jídlo: třída `volba`, uvnitř název a za ním `<span className="volba__kcal">` s počtem kalorií a slovem **kcal**.

Tlačítka zatím nic nedělají — kliknutí přidáš v dalším kroku.

Na jiných datech vypadá takový výběr takhle:

```jsx
<div className="volby">
  {barvy.map((barva) => (
    <button className="volba" key={barva.id}>{barva.nazev}</button>
  ))}
</div>
```

# --hints--

V nabídce je tlačítko pro každé jídlo z dat.

```js
const { potraviny } = await helpers.importFile('data.js');
const tlacitka = [...document.querySelectorAll('.volby .volba')];
assert.equal(tlacitka.length, potraviny.length, 'tlačítek má být tolik, kolik je jídel v datech');
potraviny.forEach((potravina, index) => {
  assert.ok(tlacitka[index].textContent.includes(potravina.nazev), `${index + 1}. tlačítko má být ${potravina.nazev}`);
});
```

U každého jídla jsou vidět kalorie za porci.

```js
const { potraviny } = await helpers.importFile('data.js');
const tlacitka = [...document.querySelectorAll('.volby .volba')];
potraviny.forEach((potravina, index) => {
  const kcal = tlacitka[index].querySelector('.volba__kcal');
  assert.ok(kcal, `u jídla ${potravina.nazev} má být <span className="volba__kcal">`);
  assert.match(kcal.textContent, new RegExp(String(potravina.kcal)), `u jídla ${potravina.nazev} má být ${potravina.kcal} kcal`);
});
assert.ok(!logs.some((zaznam) => /unique "key"/.test(zaznam.text)), 'tlačítka vykreslená z pole potřebují key');
```

Nabídka je mezi hlavičkou a zápisem a jsou to opravdu tlačítka.

```js
const sekce = [...document.querySelectorAll('.stranka > *')];
const indexNabidky = sekce.findIndex((el) => el.querySelector('.volby'));
const indexZapisu = sekce.findIndex((el) => el.querySelector('.prazdno, .zaznamy'));
assert.ok(indexNabidky > 0, 'nabídka patří pod hlavičku');
assert.ok(indexNabidky < indexZapisu, 'nabídka patří nad zápis');
assert.equal(document.querySelector('.volba').tagName, 'BUTTON', 'volba má být <button>, aby šla ovládat i klávesnicí');
```

# --help--

## --tip--

Vykreslení pole a `key` je v části [Seznam z pole a `key`](see:react-zaklady/komponenty-a-props#seznam-z-pole-a-key).

## --tip--

Novou sekci vlož mezi hlavičku a sekci se zápisem. Uvnitř `.volby` proměň pole jídel v pole tlačítek.

## --tip--

Text tlačítka je název a za ním `span` s kalorií — dva výrazy vedle sebe, mezi nimi obyčejná mezera.

## --file-- utils.js

```js
// Kolik kalorií mají všechny zapsané porce dohromady.
export function soucetKalorii(zaznamy) {
  return zaznamy.reduce((celkem, zaznam) => celkem + zaznam.kcal * zaznam.porce, 0);
}
```

## --file-- App.jsx

```jsx
import './styles.css';
import { useState } from 'react';
```
```jsx
import { DENNI_CIL, potraviny } from './data.js';
import { soucetKalorii } from './utils.js';

export default function App() {
  const [zaznamy, setZaznamy] = useState([]);

  const snedeno = soucetKalorii(zaznamy);

  return (
    <div className="stranka">
      <section className="souhrn">
        <h1 className="souhrn__nadpis">Dnes máš snědeno</h1>

      <section className="sekce">
        <h2 className="sekce__nadpis">Co jsem snědl</h2>

            </button>
          ))}
        </div>
      </section>

      <section className="sekce">
        <div className="sekce__hlavicka">
          <h2 className="sekce__nadpis">Dnešní zápis</h2>
        </div>
        {zaznamy.length === 0 && <p className="prazdno">Zatím nic. Klikni na jídlo nahoře.</p>}
      </section>
    </div>
  );
}
```
