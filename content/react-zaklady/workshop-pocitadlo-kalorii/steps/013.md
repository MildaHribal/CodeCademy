# --description--

Zkus si to: zapiš jen snídani a klikni na **Oběd**. Zápis je prázdný a stránka mlčí — uživatel neví, jestli se něco rozbilo, nebo jestli k obědu opravdu nic nemá.

**Úkol:** Uprav hlášku o prázdném zápisu tak, aby rozlišila dva případy. Když není zapsané vůbec nic, zůstane text **Zatím nic. Klikni na jídlo nahoře.** Když něco zapsané je, ale zvolený filtr nic nenašel, ukáže se místo toho **V téhle kategorii dnes nic nemáš.**

Hláška se pořád ukazuje jen tehdy, když v zápisu není vidět žádná položka.

# --hints--

Prázdný den má dál svou hlášku.

```js
const prazdno = document.querySelector('.prazdno');
assert.ok(prazdno, 'u prázdného dne má být hláška');
assert.match(prazdno.textContent, /Zatím nic/, 'u prázdného dne má hláška zůstat „Zatím nic. Klikni na jídlo nahoře."');
```

Filtr bez výsledku má vlastní hlášku.

```js
const najdi = (text) => [...document.querySelectorAll('.volba')].find((el) => el.textContent.includes(text));
const filtr = (text) => [...document.querySelectorAll('.filtr__tlacitko')].find((el) => el.textContent.trim() === text);
await helpers.click(najdi('Ovesná kaše'));
await helpers.flush();
await helpers.click(filtr('Oběd'));
await helpers.flush();
const prazdno = document.querySelector('.prazdno');
assert.ok(prazdno, 'když filtr nic nenajde, má se ukázat hláška');
assert.match(prazdno.textContent, /kategorii/, 'hláška má říct, že v téhle kategorii dnes nic není');
assert.doesNotMatch(prazdno.textContent, /Zatím nic/, 'to není tentýž případ jako prázdný den');
```

Když je co ukázat, žádná hláška se nezobrazuje.

```js
const najdi = (text) => [...document.querySelectorAll('.volba')].find((el) => el.textContent.includes(text));
const filtr = (text) => [...document.querySelectorAll('.filtr__tlacitko')].find((el) => el.textContent.trim() === text);
await helpers.click(najdi('Guláš'));
await helpers.flush();
assert.equal(document.querySelector('.prazdno'), null, 'při neprázdném zápisu se hláška ukazovat nemá');
await helpers.click(filtr('Oběd'));
await helpers.flush();
assert.equal(document.querySelector('.prazdno'), null, 'filtr s výsledkem taky žádnou hlášku nepotřebuje');
assert.equal(document.querySelectorAll('.zaznam').length, 1, 'guláš má v obědech zůstat');
```

## --tip--

Rozhodnutí mezi dvěma texty je ternární operátor — viz [Podmíněné vykreslení](see:react-zaklady/komponenty-a-props#podminene-vykresleni).

## --tip--

Podmínka „ukázat hlášku" zůstává stejná (nic není vidět). Uvnitř hlášky pak rozhodni podle toho, jestli je prázdný celý zápis, nebo jen výběr.

## --tip--

Obojí už v komponentě máš: celý zápis i to, co je po filtru vidět.

## --file-- utils.js

```js
// Kolik kalorií mají všechny zapsané porce dohromady.
export function soucetKalorii(zaznamy) {
  return zaznamy.reduce((celkem, zaznam) => celkem + zaznam.kcal * zaznam.porce, 0);
}
```

## --file-- App.jsx

```jsx
import './styles.css';
import { useState } from 'react';
import { DENNI_CIL, potraviny } from './data.js';
import { soucetKalorii } from './utils.js';

const TYPY = ['Vše', 'Snídaně', 'Oběd', 'Večeře', 'Svačina'];

export default function App() {
  const [zaznamy, setZaznamy] = useState([]);
  const [filtr, setFiltr] = useState('Vše');

  const snedeno = soucetKalorii(zaznamy);
  const zbyva = DENNI_CIL - snedeno;
  const procenta = Math.min(100, Math.round((snedeno / DENNI_CIL) * 100));
  const videt = filtr === 'Vše' ? zaznamy : zaznamy.filter((zaznam) => zaznam.typ === filtr);

  function pridej(potravina) {
    setZaznamy([...zaznamy, { ...potravina, id: crypto.randomUUID(), porce: 1 }]);
  }

  function zmenPorci(id, zmena) {
    setZaznamy(
      zaznamy.map((zaznam) =>
        zaznam.id === id ? { ...zaznam, porce: Math.max(1, zaznam.porce + zmena) } : zaznam,
      ),
    );
  }

  function smaz(id) {
    setZaznamy(zaznamy.filter((zaznam) => zaznam.id !== id));
  }

  return (
    <div className="stranka">
      <section className="souhrn">
        <h1 className="souhrn__nadpis">Dnes máš snědeno</h1>
        <p className="souhrn__cislo">{snedeno} kcal</p>
        <p className="souhrn__zbyva">{`Do cíle ${DENNI_CIL} kcal zbývá ${zbyva} kcal.`}</p>
        <div className="ukazatel" aria-hidden="true">
          <div className="ukazatel__cara" style={{ inlineSize: `${procenta}%` }} />
        </div>
      </section>

      <section className="sekce">
        <h2 className="sekce__nadpis">Co jsem snědl</h2>
        <div className="volby">
          {potraviny.map((potravina) => (
            <button className="volba" key={potravina.id} onClick={() => pridej(potravina)}>
              {potravina.nazev} <span className="volba__kcal">{potravina.kcal} kcal</span>
            </button>
          ))}
        </div>
      </section>

      <section className="sekce">
        <div className="sekce__hlavicka">
          <h2 className="sekce__nadpis">Dnešní zápis</h2>
          <div className="filtr">
            {TYPY.map((typ) => (
              <button
                className={typ === filtr ? 'filtr__tlacitko filtr__tlacitko--aktivni' : 'filtr__tlacitko'}
                key={typ}
                onClick={() => setFiltr(typ)}
              >
                {typ}
              </button>
            ))}
          </div>
        </div>
        {videt.length === 0 && <p className="prazdno">Zatím nic. Klikni na jídlo nahoře.</p>}
        <ul className="zaznamy">
          {videt.map((zaznam) => (
            <li className="zaznam" key={zaznam.id}>
              <p className="zaznam__nazev">
                {zaznam.nazev} <span className="zaznam__typ">{zaznam.typ}</span>
              </p>
              <span className="zaznam__porce">
                <button className="kulate" onClick={() => zmenPorci(zaznam.id, -1)} aria-label="Ubrat porci">−</button>
                {zaznam.porce}×
                <button className="kulate" onClick={() => zmenPorci(zaznam.id, 1)} aria-label="Přidat porci">+</button>
              </span>
              <span className="zaznam__kcal">{zaznam.kcal * zaznam.porce} kcal</span>
              <button className="smazat" onClick={() => smaz(zaznam.id)} aria-label="Smazat záznam">×</button>
            </li>
          ))}
        </ul>
      </section>
    </div>
  );
}
```
