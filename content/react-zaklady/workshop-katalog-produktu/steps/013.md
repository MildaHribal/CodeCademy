# --description--

Jedna hra v katalogu má v datech `skladem: 0`. Zákazník to z karty nepozná — a to je v e-shopu ta nejhorší varianta: naklikat se do košíku a tam se dozvědět, že zboží není.

**Úkol:** Zařiď, aby karta vyprodané hry byla na první pohled jiná než ostatní. Jak, je na tobě: ve stylech čeká třída `karta--vyprodano` (šedý podklad a čárkovaný rámeček) i druh štítku `varovani`, ale klidně přidej vlastní text nebo obojí. Karty her, které skladem jsou, se měnit nesmí.

Testy hlídají jen to, že se vyprodaná karta liší od ostatních a že u dostupných her nikde nesvítí slovo „vyprodáno".

# --hints--

Karta vyprodané hry se liší od karty hry, která je skladem.

```js
const { hry } = await helpers.importFile('data.js');
const vyprodana = hry.find((hra) => hra.skladem === 0);
const karty = [...document.querySelectorAll('.karta')];
const znak = (el) => {
  const styl = getComputedStyle(el);
  return [el.className, styl.backgroundColor, styl.borderStyle, styl.opacity, styl.color, el.querySelectorAll('.stitek').length, /vyprodán|není skladem|doprodán/i.test(el.textContent)].join('|');
};
const kartaVyprodane = karty.find((el) => el.textContent.includes(vyprodana.nazev));
const kartaDostupne = karty.find((el) => el.textContent.includes('Ubongo'));
assert.ok(kartaVyprodane, 'karta vyprodané hry má v katalogu zůstat');
assert.notEqual(znak(kartaVyprodane), znak(kartaDostupne), `karta hry ${vyprodana.nazev} (skladem 0) se má lišit od karty hry, která skladem je`);
```

Hry, které jsou skladem, nikde nehlásí vyprodáno.

```js
const { hry } = await helpers.importFile('data.js');
const vyprodana = hry.find((hra) => hra.skladem === 0);
const ostatni = [...document.querySelectorAll('.karta')].filter((el) => !el.textContent.includes(vyprodana.nazev));
ostatni.forEach((el) => {
  assert.doesNotMatch(el.textContent, /vyprodán|není skladem|doprodán/i, 'karta hry, která je skladem, nesmí hlásit vyprodáno');
  assert.ok(!el.classList.contains('karta--vyprodano'), 'třída karta--vyprodano patří jen hře, která skladem není');
});
```

Ostatní údaje na kartách zůstaly beze změny.

```js
const { hry } = await helpers.importFile('data.js');
const format = new Intl.NumberFormat('cs-CZ', { style: 'currency', currency: 'CZK', maximumFractionDigits: 0 });
const karty = [...document.querySelectorAll('.stranka .panel:last-of-type .karta')];
assert.equal(karty.length, hry.length, 'v katalogu má zůstat všech šest her');
hry.forEach((hra, index) => {
  assert.equal(karty[index].querySelector('.karta__nazev').textContent.trim(), hra.nazev, 'název hry má zůstat');
  assert.equal(karty[index].querySelector('.cena__aktualni').textContent.trim(), format.format(hra.cena), 'cena má zůstat');
});
```

```js
// Průměrné hodnocení všech her, zaokrouhlené na jedno desetinné místo.
export function prumerneHodnoceni(hry) {
  const soucet = hry.reduce((celkem, hra) => celkem + hra.hodnoceni, 0);
  return Math.round((soucet / hry.length) * 10) / 10;
}
```

## --file-- KartaHry.jsx

```jsx
import Cena from './Cena.jsx';
import Stitek from './Stitek.jsx';

export default function KartaHry({ hra }) {
  return (
    <li className={hra.skladem === 0 ? 'karta karta--vyprodano' : 'karta'}>
      <div className="karta__stitky">
        {hra.novinka && <Stitek text="Novinka" druh="novinka" />}
        {hra.puvodniCena > hra.cena && <Stitek text="Sleva" druh="sleva" />}
        {hra.skladem === 0 && <Stitek text="Vyprodáno" druh="varovani" />}
      </div>
      <h3 className="karta__nazev">{hra.nazev}</h3>
      <p className="karta__meta">
        {hra.kategorie} · {hra.hraci} hráčů · {hra.delka} min
      </p>
      <p className="karta__popis">{hra.popis}</p>
      <Cena castka={hra.cena} puvodni={hra.puvodniCena} />
    </li>
  );
}
```
