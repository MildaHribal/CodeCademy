# --description--

Když zákazník napíše „rezervace mi nešla", potřebuješ v lozích najít jeho požadavek:
kdy přišel, na jakou adresu, jak dopadl a jak dlouho trval. Proto se každý vyřízený
požadavek zapíše jedním řádkem.

**Úkol:** po odeslání každé odpovědi zapiš přes logger řádek s úrovní `info`, zprávou
`Požadavek vyřízen` a poli `method`, `path` (jen cesta, bez query stringu), `status`
(stavový kód odpovědi) a `durationMs` (jak dlouho vyřízení trvalo, číslo
v milisekundách). `createApp` k tomu dostane logger jako druhou závislost:
`createApp({ db, logger })`.

Cestu loguj bez query stringu schválně: odkaz z e-mailu s `?token=…` by jinak nechal
token v lozích.

Kandidáti: událost `finish` objektu odpovědi (nastane, když je odpověď odeslaná),
`res.statusCode`, `performance.now()` nebo `Date.now()` na měření času. Místo, kde to
jde udělat pro všechny adresy najednou, je funkce, kterou `createApp` vrací.

# --hints--

Po `GET /api/bikes?token=…` se zapíše řádek `Požadavek vyřízen` s `method`, `path` bez query stringu, `status` a číselným `durationMs`.

```js
const server = await helpers.startServer('server.js', { timeoutMs: 4000 });
await fetch(new URL('/api/bikes?token=odkaz-z-emailu-5f2a', server.url));
const parse = (line) => { try { return JSON.parse(line); } catch { return null; } };
const entry = await helpers.waitFor(
  () => server.output().split('\n').map(parse).find((item) => item?.msg === 'Požadavek vyřízen' && item.method === 'GET'),
  2000,
).catch(() => assert.fail(`Po GET /api/bikes má server zapsat řádek s msg "Požadavek vyřízen" a method "GET". Výstup serveru:\n${server.output()}`));
assert.equal(entry.level, 'info', 'Řádek o požadavku má mít level "info"');
assert.equal(entry.path, '/api/bikes', 'Pole path má být "/api/bikes" — jen cesta, bez query stringu');
assert.equal(entry.status, 200, 'Pole status má být číslo 200');
assert.equal(typeof entry.durationMs, 'number', 'Pole durationMs má být číslo');
assert.ok(entry.durationMs >= 0, 'Pole durationMs nemá být záporné');
assert.doesNotMatch(server.output(), /odkaz-z-emailu-5f2a/, 'Token z query stringu se nesmí dostat do logu');
```

Řádek se zapíše i u chybových odpovědí se skutečným stavem (`400`, `404`).

```js
const server = await helpers.startServer('server.js', { timeoutMs: 4000 });
await fetch(new URL('/api/rentals', server.url), { method: 'POST', body: 'není JSON' });
await fetch(new URL('/api/neexistuje', server.url));
const parse = (line) => { try { return JSON.parse(line); } catch { return null; } };
const entries = await helpers.waitFor(() => {
  const found = server.output().split('\n').map(parse).filter((item) => item?.msg === 'Požadavek vyřízen');
  return found.length >= 2 ? found : null;
}, 2000).catch(() => assert.fail(`Po dvou požadavcích mají být ve výstupu dva řádky "Požadavek vyřízen". Výstup serveru:\n${server.output()}`));
const post = entries.find((item) => item.method === 'POST');
assert.equal(post?.path, '/api/rentals', 'Řádek o POST /api/rentals má mít path "/api/rentals"');
assert.equal(post?.status, 400, 'POST /api/rentals s tělem, které není JSON, má mít v logu status 400');
const missing = entries.find((item) => item.path === '/api/neexistuje');
assert.equal(missing?.status, 404, 'GET /api/neexistuje má mít v logu status 404');
```

# --help--

## --tip--

Proč stálá zpráva a hodnoty v polích, je v části
[Strukturované logy](see:nasazeni-provoz/logovani-a-chyby#strukturovane-logy).

## --tip--

Čas startu si ulož na začátku funkce, kterou `createApp` vrací, a posluchač události
`finish` na `res` zaregistruj tamtéž, ještě před voláním `handle`. `server.js` musí
předat logger do `createApp`.

## --tip--

Stejný vzor pro jinou metriku: `const started = Date.now(); job.on('done', () => stats.record(job.name, Date.now() - started));`.

```js
// HTTP API půjčovny kol. Server (server.js) sem posílá každý požadavek.

async function readJson(req) {
  let body = '';
  for await (const chunk of req) body += chunk;
  return JSON.parse(body);
}

export function createApp({ db }) {
  function sendJson(res, status, data) {
    res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
    res.end(JSON.stringify(data));
  }

  async function handle(req, res) {
    const { pathname } = new URL(req.url, 'http://localhost');

    if (req.method === 'GET' && pathname === '/health') {
      try {
        db.prepare('SELECT 1').get();
        return sendJson(res, 200, { status: 'ok' });
      } catch {
        return sendJson(res, 503, { status: 'error' });
      }
    }

    if (req.method === 'GET' && pathname === '/api/bikes') {
      const bikes = db
        .prepare('SELECT id, name, type, station, price_per_day AS pricePerDay FROM bikes ORDER BY id')
        .all();
      return sendJson(res, 200, bikes);
    }

    if (req.method === 'POST' && pathname === '/api/rentals') {
      let input;
      try {
        input = await readJson(req);
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku není platný JSON.' });
      }
      const bikeId = Number(input?.bikeId);
      const days = Number(input?.days);
      const customerName = String(input?.customerName ?? '').trim();
      if (!Number.isInteger(bikeId) || !Number.isInteger(days) || days < 1 || customerName === '') {
        return sendJson(res, 400, { error: 'Vyplň kolo, jméno a počet dní (aspoň 1).' });
      }
      const bike = db.prepare('SELECT id, price_per_day AS pricePerDay FROM bikes WHERE id = ?').get(bikeId);
      if (!bike) {
        return sendJson(res, 404, { error: 'Takové kolo v půjčovně nemáme.' });
      }
      const result = db
        .prepare('INSERT INTO rentals (bike_id, customer_name, days, created_at) VALUES (?, ?, ?, ?)')
        .run(bike.id, customerName, days, new Date().toISOString());
      return sendJson(res, 201, {
        id: Number(result.lastInsertRowid),
        bikeId: bike.id,
        days,
        totalPrice: bike.pricePerDay * days,
      });
    }

    sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
  }

  return (req, res) => {
    handle(req, res);
  };
}
```

```js
import { createServer } from 'node:http';
import { loadConfig } from './config.js';
import { createLogger } from './logger.js';
import { openDatabase } from './db.js';
import { createApp } from './app.js';

const config = loadConfig(process.env);
```

## --file-- logger.js

```js
// Strukturovaný logger: jedna událost = jeden řádek JSON na standardní výstup.

const LEVELS = ['debug', 'info', 'warn', 'error'];
const SECRET_KEYS = ['password', 'token', 'authorization', 'cookie'];
const writeToStdout = (line) => process.stdout.write(`${line}\n`);

export function createLogger({ level = 'info', write = writeToStdout } = {}) {
  const minIndex = LEVELS.indexOf(level);

  function log(entryLevel, msg, fields = {}) {
    if (LEVELS.indexOf(entryLevel) < minIndex) return;
    const safeFields = {};
    for (const [key, value] of Object.entries(fields)) {
      safeFields[key] = SECRET_KEYS.includes(key) ? '[SKRYTO]' : value;
    }
    const entry = { time: new Date().toISOString(), level: entryLevel, msg, ...safeFields };
    write(JSON.stringify(entry));
  }

  return {
    debug: (msg, fields) => log('debug', msg, fields),
    info: (msg, fields) => log('info', msg, fields),
    warn: (msg, fields) => log('warn', msg, fields),
    error: (msg, fields) => log('error', msg, fields),
  };
}
```

```js
// Veškeré nastavení aplikace se čte tady, na jednom místě, při startu.

export function loadConfig(env) {
  if (env.NODE_ENV === 'production' && !env.DATABASE_FILE) {
    throw new Error('Chybí proměnná prostředí DATABASE_FILE.');
  }
  return {
    port: Number(env.PORT ?? 3000),
    databaseFile: env.DATABASE_FILE ?? 'pujcovna-dev.db',
    logLevel: env.LOG_LEVEL ?? 'info',
    shutdownTimeoutMs: Number(env.SHUTDOWN_TIMEOUT_MS ?? 8000),
  };
}
```

## --file-- db.js

```js
import { DatabaseSync } from 'node:sqlite';

const BIKES = [
  ['Superior XC 879', 'horské', 'Brno – Výstaviště', 450],
  ['Author Mission', 'městské', 'Brno – Mendlovo náměstí', 290],
  ['Kellys Tyke 24', 'dětské', 'Brno – Výstaviště', 190],
  ['Specialized Turbo Vado', 'elektrokolo', 'Brno – Lužánky', 890],
  ['Cannondale Topstone', 'gravel', 'Brno – Lužánky', 520],
];

// Otevře databázi půjčovny. Při prvním spuštění založí tabulky a vloží kola.
export function openDatabase(file) {
  const db = new DatabaseSync(file);
  db.exec(`
    CREATE TABLE IF NOT EXISTS bikes (
      id INTEGER PRIMARY KEY,
      name TEXT NOT NULL,
      type TEXT NOT NULL,
      station TEXT NOT NULL,
      price_per_day INTEGER NOT NULL
    );
    CREATE TABLE IF NOT EXISTS rentals (
      id INTEGER PRIMARY KEY,
      bike_id INTEGER NOT NULL REFERENCES bikes (id),
      customer_name TEXT NOT NULL,
      days INTEGER NOT NULL,
      created_at TEXT NOT NULL
    );
  `);
  const { count } = db.prepare('SELECT COUNT(*) AS count FROM bikes').get();
  if (count === 0) {
    const insert = db.prepare('INSERT INTO bikes (name, type, station, price_per_day) VALUES (?, ?, ?, ?)');
    for (const bike of BIKES) insert.run(...bike);
  }
  return db;
}
```

```js
        durationMs: Math.round(performance.now() - startedAt),

    handle(req, res);
  };
```
