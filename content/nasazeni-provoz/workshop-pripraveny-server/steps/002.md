# --description--

Kromě portu se mezi notebookem a hostingem liší i cesta k databázi, podrobnost logů
a to, jak dlouho smí trvat vypnutí serveru. Kdyby každý soubor četl `process.env` sám,
nikdo by nevěděl, co všechno aplikace potřebuje. Proto se nastavení čte na jednom
místě: ve funkci, která dostane proměnné prostředí jako parametr.

**Úkol:** v novém souboru `config.js` napiš a vyexportuj funkci `loadConfig(env)`.
Vrátí objekt se čtyřmi klíči:

| klíč | proměnná prostředí | výchozí hodnota | typ |
|---|---|---|---|
| `port` | `PORT` | `3000` | číslo |
| `databaseFile` | `DATABASE_FILE` | `'pujcovna-dev.db'` | řetězec |
| `logLevel` | `LOG_LEVEL` | `'info'` | řetězec |
| `shutdownTimeoutMs` | `SHUTDOWN_TIMEOUT_MS` | `8000` | číslo |

Proměnné prostředí jsou vždycky řetězce, proto čísla převeď. Server funkci zatím
nepoužívá, zapojíš ji ve čtvrtém kroku.

Stejný tvar pro e-shop s jinými proměnnými:

```js
export function loadShopConfig(env) {
  return {
    currency: env.CURRENCY ?? 'CZK',
    freeShippingFrom: Number(env.FREE_SHIPPING_FROM ?? 1500),
  };
}
```

# --hints--

`config.js` exportuje funkci `loadConfig`.

```js
const config = await helpers.importFile('config.js');
assert.equal(typeof config.loadConfig, 'function', 'config.js má exportovat funkci loadConfig — chybí před ní export?');
```

Bez proměnných prostředí vrátí `loadConfig({})` výchozí hodnoty.

```js
const { loadConfig } = await helpers.importFile('config.js');
assert.deepEqual(
  loadConfig({}),
  { port: 3000, databaseFile: 'pujcovna-dev.db', logLevel: 'info', shutdownTimeoutMs: 8000 },
  'loadConfig({}) má vrátit výchozí hodnoty všech čtyř klíčů',
);
```

Hodnoty z prostředí přebijí výchozí a `port` se `shutdownTimeoutMs` jsou čísla.

```js
const { loadConfig } = await helpers.importFile('config.js');
const env = { PORT: '8080', DATABASE_FILE: '/data/pujcovna.db', LOG_LEVEL: 'warn', SHUTDOWN_TIMEOUT_MS: '5000' };
assert.deepEqual(
  loadConfig(env),
  { port: 8080, databaseFile: '/data/pujcovna.db', logLevel: 'warn', shutdownTimeoutMs: 5000 },
  'loadConfig({ PORT: "8080", DATABASE_FILE: "/data/pujcovna.db", LOG_LEVEL: "warn", SHUTDOWN_TIMEOUT_MS: "5000" }) má vrátit hodnoty z prostředí, port a shutdownTimeoutMs jako čísla',
);
```

`loadConfig` čte jen z parametru `env`, ne z `process.env`.

```js
const { loadConfig } = await helpers.importFile('config.js');
process.env.PORT = '4444';
process.env.LOG_LEVEL = 'debug';
const config = loadConfig({});
assert.equal(config.port, 3000, 'loadConfig({}) má vrátit port 3000, i když process.env.PORT je 4444 — čti z parametru env');
assert.equal(config.logLevel, 'info', 'loadConfig({}) má vrátit logLevel "info", i když process.env.LOG_LEVEL je "debug"');
```

## --file-- app.js

```js
// HTTP API půjčovny kol. Server (server.js) sem posílá každý požadavek.

async function readJson(req) {
  let body = '';
  for await (const chunk of req) body += chunk;
  return JSON.parse(body);
}

export function createApp({ db }) {
  function sendJson(res, status, data) {
    res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
    res.end(JSON.stringify(data));
  }

  async function handle(req, res) {
    const { pathname } = new URL(req.url, 'http://localhost');

    if (req.method === 'GET' && pathname === '/api/bikes') {
      const bikes = db
        .prepare('SELECT id, name, type, station, price_per_day AS pricePerDay FROM bikes ORDER BY id')
        .all();
      return sendJson(res, 200, bikes);
    }

    if (req.method === 'POST' && pathname === '/api/rentals') {
      let input;
      try {
        input = await readJson(req);
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku není platný JSON.' });
      }
      const bikeId = Number(input?.bikeId);
      const days = Number(input?.days);
      const customerName = String(input?.customerName ?? '').trim();
      if (!Number.isInteger(bikeId) || !Number.isInteger(days) || days < 1 || customerName === '') {
        return sendJson(res, 400, { error: 'Vyplň kolo, jméno a počet dní (aspoň 1).' });
      }
      const bike = db.prepare('SELECT id, price_per_day AS pricePerDay FROM bikes WHERE id = ?').get(bikeId);
      if (!bike) {
        return sendJson(res, 404, { error: 'Takové kolo v půjčovně nemáme.' });
      }
      const result = db
        .prepare('INSERT INTO rentals (bike_id, customer_name, days, created_at) VALUES (?, ?, ?, ?)')
        .run(bike.id, customerName, days, new Date().toISOString());
      return sendJson(res, 201, {
        id: Number(result.lastInsertRowid),
        bikeId: bike.id,
        days,
        totalPrice: bike.pricePerDay * days,
      });
    }

    sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
  }

  return (req, res) => {
    handle(req, res);
  };
}
```

## --file-- db.js

```js
import { DatabaseSync } from 'node:sqlite';

const BIKES = [
  ['Superior XC 879', 'horské', 'Brno – Výstaviště', 450],
  ['Author Mission', 'městské', 'Brno – Mendlovo náměstí', 290],
  ['Kellys Tyke 24', 'dětské', 'Brno – Výstaviště', 190],
  ['Specialized Turbo Vado', 'elektrokolo', 'Brno – Lužánky', 890],
  ['Cannondale Topstone', 'gravel', 'Brno – Lužánky', 520],
];

// Otevře databázi půjčovny. Při prvním spuštění založí tabulky a vloží kola.
export function openDatabase(file) {
  const db = new DatabaseSync(file);
  db.exec(`
    CREATE TABLE IF NOT EXISTS bikes (
      id INTEGER PRIMARY KEY,
      name TEXT NOT NULL,
      type TEXT NOT NULL,
      station TEXT NOT NULL,
      price_per_day INTEGER NOT NULL
    );
    CREATE TABLE IF NOT EXISTS rentals (
      id INTEGER PRIMARY KEY,
      bike_id INTEGER NOT NULL REFERENCES bikes (id),
      customer_name TEXT NOT NULL,
      days INTEGER NOT NULL,
      created_at TEXT NOT NULL
    );
  `);
  const { count } = db.prepare('SELECT COUNT(*) AS count FROM bikes').get();
  if (count === 0) {
    const insert = db.prepare('INSERT INTO bikes (name, type, station, price_per_day) VALUES (?, ?, ?, ?)');
    for (const bike of BIKES) insert.run(...bike);
  }
  return db;
}
```
