# --description--

Když databáze uprostřed provozu selže (zamčený nebo poškozený soubor), `db.prepare`
v `handle` vyhodí výjimku. Z `async` funkce se stane odmítnutý Promise, který nikdo
nechytá, a Node celý proces ukončí. Jedna chyba shodí půjčovnu všem.

Síť proti pádu jsi v sekci o Node už stavěl. Teď k ní přidáš to, co v provozu
potřebuješ, abys chybu našel.

**Úkol:** když obsluha požadavku vyhodí nečekanou chybu:

- klient dostane stav `500` a JSON s klíči `error` (obecná česká věta) a `requestId`
  (stejné id jako v hlavičce),
- text výjimky ani stack trace v odpovědi nebudou,
- logger zapíše řádek s úrovní `error`, polem `requestId` a polem `error`, ve kterém
  je stack trace jako řetězec,
- server běží dál a další požadavky obslouží.

Pozor na odpověď, která se už začala posílat: druhou poslat nejde.

# --hints--

Když databáze selže, `GET /api/bikes` dostane `500` s klíči `error` a `requestId`, bez textu výjimky.

```js
const { createServer } = await import('node:http');
const { DatabaseSync } = await import('node:sqlite');
const { createApp } = await helpers.importFile('app.js');
process.on('unhandledRejection', () => {});
const db = new DatabaseSync(':memory:');
db.close();
const logger = { debug() {}, info() {}, warn() {}, error() {} };
const server = createServer(createApp({ db, logger }));
await new Promise((resolve) => server.listen(0, '127.0.0.1', resolve));
try {
  const res = await fetch(`http://127.0.0.1:${server.address().port}/api/bikes`, { signal: AbortSignal.timeout(3000) })
    .catch(() => assert.fail('GET /api/bikes se zavřenou databází neodpověděl — chytáš odmítnutý Promise z handle?'));
  assert.equal(res.status, 500, 'GET /api/bikes se zavřenou databází má vrátit stav 500');
  const text = await res.text();
  const body = JSON.parse(text);
  assert.equal(typeof body.error, 'string', 'Tělo odpovědi 500 má mít klíč error s větou pro uživatele');
  assert.equal(body.requestId, res.headers.get('x-request-id'), 'Tělo odpovědi 500 má mít requestId stejné jako hlavička X-Request-Id');
  assert.doesNotMatch(text, /not open|at |DatabaseSync/, 'Odpověď 500 nesmí obsahovat text výjimky ani stack trace');
} finally {
  server.closeAllConnections();
  server.close();
}
```

Logger zapíše chybu s úrovní `error`, `requestId` a stack tracem v poli `error`.

```js
const { createServer } = await import('node:http');
const { DatabaseSync } = await import('node:sqlite');
const { createApp } = await helpers.importFile('app.js');
process.on('unhandledRejection', () => {});
const db = new DatabaseSync(':memory:');
db.close();
const entries = [];
const logger = {
  debug: (msg, fields) => entries.push({ level: 'debug', msg, ...fields }),
  info: (msg, fields) => entries.push({ level: 'info', msg, ...fields }),
  warn: (msg, fields) => entries.push({ level: 'warn', msg, ...fields }),
  error: (msg, fields) => entries.push({ level: 'error', msg, ...fields }),
};
const server = createServer(createApp({ db, logger }));
await new Promise((resolve) => server.listen(0, '127.0.0.1', resolve));
try {
  const res = await fetch(`http://127.0.0.1:${server.address().port}/api/bikes`, {
    headers: { 'X-Request-Id': 'podpora-3391' },
    signal: AbortSignal.timeout(3000),
  }).catch(() => assert.fail('GET /api/bikes se zavřenou databází neodpověděl'));
  await res.text();
  const entry = await helpers.waitFor(() => entries.find((item) => item.level === 'error'), 1000)
    .catch(() => assert.fail('Při nečekané chybě má logger zapsat řádek přes logger.error'));
  assert.equal(entry.requestId, 'podpora-3391', 'Řádek s chybou má mít requestId "podpora-3391" z hlavičky požadavku');
  assert.equal(typeof entry.error, 'string', 'Pole error má být řetězec (stack trace), ne objekt Error — JSON.stringify by z něj udělal {}');
  assert.match(entry.error, /not open/, 'Pole error má obsahovat zprávu výjimky ze stack trace');
} finally {
  server.closeAllConnections();
  server.close();
}
```

Po nečekané chybě server běží dál a obslouží další požadavek.

```js
const server = await helpers.startServer('server.js', { timeoutMs: 4000 });
const fs = await import('node:fs/promises');
const dbFile = `${helpers.dir}/pujcovna-dev.db`;
await fs.writeFile(dbFile, 'tohle není databáze');
await fetch(new URL('/api/bikes', server.url)).catch(() => {});
await helpers.wait(200);
const res = await fetch(new URL('/api/neexistuje', server.url)).catch(() =>
  assert.fail(`Po chybě databáze server přestal odpovídat (spadl?). Výstup serveru:\n${server.output()}`));
assert.equal(res.status, 404, 'Po nečekané chybě má server dál odpovídat, GET /api/neexistuje má vrátit 404');
```

## --tip--

Síť proti pádu a proč nesmí prozradit podrobnosti, je v části
[Sledování chyb](see:nasazeni-provoz/logovani-a-chyby#sledovani-chyb).

## --tip--

`handle` vrací Promise — chyba se chytá tam, kde se `handle` volá, a `requestId` je
v té funkci po ruce. Než pošleš `500`, zkontroluj `res.headersSent`.

## --file-- app.js

```js
import { randomUUID } from 'node:crypto';

// HTTP API půjčovny kol. Server (server.js) sem posílá každý požadavek.

async function readJson(req) {
  let body = '';
  for await (const chunk of req) body += chunk;
  return JSON.parse(body);
}

export function createApp({ db, logger }) {
  function sendJson(res, status, data) {
    res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
    res.end(JSON.stringify(data));
  }

  async function handle(req, res) {
    const { pathname } = new URL(req.url, 'http://localhost');

    if (req.method === 'GET' && pathname === '/health') {
      try {
        db.prepare('SELECT 1').get();
        return sendJson(res, 200, { status: 'ok' });
      } catch {
        return sendJson(res, 503, { status: 'error' });
      }
    }

    if (req.method === 'GET' && pathname === '/api/bikes') {
      const bikes = db
        .prepare('SELECT id, name, type, station, price_per_day AS pricePerDay FROM bikes ORDER BY id')
        .all();
      return sendJson(res, 200, bikes);
    }

    if (req.method === 'POST' && pathname === '/api/rentals') {
      let input;
      try {
        input = await readJson(req);
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku není platný JSON.' });
      }
      const bikeId = Number(input?.bikeId);
      const days = Number(input?.days);
      const customerName = String(input?.customerName ?? '').trim();
      if (!Number.isInteger(bikeId) || !Number.isInteger(days) || days < 1 || customerName === '') {
        return sendJson(res, 400, { error: 'Vyplň kolo, jméno a počet dní (aspoň 1).' });
      }
      const bike = db.prepare('SELECT id, price_per_day AS pricePerDay FROM bikes WHERE id = ?').get(bikeId);
      if (!bike) {
        return sendJson(res, 404, { error: 'Takové kolo v půjčovně nemáme.' });
      }
      const result = db
        .prepare('INSERT INTO rentals (bike_id, customer_name, days, created_at) VALUES (?, ?, ?, ?)')
        .run(bike.id, customerName, days, new Date().toISOString());
      return sendJson(res, 201, {
        id: Number(result.lastInsertRowid),
        bikeId: bike.id,
        days,
        totalPrice: bike.pricePerDay * days,
      });
    }

    sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
  }

  return (req, res) => {
    const startedAt = performance.now();
    const requestId = req.headers['x-request-id'] ?? randomUUID();
    res.setHeader('X-Request-Id', requestId);
    res.on('finish', () => {
      logger.info('Požadavek vyřízen', {
        requestId,
        method: req.method,
        path: new URL(req.url, 'http://localhost').pathname,
        status: res.statusCode,
        durationMs: Math.round(performance.now() - startedAt),
      });
    });
    handle(req, res);
  };
}
```

## --file-- logger.js

```js
// Strukturovaný logger: jedna událost = jeden řádek JSON na standardní výstup.

const LEVELS = ['debug', 'info', 'warn', 'error'];
const SECRET_KEYS = ['password', 'token', 'authorization', 'cookie'];
const writeToStdout = (line) => process.stdout.write(`${line}\n`);

export function createLogger({ level = 'info', write = writeToStdout } = {}) {
  const minIndex = LEVELS.indexOf(level);

  function log(entryLevel, msg, fields = {}) {
    if (LEVELS.indexOf(entryLevel) < minIndex) return;
    const safeFields = {};
    for (const [key, value] of Object.entries(fields)) {
      safeFields[key] = SECRET_KEYS.includes(key) ? '[SKRYTO]' : value;
    }
    const entry = { time: new Date().toISOString(), level: entryLevel, msg, ...safeFields };
    write(JSON.stringify(entry));
  }

  return {
    debug: (msg, fields) => log('debug', msg, fields),
    info: (msg, fields) => log('info', msg, fields),
    warn: (msg, fields) => log('warn', msg, fields),
    error: (msg, fields) => log('error', msg, fields),
  };
}
```

```js
export function loadConfig(env) {
  if (env.NODE_ENV === 'production' && !env.DATABASE_FILE) {
    throw new Error('Chybí proměnná prostředí DATABASE_FILE.');
  }
  return {
    port: Number(env.PORT ?? 3000),
    databaseFile: env.DATABASE_FILE ?? 'pujcovna-dev.db',
    logLevel: env.LOG_LEVEL ?? 'info',
    shutdownTimeoutMs: Number(env.SHUTDOWN_TIMEOUT_MS ?? 8000),
  };
}
```

```js
import { DatabaseSync } from 'node:sqlite';

const BIKES = [
  ['Superior XC 879', 'horské', 'Brno – Výstaviště', 450],
  ['Author Mission', 'městské', 'Brno – Mendlovo náměstí', 290],
  ['Kellys Tyke 24', 'dětské', 'Brno – Výstaviště', 190],
  ['Specialized Turbo Vado', 'elektrokolo', 'Brno – Lužánky', 890],
  ['Cannondale Topstone', 'gravel', 'Brno – Lužánky', 520],
];

// Otevře databázi půjčovny. Při prvním spuštění založí tabulky a vloží kola.
export function openDatabase(file) {
  const db = new DatabaseSync(file);
  db.exec(`
    CREATE TABLE IF NOT EXISTS bikes (
      id INTEGER PRIMARY KEY,
      name TEXT NOT NULL,
      type TEXT NOT NULL,
      station TEXT NOT NULL,
      price_per_day INTEGER NOT NULL
    );
    CREATE TABLE IF NOT EXISTS rentals (
      id INTEGER PRIMARY KEY,
      bike_id INTEGER NOT NULL REFERENCES bikes (id),
      customer_name TEXT NOT NULL,
      days INTEGER NOT NULL,
      created_at TEXT NOT NULL
    );
  `);
  const { count } = db.prepare('SELECT COUNT(*) AS count FROM bikes').get();
  if (count === 0) {
    const insert = db.prepare('INSERT INTO bikes (name, type, station, price_per_day) VALUES (?, ?, ?, ?)');
    for (const bike of BIKES) insert.run(...bike);
  }
  return db;
}
```
