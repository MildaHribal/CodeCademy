# --description--

`/health` teď vrací `200` natvrdo. Prozradí jen, že proces běží. Když se ale rozbije
databáze (smazaný soubor, plný disk), půjčovna nezvládne jedinou rezervaci — a health
check dál tvrdí, že je všechno v pořádku. Docker ani hosting pak nemají důvod cokoli
dělat.

**Úkol:** `GET /health` ověří, že databáze odpovídá na jednoduchý dotaz. Když ano,
odpoví `200` a `{ "status": "ok" }`. Když dotaz vyhodí chybu, odpoví `503`
a `{ "status": "error" }`. Text chyby do odpovědi nedávej: adresa je veřejná.

Co se hodí: dotaz, který nic nečte z tabulek (`SELECT 1`), metody databáze `prepare`
s `get`, nebo `exec`, a `try`/`catch` kolem nich.

# --hints--

S funkční databází vrátí `GET /health` stav `200` a `{ "status": "ok" }`.

```js
const { createServer } = await import('node:http');
const { DatabaseSync } = await import('node:sqlite');
const { createApp } = await helpers.importFile('app.js');
const db = new DatabaseSync(':memory:');
const server = createServer(createApp({ db }));
await new Promise((resolve) => server.listen(0, '127.0.0.1', resolve));
try {
  const res = await fetch(`http://127.0.0.1:${server.address().port}/health`, { signal: AbortSignal.timeout(3000) });
  assert.equal(res.status, 200, 'GET /health s funkční databází má vrátit stav 200');
  assert.deepEqual(await res.json(), { status: 'ok' }, 'GET /health s funkční databází má vrátit { status: "ok" }');
} finally {
  server.closeAllConnections();
  server.close();
}
```

Když databáze nefunguje, vrátí `GET /health` stav `503` a `{ "status": "error" }` bez textu chyby.

```js
const { createServer } = await import('node:http');
const { DatabaseSync } = await import('node:sqlite');
const { createApp } = await helpers.importFile('app.js');
const db = new DatabaseSync(':memory:');
db.close();
const server = createServer(createApp({ db }));
await new Promise((resolve) => server.listen(0, '127.0.0.1', resolve));
process.on('unhandledRejection', () => {});
try {
  const res = await fetch(`http://127.0.0.1:${server.address().port}/health`, { signal: AbortSignal.timeout(3000) })
    .catch(() => assert.fail('GET /health se zavřenou databází neodpověděl — chytáš chybu dotazu přes try/catch?'));
  assert.equal(res.status, 503, 'GET /health se zavřenou databází má vrátit stav 503');
  const body = await res.text();
  assert.deepEqual(JSON.parse(body), { status: 'error' }, 'GET /health se zavřenou databází má vrátit { status: "error" }');
} finally {
  server.closeAllConnections();
  server.close();
}
```

# --help--

## --tip--

Health check má ověřit to, bez čeho aplikace nefunguje, a nic pomalého navíc — viz
[Health check](see:nasazeni-provoz/produkcni-rezim#health-check).

## --tip--

Dotaz do databáze patří do `try` bloku v obsluze `/health`. Odpověď `200` pošli
až za dotazem, odpověď `503` v `catch`.

## --tip--

Stejný vzor u jiné závislosti: `try { cache.ping(); return reply(200); } catch { return reply(503); }`.

# --explain--

Vysvětli, proč health check půjčovny ověřuje vlastní databázi, ale nevolá API
platební brány, přes kterou zákazníci platí.

## --model--

Bez databáze aplikace nezvládne nic a restart nebo přesměrování provozu na jinou
instanci může pomoct. Výpadek platební brány restart nespraví. Kdyby health check na
bráně závisel, Docker by při jejím výpadku restartoval všechny zdravé kontejnery
a nešel by ani seznam kol. Navíc se health check volá často, takže má být rychlý.

## --checklist--

- Health check ověřuje to, co je v moci aplikace.
- Výpadek cizí služby restart aplikace nespraví.
- Závislost na cizí službě by restartovala zdravé instance.
- Health check se volá často, proto nesmí být pomalý.

// Veškeré nastavení aplikace se čte tady, na jednom místě, při startu.

export function loadConfig(env) {
  if (env.NODE_ENV === 'production' && !env.DATABASE_FILE) {
    throw new Error('Chybí proměnná prostředí DATABASE_FILE.');
  }
  return {
    port: Number(env.PORT ?? 3000),
    databaseFile: env.DATABASE_FILE ?? 'pujcovna-dev.db',
    logLevel: env.LOG_LEVEL ?? 'info',
    shutdownTimeoutMs: Number(env.SHUTDOWN_TIMEOUT_MS ?? 8000),
  };
}

## --file-- app.js

```js
// HTTP API půjčovny kol. Server (server.js) sem posílá každý požadavek.

async function readJson(req) {
  let body = '';
  for await (const chunk of req) body += chunk;
  return JSON.parse(body);
}

export function createApp({ db }) {
  function sendJson(res, status, data) {
    res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
    res.end(JSON.stringify(data));
  }

  async function handle(req, res) {
    const { pathname } = new URL(req.url, 'http://localhost');

    if (req.method === 'GET' && pathname === '/health') {
      return sendJson(res, 200, { status: 'ok' });
    }

    if (req.method === 'GET' && pathname === '/api/bikes') {
      const bikes = db
        .prepare('SELECT id, name, type, station, price_per_day AS pricePerDay FROM bikes ORDER BY id')
        .all();
      return sendJson(res, 200, bikes);
    }

    if (req.method === 'POST' && pathname === '/api/rentals') {
      let input;
      try {
        input = await readJson(req);
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku není platný JSON.' });
      }
      const bikeId = Number(input?.bikeId);
      const days = Number(input?.days);
      const customerName = String(input?.customerName ?? '').trim();
      if (!Number.isInteger(bikeId) || !Number.isInteger(days) || days < 1 || customerName === '') {
        return sendJson(res, 400, { error: 'Vyplň kolo, jméno a počet dní (aspoň 1).' });
      }
      const bike = db.prepare('SELECT id, price_per_day AS pricePerDay FROM bikes WHERE id = ?').get(bikeId);
      if (!bike) {
        return sendJson(res, 404, { error: 'Takové kolo v půjčovně nemáme.' });
      }
      const result = db
        .prepare('INSERT INTO rentals (bike_id, customer_name, days, created_at) VALUES (?, ?, ?, ?)')
        .run(bike.id, customerName, days, new Date().toISOString());
      return sendJson(res, 201, {
        id: Number(result.lastInsertRowid),
        bikeId: bike.id,
        days,
        totalPrice: bike.pricePerDay * days,
      });
    }

    sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
  }

  return (req, res) => {
    handle(req, res);
  };
}
```

## --file-- db.js

```js
import { DatabaseSync } from 'node:sqlite';

const BIKES = [
  ['Superior XC 879', 'horské', 'Brno – Výstaviště', 450],
  ['Author Mission', 'městské', 'Brno – Mendlovo náměstí', 290],
  ['Kellys Tyke 24', 'dětské', 'Brno – Výstaviště', 190],
  ['Specialized Turbo Vado', 'elektrokolo', 'Brno – Lužánky', 890],
  ['Cannondale Topstone', 'gravel', 'Brno – Lužánky', 520],
];

// Otevře databázi půjčovny. Při prvním spuštění založí tabulky a vloží kola.
export function openDatabase(file) {
  const db = new DatabaseSync(file);
  db.exec(`
    CREATE TABLE IF NOT EXISTS bikes (
      id INTEGER PRIMARY KEY,
      name TEXT NOT NULL,
      type TEXT NOT NULL,
      station TEXT NOT NULL,
      price_per_day INTEGER NOT NULL
    );
    CREATE TABLE IF NOT EXISTS rentals (
      id INTEGER PRIMARY KEY,
      bike_id INTEGER NOT NULL REFERENCES bikes (id),
      customer_name TEXT NOT NULL,
      days INTEGER NOT NULL,
      created_at TEXT NOT NULL
    );
  `);
  const { count } = db.prepare('SELECT COUNT(*) AS count FROM bikes').get();
  if (count === 0) {
    const insert = db.prepare('INSERT INTO bikes (name, type, station, price_per_day) VALUES (?, ?, ?, ?)');
    for (const bike of BIKES) insert.run(...bike);
  }
  return db;
}
```
