# --description--

Za pár kroků bude aplikace logovat každý požadavek. Stačí, aby někdo jednou zapsal
do polí celé hlavičky nebo tělo formuláře, a hesla i tokeny skončí v lozích na
měsíce. Spolehlivější než hlídat každé volání je naučit logger, že některé klíče
nikdy nevypíše.

**Úkol:** logger nahradí hodnotu pole s klíčem `password`, `token`, `authorization`
nebo `cookie` textem `[SKRYTO]`. Ostatní pole nechá, jak jsou. Objekt s poli, který
logger dostal, se nesmí změnit: volající ho může dál potřebovat (třeba hlavičky
požadavku).

Jak to uděláš, je na tobě.

# --hints--

Pole `password`, `token`, `authorization` a `cookie` mají v zapsaném řádku hodnotu `[SKRYTO]`.

```js
const { createLogger } = await helpers.importFile('logger.js');
const lines = [];
const logger = createLogger({ write: (line) => lines.push(line) });
logger.warn('Neúspěšné přihlášení', { password: 'Jaro2026!', token: 'b7f3e1', authorization: 'Bearer eyJhbGci', cookie: 'sid=91ac' });
assert.equal(lines.length, 1, 'logger.warn(…) má zapsat právě jeden řádek');
const entry = JSON.parse(lines[0]);
for (const key of ['password', 'token', 'authorization', 'cookie']) {
  assert.equal(entry[key], '[SKRYTO]', `Pole ${key} má být v řádku logu nahrazené textem [SKRYTO]`);
}
assert.doesNotMatch(lines[0], /Jaro2026!|b7f3e1|eyJhbGci|sid=91ac/, 'Řádek logu nesmí obsahovat žádnou z tajných hodnot');
```

Ostatní pole, úroveň a zpráva zůstanou beze změny.

```js
const { createLogger } = await helpers.importFile('logger.js');
const lines = [];
const logger = createLogger({ write: (line) => lines.push(line) });
logger.info('Přihlášení', { userId: 42, path: '/login', password: 'Jaro2026!' });
const entry = JSON.parse(lines[0]);
assert.equal(entry.level, 'info', 'Řádek má mít dál level "info"');
assert.equal(entry.msg, 'Přihlášení', 'Řádek má mít dál msg "Přihlášení"');
assert.equal(entry.userId, 42, 'Pole userId: 42 má zůstat beze změny');
assert.equal(entry.path, '/login', 'Pole path: "/login" má zůstat beze změny');
```

Objekt s poli, který logger dostal, zůstane beze změny.

```js
const { createLogger } = await helpers.importFile('logger.js');
const logger = createLogger({ write: () => {} });
const headers = { authorization: 'Bearer eyJhbGci', 'user-agent': 'Firefox' };
logger.info('Požadavek', headers);
assert.deepEqual(headers, { authorization: 'Bearer eyJhbGci', 'user-agent': 'Firefox' }, 'logger.info(…, headers) nemá měnit objekt headers — skryj hodnotu jen v tom, co se zapíše');
```

Logger bez polí (`logger.info('Server poslouchá')`) dál funguje.

```js
const { createLogger } = await helpers.importFile('logger.js');
const lines = [];
const logger = createLogger({ write: (line) => lines.push(line) });
logger.info('Server poslouchá');
assert.equal(JSON.parse(lines[0]).msg, 'Server poslouchá', 'logger.info("Server poslouchá") bez polí má zapsat řádek se zprávou');
```

# --help--

## --tip--

Co do logu nepatří a proč to hlídat na jednom místě, je v části
[Co do logu nepatří](see:nasazeni-provoz/logovani-a-chyby#co-do-logu-nepatri).

## --tip--

Změna patří do funkce `log` před složením řádku. Tajné klíče si drž v konstantě vedle
`LEVELS`. Hodnoty nepřepisuj přímo v `fields` — vyrob si nový objekt.

## --tip--

Stejná myšlenka na jiných datech: z objektu produktu vyrobit kopii, kde je `price`
zaokrouhlená, a originál nechat být — přes cyklus nad `Object.entries`, nebo přes
`Object.fromEntries` s `map`.

## --file-- logger.js

```js
// Strukturovaný logger: jedna událost = jeden řádek JSON na standardní výstup.

const LEVELS = ['debug', 'info', 'warn', 'error'];
const writeToStdout = (line) => process.stdout.write(`${line}\n`);

export function createLogger({ level = 'info', write = writeToStdout } = {}) {
  const minIndex = LEVELS.indexOf(level);

  function log(entryLevel, msg, fields = {}) {
    if (LEVELS.indexOf(entryLevel) < minIndex) return;
    const entry = { time: new Date().toISOString(), level: entryLevel, msg, ...fields };
    write(JSON.stringify(entry));
  }

  return {
    debug: (msg, fields) => log('debug', msg, fields),
    info: (msg, fields) => log('info', msg, fields),
    warn: (msg, fields) => log('warn', msg, fields),
    error: (msg, fields) => log('error', msg, fields),
  };
}
```

```js
export function loadConfig(env) {
  if (env.NODE_ENV === 'production' && !env.DATABASE_FILE) {
    throw new Error('Chybí proměnná prostředí DATABASE_FILE.');
  }
  return {
    port: Number(env.PORT ?? 3000),
    databaseFile: env.DATABASE_FILE ?? 'pujcovna-dev.db',
    logLevel: env.LOG_LEVEL ?? 'info',
    shutdownTimeoutMs: Number(env.SHUTDOWN_TIMEOUT_MS ?? 8000),
  };
}
```

## --file-- app.js

```js
// HTTP API půjčovny kol. Server (server.js) sem posílá každý požadavek.

async function readJson(req) {
  let body = '';
  for await (const chunk of req) body += chunk;
  return JSON.parse(body);
}

export function createApp({ db }) {
  function sendJson(res, status, data) {
    res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
    res.end(JSON.stringify(data));
  }

  async function handle(req, res) {
    const { pathname } = new URL(req.url, 'http://localhost');

    if (req.method === 'GET' && pathname === '/health') {
      try {
        db.prepare('SELECT 1').get();
        return sendJson(res, 200, { status: 'ok' });
      } catch {
        return sendJson(res, 503, { status: 'error' });
      }
    }

    if (req.method === 'GET' && pathname === '/api/bikes') {
      const bikes = db
        .prepare('SELECT id, name, type, station, price_per_day AS pricePerDay FROM bikes ORDER BY id')
        .all();
      return sendJson(res, 200, bikes);
    }

    if (req.method === 'POST' && pathname === '/api/rentals') {
      let input;
      try {
        input = await readJson(req);
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku není platný JSON.' });
      }
      const bikeId = Number(input?.bikeId);
      const days = Number(input?.days);
      const customerName = String(input?.customerName ?? '').trim();
      if (!Number.isInteger(bikeId) || !Number.isInteger(days) || days < 1 || customerName === '') {
        return sendJson(res, 400, { error: 'Vyplň kolo, jméno a počet dní (aspoň 1).' });
      }
      const bike = db.prepare('SELECT id, price_per_day AS pricePerDay FROM bikes WHERE id = ?').get(bikeId);
      if (!bike) {
        return sendJson(res, 404, { error: 'Takové kolo v půjčovně nemáme.' });
      }
      const result = db
        .prepare('INSERT INTO rentals (bike_id, customer_name, days, created_at) VALUES (?, ?, ?, ?)')
        .run(bike.id, customerName, days, new Date().toISOString());
      return sendJson(res, 201, {
        id: Number(result.lastInsertRowid),
        bikeId: bike.id,
        days,
        totalPrice: bike.pricePerDay * days,
      });
    }

    sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
  }

  return (req, res) => {
    handle(req, res);
  };
}
```

```js
import { DatabaseSync } from 'node:sqlite';

const BIKES = [
  ['Superior XC 879', 'horské', 'Brno – Výstaviště', 450],
  ['Author Mission', 'městské', 'Brno – Mendlovo náměstí', 290],
  ['Kellys Tyke 24', 'dětské', 'Brno – Výstaviště', 190],
  ['Specialized Turbo Vado', 'elektrokolo', 'Brno – Lužánky', 890],
  ['Cannondale Topstone', 'gravel', 'Brno – Lužánky', 520],
];

// Otevře databázi půjčovny. Při prvním spuštění založí tabulky a vloží kola.
export function openDatabase(file) {
  const db = new DatabaseSync(file);
  db.exec(`
    CREATE TABLE IF NOT EXISTS bikes (
      id INTEGER PRIMARY KEY,
      name TEXT NOT NULL,
      type TEXT NOT NULL,
      station TEXT NOT NULL,
      price_per_day INTEGER NOT NULL
    );
    CREATE TABLE IF NOT EXISTS rentals (
      id INTEGER PRIMARY KEY,
      bike_id INTEGER NOT NULL REFERENCES bikes (id),
      customer_name TEXT NOT NULL,
      days INTEGER NOT NULL,
      created_at TEXT NOT NULL
    );
  `);
  const { count } = db.prepare('SELECT COUNT(*) AS count FROM bikes').get();
  if (count === 0) {
    const insert = db.prepare('INSERT INTO bikes (name, type, station, price_per_day) VALUES (?, ?, ?, ?)');
    for (const bike of BIKES) insert.run(...bike);
  }
  return db;
}
```

## --file-- logger.js

```js
// Strukturovaný logger: jedna událost = jeden řádek JSON na standardní výstup.

const LEVELS = ['debug', 'info', 'warn', 'error'];
const SECRET_KEYS = ['password', 'token', 'authorization', 'cookie'];
const writeToStdout = (line) => process.stdout.write(`${line}\n`);

export function createLogger({ level = 'info', write = writeToStdout } = {}) {
  const minIndex = LEVELS.indexOf(level);

  function log(entryLevel, msg, fields = {}) {
    if (LEVELS.indexOf(entryLevel) < minIndex) return;
    const safeFields = {};
    for (const [key, value] of Object.entries(fields)) {
      safeFields[key] = SECRET_KEYS.includes(key) ? '[SKRYTO]' : value;
    }
    const entry = { time: new Date().toISOString(), level: entryLevel, msg, ...safeFields };
    write(JSON.stringify(entry));
  }

  return {
    debug: (msg, fields) => log('debug', msg, fields),
    info: (msg, fields) => log('info', msg, fields),
    warn: (msg, fields) => log('warn', msg, fields),
    error: (msg, fields) => log('error', msg, fields),
  };
}
```
