# --description--

Výchozí `pujcovna-dev.db` je pohodlná na notebooku. V produkci je nebezpečná: když
na hostingu zapomeneš nastavit `DATABASE_FILE`, aplikace si potichu založí novou
prázdnou databázi a rezervace se začnou ukládat jinam než včera. Nikdo si toho
nevšimne, dokud zákazníci nezačnou volat.

Lepší je [[rychlé selhání]]: když v produkci povinná hodnota chybí, aplikace odmítne
vůbec nastartovat.

**Úkol:** uprav `loadConfig(env)` tak, aby při `env.NODE_ENV === 'production'`
a chybějící `DATABASE_FILE` vyhodila `Error` se zprávou, která jmenuje chybějící
proměnnou (například `Chybí proměnná prostředí DATABASE_FILE.`). Mimo produkci
zůstane výchozí hodnota.

Stejná kontrola pro jinou aplikaci:

```js
if (env.NODE_ENV === 'production' && !env.SMTP_PASSWORD) {
  throw new Error('Chybí proměnná prostředí SMTP_PASSWORD.');
}
```

# --hints--

V produkci bez `DATABASE_FILE` vyhodí `loadConfig` chybu, která jmenuje `DATABASE_FILE`.

```js
const { loadConfig } = await helpers.importFile('config.js');
assert.throws(
  () => loadConfig({ NODE_ENV: 'production', PORT: '8080' }),
  /DATABASE_FILE/,
  'loadConfig({ NODE_ENV: "production", PORT: "8080" }) má vyhodit Error se zprávou, která obsahuje DATABASE_FILE',
);
```

Prázdná hodnota `DATABASE_FILE=` v produkci se počítá jako chybějící.

```js
const { loadConfig } = await helpers.importFile('config.js');
assert.throws(
  () => loadConfig({ NODE_ENV: 'production', DATABASE_FILE: '' }),
  /DATABASE_FILE/,
  'loadConfig({ NODE_ENV: "production", DATABASE_FILE: "" }) má vyhodit chybu — prázdný řetězec není cesta k databázi',
);
```

V produkci s `DATABASE_FILE` i mimo produkci bez ní vrátí `loadConfig` konfiguraci jako dřív.

```js
const { loadConfig } = await helpers.importFile('config.js');
const production = loadConfig({ NODE_ENV: 'production', DATABASE_FILE: '/data/pujcovna.db' });
assert.equal(production.databaseFile, '/data/pujcovna.db', 'loadConfig({ NODE_ENV: "production", DATABASE_FILE: "/data/pujcovna.db" }) má vrátit databaseFile "/data/pujcovna.db"');
let development;
try {
  development = loadConfig({});
} catch (error) {
  assert.fail(`loadConfig({}) mimo produkci nemá vyhodit chybu, vyhodila: ${error.message}`);
}
assert.deepEqual(
  development,
  { port: 3000, databaseFile: 'pujcovna-dev.db', logLevel: 'info', shutdownTimeoutMs: 8000 },
  'loadConfig({}) mimo produkci nemá vyhodit chybu a má vrátit výchozí hodnoty',
);
```

# --explain--

Vysvětli, proč je lepší, když aplikace bez `DATABASE_FILE` v produkci vůbec
nenastartuje, než aby použila výchozí soubor.

## --model--

S výchozím souborem by aplikace v produkci běžela nad novou prázdnou databází a chyba
by se ukázala až zákazníkům, třeba jako zmizelé rezervace. Když konfigurace chybí
a aplikace spadne při startu, nasazení selže hned, stará verze běží dál a v logu je
přesně napsané, která proměnná chybí.

## --checklist--

- Výchozí hodnota by v produkci potichu použila špatnou databázi.
- Chyba při startu se ukáže při nasazení, ne až uživatelům.
- Nová verze nenastartuje, takže stará běží dál.
- Zpráva chyby jmenuje proměnnou, kterou je potřeba doplnit.

import { createServer } from 'node:http';
import { openDatabase } from './db.js';
import { createApp } from './app.js';

const db = openDatabase('pujcovna-dev.db');
const server = createServer(createApp({ db }));

const port = Number(process.env.PORT ?? 3000);
server.listen(port, () => {
  console.log(`Půjčovna kol běží na portu ${port}`);
});

## --file-- app.js

```js
// HTTP API půjčovny kol. Server (server.js) sem posílá každý požadavek.

async function readJson(req) {
  let body = '';
  for await (const chunk of req) body += chunk;
  return JSON.parse(body);
}

export function createApp({ db }) {
  function sendJson(res, status, data) {
    res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
    res.end(JSON.stringify(data));
  }

  async function handle(req, res) {
    const { pathname } = new URL(req.url, 'http://localhost');

    if (req.method === 'GET' && pathname === '/api/bikes') {
      const bikes = db
        .prepare('SELECT id, name, type, station, price_per_day AS pricePerDay FROM bikes ORDER BY id')
        .all();
      return sendJson(res, 200, bikes);
    }

    if (req.method === 'POST' && pathname === '/api/rentals') {
      let input;
      try {
        input = await readJson(req);
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku není platný JSON.' });
      }
      const bikeId = Number(input?.bikeId);
      const days = Number(input?.days);
      const customerName = String(input?.customerName ?? '').trim();
      if (!Number.isInteger(bikeId) || !Number.isInteger(days) || days < 1 || customerName === '') {
        return sendJson(res, 400, { error: 'Vyplň kolo, jméno a počet dní (aspoň 1).' });
      }
      const bike = db.prepare('SELECT id, price_per_day AS pricePerDay FROM bikes WHERE id = ?').get(bikeId);
      if (!bike) {
        return sendJson(res, 404, { error: 'Takové kolo v půjčovně nemáme.' });
      }
      const result = db
        .prepare('INSERT INTO rentals (bike_id, customer_name, days, created_at) VALUES (?, ?, ?, ?)')
        .run(bike.id, customerName, days, new Date().toISOString());
      return sendJson(res, 201, {
        id: Number(result.lastInsertRowid),
        bikeId: bike.id,
        days,
        totalPrice: bike.pricePerDay * days,
      });
    }

    sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
  }

  return (req, res) => {
    handle(req, res);
  };
}
```

## --file-- db.js

```js
import { DatabaseSync } from 'node:sqlite';

const BIKES = [
  ['Superior XC 879', 'horské', 'Brno – Výstaviště', 450],
  ['Author Mission', 'městské', 'Brno – Mendlovo náměstí', 290],
  ['Kellys Tyke 24', 'dětské', 'Brno – Výstaviště', 190],
  ['Specialized Turbo Vado', 'elektrokolo', 'Brno – Lužánky', 890],
  ['Cannondale Topstone', 'gravel', 'Brno – Lužánky', 520],
];

// Otevře databázi půjčovny. Při prvním spuštění založí tabulky a vloží kola.
export function openDatabase(file) {
  const db = new DatabaseSync(file);
  db.exec(`
    CREATE TABLE IF NOT EXISTS bikes (
      id INTEGER PRIMARY KEY,
      name TEXT NOT NULL,
      type TEXT NOT NULL,
      station TEXT NOT NULL,
      price_per_day INTEGER NOT NULL
    );
    CREATE TABLE IF NOT EXISTS rentals (
      id INTEGER PRIMARY KEY,
      bike_id INTEGER NOT NULL REFERENCES bikes (id),
      customer_name TEXT NOT NULL,
      days INTEGER NOT NULL,
      created_at TEXT NOT NULL
    );
  `);
  const { count } = db.prepare('SELECT COUNT(*) AS count FROM bikes').get();
  if (count === 0) {
    const insert = db.prepare('INSERT INTO bikes (name, type, station, price_per_day) VALUES (?, ?, ?, ?)');
    for (const bike of BIKES) insert.run(...bike);
  }
  return db;
}
```

## --file-- config.js

```js
// Veškeré nastavení aplikace se čte tady, na jednom místě, při startu.

export function loadConfig(env) {
  if (env.NODE_ENV === 'production' && !env.DATABASE_FILE) {
    throw new Error('Chybí proměnná prostředí DATABASE_FILE.');
  }
  return {
    port: Number(env.PORT ?? 3000),
    databaseFile: env.DATABASE_FILE ?? 'pujcovna-dev.db',
    logLevel: env.LOG_LEVEL ?? 'info',
    shutdownTimeoutMs: Number(env.SHUTDOWN_TIMEOUT_MS ?? 8000),
  };
}
```
