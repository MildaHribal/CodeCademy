# --description--

Hosting, Docker i služba na hlídání dostupnosti se potřebují zeptat: „Žiješ?" Na to
slouží [[health check]], v téhle aplikaci na adrese `/health`.

**Úkol:** v `app.js` přidej do funkce `handle` obsluhu `GET /health`, která odpoví
stavem `200` a JSON `{ "status": "ok" }`. Dej ji na začátek, před ostatní adresy.
Databázi zatím neověřuj, to přijde v dalším kroku.

Stejný tvar má v jiném API adresa s verzí:

```js
if (req.method === 'GET' && pathname === '/version') {
  return sendJson(res, 200, { version: '2.4.0' });
}
```

# --hints--

`GET /health` vrátí stav `200` a JSON `{ "status": "ok" }`.

```js
const server = await helpers.startServer('server.js', { timeoutMs: 4000 });
const res = await fetch(new URL('/health', server.url));
assert.equal(res.status, 200, 'GET /health má vrátit stav 200');
assert.deepEqual(await res.json(), { status: 'ok' }, 'GET /health má vrátit JSON { status: "ok" }');
```

Ostatní adresy fungují jako dřív.

```js
const server = await helpers.startServer('server.js', { timeoutMs: 4000 });
const bikes = await fetch(new URL('/api/bikes', server.url));
assert.equal((await bikes.json()).length, 5, 'GET /api/bikes má dál vracet pět kol');
const missing = await fetch(new URL('/healthz', server.url));
assert.equal(missing.status, 404, 'GET /healthz (neexistující adresa) má dál vracet 404');
```

const BIKES = [

  ['Cannondale Topstone', 'gravel', 'Brno – Lužánky', 520],
];

// Otevře databázi půjčovny. Při prvním spuštění založí tabulky a vloží kola.
export function openDatabase(file) {
  const db = new DatabaseSync(file);
  db.exec(`
    CREATE TABLE IF NOT EXISTS bikes (
      id INTEGER PRIMARY KEY,
      name TEXT NOT NULL,
      type TEXT NOT NULL,
      station TEXT NOT NULL,
      price_per_day INTEGER NOT NULL
    );
    CREATE TABLE IF NOT EXISTS rentals (
      id INTEGER PRIMARY KEY,
      bike_id INTEGER NOT NULL REFERENCES bikes (id),
      customer_name TEXT NOT NULL,
      days INTEGER NOT NULL,
      created_at TEXT NOT NULL
    );
  `);
  const { count } = db.prepare('SELECT COUNT(*) AS count FROM bikes').get();
  if (count === 0) {
    const insert = db.prepare('INSERT INTO bikes (name, type, station, price_per_day) VALUES (?, ?, ?, ?)');
    for (const bike of BIKES) insert.run(...bike);
  }
  return db;
}

async function readJson(req) {
  let body = '';
  for await (const chunk of req) body += chunk;
  return JSON.parse(body);
}

export function createApp({ db }) {
  function sendJson(res, status, data) {
    res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
    res.end(JSON.stringify(data));
  }

  async function handle(req, res) {
    const { pathname } = new URL(req.url, 'http://localhost');

    if (req.method === 'GET' && pathname === '/health') {
      return sendJson(res, 200, { status: 'ok' });
    }

    if (req.method === 'GET' && pathname === '/api/bikes') {
      const bikes = db
        .prepare('SELECT id, name, type, station, price_per_day AS pricePerDay FROM bikes ORDER BY id')
