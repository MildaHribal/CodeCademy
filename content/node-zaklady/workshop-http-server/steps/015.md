# --description--

Knihovna bude přijímat nové knihy: klient pošle `POST /api/books` a v těle požadavku
JSON s knihou. [[Tělo požadavku]] ale nepřijde najednou — `req` je [[proud]] kousků
bajtů, které se musí posbírat a teprve celek převést na text.

Funkci, která to umí, dáš do nového souboru `body.js`, ať ji může použít každá routa.

**Úkol:** seřaď řádky tak, aby `readBody(req)` prošla všechny kousky těla a vrátila
celé tělo jako text. Odsazení nastavíš klávesou Tab. Dva řádky jsou navíc a na
jednom místě doplníš kódování.

> [!PITFALL]
> Kousky se na text převádějí až spojené. Písmeno `Č` má v UTF-8 dva bajty a hranice
> kousku může padnout mezi ně — převod po kouscích z něj udělá `��`.

# --hints--

`body.js` exportuje asynchronní funkci `readBody`, která vrací Promise.

```js
const module = await helpers.importFile('body.js');
assert.equal(typeof module.readBody, 'function', 'body.js má exportovat funkci readBody');
async function* empty() {}
const result = module.readBody(empty());
assert.ok(result instanceof Promise, 'readBody(…) má vracet Promise — je funkce async?');
await result;
```

`readBody` projde všechny kousky a vrátí je spojené do jednoho textu.

```js
const { readBody } = await helpers.importFile('body.js');
async function* pieces() {
  yield Buffer.from('{"title":');
  yield Buffer.from('"Máj",');
  yield Buffer.from('"year":1836}');
}
assert.equal(await readBody(pieces()), '{"title":"Máj","year":1836}', 'readBody(…) se třemi kousky má vrátit celý text {"title":"Máj","year":1836}');
```

Písmeno rozdělené mezi dva kousky zůstane celé.

```js
const { readBody } = await helpers.importFile('body.js');
const bytes = Buffer.from('Karel Čapek');
// Hranice kousku padne doprostřed dvou bajtů písmene Č.
const cut = bytes.indexOf(0xc4) + 1;
async function* pieces() {
  yield bytes.subarray(0, cut);
  yield bytes.subarray(cut);
}
assert.equal(await readBody(pieces()), 'Karel Čapek', 'readBody(…) s Č rozděleným mezi dva kousky má vrátit „Karel Čapek" — převádíš na text až spojené bajty?');
```

Prázdné tělo vrátí prázdný řetězec.

```js
const { readBody } = await helpers.importFile('body.js');
async function* empty() {}
assert.equal(await readBody(empty()), '', 'readBody(…) bez kousků má vrátit prázdný řetězec ""');
```

# --explain--

Vysvětli vlastními slovy, proč `readBody` kousky nejdřív spojí přes `Buffer.concat` a na text je převede až potom.

## --model--

Tělo požadavku přichází jako proud kousků bajtů a hranice kousků nerespektuje
znaky. Znak jako `Č` zabírá v UTF-8 dva bajty a může se rozdělit mezi dva kousky.
Kdybych každý kousek převáděl na text zvlášť, z půlek by vznikly neplatné znaky.
Spojené bajty už obsahují celé znaky, takže převod celku je správně.

## --checklist--

- Tělo přichází po kouscích bajtů (proud).
- Znak v UTF-8 může mít víc bajtů a rozdělit se mezi dva kousky.
- Převod půlky znaku na text dá neplatný znak.
- Na text se proto převádí až spojené bajty.

export async function readBody(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString(__1__);
}

## --file-- server.js

```js
import { createServer } from 'node:http';
import { findBook, loadBooks } from './library.js';
import { handleStats } from './stats.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

const server = createServer(async (req, res) => {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      const books = await loadBooks();
      const author = searchParams.get('author');
      if (author === null) {
        return sendJson(res, 200, books);
      }
      return sendJson(res, 200, books.filter((book) => book.author === author));
    }
    res.setHeader('Allow', 'GET');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const book = findBook(await loadBooks(), id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    return sendJson(res, 200, book);
  }

  if (pathname === '/api/authors') {
    const books = await loadBooks();
    const authors = [...new Set(books.map((book) => book.author))];
    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
  }

  if (pathname === '/api/stats') {
    return handleStats(req, res, await loadBooks());
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
});

server.listen(port, () => {
  console.log(`Server běží na http://localhost:${port}`);
});
```

## --file-- library.js

```js
import { readFile, writeFile } from 'node:fs/promises';

// Funkce pro práci s knihami. Server je bude importovat.

// books.json leží vedle tohoto souboru. Cesta postavená od import.meta.url
// funguje, ať spustíš node z jakékoli složky.
const booksFile = new URL('./books.json', import.meta.url);

export async function loadBooks() {
  const text = await readFile(booksFile, 'utf8');
  return JSON.parse(text);
}

export async function saveBooks(books) {
  await writeFile(booksFile, JSON.stringify(books, null, 2));
}

export function findBook(books, id) {
  return books.find((book) => book.id === id);
}
```

## --file-- stats.js

```js
// Statistiky knihovny pro GET /api/stats — dopsal kolega Tomáš.
// Nepovinný filtr ?author=… spočítá statistiky jen z knih jednoho autora.

function respond(res, code, payload) {
  res.writeHead(code, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(payload));
}

export function handleStats(req, res, allBooks) {
  const params = new URL(req.url, 'http://localhost').searchParams;
  const author = params.get('author');

  const picked = [];
  for (const b of allBooks) {
    if (author === null || b.author === author) {
      picked.push(b);
    }
  }

  if (picked.length === 0) {
    return respond(res, 404, { error: 'Tento autor v knihovně nemá žádnou knihu.' });
  }

  const years = [];
  for (const b of picked) {
    years.push(b.year);
  }

  respond(res, 200, {
    count: picked.length,
    firstYear: Math.min(...years),
    lastYear: Math.max(...years),
  });
}
```
