# --description--

Knihovna má přijímat nové knihy: klient pošle `POST /api/books` a v **těle
požadavku** JSON s knihou. V tomhle kroku tělo přečteš a vytvoříš z něj novou
knihu. Uložení do souboru přidáš v dalším kroku.

U odpovědi jsi tělo posílal přes `res.end`. U požadavku ho musíš **přečíst** a to
v Node není jedna vlastnost. Tělo může být velké (třeba nahrávaný soubor), proto
nepřijde najednou, ale jako **proud** (*stream*) kousků bajtů (*chunks*). Objekt
`req` je takový proud a projdeš ho cyklem `for await…of`: každý průchod čeká na
další kousek, a když tělo skončí, cyklus skončí taky.

Na obecné funkci, kterou použiješ pro jakýkoli požadavek:

```js
async function readBody(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString('utf8');
}
```

Kousky jsou `Buffer` (bajty). Nejdřív je spoj a teprve celek převeď na text.
Kdybys každý kousek hned lepil do řetězce, písmeno s háčkem rozdělené mezi dva
kousky by se rozbilo.

V `server.js`:

1. přidej funkci `readBody(req)` (klidně přesně tu z příkladu),
2. v routě `/api/books` obsluž metodu `POST`: přečti tělo, převeď ho přes
   `JSON.parse` a vytvoř z něj novou knihu,
3. odpověz stavem `201 Created` a nově vytvořenou knihou,
4. hlavička `Allow` u odpovědi `405` teď musí uvádět `GET, POST`.

Nová kniha má klíče `title`, `author` a `year` z těla požadavku a `id`, které
**přidělí server**: o jedna vyšší než nejvyšší id v knihovně. Id z těla požadavku
nepoužívej. Klient by mohl poslat id existující knihy a v knihovně by pak byly dvě
se stejným id. Proto knihu skládej po klíčích, ne jako `{ ...input }`.

Na jiných datech (komentáře):

```js
if (req.method === 'POST') {
  const input = JSON.parse(await readBody(req));
  const comments = await loadComments();
  const comment = {
    id: Math.max(0, ...comments.map((item) => item.id)) + 1,
    text: input.text,
  };
  return sendJson(res, 201, comment);
}
```

`Math.max` nebere pole, ale jednotlivé argumenty, proto se pole id rozprostře
třemi tečkami. Nula na začátku zajistí, že i v prázdné knihovně dostane první
kniha id `1` (samotné `Math.max()` bez argumentů je `-Infinity`).

Kniha se zatím nikam neukládá: server odpoví, jako by ji vytvořil, ale
`GET /api/books` ji neukáže. To napravíš hned v dalším kroku.

# --hints--

`POST /api/books` s knihou v těle vrátí stav `201` a vytvořenou knihu s číselným `id`.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const input = { title: 'R.U.R.', author: 'Karel Čapek', year: 1920 };
  const res = await fetch(new URL('/api/books', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  });
  assert.equal(res.status, 201, 'POST /api/books má vrátit 201');
  const book = await res.json();
  assert.equal(typeof book.id, 'number', 'Vytvořená kniha má mít číselné id');
  assert.equal(book.title, input.title);
  assert.equal(book.author, input.author);
  assert.equal(book.year, input.year);
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

Id nové knihy přidělí server — je vyšší než všechna existující, i když klient pošle vlastní `id`.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const res = await fetch(new URL('/api/books', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ id: 1, title: 'Bílá nemoc', author: 'Karel Čapek', year: 1937 }),
  });
  const book = await res.json();
  const maxId = Math.max(...JSON.parse(files['books.json']).map((item) => item.id));
  assert.equal(book.id, maxId + 1, `Nová kniha má dostat id ${maxId + 1}, dostala ${book.id}`);
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

Odpověď `405` (třeba na `DELETE /api/books`) má v hlavičce `Allow` metody `GET` i `POST`.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(new URL('/api/books', server.url), { method: 'DELETE' });
assert.equal(res.status, 405);
const allow = res.headers.get('allow') ?? '';
assert.match(allow, /\bGET\b/, `Allow má obsahovat GET, přišlo: ${allow}`);
assert.match(allow, /\bPOST\b/, `Allow má obsahovat POST, přišlo: ${allow}`);
```

```js
import { readFile, writeFile } from 'node:fs/promises';

// books.json leží vedle tohoto souboru. Cesta postavená od import.meta.url
// funguje, ať spustíš node z jakékoli složky.
const booksFile = new URL('./books.json', import.meta.url);

export async function loadBooks() {
  const text = await readFile(booksFile, 'utf8');
  return JSON.parse(text);
}

export async function saveBooks(books) {
  await writeFile(booksFile, JSON.stringify(books, null, 2));
}
```

# --solution--

## --file-- server.js

```js
import { createServer } from 'node:http';
import { loadBooks } from './library.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Přečte celé tělo požadavku jako text. Tělo přichází po kouscích (stream).
async function readBody(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString('utf8');
}

const server = createServer(async (req, res) => {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      return sendJson(res, 200, await loadBooks());
    }
    if (req.method === 'POST') {
      const input = JSON.parse(await readBody(req));
      const books = await loadBooks();
      const book = {
        id: Math.max(0, ...books.map((item) => item.id)) + 1,
        title: input.title,
        author: input.author,
        year: input.year,
      };
      return sendJson(res, 201, book);
    }
    res.setHeader('Allow', 'GET, POST');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const books = await loadBooks();
    const book = books.find((item) => item.id === id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    return sendJson(res, 200, book);
  }

  if (pathname === '/api/authors') {
    const books = await loadBooks();
    const authors = [...new Set(books.map((book) => book.author))];
    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
});

server.listen(port, () => {
  console.log(`Server běží na http://localhost:${port}`);
});
```
