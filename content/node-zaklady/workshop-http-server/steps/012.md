# --description--

Server knihu vytvoří a hned zapomene. Teď ji uložíš funkcí `saveBooks`, kterou
máš v `library.js` z desátého kroku.

V `server.js`:

1. do importu z `./library.js` přidej `saveBooks`,
2. v obsluze `POST` přidej novou knihu do pole `books` a celé pole ulož přes
   `await saveBooks(books)` — **dřív**, než pošleš odpověď.

Proč až před odpovědí a proč s `await`? Stav `201` klientovi tvrdí, že kniha
existuje. Kdybys odpověděl a zápis by pak selhal (plný disk, chybějící práva),
klient by si myslel, že je kniha uložená, a ona by nebyla.

`push` tu nevadí, přestože pole jinak zbytečně neměníme: pole `books` si handler
právě načetl ze souboru a nikdo jiný ho nepoužívá. Stejně dobře poslouží
`saveBooks([...books, book])`.

Na jiných datech (komentáře):

```js
const comments = await loadComments();
const comment = { id: 4, text: input.text };
comments.push(comment);
await saveComments(comments);
return sendJson(res, 201, comment);
```

Ve vlastním terminálu si to vyzkoušíš takhle:

```sh
curl -X POST http://localhost:3000/api/books \
  -H 'Content-Type: application/json' \
  -d '{"title": "R.U.R.", "author": "Karel Čapek", "year": 1920}'
```

Pak otevři `http://localhost:3000/api/books` nebo soubor `books.json` — nová kniha
tam je a zůstane tam i po restartu serveru.

# --hints--

Nová kniha se uloží: je v `books.json` a vrátí ji i `GET /api/books`.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const input = { title: 'Povídky z jedné kapsy', author: 'Karel Čapek', year: 1929 };
  await fetch(new URL('/api/books', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  });
  const saved = JSON.parse(await fs.readFile(booksPath, 'utf8'));
  assert.equal(saved.length, JSON.parse(files['books.json']).length + 1, 'books.json má mít o jednu knihu víc');
  assert.ok(saved.some((book) => book.title === input.title), 'Nová kniha má být v books.json');
  const list = await (await fetch(new URL('/api/books', server.url))).json();
  assert.ok(list.some((book) => book.title === input.title), 'GET /api/books má vrátit i novou knihu');
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

`POST /api/books` s knihou v těle vrátí stav `201` a vytvořenou knihu s číselným `id`.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const input = { title: 'R.U.R.', author: 'Karel Čapek', year: 1920 };
  const res = await fetch(new URL('/api/books', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  });
  assert.equal(res.status, 201, 'POST /api/books má vrátit 201');
  const book = await res.json();
  assert.equal(typeof book.id, 'number', 'Vytvořená kniha má mít číselné id');
  assert.equal(book.title, input.title);
  assert.equal(book.author, input.author);
  assert.equal(book.year, input.year);
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

```js
import { readFile, writeFile } from 'node:fs/promises';

// books.json leží vedle tohoto souboru. Cesta postavená od import.meta.url
// funguje, ať spustíš node z jakékoli složky.
const booksFile = new URL('./books.json', import.meta.url);

export async function loadBooks() {
  const text = await readFile(booksFile, 'utf8');
  return JSON.parse(text);
}

export async function saveBooks(books) {
  await writeFile(booksFile, JSON.stringify(books, null, 2));
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
```
```js
import { loadBooks, saveBooks } from './library.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Přečte celé tělo požadavku jako text. Tělo přichází po kouscích (stream).
async function readBody(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString('utf8');
}

const server = createServer(async (req, res) => {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      return sendJson(res, 200, await loadBooks());
    }
    if (req.method === 'POST') {
      const input = JSON.parse(await readBody(req));
      const books = await loadBooks();
      const book = {
        id: Math.max(0, ...books.map((item) => item.id)) + 1,
        title: input.title,
        author: input.author,
        year: input.year,
      };
      books.push(book);
      await saveBooks(books);
      return sendJson(res, 201, book);
    }
    res.setHeader('Allow', 'GET, POST');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const books = await loadBooks();
    const book = books.find((item) => item.id === id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    return sendJson(res, 200, book);
  }

  if (pathname === '/api/authors') {
    const books = await loadBooks();
    const authors = [...new Set(books.map((book) => book.author))];
    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
});

server.listen(port, () => {
  console.log(`Server běží na http://localhost:${port}`);
});
```
