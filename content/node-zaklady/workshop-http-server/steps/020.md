# --description--

Opakování bez návodu — všechno, co potřebuješ, už v serveru máš.

**Úkol:** doplň do `server.js` mazání knih:

- `DELETE /api/books/:id` knihu smaže ze souboru a odpoví stavem `204` bez těla,
- když kniha s tím id neexistuje, odpoví `404` a JSON s klíčem `error` a soubor nezmění,
- `GET /api/books/:id` funguje dál,
- jiná metoda na detailu knihy (třeba `PATCH`) dostane `405` a hlavičku `Allow`
  s metodami `GET` a `DELETE`.

Odpověď `204` nemá tělo, a tak nepotřebuje ani hlavičku `Content-Type`.

# --hints--

`DELETE /api/books/:id` vrátí `204` bez těla a kniha zmizí ze souboru i z detailu.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const res = await fetch(new URL('/api/books/2', server.url), { method: 'DELETE' });
  assert.equal(res.status, 204, 'DELETE /api/books/2 má vrátit stav 204');
  assert.equal(await res.text(), '', 'Odpověď 204 na DELETE /api/books/2 nemá mít tělo');
  const saved = JSON.parse(await fs.readFile(booksPath, 'utf8'));
  assert.deepEqual(saved.map((book) => book.id), [1, 3, 4, 5], 'Po DELETE /api/books/2 mají v books.json zůstat knihy 1, 3, 4 a 5');
  const detail = await fetch(new URL('/api/books/2', server.url));
  assert.equal(detail.status, 404, 'Po smazání má GET /api/books/2 vrátit stav 404');
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

`DELETE` neexistující knihy vrátí `404` s klíčem `error` a soubor nezmění.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const res = await fetch(new URL('/api/books/99', server.url), { method: 'DELETE' });
  assert.equal(res.status, 404, 'DELETE /api/books/99 má vrátit stav 404');
  assert.equal(typeof (await res.json()).error, 'string', 'Tělo odpovědi na DELETE /api/books/99 má mít klíč error');
  const saved = JSON.parse(await fs.readFile(booksPath, 'utf8'));
  assert.deepEqual(saved, JSON.parse(files['books.json']), 'Po DELETE /api/books/99 se books.json nemá změnit');
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

`GET /api/books/:id` dál vrací knihu.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(new URL('/api/books/3', server.url));
assert.equal(res.status, 200, 'GET /api/books/3 má dál vracet stav 200');
assert.equal((await res.json()).title, 'Válka s mloky', 'GET /api/books/3 má dál vracet Válku s mloky');
```

Jiná metoda na detailu knihy vrátí `405` s hlavičkou `Allow: GET, DELETE`.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const res = await fetch(new URL('/api/books/1', server.url), { method: 'PATCH' });
  assert.equal(res.status, 405, 'PATCH /api/books/1 má vrátit stav 405');
  const allow = res.headers.get('allow') ?? '';
  assert.match(allow, /\bGET\b/, `Allow u PATCH /api/books/1 má obsahovat GET, přišlo: ${allow}`);
  assert.match(allow, /\bDELETE\b/, `Allow u PATCH /api/books/1 má obsahovat DELETE, přišlo: ${allow}`);
  const saved = JSON.parse(await fs.readFile(booksPath, 'utf8'));
  assert.deepEqual(saved, JSON.parse(files['books.json']), 'PATCH /api/books/1 nemá books.json změnit');
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

# --help--

## --tip--

`204` a `405` najdeš ve [Stavových kódech](see:node-zaklady/http-v-node#stavove-kody);
tvar routy podle metody máš u `/api/books` z desátého kroku.

## --tip--

V routě detailu rozhoduj podle `req.method`. Pro mazání vyrob `filter` pole bez knihy
a ulož ho přes `saveBooks`; odpověď `204` pošleš přes `res.writeHead` a prázdné `res.end`.

## --tip--

Stejné mazání u objednávek: `await saveOrders(orders.filter((order) => order.id !== orderId));`
a pak odpověď bez těla.

```js
import { createServer } from 'node:http';
import { readBody } from './body.js';
import { findBook, loadBooks, saveBooks } from './library.js';
import { handleStats } from './stats.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Vrátí popis první chyby ve vstupu, nebo null, když je kniha v pořádku.
function validateBook(input) {
  if (typeof input !== 'object' || input === null) {
    return 'Tělo požadavku musí být JSON objekt.';
  }
  if (typeof input.title !== 'string' || input.title.trim() === '') {
    return 'Kniha musí mít název (title).';
  }
  if (typeof input.author !== 'string' || input.author.trim() === '') {
    return 'Kniha musí mít autora (author).';
  }
  if (!Number.isInteger(input.year)) {
    return 'Rok vydání (year) musí být celé číslo.';
  }
  return null;
}

const server = createServer(async (req, res) => {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      const books = await loadBooks();
      const author = searchParams.get('author');
      if (author === null) {
        return sendJson(res, 200, books);
      }
      return sendJson(res, 200, books.filter((book) => book.author === author));
    }
    if (req.method === 'POST') {
      let input;
      try {
        input = JSON.parse(await readBody(req));
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku musí být platný JSON.' });
      }

      const problem = validateBook(input);
      if (problem) {
        return sendJson(res, 400, { error: problem });
      }

      const books = await loadBooks();
      const book = {
        id: Math.max(0, ...books.map((item) => item.id)) + 1,
        title: input.title,
        author: input.author,
        year: input.year,
      };
      books.push(book);
      await saveBooks(books);
      return sendJson(res, 201, book);
    }
    res.setHeader('Allow', 'GET, POST');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const book = findBook(await loadBooks(), id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    return sendJson(res, 200, book);
  }

  if (pathname === '/api/authors') {
    const books = await loadBooks();
    const authors = [...new Set(books.map((book) => book.author))];
    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
  }

  if (pathname === '/api/stats') {
    return handleStats(req, res, await loadBooks());
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
});

server.listen(port, () => {
  console.log(`Server běží na http://localhost:${port}`);
});
```

## --file-- library.js

```js
import { readFile, writeFile } from 'node:fs/promises';

// Funkce pro práci s knihami. Server je bude importovat.

// books.json leží vedle tohoto souboru. Cesta postavená od import.meta.url
// funguje, ať spustíš node z jakékoli složky.
const booksFile = new URL('./books.json', import.meta.url);

export async function loadBooks() {
  const text = await readFile(booksFile, 'utf8');
  return JSON.parse(text);
}

export async function saveBooks(books) {
  await writeFile(booksFile, JSON.stringify(books, null, 2));
}

export function findBook(books, id) {
  return books.find((book) => book.id === id);
}
```

```js
// Statistiky knihovny pro GET /api/stats — dopsal kolega Tomáš.
// Nepovinný filtr ?author=… spočítá statistiky jen z knih jednoho autora.

function respond(res, code, payload) {
  res.writeHead(code, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(payload));
}

export function handleStats(req, res, allBooks) {
  const params = new URL(req.url, 'http://localhost').searchParams;
  const author = params.get('author');

  const picked = [];
  for (const b of allBooks) {
    if (author === null || b.author === author) {
      picked.push(b);
    }
  }

  if (picked.length === 0) {
    return respond(res, 404, { error: 'Tento autor v knihovně nemá žádnou knihu.' });
  }

  const years = [];
  for (const b of picked) {
    years.push(b.year);
  }

  respond(res, 200, {
    count: picked.length,
    firstYear: Math.min(...years),
    lastYear: Math.max(...years),
  });
}
```

```js
// Přečte celé tělo požadavku jako text. Tělo přichází po kouscích (proud).
export async function readBody(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString('utf8');
}
```

```js
import { createServer } from 'node:http';
import { readBody } from './body.js';
import { findBook, loadBooks, saveBooks } from './library.js';
import { handleStats } from './stats.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Vrátí popis první chyby ve vstupu, nebo null, když je kniha v pořádku.
function validateBook(input) {
  if (typeof input !== 'object' || input === null) {
    return 'Tělo požadavku musí být JSON objekt.';
  }
  if (typeof input.title !== 'string' || input.title.trim() === '') {
    return 'Kniha musí mít název (title).';
  }
  if (typeof input.author !== 'string' || input.author.trim() === '') {
    return 'Kniha musí mít autora (author).';
  }
  if (!Number.isInteger(input.year)) {
    return 'Rok vydání (year) musí být celé číslo.';
  }
  return null;
}

const server = createServer(async (req, res) => {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      const books = await loadBooks();
      const author = searchParams.get('author');
      if (author === null) {
        return sendJson(res, 200, books);
      }
      return sendJson(res, 200, books.filter((book) => book.author === author));
    }
    if (req.method === 'POST') {
      let input;
      try {
        input = JSON.parse(await readBody(req));
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku musí být platný JSON.' });
      }

      const problem = validateBook(input);
      if (problem) {
        return sendJson(res, 400, { error: problem });
      }

      const books = await loadBooks();
      const book = {
        id: Math.max(0, ...books.map((item) => item.id)) + 1,
        title: input.title,
        author: input.author,
        year: input.year,
      };
      books.push(book);
      await saveBooks(books);
      return sendJson(res, 201, book);
    }
    res.setHeader('Allow', 'GET, POST');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const books = await loadBooks();
    const book = findBook(books, id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    if (req.method === 'GET') {
      return sendJson(res, 200, book);
    }
    if (req.method === 'DELETE') {
      await saveBooks(books.filter((item) => item.id !== id));
      res.writeHead(204);
      return res.end();
    }
    res.setHeader('Allow', 'GET, DELETE');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname === '/api/authors') {
    const books = await loadBooks();

    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
```
