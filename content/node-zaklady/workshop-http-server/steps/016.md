# --description--

Funkci `readBody` teď použiješ.

**Úkol:** `POST /api/books` s knihou v těle (JSON s klíči `title`, `author` a `year`)
vytvoří novou knihu a odpoví stavem `201` a touto knihou. Knihu zatím nikam
neukládej — to je další krok.

- `id` nové knihy přidělí **server**: o jedna vyšší než nejvyšší id v knihovně.
  Id, které pošle klient, se nepoužije.
- Nová kniha má přesně klíče `id`, `title`, `author` a `year` — nic dalšího, co
  klient v těle poslal, do ní nepatří.
- Hlavička `Allow` u odpovědi `405` teď uvádí `GET, POST`.

> [!PITFALL]
> Kopie celého těla do nové knihy by pustila do knihovny cokoli, co klient pošle —
> třeba vlastní `id` nebo klíč `"admin": true`. Knihu skládej po klíčích.

> [!NOTE]
> „Nejvyšší id plus jedna" po smazání poslední knihy přidělí její id znovu. Pro
> knihovnu to stačí; v projektu na konci sekce potkáš lepší řešení.

Vyzkoušej to v HTTP klientovi: metoda `POST`, cesta `/api/books` a v části
**Hlavičky a tělo** tělo `{"title": "R.U.R.", "author": "Karel Čapek", "year": 1920}`.

# --hints--

`POST /api/books` s knihou v těle vrátí stav `201` a vytvořenou knihu s číselným `id`.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const postBook = (body) => fetch(new URL('/api/books', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: typeof body === 'string' ? body : JSON.stringify(body),
  });
  const input = { title: 'R.U.R.', author: 'Karel Čapek', year: 1920 };
  const res = await postBook(input);
  assert.equal(res.status, 201, `POST /api/books s ${JSON.stringify(input)} má vrátit stav 201`);
  const book = await res.json();
  assert.equal(typeof book.id, 'number', `Vytvořená kniha má mít číselné id, přišlo: ${JSON.stringify(book)}`);
  assert.equal(book.title, input.title, 'Vytvořená kniha má mít title z těla požadavku');
  assert.equal(book.author, input.author, 'Vytvořená kniha má mít author z těla požadavku');
  assert.equal(book.year, input.year, 'Vytvořená kniha má mít year z těla požadavku');
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

Id nové knihy přidělí server — o jedna vyšší než nejvyšší, i když klient pošle vlastní `id`.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const postBook = (body) => fetch(new URL('/api/books', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: typeof body === 'string' ? body : JSON.stringify(body),
  });
  const res = await postBook({ id: 1, title: 'Bílá nemoc', author: 'Karel Čapek', year: 1937 });
  const book = await res.json();
  const maxId = Math.max(...JSON.parse(files['books.json']).map((item) => item.id));
  assert.equal(book.id, maxId + 1, `Nová kniha má dostat id ${maxId + 1} (nejvyšší id + 1), dostala ${book.id}`);
  // Knihovna s mezerami v id: počet knih + 1 by dal id, které už existuje.
  await fs.writeFile(booksPath, JSON.stringify([
    { id: 1, title: 'Saturnin', author: 'Zdeněk Jirotka', year: 1942 },
    { id: 8, title: 'Krakatit', author: 'Karel Čapek', year: 1924 },
  ]));
  const next = await (await postBook({ title: 'Máj', author: 'Karel Hynek Mácha', year: 1836 })).json();
  assert.equal(next.id, 9, `V knihovně s id 1 a 8 má nová kniha dostat id 9 (nejvyšší + 1), dostala ${next.id}`);
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

Nová kniha má jen klíče `id`, `title`, `author` a `year`.

```js

  assert.deepEqual(Object.keys(book).toSorted(), ['author', 'id', 'title', 'year'], `POST s klíčem admin navíc má vytvořit knihu jen s id, title, author a year, přišlo: ${JSON.stringify(book)}`);

  await fs.writeFile(booksPath, files['books.json']);
}
```

Odpověď `405` (třeba na `DELETE /api/books`) má v hlavičce `Allow` metody `GET` i `POST`.

```js
const server = await helpers.startServer('server.js');

assert.equal(res.status, 405, 'DELETE /api/books má vrátit stav 405');

assert.match(allow, /\bGET\b/, `Allow u DELETE /api/books má obsahovat GET, přišlo: ${allow}`);
```

## --tip--

V routě `/api/books` přidej větev pro `req.method` `'POST'` před odpověď `405`.
V ní text z `readBody` (importuj ji z `./body.js`) převeď `JSON.parse`, načti knihy
a nejvyšší id najdi přes `Math.max` s rozprostřeným polem id. Pozor:
`Math.max()` bez argumentů je `-Infinity`.

## --tip--

Stejný postup pro komentáře:

```js

return sendJson(response, 201, comment);
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { findBook, loadBooks } from './library.js';
import { handleStats } from './stats.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

const server = createServer(async (req, res) => {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      const books = await loadBooks();
      const author = searchParams.get('author');
      if (author === null) {
        return sendJson(res, 200, books);
      }
      return sendJson(res, 200, books.filter((book) => book.author === author));
    }
    res.setHeader('Allow', 'GET');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const book = findBook(await loadBooks(), id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    return sendJson(res, 200, book);
  }

  if (pathname === '/api/authors') {
    const books = await loadBooks();
    const authors = [...new Set(books.map((book) => book.author))];
    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
  }

  if (pathname === '/api/stats') {
    return handleStats(req, res, await loadBooks());
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
});

server.listen(port, () => {
  console.log(`Server běží na http://localhost:${port}`);
});
```

## --file-- library.js

```js
import { readFile, writeFile } from 'node:fs/promises';

// Funkce pro práci s knihami. Server je bude importovat.

// books.json leží vedle tohoto souboru. Cesta postavená od import.meta.url
// funguje, ať spustíš node z jakékoli složky.
const booksFile = new URL('./books.json', import.meta.url);

export async function loadBooks() {
  const text = await readFile(booksFile, 'utf8');
  return JSON.parse(text);
}

export async function saveBooks(books) {
  await writeFile(booksFile, JSON.stringify(books, null, 2));
}

export function findBook(books, id) {
  return books.find((book) => book.id === id);
}
```

```js
// Statistiky knihovny pro GET /api/stats — dopsal kolega Tomáš.
// Nepovinný filtr ?author=… spočítá statistiky jen z knih jednoho autora.

function respond(res, code, payload) {
  res.writeHead(code, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(payload));
}

export function handleStats(req, res, allBooks) {
  const params = new URL(req.url, 'http://localhost').searchParams;
  const author = params.get('author');

  const picked = [];
  for (const b of allBooks) {
    if (author === null || b.author === author) {
      picked.push(b);
    }
  }

  if (picked.length === 0) {
    return respond(res, 404, { error: 'Tento autor v knihovně nemá žádnou knihu.' });
  }

  const years = [];
  for (const b of picked) {
    years.push(b.year);
  }

  respond(res, 200, {
    count: picked.length,
    firstYear: Math.min(...years),
    lastYear: Math.max(...years),
  });
}
```

```js
// Přečte celé tělo požadavku jako text. Tělo přichází po kouscích (proud).
export async function readBody(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString('utf8');
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { readBody } from './body.js';
import { findBook, loadBooks } from './library.js';
import { handleStats } from './stats.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

const server = createServer(async (req, res) => {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      const books = await loadBooks();
      const author = searchParams.get('author');
      if (author === null) {
        return sendJson(res, 200, books);
      }
      return sendJson(res, 200, books.filter((book) => book.author === author));
    }
    if (req.method === 'POST') {
      const input = JSON.parse(await readBody(req));
      const books = await loadBooks();
      const book = {
        id: Math.max(0, ...books.map((item) => item.id)) + 1,
        title: input.title,
        author: input.author,
        year: input.year,
      };
      return sendJson(res, 201, book);
    }
    res.setHeader('Allow', 'GET, POST');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const book = findBook(await loadBooks(), id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    return sendJson(res, 200, book);
  }

  if (pathname === '/api/authors') {
    const books = await loadBooks();
    const authors = [...new Set(books.map((book) => book.author))];
    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
  }

  if (pathname === '/api/stats') {
    return handleStats(req, res, await loadBooks());
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
});

server.listen(port, () => {
  console.log(`Server běží na http://localhost:${port}`);
});
```
