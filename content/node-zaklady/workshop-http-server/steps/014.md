# --description--

Než server začne přijímat nové knihy, potřebuje je umět **uložit**. Databázi nemáme,
takže data půjdou zpátky do `books.json`.

**Úkol:** v souboru `library.js` napiš a vyexportuj asynchronní funkci
`saveBooks(books)`, která zapíše celé pole knih do `books.json` jako JSON čitelně
odsazený dvěma mezerami.

Modul `node:fs/promises` má na zápis víc funkcí: `writeFile` soubor přepíše,
`appendFile` přidá text na konec. Vyber tu, se kterou po zápisu zůstane v souboru
platný JSON. Odsazení umí `JSON.stringify` sám, když mu dáš další argumenty —
které, najdeš na MDN v [JSON.stringify()](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/JSON/stringify).

Proč odsazení, když stroji je to jedno? Soubor s daty budeš číst ty, při ladění
i v gitu. Změna jedné knihy je pak v rozdílu vidět jako jeden řádek.

`library.js` exportuje funkci `saveBooks`.

```js
const library = await helpers.importFile('library.js');
assert.equal(typeof library.saveBooks, 'function', 'library.js má exportovat funkci saveBooks');
```

Po `await saveBooks(books)` je v `books.json` přesně to pole a `loadBooks()` ho vrátí.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
try {
  const { saveBooks, loadBooks } = await helpers.importFile('library.js');
  const books = [{ id: 10, title: 'Zkušební kniha', author: 'Test Testovič', year: 2001 }];
  await saveBooks(books);
  let saved;
  try {
    saved = JSON.parse(await fs.readFile(booksPath, 'utf8'));
  } catch {
    assert.fail('Po saveBooks(books) není books.json platný JSON — nepřidáváš text na konec místo přepsání?');
  }
  assert.deepEqual(saved, books, 'Po saveBooks(books) má books.json obsahovat právě uložené pole');
  assert.deepEqual(await loadBooks(), books, 'Po saveBooks(books) má loadBooks() vrátit uložené pole');
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

Soubor je čitelně odsazený dvěma mezerami.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
try {
  const { saveBooks } = await helpers.importFile('library.js');
  const books = [{ id: 10, title: 'Zkušební kniha', author: 'Test Testovič', year: 2001 }];
  await saveBooks(books);
  const text = await fs.readFile(booksPath, 'utf8');
  assert.equal(text.trim(), JSON.stringify(books, null, 2), 'Po saveBooks(books) má být books.json odsazený dvěma mezerami na úroveň');
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

`loadBooks` a `findBook` fungují dál.

```js
const { loadBooks, findBook } = await helpers.importFile('library.js');
const books = await loadBooks();
assert.deepEqual(books, JSON.parse(files['books.json']), 'loadBooks() má dál vracet knihy z books.json');
assert.equal(findBook(books, 3)?.title, 'Válka s mloky', 'findBook(knihy, 3) má dál vracet Válku s mloky');
```

# --help--

## --tip--

Zápis souboru je v modulu `node:fs/promises` stejně jako čtení — viz
[Soubory: `node:fs/promises`](see:node-zaklady/co-je-node#soubory-node-fs-promises).

## --tip--

Funkci pro zápis přidej do existujícího importu nahoře v `library.js`. Dostane cestu
(`booksFile`) a text; text vyrobí `JSON.stringify` s třetím argumentem pro odsazení.

## --tip--

Stejný zápis s jiným souborem a čtyřmi mezerami:
`await writeFile(settingsFile, JSON.stringify(settings, null, 4));`

## --file-- server.js

```js
import { createServer } from 'node:http';
import { findBook, loadBooks } from './library.js';
import { handleStats } from './stats.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

const server = createServer(async (req, res) => {
  const { pathname, searchParams } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      const books = await loadBooks();
      const author = searchParams.get('author');
      if (author === null) {
        return sendJson(res, 200, books);
      }
      return sendJson(res, 200, books.filter((book) => book.author === author));
    }
    res.setHeader('Allow', 'GET');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const book = findBook(await loadBooks(), id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    return sendJson(res, 200, book);
  }

  if (pathname === '/api/authors') {
    const books = await loadBooks();
    const authors = [...new Set(books.map((book) => book.author))];
    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
  }

  if (pathname === '/api/stats') {
    return handleStats(req, res, await loadBooks());
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
});

server.listen(port, () => {
  console.log(`Server běží na http://localhost:${port}`);
});
```

```js
// Statistiky knihovny pro GET /api/stats — dopsal kolega Tomáš.
// Nepovinný filtr ?author=… spočítá statistiky jen z knih jednoho autora.

function respond(res, code, payload) {
  res.writeHead(code, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(payload));
}

export function handleStats(req, res, allBooks) {
  const params = new URL(req.url, 'http://localhost').searchParams;
  const author = params.get('author');

  const picked = [];
  for (const b of allBooks) {
    if (author === null || b.author === author) {
      picked.push(b);
    }
  }

  if (picked.length === 0) {
    return respond(res, 404, { error: 'Tento autor v knihovně nemá žádnou knihu.' });
  }

  const years = [];
  for (const b of picked) {
    years.push(b.year);
  }

  respond(res, 200, {
    count: picked.length,
    firstYear: Math.min(...years),
    lastYear: Math.max(...years),
  });
}
```

## --file-- library.js

```js
import { readFile, writeFile } from 'node:fs/promises';

// Funkce pro práci s knihami. Server je bude importovat.

// books.json leží vedle tohoto souboru. Cesta postavená od import.meta.url
// funguje, ať spustíš node z jakékoli složky.
const booksFile = new URL('./books.json', import.meta.url);

export async function loadBooks() {
  const text = await readFile(booksFile, 'utf8');
  return JSON.parse(text);
}

export async function saveBooks(books) {
  await writeFile(booksFile, JSON.stringify(books, null, 2));
}

export function findBook(books, id) {
  return books.find((book) => book.id === id);
}
```
