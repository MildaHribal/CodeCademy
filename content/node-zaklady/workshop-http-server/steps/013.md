# --description--

Server teď věří všemu, co přijde. Pošli mu tělo `{"title": ` (useknutý JSON) a
`JSON.parse` vyhodí výjimku. V asynchronním handleru ji nikdo nechytí a **celý
server spadne** — pro všechny, nejen pro toho, kdo poslal nesmysl.

Pravidlo serveru zní: **vstup zvenku je nedůvěryhodný, dokud ho neověříš.**
A když je špatný, odpovíš stavem `400 Bad Request` a zprávou, co je špatně. Tím
říkáš „chyba je v tvém požadavku, oprav si ho". Stav `500` patří jen chybám
serveru.

V routě `POST /api/books` v `server.js` obal `JSON.parse` do `try`/`catch`. Když
tělo není platný JSON, odpověz `400` a JSON objektem s klíčem `error`, například
`{ "error": "Tělo požadavku musí být platný JSON." }`.

Na jiných datech — nastavení ze souboru, které může být rozbité:

```js
let settings;
try {
  settings = JSON.parse(await readFile(settingsFile, 'utf8'));
} catch {
  settings = { theme: 'light' }; // rozbitý soubor: použij výchozí nastavení
}
console.log(settings.theme);
```

Dvě věci, na které se tu naráží:

- Proměnná je deklarovaná přes `let` **před** `try`. Kdybys napsal
  `const settings = …` uvnitř bloku `try`, za blokem by neexistovala.
- `catch` bez `(error)` je v pořádku, když chybu nepotřebuješ číst — stačí vědět,
  že nastala. V obsluze požadavku z `catch` rovnou pošli odpověď přes
  `return sendJson(…)`, ať se kód za blokem už nespustí.

# --hints--

Tělo, které není platný JSON, vrátí stav `400` a JSON s klíčem `error`.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  const res = await fetch(new URL('/api/books', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: '{"title": ',
  }).catch(() => assert.fail('Server na neplatný JSON neodpověděl — spadl na výjimce z JSON.parse?'));
  assert.equal(res.status, 400, 'Neplatný JSON má vrátit 400');
  const body = await res.json();
  assert.equal(typeof body.error, 'string', 'Tělo 400 má mít klíč error s textem');
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

Po neplatném JSON server běží dál a odpovídá.

```js
const fs = await import('node:fs/promises');
const booksPath = `${helpers.dir}/books.json`;
const server = await helpers.startServer('server.js');
try {
  await fetch(new URL('/api/books', server.url), { method: 'POST', body: 'tohle není json' }).catch(() => {});
  const res = await fetch(new URL('/api/books', server.url));
  assert.equal(res.status, 200, 'Server má po neplatném požadavku dál odpovídat');
} catch (error) {
  assert.fail(`Server po neplatném JSON přestal odpovídat (spadl?): ${error.message}\n${server.output()}`);
} finally {
  await fs.writeFile(booksPath, files['books.json']);
}
```

## --file-- library.js

```js
import { readFile, writeFile } from 'node:fs/promises';

// books.json leží vedle tohoto souboru. Cesta postavená od import.meta.url
// funguje, ať spustíš node z jakékoli složky.
const booksFile = new URL('./books.json', import.meta.url);

export async function loadBooks() {
  const text = await readFile(booksFile, 'utf8');
  return JSON.parse(text);
}

export async function saveBooks(books) {
  await writeFile(booksFile, JSON.stringify(books, null, 2));
}
```

import { createServer } from 'node:http';
import { loadBooks, saveBooks } from './library.js';

// Port z prostředí. Proměnné prostředí jsou řetězce, proto Number().
export const port = Number(process.env.PORT ?? 3000);

// Pošle data jako JSON se stavovým kódem a hlavičkou, podle které klient pozná typ.
function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

// Přečte celé tělo požadavku jako text. Tělo přichází po kouscích (stream).
async function readBody(req) {
  const chunks = [];
  for await (const chunk of req) {
    chunks.push(chunk);
  }
  return Buffer.concat(chunks).toString('utf8');
}

const server = createServer(async (req, res) => {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (pathname === '/api/books') {
    if (req.method === 'GET') {
      return sendJson(res, 200, await loadBooks());
    }
    if (req.method === 'POST') {
      let input;
      try {
        input = JSON.parse(await readBody(req));
      } catch {
        return sendJson(res, 400, { error: 'Tělo požadavku musí být platný JSON.' });
      }

      const books = await loadBooks();
      const book = {
        id: Math.max(0, ...books.map((item) => item.id)) + 1,
        title: input.title,
        author: input.author,
        year: input.year,
      };
      books.push(book);
      await saveBooks(books);
      return sendJson(res, 201, book);
    }
    res.setHeader('Allow', 'GET, POST');
    return sendJson(res, 405, { error: 'Tahle metoda tu není povolená.' });
  }

  if (pathname.startsWith('/api/books/')) {
    const id = Number(pathname.slice('/api/books/'.length));
    const books = await loadBooks();
    const book = books.find((item) => item.id === id);
    if (!book) {
      return sendJson(res, 404, { error: 'Kniha nenalezena.' });
    }
    return sendJson(res, 200, book);
  }

  if (pathname === '/api/authors') {

    const authors = [...new Set(books.map((book) => book.author))];
    authors.sort((a, b) => a.localeCompare(b, 'cs'));
    return sendJson(res, 200, authors);
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
});

server.listen(port, () => {
  console.log(`Server běží na http://localhost:${port}`);
});
