# --description--

**Úkol:** V seedu přibyl `components/zruseni-predplatneho.tsx` — hotový dialog od kolegyně. Čeká
na tři props: `otevreno`, `onOtevreno` a `onPotvrdit`. Napoj ho:

1. přidej stav `ruseniOtevreno` (výchozí `false`),
2. tlačítko „Zrušit předplatné" ho přepne na `true`,
3. komponentu vykresli pod tlačítkem, `onPotvrdit` zatím jen dialog zavře.

Proč řízený režim, když by stačil `Dialog.Trigger`? Protože o otevření chceme rozhodovat my — za
chvíli bude dialog přepínat i stav předplatného.

# --hints--

Kliknutí na „Zrušit předplatné" otevře dialog.

```js
const zalozky = [...document.querySelectorAll('[role="tab"]')];
await helpers.click(zalozky[1]);
await helpers.flush();
assert.equal(document.querySelector('[data-testid="dialog-ruseni"]'), null, 'Na začátku dialog otevřený být nemá');
const zrusit = [...document.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Zrušit předplatné');
await helpers.click(zrusit);
await helpers.flush();
const dialog = document.querySelector('[data-testid="dialog-ruseni"]');
assert.ok(dialog, 'Po kliknutí má být dialog v dokumentu');
assert.equal(dialog.getAttribute('data-state'), 'open', 'Dialog má mít data-state="open"');
```

Dialog jde zavřít Escapem i tlačítkem „Nechat předplatné".

```js
const zalozky = [...document.querySelectorAll('[role="tab"]')];
await helpers.click(zalozky[1]);
await helpers.flush();
const zrusit = [...document.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Zrušit předplatné');
await helpers.click(zrusit);
await helpers.flush();
await helpers.press(document.querySelector('[data-testid="dialog-ruseni"]'), 'Escape');
await helpers.flush();
assert.equal(document.querySelector('[data-testid="dialog-ruseni"]'), null, 'Escape má dialog zavřít');
await helpers.click(zrusit);
await helpers.flush();
const nechat = [...document.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Nechat předplatné');
await helpers.click(nechat);
await helpers.flush();
assert.equal(document.querySelector('[data-testid="dialog-ruseni"]'), null, 'Tlačítko „Nechat předplatné" má dialog zavřít');
```

Potvrzení dialog zavře a fokus se vrátí na tlačítko, které ho otevřelo.

```js
const zalozky = [...document.querySelectorAll('[role="tab"]')];
await helpers.click(zalozky[1]);
await helpers.flush();
const zrusit = [...document.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Zrušit předplatné');
await helpers.click(zrusit);
await helpers.flush();
const dialog = document.querySelector('[data-testid="dialog-ruseni"]');
const potvrdit = [...dialog.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Zrušit předplatné');
await helpers.click(potvrdit);
await helpers.flush();
assert.equal(document.querySelector('[data-testid="dialog-ruseni"]'), null, 'Po potvrzení má dialog zmizet');
await helpers.waitFor(() => document.activeElement === zrusit, 2000).catch(() => {});
assert.equal(document.activeElement, zrusit, 'Fokus se má vrátit na tlačítko, kterým se dialog otevřel');
```

# --help--

## --tip--

Řízený režim znamená, že o otevření rozhoduje stav v tvojí komponentě — viz [Řízený a neřízený režim](see:react-ui-knihovny/shadcn-a-radix#rizeny-a-nerizeny-rezim).

## --tip--

Komponenta v seedu očekává tři props; podívej se do ní, co s nimi dělá, a dodej je z `App.tsx`.

## --tip--

Zavření dialogu není jen kliknutí na křížek: primitivum hlásí každou změnu (Escape, klik mimo) jednou obsluhou, které pošle novou hodnotu.

```tsx
import type { ComponentProps } from 'react';
import { cva, type VariantProps } from 'class-variance-authority';
import { Slot } from 'radix-ui';
import { cn } from '../../lib/utils';

export const buttonVariants = cva(
  'inline-flex items-center justify-center rounded-lg font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-znacka disabled:pointer-events-none disabled:opacity-50',
  {
    variants: {
      variant: {
        plna: 'bg-znacka text-white hover:bg-znacka-tmava',
        ticha: 'bg-transparent text-slate-700 hover:bg-tlumena',
        nebezpeci: 'bg-nebezpeci text-white hover:bg-rose-800',
      },
      size: {
        sm: 'h-8 px-3 text-xs',
        md: 'h-10 px-4 text-sm',
      },
    },
    defaultVariants: { variant: 'plna', size: 'md' },
  },
);

type ButtonProps = ComponentProps<'button'> &
  VariantProps<typeof buttonVariants> & { asChild?: boolean };

export function Button({ className, variant, size, asChild = false, ...props }: ButtonProps) {
  const Prvek = asChild ? Slot.Root : 'button';
  return <Prvek className={cn(buttonVariants({ variant, size }), className)} {...props} />;
}
```

```tsx
import { Dialog } from 'radix-ui';
import { Button } from './ui/button';

type Props = {
  otevreno: boolean;
  onOtevreno: (hodnota: boolean) => void;
  onPotvrdit: () => void;
};

export function ZruseniPredplatneho({ otevreno, onOtevreno, onPotvrdit }: Props) {
  return (
    <Dialog.Root open={otevreno} onOpenChange={onOtevreno}>
      <Dialog.Portal>
        <Dialog.Overlay className="fixed inset-0 bg-slate-900/50" />
        <Dialog.Content
          data-testid="dialog-ruseni"
          className="fixed left-1/2 top-1/2 w-80 -translate-x-1/2 -translate-y-1/2 rounded-karta bg-white p-6 shadow-xl"
        >
          <p className="text-lg font-semibold">Zrušit předplatné</p>
          <Dialog.Description className="mt-2 text-sm text-popisek">
            Hudbu si pustíš do konce zaplaceného období, pak se účet přepne na verzi s reklamami.
          </Dialog.Description>
          <div className="mt-6 flex justify-end gap-3">
            <Dialog.Close asChild>
              <Button variant="ticha" size="sm">
                Nechat předplatné
              </Button>
            </Dialog.Close>
            <Button variant="nebezpeci" size="sm" onClick={onPotvrdit}>
              Zrušit předplatné
            </Button>
          </div>
        </Dialog.Content>
      </Dialog.Portal>
    </Dialog.Root>
  );
```
