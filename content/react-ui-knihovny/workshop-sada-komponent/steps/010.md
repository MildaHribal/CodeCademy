# --description--

**Úkol:** Ručně psaný typ props se s každou novou variantou rozchází s realitou. Knihovna `cva`
umí typ variant odvodit sama: pomocný typ `VariantProps<typeof buttonVariants>` obsahuje přesně
ty klíče a hodnoty, které jsou v `variants`.

Nahraď ručně vypsané typy props v `components/ui/button.tsx` i v `components/ui/badge.tsx` tímhle
zápisem. Typ props tak vznikne ze dvou zdrojů: z prvku HTML a z variant.

V náhledu se nic nezmění — typy Sucrase při překladu zahodí a stránka běží dál. Rozdíl uvidíš
v editoru a v projektu, kde běží `tsc`: od téhle chvíle překlep `variant="tich"` neprojde kontrolou
typů a nová varianta se v nápovědě objeví sama.

# --hints--

Typ props tlačítka vzniká z `ComponentProps<'button'>` a z `VariantProps<typeof buttonVariants>`.

```js
const zdroj = helpers.stripComments(files['components/ui/button.tsx'], 'js');
assert.match(zdroj, /VariantProps\s*<\s*typeof\s+buttonVariants\s*>/, 'Typ props tlačítka má vzniknout z VariantProps<typeof buttonVariants>');
assert.match(zdroj, /ComponentProps(WithoutRef)?\s*<\s*['"]button['"]\s*>/, "Typ props tlačítka má pořád obsahovat ComponentProps<'button'>");
assert.doesNotMatch(zdroj, /variant\?:\s*'plna'/, 'Ručně vypsané hodnoty variant už v typu být nemají');
```

Typ props štítku vzniká stejně z `badgeVariants`.

```js
const zdroj = helpers.stripComments(files['components/ui/badge.tsx'], 'js');
assert.match(zdroj, /VariantProps\s*<\s*typeof\s+badgeVariants\s*>/, 'Typ props štítku má vzniknout z VariantProps<typeof badgeVariants>');
assert.match(zdroj, /ComponentProps(WithoutRef)?\s*<\s*['"]span['"]\s*>/, "Typ props štítku má pořád obsahovat ComponentProps<'span'>");
```

Stránka se chová pořád stejně: varianty i props zvenku fungují.

```js
const karta = document.querySelector('[data-testid="karta-jana"]');
const rezervovat = [...karta.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Rezervovat');
await helpers.waitFor(() => getComputedStyle(rezervovat).backgroundColor === 'rgb(15, 118, 110)', 3000).catch(() => {});
assert.equal(getComputedStyle(rezervovat).backgroundColor, 'rgb(15, 118, 110)', 'Tlačítko „Rezervovat" má zůstat zelené');
const tereza = document.querySelector('[data-testid="karta-tereza"]');
const stitek = [...tereza.querySelectorAll('span')].find((el) => el.textContent.trim() === 'Dnes obsazeno');
assert.equal(getComputedStyle(stitek).backgroundColor, 'rgb(226, 232, 240)', 'Štítek „Dnes obsazeno" má zůstat šedý');
assert.equal(stitek.getAttribute('title'), 'Dnes už má plno', 'Štítek má pořád posílat zbytek props dál');
```

# --help--

## --tip--

Typ variant nemusíš psát dvakrát — knihovna `cva` ho umí odvodit z toho, co jsi napsal do `variants`.

## --tip--

Pomocný typ se jmenuje `VariantProps` a importuje se ze stejného balíčku jako `cva`. Bere v lomených závorkách typ tvojí funkce variant (`typeof …`).

## --tip--

Na jiných datech: u komponenty `Alert` s `alertVariants` by typ props vznikl z `ComponentProps<'div'>` a z variant té komponenty, spojených `&`.

import { useState, type FormEvent } from 'react';
import { Badge } from './components/ui/badge';
import { Button } from './components/ui/button';
import { kadernice } from './data';
import './styles.css';

export default function App() {
  const [jmeno, setJmeno] = useState('');
  const [chyba, setChyba] = useState('');

  function odeslat(event: FormEvent<HTMLFormElement>) {

    setChyba(jmeno.trim() === '' ? 'Vyplň jméno, ať víme, komu se ozvat.' : '');

        </header>

        <div className="mb-6 flex items-center justify-between gap-4 rounded-karta border border-slate-200 bg-white px-6 py-4">
          <p className="text-sm">Máš rezervaci na 16:00 u Marka Dvořáka.</p>
          <Button variant="ticha" tone="nebezpeci" size="sm">
            Zrušit rezervaci
          </Button>
        </div>

        <ul className="flex flex-col gap-4">
          {kadernice.map((osoba) => (
            <li
              key={osoba.id}
              data-testid={'karta-' + osoba.id}
              className="rounded-karta border border-slate-200 bg-white p-6"
            >
              <div className="flex items-start justify-between gap-4">
                <div>
                  <h2 className="text-lg font-semibold">{osoba.jmeno}</h2>
                  <p className="text-sm text-slate-600">{osoba.specializace}</p>
                </div>
                {osoba.volneTerminy.length > 0 ? (
                  <Badge title={'Nejbližší termín ' + osoba.volneTerminy[0]}>Volno dnes</Badge>
                ) : (
                  <Badge stav="obsazeno" title="Dnes už má plno">
                    Dnes obsazeno
                  </Badge>
                )}
              </div>

              <div className="mt-4">
                <p className="text-sm text-slate-600">Cena od {osoba.cenaOd} Kč</p>
                <div className="mt-3 flex flex-wrap gap-2">
                  {osoba.volneTerminy.map((cas) => (
                    <Button key={cas} variant="ticha" size="sm" className="border border-slate-200">
                      {cas}
                    </Button>
                  ))}
                </div>
              </div>

import { cva } from 'class-variance-authority';
import { cn } from '../../lib/utils';

export const badgeVariants = cva('inline-flex items-center rounded-full px-3 py-1 text-xs font-medium', {
  variants: {
    stav: {
      volno: 'bg-znacka-svetla text-znacka',
      obsazeno: 'bg-tlumena text-popisek',
    },
  },
  defaultVariants: { stav: 'volno' },
});

type BadgeProps = ComponentProps<'span'> & { stav?: 'volno' | 'obsazeno' };

export function Badge({ className, stav, ...props }: BadgeProps) {
  return <span className={cn(badgeVariants({ stav }), className)} {...props} />;

import type { ComponentProps } from 'react';
import { cn } from '../../lib/utils';

import { cva, type VariantProps } from 'class-variance-authority';
export const buttonVariants = cva('inline-flex items-center justify-center rounded-lg font-medium transition-colors', {
  variants: {
    variant: {
      plna: 'bg-znacka text-white hover:bg-znacka-tmava',
      ticha: 'bg-transparent text-slate-700 hover:bg-slate-100',
    },
    size: {
      sm: 'h-8 px-3 text-xs',
      md: 'h-10 px-4 text-sm',
    },
    tone: {
      normalni: '',
      nebezpeci: '',
    },
  },
  compoundVariants: [
    { variant: 'plna', tone: 'nebezpeci', class: 'bg-nebezpeci text-white hover:bg-rose-800' },
    { variant: 'ticha', tone: 'nebezpeci', class: 'text-nebezpeci hover:bg-rose-50' },
  ],
  defaultVariants: { variant: 'plna', size: 'md', tone: 'normalni' },
});

type ButtonProps = ComponentProps<'button'> & VariantProps<typeof buttonVariants>;

export function Button({ className, variant, size, tone, ...props }: ButtonProps) {
  return <button className={cn(buttonVariants({ variant, size, tone }), className)} {...props} />;
}
