# --description--

Filtr výpůjček (Vše, Aktivní, Po termínu) má stejný problém jako menu: bílé pozadí vybrané
volby přeskakuje. Postup znáš z minulého kroku, teď ho napiš sám.

**Úkol:** V `components/RentalList.jsx` zařiď, aby bílé pozadí vybrané volby filtru přejíždělo
mezi volbami. Ukazatel filtru si musí zachovat svůj `data-testid`.

Dej pozor na jednu věc: menu i filtr jsou na stránce zároveň. Rozmysli si, co by se stalo,
kdyby oba ukazatele měly stejné jméno.

# --hints--

Po kliknutí na „Po termínu" je ukazatel filtru jen jeden a stojí v téhle volbě.

```js
const group = document.querySelector('[aria-label="Filtr výpůjček"]');
assert.ok(group, 'na stránce má být skupina „Filtr výpůjček"');
const option = [...group.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Po termínu');
await helpers.click(option);
await helpers.flush();
const indicators = document.querySelectorAll('[data-testid="filter-indicator"]');
assert.equal(indicators.length, 1, 'ukazatel filtru má být na stránce jen jednou');
assert.ok(option.contains(indicators[0]), 'ukazatel filtru má být uvnitř volby „Po termínu"');
```

Ukazatel filtru na novou volbu přejede z té původní.

```js
const group = document.querySelector('[aria-label="Filtr výpůjček"]');
const option = [...group.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Aktivní');
await helpers.click(option);
await helpers.flush();
const indicator = option.querySelector('[data-testid="filter-indicator"]');
assert.ok(indicator, 'po kliknutí má být ukazatel filtru ve volbě „Aktivní"');
let moved = false;
const start = performance.now();
while (!moved && performance.now() - start < 600) {
  const transform = getComputedStyle(indicator).transform;
  if (transform !== 'none' && transform !== 'matrix(1, 0, 0, 1, 0, 0)') moved = true;
  await new Promise((resolve) => requestAnimationFrame(resolve));
}
assert.ok(moved, 'ukazatel filtru má po kliknutí vyjet z pozice předchozí volby');
```

Přepnutí filtru nepohne ukazatelem v menu.

```js
const menuIndicator = document.querySelector('[data-testid="menu-indicator"]');
assert.ok(menuIndicator, 'v menu má zůstat ukazatel aktivní položky');
const before = menuIndicator.getBoundingClientRect();
const group = document.querySelector('[aria-label="Filtr výpůjček"]');
const option = [...group.querySelectorAll('button')].find((el) => el.textContent.trim() === 'Po termínu');
await helpers.click(option);
await helpers.flush();
let steady = true;
const start = performance.now();
while (performance.now() - start < 500) {
  const current = document.querySelector('[data-testid="menu-indicator"]');
  const box = current ? current.getBoundingClientRect() : null;
  const visible = current && Number(getComputedStyle(current).opacity) > 0.99;
  if (!box || !visible || Math.abs(box.top - before.top) > 1 || Math.abs(box.left - before.left) > 1) steady = false;
  await new Promise((resolve) => requestAnimationFrame(resolve));
}
assert.ok(steady, 'ukazatel v menu má při přepnutí filtru zůstat viditelný na svém místě — každý ukazatel potřebuje vlastní jméno');
```

# --help--

## --tip--

Je to stejný mechanismus jako u menu: [layoutId](see:react-ui-knihovny/motion-for-react#layout-animace-a-layoutid) spojí zanikající a vznikající prvek.

## --tip--

Prvek ukazatele musí být `motion` komponenta, jinak props Motionu nezná. Jméno vyber jiné než u menu — prvky se stejným jménem Motion propojí napříč celou stránkou.

## --tip--

U přepínače tarifu by to byl ukazatel `period-highlight` a u výběru měny `currency-highlight`: co přepínač, to vlastní jméno.

```jsx
import { useState } from 'react';
import { BikePanel } from './components/BikePanel';
import { RentalList } from './components/RentalList';
import { Sidebar } from './components/Sidebar';
import { StatCards } from './components/StatCards';
import { Toast } from './components/Toast';
import { bikes, initialRentals, revenueBeforeToday, walkInRental } from './data';
import './styles.css';

export default function App() {
  const [page, setPage] = useState('Přehled');
  const [rentals, setRentals] = useState(initialRentals);
  const [revenue, setRevenue] = useState(revenueBeforeToday);
  const [selectedId, setSelectedId] = useState(null);
  const [message, setMessage] = useState(null);
  const [nextNumber, setNextNumber] = useState(1045);

  const lowBatteryCount = bikes.filter((bike) => bike.battery < 20).length;
  const selectedRental = rentals.find((rental) => rental.id === selectedId) ?? null;
  const selectedBike = selectedRental ? bikes.find((bike) => bike.id === selectedRental.bikeId) : null;

  function endRental(id) {
    const rental = rentals.find((item) => item.id === id);
    setRentals(rentals.filter((item) => item.id !== id));
    setRevenue(revenue + rental.price);
    setMessage(`Výpůjčka ${id} ukončena, tržba +${rental.price} Kč`);
  }

  function addRental() {
    setRentals([{ ...walkInRental, id: `V-${nextNumber}` }, ...rentals]);
    setNextNumber(nextNumber + 1);
  }

  return (
    <div className="flex min-h-screen bg-surface text-slate-900">
      <Sidebar active={page} onSelect={setPage} />
      <main className="flex-1 space-y-6 p-8">
        <h1 className="text-2xl font-semibold">{page}</h1>
        <StatCards rentedCount={rentals.length} revenue={revenue} lowBatteryCount={lowBatteryCount} />
        <RentalList rentals={rentals} onEnd={endRental} onDetail={setSelectedId} onAdd={addRental} />
      </main>
      <BikePanel rental={selectedRental} bike={selectedBike} onClose={() => setSelectedId(null)} />
      <div className="fixed right-6 bottom-6">
        <Toast message={message} onClose={() => setMessage(null)} />
      </div>
    </div>
  );
}
```

```jsx
import { useState } from 'react';
import { Button } from './ui/button';

const filters = [
  { value: 'all', label: 'Vše' },
  { value: 'active', label: 'Aktivní' },
  { value: 'late', label: 'Po termínu' },
];

export function RentalList({ rentals, onEnd, onDetail, onAdd }) {
  const [filter, setFilter] = useState('all');
  const visibleRentals = filter === 'all' ? rentals : rentals.filter((rental) => rental.status === filter);

  return (
    <section aria-labelledby="rentals-heading" className="rounded-2xl border border-slate-200 bg-white p-5">
      <div className="flex items-center justify-between gap-4">
        <h2 id="rentals-heading" className="text-lg font-semibold">Výpůjčky</h2>
        <Button onClick={onAdd}>Nová výpůjčka</Button>
      </div>

      <div role="group" aria-label="Filtr výpůjček" className="mt-4 inline-flex rounded-lg bg-slate-100 p-1">
        {filters.map((option) => (
          <button
            key={option.value}
            aria-pressed={filter === option.value}
            onClick={() => setFilter(option.value)}
            className="relative rounded-md px-3 py-1.5 text-sm font-medium text-slate-700"
          >
            {filter === option.value && (
              <span data-testid="filter-indicator" className="absolute inset-0 rounded-md bg-white shadow-sm" />
            )}
            <span className="relative">{option.label}</span>
          </button>
        ))}
      </div>

      <ul className="relative mt-4">
        {visibleRentals.map((rental) => (
          <li
            key={rental.id}
            data-testid="rental-row"
            className="flex w-full items-center justify-between gap-4 border-b border-slate-100 bg-white py-3"
          >
            <div>
              <p className="text-sm font-medium">{rental.customer}</p>
              <p className="text-xs text-muted">
                {rental.id} · kolo {rental.bikeId} · od {rental.from}
              </p>
            </div>
            <div className="flex gap-2">
              <Button variant="ghost" size="sm" onClick={() => onDetail(rental.id)}>
                Detail
              </Button>
              <Button variant="danger" size="sm" onClick={() => onEnd(rental.id)}>
                Ukončit
              </Button>
            </div>
          </li>
        ))}
      </ul>
    </section>
  );
}
```

```jsx
import { useEffect } from 'react';
import { animate, motion, useMotionValue, useTransform } from 'motion/react';
import { formatCzk } from '../lib/format';

export function RevenueCounter({ value }) {
  const count = useMotionValue(0);
  const text = useTransform(() => formatCzk(Math.round(count.get())));

  useEffect(() => {
    const controls = animate(count, value, { duration: 0.8, ease: 'easeOut' });
    return () => controls.stop();
  }, [count, value]);

  return <motion.span>{text}</motion.span>;
}
```

```jsx
import { motion } from 'motion/react';
import { Card } from './ui/card';
import { RevenueCounter } from './RevenueCounter';

const listVariants = {
  hidden: {},
  visible: { transition: { staggerChildren: 0.1 } },
};

const cardVariants = {
  hidden: { opacity: 0, y: 16 },
  visible: { opacity: 1, y: 0 },
};

export function StatCards({ rentedCount, revenue, lowBatteryCount }) {
  const stats = [
    { id: 'rented', label: 'Vypůjčená kola', value: rentedCount },
    { id: 'revenue', label: 'Tržba dne', value: <RevenueCounter value={revenue} /> },
    { id: 'battery', label: 'Kola s vybitou baterií', value: lowBatteryCount },
  ];

  return (
    <motion.ul variants={listVariants} initial="hidden" animate="visible" className="grid gap-4 sm:grid-cols-3">
      {stats.map((stat) => (
        <motion.li key={stat.id} variants={cardVariants} data-testid={`stat-${stat.id}`}>
          <Card>
            <p className="text-sm text-muted">{stat.label}</p>
            <p className="mt-1 text-2xl font-semibold">{stat.value}</p>
          </Card>
        </motion.li>
      ))}
    </motion.ul>
  );
}
```
