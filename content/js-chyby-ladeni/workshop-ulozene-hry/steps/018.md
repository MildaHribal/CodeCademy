# --description--

Hráč vloží kód pozice od kamaráda do slotu. Když je kód neplatný, jeho dosavadní hra v tom slotu nesmí zmizet.

**Úkol:** Napiš funkci `importSave(slotId, text)`:

- Když text jde načíst a projde kontrolou, uloží pozici do `savedGames[slotId]` jako text přes `JSON.stringify` a vrátí ji.
- Když ne, vyhodí `Error` se zprávou `Import do slot-1 se nepovedl` (se skutečným slotem) a původní chybou jako příčinou. Slot přitom zůstane beze změny.

> [!PITFALL]
> Proměnná založená přes `const` uvnitř `try` za blokem neexistuje — `ReferenceError: save is not defined`. Když pozici potřebuješ i za blokem, založ proměnnou před ním.

# --hints--

`importSave` uloží platnou pozici do slotu a vrátí ji.

```js
const text = '{"player":"Kuba","level":20,"coins":990,"savedAt":"2026-09-14T11:00:00"}';
const save = importSave('slot-5', text);
assert.equal(save?.player, 'Kuba', "importSave('slot-5', Kubova pozice) má vrátit Kubovu pozici");
assert.deepEqual(JSON.parse(savedGames['slot-5']), JSON.parse(text), "po importSave('slot-5', …) má savedGames['slot-5'] obsahovat Kubovu pozici");
```

`importSave` přepíše i existující slot.

```js
importSave('slot-1', '{"player":"Kuba","level":20,"coins":990,"savedAt":"2026-09-14T11:00:00"}');
assert.equal(JSON.parse(savedGames['slot-1']).player, 'Kuba', "po importSave('slot-1', Kubova pozice) má slot-1 patřit Kubovi");
```

Neplatný text vyhodí chybu se zprávou o slotu a příčinou.

```js
let error;
try {
  importSave('slot-1', '{"player":"Kuba","level":');
} catch (caught) {
  error = caught;
}
assert.ok(error instanceof Error, "importSave('slot-1', uříznutý text) má vyhodit objekt chyby");
assert.equal(error.message, 'Import do slot-1 se nepovedl', "zpráva má být 'Import do slot-1 se nepovedl'");
assert.ok(error.cause instanceof SyntaxError, 'příčina (cause) má být SyntaxError z JSON.parse');
```

Po neplatném importu zůstane slot beze změny — i když text jde přečíst, ale neprojde kontrolou.

```js
const before = savedGames['slot-1'];
assert.throws(() => importSave('slot-1', '{"player":"Kuba","level":99,"coins":1}'), Error, "importSave('slot-1', pozice s úrovní 99) má vyhodit chybu");
assert.equal(savedGames['slot-1'], before, 'po neplatném importu má slot-1 zůstat beze změny');
let error;
try {
  importSave('slot-1', '{"player":"Kuba","level":99,"coins":1}');
} catch (caught) {
  error = caught;
}
assert.ok(error?.cause instanceof RangeError, 'u úrovně 99 má být příčinou RangeError z checkLevel');
```

# --help--

## --tip--

Proměnná, která má žít i za blokem `try` — viz [Proměnná z `try` není za blokem vidět](see:js-chyby-ladeni/vyjimky#promenna-z-try-neni-za-blokem-videt). Obalení chyby znáš z `loadGame`.

## --tip--

V `try` jen načti a zkontroluj pozici. Zápis do `savedGames` dej až za celý blok — k němu se běh dostane, jen když `catch` nic nevyhodil.

## --tip--

Kostra: `// 1. let pro pozici`, `// 2. try: načti a zkontroluj`, `// 3. catch: vyhoď obalenou chybu`, `// 4. ulož a vrať`.

## --file-- script.js

```js
// Uložené pozice hry Farma Lísková. Klíč je slot, hodnota text ve formátu JSON —
// přesně tak, jak hra pozice ukládá do úložiště prohlížeče.
const savedGames = {
  'slot-1': '{"player":"Eliška","level":12,"coins":340,"savedAt":"2026-09-12T18:40:00"}',
  'slot-2': '{"player":"Tonda","level":3,"coins":',
  'slot-3': '{"player":"Mája","level":0,"coins":25,"savedAt":"2026-09-01T09:15:00"}',
  'slot-4': '{"player":"Radek","level":7,"coins":-20,"savedAt":"2026-09-11T16:30:00"}',
  autosave: '{"player":"Eliška","level":12,"coins":355,"savedAt":"2026-09-13T21:05:00"}',
};

// Když je loader.busy true, hra ukazuje točící se ikonu nahrávání.
const loader = { busy: false };

// ----- Od kolegy -----

// Zkontroluje počet mincí v uložené hře a vrátí ho.
function checkCoins(coins) {
  if (!Number.isInteger(coins) || coins < 0) {
    throw new RangeError(`Počet mincí musí být nezáporné celé číslo, ne ${coins}`);
  }
  return coins;
}

// Zkopíruje pozici do zálohy (slot-1 → slot-1-backup).
// Vrátí true, když se záloha povedla, jinak false.
function backupSlot(slotId) {
  try {
    const save = validateSave(readSave(savedGames[slotId]));
    savedGames[`${slotId}-backup`] = JSON.stringify(save);
    return true;
  } catch {
    return false;
  }
}

// ----- Tvoje funkce -----

// Z textu uložené pozice vrátí objekt hry.
function readSave(text) {
  return JSON.parse(text);
}

console.log(readSave(savedGames['slot-1']));

// Zkontroluje úroveň hráče (celé číslo 1–50) a vrátí ji.
function checkLevel(level) {
  if (!Number.isInteger(level) || level < 1 || level > 50) {
    throw new RangeError(`Úroveň musí být celé číslo 1–50, ne ${level}`);
  }
  return level;
}

// Zkontroluje celou uloženou hru a vrátí ji.
function validateSave(save) {
  if (typeof save.player !== 'string' || save.player.trim() === '') {
    throw new TypeError('Uložená hra nemá jméno hráče');
  }
  checkLevel(save.level);
  checkCoins(save.coins);
  return save;
}

// Vrátí uloženou hru, nebo null, když ji nejde načíst.
function tryReadSave(text) {
  try {
    return validateSave(readSave(text));
  } catch {
    return null;
  }
}

console.log('slot-2:', tryReadSave(savedGames['slot-2']));

// Popisek pozice do menu, nebo důvod, proč ji nejde načíst.
function slotLabel(slotId) {
  try {
    const save = validateSave(readSave(savedGames[slotId]));
    return `${save.player} · úroveň ${save.level} · ${save.coins} mincí`;
  } catch (error) {
    return `Nejde načíst: ${error.message}`;
  }
}

console.log(slotLabel('slot-1'));
console.log(slotLabel('slot-3'));

function loadSlot(slotId) {
  loader.busy = true;
  try {
    return validateSave(readSave(savedGames[slotId]));
  } finally {
    loader.busy = false;
  }
}

// Uloží hru do slotu. Neplatnou hru neuloží a vyhodí chybu.
function saveSlot(slotId, save) {
  loader.busy = true;
  try {
    validateSave(save);
    savedGames[slotId] = JSON.stringify(save);
  } finally {
    loader.busy = false;
  }
}

// Načte hru ze slotu. Když to nejde, vyhodí chybu s původní chybou v cause.
function loadGame(slotId) {
  if (!Object.hasOwn(savedGames, slotId)) {
    throw new Error(`Pozice ${slotId} neexistuje`);
  }
  try {
    return loadSlot(slotId);
  } catch (error) {
    throw new Error(`Pozici ${slotId} nejde načíst`, { cause: error });
  }
}

try {
  loadGame('slot-2');
} catch (error) {
  console.log(error.message, '←', error.cause.message);
}

// Pozice vložená ze schránky. Prázdný nebo poškozený text vrátí null,
// jiná chyba (třeba volání s číslem místo textu) vyletí dál.
function readSaveOrNull(text) {
  try {
    return JSON.parse(text.trim());
  } catch (error) {
    if (error instanceof SyntaxError) {
      return null;
    }
    throw error;
  }
}

// Sloty, jejichž pozici nejde načíst.
function brokenSlots() {
  return Object.keys(savedGames).filter((slotId) => tryReadSave(savedGames[slotId]) === null);
}

console.log('Poškozené pozice:', brokenSlots());

// Načte všechny zadané sloty. Když některé nejdou, vyhodí AggregateError se všemi chybami.
function loadAll(slotIds) {
  const saves = [];
  const errors = [];
  for (const slotId of slotIds) {
    try {
      saves.push(loadGame(slotId));
    } catch (error) {
      errors.push(error);
    }
  }
  if (errors.length > 0) {
    throw new AggregateError(errors, `Nejde načíst ${errors.length} z ${slotIds.length} uložených her`);
  }
  return saves;
}

try {
  loadAll(Object.keys(savedGames));
} catch (error) {
  console.log(error.message);
}

// Naposledy uložená hra ze všech, které jdou načíst.
function newestSave() {
  let newest;
  for (const text of Object.values(savedGames)) {
    const save = tryReadSave(text);
    if (save !== null && (newest === undefined || save.savedAt > newest.savedAt)) {
      newest = save;
    }
  }
  return newest;
}

console.log('Pokračovat ve hře:', newestSave());

// Tlačítko Hrát: buď obrazovka farmy s hrou, nebo menu s hláškou.
function startGame(slotId) {
  try {
    return { screen: 'farm', save: loadGame(slotId) };
  } catch (error) {
    const reason = error.cause ? ` (${error.cause.message})` : '';
    return { screen: 'menu', notice: `${error.message}${reason}` };
  }
}

console.log(startGame('slot-3'));

// Načte hru ze slotu; když je poškozená, obnoví ji ze zálohy.
function restoreBackup(slotId) {
  try {
    return loadGame(slotId);
  } catch (originalError) {
    try {
      const save = loadGame(`${slotId}-backup`);
      savedGames[slotId] = savedGames[`${slotId}-backup`];
      return save;
    } catch (backupError) {
      throw new AggregateError([originalError, backupError], `Pozici ${slotId} ani její zálohu nejde načíst`);
    }
  }
}
```
