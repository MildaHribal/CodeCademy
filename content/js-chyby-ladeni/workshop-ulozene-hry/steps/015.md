# --description--

Všechny funkce, které zatím načítají pozici, chybu vyhazují nebo obalují a rozhodnutí nechávají na volajícím. Teď přichází místo, které rozhodnout **umí**: tlačítko **Hrát** u slotu. Buď otevře farmu, nebo vrátí hráče do menu s hláškou.

**Úkol:** Napiš funkci `startGame(slotId)`, která:

- když `loadGame` pozici načte, vrátí objekt s `screen: 'farm'` a načtenou pozicí pod klíčem `save`,
- když `loadGame` vyhodí chybu, vrátí objekt s `screen: 'menu'` a hláškou pod klíčem `notice`. Hláška je zpráva chyby, a když má chyba příčinu, za ní v závorce ještě zpráva příčiny: `Pozici slot-3 nejde načíst (Úroveň musí být celé číslo 1–50, ne 0)`. U chyby bez příčiny žádná závorka není.

Pod funkci vypiš výsledek pro `slot-3`.

# --hints--

`startGame` pro platný slot vrátí obrazovku farmy s pozicí.

```js
const result = startGame('slot-1');
assert.equal(result?.screen, 'farm', "startGame('slot-1').screen má být 'farm'");
assert.equal(result?.save?.player, 'Eliška', "startGame('slot-1').save má být pozice Elišky");
```

`startGame` pro poškozený slot vrátí menu s hláškou a příčinou v závorce.

```js
let cause;
try {
  checkLevel(0);
} catch (error) {
  cause = error.message;
}
const result = startGame('slot-3');
assert.equal(result?.screen, 'menu', "startGame('slot-3').screen má být 'menu'");
assert.equal(result?.notice, `Pozici slot-3 nejde načíst (${cause})`, `startGame('slot-3').notice má být 'Pozici slot-3 nejde načíst (${cause})'`);
```

`startGame` pro neexistující slot vrátí menu s hláškou bez závorky.

```js
const result = startGame('slot-9');
assert.equal(result?.screen, 'menu', "startGame('slot-9').screen má být 'menu'");
assert.equal(result?.notice, 'Pozice slot-9 neexistuje', "startGame('slot-9').notice má být 'Pozice slot-9 neexistuje' — bez závorky, chyba nemá cause");
```

`startGame` nevyhodí výjimku ani pro uříznutý text.

```js
assert.doesNotThrow(() => startGame('slot-2'), "startGame('slot-2') nemá vyhodit výjimku — tady se chyba chytá");
assert.match(startGame('slot-2')?.notice ?? '', /^Pozici slot-2 nejde načíst \(.+\)$/, "startGame('slot-2').notice má mít zprávu o slotu a v závorce zprávu z JSON.parse");
```

# --help--

## --tip--

Tohle je místo „nahoře", kde se chyba chytá a mění na hlášku — viz [Kde chybu chytat](see:js-chyby-ladeni/vyjimky#kde-chybu-chytat).

## --tip--

Jeden `try…catch` kolem volání `loadGame`. V `catch` si nejdřív připrav text závorky (prázdný, když `error.cause` chybí) a pak vrať objekt menu.

## --tip--

Na jiných datech by hláška platby vznikla z `problem.message` a z `problem.cause?.message`, když existuje.

# --explain--

Vysvětli vlastními slovy, proč se chyba chytá až ve `startGame`, a ne už v `loadSlot` nebo `loadGame`.

## --model--

`loadSlot` a `loadGame` nevědí, co s chybou dělat — jestli ukázat hlášku, zkusit zálohu, nebo nahlásit všechny chyby najednou. Proto chybu jen vyhodí nebo obalí. `startGame` je obsluha tlačítka a ví přesně, co hráč potřebuje: vrátit se do menu s hláškou. Kdyby chybu chytala už `loadSlot`, nemohla by ji `loadAll` sbírat ani `restoreBackup` reagovat zálohou.

## --checklist--

- Nízkoúrovňové funkce nevědí, co s chybou udělat.
- Proto chybu vyhodí nebo obalí a nechají ji letět nahoru.
- Obsluha tlačítka ví, co hráč potřebuje, a tam se chyba chytá.
- Když by chybu chytala už nízká funkce, další volající by na ni nemohli reagovat po svém.

```js
// Uložené pozice hry Farma Lísková. Klíč je slot, hodnota text ve formátu JSON —
// přesně tak, jak hra pozice ukládá do úložiště prohlížeče.
const savedGames = {
  'slot-1': '{"player":"Eliška","level":12,"coins":340,"savedAt":"2026-09-12T18:40:00"}',
  'slot-2': '{"player":"Tonda","level":3,"coins":',
  'slot-3': '{"player":"Mája","level":0,"coins":25,"savedAt":"2026-09-01T09:15:00"}',
  'slot-4': '{"player":"Radek","level":7,"coins":-20,"savedAt":"2026-09-11T16:30:00"}',
  autosave: '{"player":"Eliška","level":12,"coins":355,"savedAt":"2026-09-13T21:05:00"}',
};

// Když je loader.busy true, hra ukazuje točící se ikonu nahrávání.
const loader = { busy: false };

// ----- Od kolegy -----

// Zkontroluje počet mincí v uložené hře a vrátí ho.
function checkCoins(coins) {
  if (!Number.isInteger(coins) || coins < 0) {
    throw new RangeError(`Počet mincí musí být nezáporné celé číslo, ne ${coins}`);
  }
  return coins;
}

// Zkopíruje pozici do zálohy (slot-1 → slot-1-backup).
// Vrátí true, když se záloha povedla, jinak false.
function backupSlot(slotId) {
  try {
    const save = validateSave(readSave(savedGames[slotId]));
    savedGames[`${slotId}-backup`] = JSON.stringify(save);
    return true;
  } catch {
    return false;
  } finally {
    // ať funkce vždycky něco vrátí
    return true;
  }
}

// ----- Tvoje funkce -----

// Z textu uložené pozice vrátí objekt hry.
function readSave(text) {
  return JSON.parse(text);
}

console.log(readSave(savedGames['slot-1']));

// Zkontroluje úroveň hráče (celé číslo 1–50) a vrátí ji.
function checkLevel(level) {
  if (!Number.isInteger(level) || level < 1 || level > 50) {
    throw new RangeError(`Úroveň musí být celé číslo 1–50, ne ${level}`);
  }
  return level;
}

// Zkontroluje celou uloženou hru a vrátí ji.
function validateSave(save) {
  if (typeof save.player !== 'string' || save.player.trim() === '') {
    throw new TypeError('Uložená hra nemá jméno hráče');
  }
  checkLevel(save.level);
  checkCoins(save.coins);
  return save;
}

// Vrátí uloženou hru, nebo null, když ji nejde načíst.
function tryReadSave(text) {
  try {
    return validateSave(readSave(text));
  } catch {
    return null;
  }
}

console.log('slot-2:', tryReadSave(savedGames['slot-2']));

// Popisek pozice do menu, nebo důvod, proč ji nejde načíst.
function slotLabel(slotId) {
  try {
    const save = validateSave(readSave(savedGames[slotId]));
    return `${save.player} · úroveň ${save.level} · ${save.coins} mincí`;
  } catch (error) {
    return `Nejde načíst: ${error.message}`;
  }
}

console.log(slotLabel('slot-1'));
console.log(slotLabel('slot-3'));

function loadSlot(slotId) {
  loader.busy = true;
  try {
    return validateSave(readSave(savedGames[slotId]));
  } finally {
    loader.busy = false;
  }
}

// Uloží hru do slotu. Neplatnou hru neuloží a vyhodí chybu.
function saveSlot(slotId, save) {
  loader.busy = true;
  try {
    validateSave(save);
    savedGames[slotId] = JSON.stringify(save);
  } finally {
    loader.busy = false;
  }
}

// Načte hru ze slotu. Když to nejde, vyhodí chybu s původní chybou v cause.
function loadGame(slotId) {
  if (!Object.hasOwn(savedGames, slotId)) {
    throw new Error(`Pozice ${slotId} neexistuje`);
  }
  try {
    return loadSlot(slotId);
  } catch (error) {
    throw new Error(`Pozici ${slotId} nejde načíst`, { cause: error });
  }
}

try {
  loadGame('slot-2');
} catch (error) {
  console.log(error.message, '←', error.cause.message);
}

// Pozice vložená ze schránky. Prázdný nebo poškozený text vrátí null,
// jiná chyba (třeba volání s číslem místo textu) vyletí dál.
function readSaveOrNull(text) {
  try {
    return JSON.parse(text.trim());
  } catch (error) {
    if (error instanceof SyntaxError) {
      return null;
    }
    throw error;
  }
}

// Sloty, jejichž pozici nejde načíst.
function brokenSlots() {
  return Object.keys(savedGames).filter((slotId) => tryReadSave(savedGames[slotId]) === null);
}

console.log('Poškozené pozice:', brokenSlots());

// Načte všechny zadané sloty. Když některé nejdou, vyhodí AggregateError se všemi chybami.
function loadAll(slotIds) {
  const saves = [];
  const errors = [];
  for (const slotId of slotIds) {
    try {
      saves.push(loadGame(slotId));
    } catch (error) {
      errors.push(error);
    }
  }
  if (errors.length > 0) {
    throw new AggregateError(errors, `Nejde načíst ${errors.length} z ${slotIds.length} uložených her`);
  }
  return saves;
}

try {
  loadAll(Object.keys(savedGames));
} catch (error) {
  console.log(error.message);
}

// Naposledy uložená hra ze všech, které jdou načíst.
function newestSave() {
  let newest;
  for (const text of Object.values(savedGames)) {
    const save = tryReadSave(text);
    if (save !== null && (newest === undefined || save.savedAt > newest.savedAt)) {
      newest = save;
    }
  }
  return newest;
}

console.log('Pokračovat ve hře:', newestSave());
```
