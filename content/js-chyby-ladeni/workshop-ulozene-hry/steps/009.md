# --description--

Zkus si zavolat `loadGame('slot-9')`. Chyba přijde, ale její příčina je matoucí: `"undefined" is not valid JSON`. Slot `slot-9` přitom není poškozený — vůbec neexistuje. To je jiná situace a zaslouží si jinou zprávu.

**Úkol:** Uprav `loadGame` tak, aby pro slot, který v `savedGames` není, hned na začátku vyhodila `Error` se zprávou `Pozice slot-9 neexistuje` (se skutečným slotem) — **bez** `cause`, protože žádná původní chyba není. Existující sloty se načítají jako dosud.

Kontrolu na začátku funkce, která ji předčasně ukončí, znáš ze sekce o funkcích. Tentokrát bez návodu.

# --hints--

`loadGame` pro neexistující slot vyhodí chybu se zprávou, že pozice neexistuje.

```js
let error;
try {
  loadGame('slot-9');
} catch (caught) {
  error = caught;
}
assert.ok(error instanceof Error, "loadGame('slot-9') má vyhodit objekt chyby");
assert.match(error.message, /slot-9/, "zpráva chyby z loadGame('slot-9') má obsahovat slot-9");
assert.match(error.message, /neexistuje/, "zpráva chyby z loadGame('slot-9') má říct, že pozice neexistuje");
```

Chyba neexistujícího slotu nemá `cause`.

```js
let error;
try {
  loadGame('slot-9');
} catch (caught) {
  error = caught;
}
assert.equal(error?.cause, undefined, "chyba z loadGame('slot-9') nemá mít cause — kontrola má proběhnout dřív než načítání");
```

Existující sloty se načítají dál: platný vrátí pozici, poškozený vyhodí chybu s `cause`.

```js
assert.equal(loadGame('autosave').coins, 355, "loadGame('autosave').coins má být 355");
let error;
try {
  loadGame('slot-2');
} catch (caught) {
  error = caught;
}
assert.ok(error?.cause instanceof SyntaxError, "loadGame('slot-2') má dál vyhazovat chybu s SyntaxError v cause");
assert.doesNotMatch(error.message, /neexistuje/, "slot-2 existuje, jen je poškozený — zpráva nemá tvrdit, že neexistuje");
```

# --help--

## --tip--

Jde o [[guard clause]] — kontrolu na začátku funkce, která ji ukončí dřív, než se pustí do práce. Chybu vyhodíš stejně jako v `checkLevel`.

## --tip--

Kontrola patří **před** `try`, jinak by ji `catch` chytil a obalil. Jestli objekt klíč má, zjistí `Object.hasOwn` nebo operátor `in`.

## --tip--

Kostra funkce: `// 1. slot neexistuje → vyhoď chybu a skonči`, `// 2. try: načti slot`, `// 3. catch: obal chybu přes cause`.

```js
// Uložené pozice hry Farma Lísková. Klíč je slot, hodnota text ve formátu JSON —
// přesně tak, jak hra pozice ukládá do úložiště prohlížeče.
const savedGames = {
  'slot-1': '{"player":"Eliška","level":12,"coins":340,"savedAt":"2026-09-12T18:40:00"}',
  'slot-2': '{"player":"Tonda","level":3,"coins":',
  'slot-3': '{"player":"Mája","level":0,"coins":25,"savedAt":"2026-09-01T09:15:00"}',
  'slot-4': '{"player":"Radek","level":7,"coins":-20,"savedAt":"2026-09-11T16:30:00"}',
  autosave: '{"player":"Eliška","level":12,"coins":355,"savedAt":"2026-09-13T21:05:00"}',
};

// Když je loader.busy true, hra ukazuje točící se ikonu nahrávání.
const loader = { busy: false };

// ----- Od kolegy -----

// Zkontroluje počet mincí v uložené hře a vrátí ho.
function checkCoins(coins) {
  if (!Number.isInteger(coins) || coins < 0) {
    throw 'Počet mincí musí být nezáporné celé číslo';
  }
  return coins;
}

// Zkopíruje pozici do zálohy (slot-1 → slot-1-backup).
// Vrátí true, když se záloha povedla, jinak false.
function backupSlot(slotId) {
  try {
    const save = validateSave(readSave(savedGames[slotId]));
    savedGames[`${slotId}-backup`] = JSON.stringify(save);
    return true;
  } catch {
    return false;
  } finally {
    // ať funkce vždycky něco vrátí
    return true;
  }
}

// ----- Tvoje funkce -----

// Z textu uložené pozice vrátí objekt hry.
function readSave(text) {
  return JSON.parse(text);
}

console.log(readSave(savedGames['slot-1']));

// Zkontroluje úroveň hráče (celé číslo 1–50) a vrátí ji.
function checkLevel(level) {
  if (!Number.isInteger(level) || level < 1 || level > 50) {
    throw new RangeError(`Úroveň musí být celé číslo 1–50, ne ${level}`);
  }
  return level;
}

// Zkontroluje celou uloženou hru a vrátí ji.
function validateSave(save) {
  if (typeof save.player !== 'string' || save.player.trim() === '') {
    throw new TypeError('Uložená hra nemá jméno hráče');
  }
  checkLevel(save.level);
  checkCoins(save.coins);
  return save;
}

// Vrátí uloženou hru, nebo null, když ji nejde načíst.
function tryReadSave(text) {
  try {
    return validateSave(readSave(text));
  } catch {
    return null;
  }
}

console.log('slot-2:', tryReadSave(savedGames['slot-2']));

// Popisek pozice do menu, nebo důvod, proč ji nejde načíst.
function slotLabel(slotId) {
  try {
    const save = validateSave(readSave(savedGames[slotId]));
    return `${save.player} · úroveň ${save.level} · ${save.coins} mincí`;
  } catch (error) {
    return `Nejde načíst: ${error.message}`;
  }
}

console.log(slotLabel('slot-1'));
console.log(slotLabel('slot-3'));

function loadSlot(slotId) {
  loader.busy = true;
  try {
    return validateSave(readSave(savedGames[slotId]));
  } finally {
    loader.busy = false;
  }
}

// Uloží hru do slotu. Neplatnou hru neuloží a vyhodí chybu.
function saveSlot(slotId, save) {
  loader.busy = true;
  try {
    validateSave(save);
    savedGames[slotId] = JSON.stringify(save);
  } finally {
    loader.busy = false;
  }
}

--edit--
// Načte hru ze slotu. Když to nejde, vyhodí chybu s původní chybou v cause.
function loadGame(slotId) {
  try {
    return loadSlot(slotId);
  } catch (error) {
    throw new Error(`Pozici ${slotId} nejde načíst`, { cause: error });
  }
}

try {
  loadGame('slot-2');
} catch (error) {
  console.log(error.message, '←', error.cause.message);
}
--edit--
```
