# --description--

Stejnou ikonu hra ukazuje při ukládání. Teď napíšeš vzor s `finally` sám.

**Úkol:** Napiš funkci `saveSlot(slotId, save)`, která:

1. zapne ikonu (`loader.busy` na `true`),
2. zkontroluje pozici funkcí `validateSave`,
3. uloží ji do `savedGames[slotId]` jako text přes `JSON.stringify`,
4. v každém případě ikonu vypne.

Neplatnou pozici funkce neuloží — chyba z kontroly vyletí ven dřív, než se k uložení dostane, a slot zůstane beze změny.

Stejný tvar jako v minulém kroku, jen bez `return`. Na jiných datech:

```js
function sendScore(score) {
  // 1. zapni stav
  form.sending = true;
  try {
    // 2. práce, která může selhat
    checkScore(score);
    scores.push(score);
  } finally {
    // 3. úklid v každém případě
    form.sending = false;
  }
}
```

Konzole se nezmění, dokud `saveSlot` nezavoláš. Zkus si pod ni zavolat `saveSlot('slot-5', { player: 'Ondra', level: 1, coins: 0, savedAt: '2026-09-14T08:00:00' })` a vypsat `savedGames`.

# --hints--

`saveSlot` uloží platnou pozici do slotu jako text ve formátu JSON.

```js
const save = { player: 'Ondra', level: 1, coins: 0, savedAt: '2026-09-14T08:00:00' };
saveSlot('slot-5', save);
assert.equal(typeof savedGames['slot-5'], 'string', "po saveSlot('slot-5', …) má být savedGames['slot-5'] text");
assert.deepEqual(JSON.parse(savedGames['slot-5']), save, "savedGames['slot-5'] má po načtení obsahovat Ondrovu pozici");
assert.equal(loader.busy, false, 'po uložení má být loader.busy false');
```

Neplatnou pozici `saveSlot` neuloží: vyhodí chybu a slot zůstane beze změny.

```js
const before = savedGames['slot-1'];
assert.throws(() => saveSlot('slot-1', { player: 'Eliška', level: 99, coins: 340 }), "saveSlot('slot-1', pozice s úrovní 99) má vyhodit chybu");
assert.equal(savedGames['slot-1'], before, "po neplatném uložení má slot-1 zůstat beze změny");
```

Po chybě je `loader.busy` zase `false`.

```js
try {
  saveSlot('slot-6', { player: '', level: 3, coins: 10 });
} catch {
  // chyba je tu v pořádku, testujeme úklid
}
assert.equal(loader.busy, false, "po chybě v saveSlot('slot-6', pozice bez jména) má být loader.busy false");
assert.equal(Object.hasOwn(savedGames, 'slot-6'), false, 'neplatná pozice nemá v savedGames založit slot-6');
```

# --help--

## --tip--

Úklid, který proběhne vždycky, patří do `finally` — viz [`finally`: úklid, ať to dopadne jakkoli](see:js-chyby-ladeni/vyjimky#finally-uklid-at-to-dopadne-jakkoli).

## --tip--

Zapnutí ikony je před `try`, kontrola a zápis do `savedGames[slotId]` uvnitř `try` (kontrola první), vypnutí ve `finally`. `catch` tu není.

```js
// Uložené pozice hry Farma Lísková. Klíč je slot, hodnota text ve formátu JSON —
// přesně tak, jak hra pozice ukládá do úložiště prohlížeče.
const savedGames = {
  'slot-1': '{"player":"Eliška","level":12,"coins":340,"savedAt":"2026-09-12T18:40:00"}',
  'slot-2': '{"player":"Tonda","level":3,"coins":',
  'slot-3': '{"player":"Mája","level":0,"coins":25,"savedAt":"2026-09-01T09:15:00"}',
  'slot-4': '{"player":"Radek","level":7,"coins":-20,"savedAt":"2026-09-11T16:30:00"}',
  autosave: '{"player":"Eliška","level":12,"coins":355,"savedAt":"2026-09-13T21:05:00"}',
};

// Když je loader.busy true, hra ukazuje točící se ikonu nahrávání.
const loader = { busy: false };

// ----- Od kolegy -----

// Zkontroluje počet mincí v uložené hře a vrátí ho.
function checkCoins(coins) {
  if (!Number.isInteger(coins) || coins < 0) {
    throw 'Počet mincí musí být nezáporné celé číslo';
  }
  return coins;
}

// Zkopíruje pozici do zálohy (slot-1 → slot-1-backup).
// Vrátí true, když se záloha povedla, jinak false.
function backupSlot(slotId) {
  try {
    const save = validateSave(readSave(savedGames[slotId]));
    savedGames[`${slotId}-backup`] = JSON.stringify(save);
    return true;
  } catch {
    return false;
  } finally {
    // ať funkce vždycky něco vrátí
    return true;
  }
}

// ----- Tvoje funkce -----

// Z textu uložené pozice vrátí objekt hry.
function readSave(text) {
  return JSON.parse(text);
}

console.log(readSave(savedGames['slot-1']));

// Zkontroluje úroveň hráče (celé číslo 1–50) a vrátí ji.
function checkLevel(level) {
  if (!Number.isInteger(level) || level < 1 || level > 50) {
    throw new RangeError(`Úroveň musí být celé číslo 1–50, ne ${level}`);
  }
  return level;
}

// Zkontroluje celou uloženou hru a vrátí ji.
function validateSave(save) {
  if (typeof save.player !== 'string' || save.player.trim() === '') {
    throw new TypeError('Uložená hra nemá jméno hráče');
  }
  checkLevel(save.level);
  checkCoins(save.coins);
  return save;
}

// Vrátí uloženou hru, nebo null, když ji nejde načíst.
function tryReadSave(text) {
  try {
    return validateSave(readSave(text));
  } catch {
    return null;
  }
}

console.log('slot-2:', tryReadSave(savedGames['slot-2']));

// Popisek pozice do menu, nebo důvod, proč ji nejde načíst.
function slotLabel(slotId) {
  try {
    const save = validateSave(readSave(savedGames[slotId]));
    return `${save.player} · úroveň ${save.level} · ${save.coins} mincí`;
  } catch (error) {
    return `Nejde načíst: ${error.message}`;
  }
}

console.log(slotLabel('slot-1'));
console.log(slotLabel('slot-3'));

function loadSlot(slotId) {
  loader.busy = true;
  try {
    return validateSave(readSave(savedGames[slotId]));
  } finally {
    loader.busy = false;
  }
}
```
