# --description--

## Hlášení

Po opravě `checkField` už objednávky s `NaN` neprocházejí. Místo toho `submitOrder({ ...formData, coupon: 'kava10' })` spadne s hláškou:

`TypeError: code.trim(...).toUppercase is not a function`

Tuhle chybu `checkField` celou dobu schovával. Objednávky bez slevového kódu fungují.

## Úkol

Oprav kolegovu funkci `validateCoupon`, aby slevový kód fungoval bez ohledu na velká a malá písmena a mezery okolo. Změň co nejmenší kus kódu.

# --hints--

`validateCoupon` vrátí slevu pro známý kód psaný malými písmeny i s mezerami.

```js
assert.equal(validateCoupon('kava10'), 10, "validateCoupon('kava10') má vrátit 10");
assert.equal(validateCoupon(' Letna15 '), 15, "validateCoupon(' Letna15 ') má vrátit 15");
```

Neznámý kód dál vyhodí `ValidationError` s políčkem `coupon`, prázdný kód dá slevu 0.

```js
let error;
try {
  validateCoupon('zdarma');
} catch (caught) {
  error = caught;
}
assert.ok(error instanceof ValidationError, "validateCoupon('zdarma') má vyhodit ValidationError");
assert.equal(error.field, 'coupon', "chyba z validateCoupon('zdarma') má mít field 'coupon'");
assert.equal(validateCoupon(''), 0, "validateCoupon('') má vrátit 0");
assert.equal(validateCoupon(undefined), 0, 'validateCoupon(undefined) má vrátit 0');
```

Objednávka se slevovým kódem má správnou cenu.

```js
const result = submitOrder({ ...formData, coupon: 'letna15' });
assert.equal(result.ok, true, "submitOrder s kódem 'letna15' má vrátit ok: true");
assert.equal(result.order.total, 841, "submitOrder s kódem 'letna15' má vrátit total 841 — 897 Kč − 15 % = 762 Kč, plus 79 Kč doprava");
```

# --help--

## --tip--

Tohle je chyba programu, ne zákazníka — viz [Chyba uživatele a chyba programu](see:js-chyby-ladeni/vyjimky#chyba-uzivatele-a-chyba-programu). Hláška říká přesně, co není funkce.

## --tip--

Podívej se pozorně na jméno metody na řádku s `coupons[…]`. Jak se přesně jmenuje metoda řetězce, která převádí na velká písmena?

## --tip--

Stejný druh chyby na jiných datech: `name.toLowercase()` skončí hláškou `name.toLowercase is not a function`, protože metoda se jmenuje `toLowerCase`.

// Katalog: kód produktu → název a cena za balení v Kč.
const catalog = {
  'ETH-250': { name: 'Etiopie Yirgacheffe 250 g', price: 289 },
  'COL-250': { name: 'Kolumbie Huila 250 g', price: 259 },
  'KEN-250': { name: 'Keňa Nyeri 250 g', price: 319 },
  'BRA-1000': { name: 'Brazílie Santos 1 kg', price: 749 },
};

// Způsoby dopravy a jejich cena v Kč.
const shippingOptions = [
  { id: 'zasilkovna', label: 'Zásilkovna', price: 79 },
  { id: 'ppl', label: 'PPL na adresu', price: 129 },
  { id: 'osobne', label: 'Osobní odběr v pražírně', price: 0 },
];

// Slevové kódy: kód → sleva v procentech.
const coupons = { KAVA10: 10, LETNA15: 15 };

// Pořadí políček ve formuláři a jejich popisky (použiješ na konci workshopu).
const fieldOrder = ['name', 'email', 'street', 'city', 'zip', 'shipping', 'coupon', 'items'];
const fieldLabels = {
  name: 'Jméno',
  email: 'E-mail',
  street: 'Ulice',
  city: 'Město',
  zip: 'PSČ',
  shipping: 'Doprava',
  coupon: 'Slevový kód',
  items: 'Košík',
  form: 'Objednávka',
};

// Data z formuláře objednávky tak, jak je poslal prohlížeč: všechno jsou texty.
const formData = {
  name: '  Jana   Nováková ',
  email: ' Jana.Novakova@Example.cz',
  street: 'Korunní 12',
  city: 'Praha',
  zip: '120 00',
  shipping: 'zasilkovna',
  coupon: '',
  items: [
    { sku: 'ETH-250', quantity: '2' },
    { sku: 'KEN-250', quantity: '1' },
  ],
};

// Spustí kontrolu jednoho políčka. Když kontrola vyhodí chybu políčka,
// zapíše její zprávu do fieldErrors pod jméno políčka a vrátí undefined.
function checkField(fieldErrors, validate) {
  try {
    return validate();
  } catch (error) {
    if (!(error instanceof ValidationError)) {
      throw error;
    }
    fieldErrors[error.field] = error.message;
  }
}

function validateCoupon(code) {

// Ořízne mezery na okrajích a víc mezer uvnitř nahradí jednou.
function normalizeSpaces(text) {
  return text.trim().replace(/\s+/g, ' ');
}

console.log(normalizeSpaces(formData.name));

class ValidationError extends Error {
  constructor(field, message, options) {
    super(message, options);
    this.name = 'ValidationError';
    this.field = field;
  }
}

// Produkt, který v katalogu není.
class NotFoundError extends Error {
  constructor(sku, options) {
    super(`Produkt ${sku} v katalogu není`, options);
    this.name = 'NotFoundError';
    this.sku = sku;
  }
}

console.log(`${new NotFoundError('GUA-250')}`);

// Povinné textové políčko: vrátí text bez zbytečných mezer, prázdné vyhodí ValidationError.
function requireText(value, field) {
  if (typeof value !== 'string' || value.trim() === '') {
    throw new ValidationError(field, 'Vyplň toto políčko');
  }
  return normalizeSpaces(value);
}

console.log(requireText(formData.street, 'street'));

// E-mail: povinný, ve tvaru jana@example.cz; vrátí ho malými písmeny.
function validateEmail(value) {
  const email = requireText(value, 'email').toLowerCase();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    throw new ValidationError('email', 'Zadej e-mail ve tvaru jana@example.cz');
  }
  return email;
}

console.log(validateEmail(formData.email));

// PSČ: pět číslic, mezery jsou dovolené; vrátí ho ve tvaru „120 00".
function validateZip(value) {
  const digits = requireText(value, 'zip').replaceAll(' ', '');
  if (!/^\d{5}$/.test(digits)) {
    throw new ValidationError('zip', 'PSČ má pět číslic, třeba 120 00');
  }
  return `${digits.slice(0, 3)} ${digits.slice(3)}`;
}

console.log(validateZip('12000'));

// Počet kusů: celé číslo 1–20. Z formuláře přijde text, z mobilní aplikace číslo.
function validateQuantity(value, field) {
  const quantity = Number(value);
  if (!Number.isInteger(quantity) || quantity < 1 || quantity > 20) {
    throw new ValidationError(field, 'Zadej počet kusů od 1 do 20');
  }
  return quantity;
}

console.log(validateQuantity(formData.items[0].quantity, 'items.0.quantity'));

// Produkt z katalogu; neznámý kód vyhodí NotFoundError.
function findProduct(sku) {
  if (!Object.hasOwn(catalog, sku)) {
    throw new NotFoundError(sku);
  }
  return catalog[sku];
}

console.log(findProduct('KEN-250'));

// Způsob dopravy z nabídky; neznámý vyhodí ValidationError.
function validateShipping(id) {
  const option = shippingOptions.find((option) => option.id === id);
  if (option === undefined) {
    throw new ValidationError('shipping', 'Vyber způsob dopravy');
  }
  return option;
}

console.log(validateShipping(formData.shipping));

// Kontaktní údaje. Chyby políček zapíše do fieldErrors.
function validateContact(data, fieldErrors) {
  return {
    name: checkField(fieldErrors, () => requireText(data.name, 'name')),
    email: checkField(fieldErrors, () => validateEmail(data.email)),
  };
}

const contactErrors = {};
console.log(validateContact({ name: 'Jana', email: 'jana.example.cz' }, contactErrors), contactErrors);

// Doručovací adresa. Chyby políček zapíše do fieldErrors.
function validateAddress(data, fieldErrors) {
  return {
    street: checkField(fieldErrors, () => requireText(data.street, 'street')),
    city: checkField(fieldErrors, () => requireText(data.city, 'city')),
    zip: checkField(fieldErrors, () => validateZip(data.zip)),
  };
}

console.log(validateAddress(formData, {}));

// Řádek objednávky z položky košíku.
function toOrderLine(item, index) {
  let product;
  try {
    product = findProduct(item.sku);
  } catch (error) {
    if (error instanceof NotFoundError) {
      throw new ValidationError(`items.${index}.sku`, 'Tenhle produkt už nenabízíme', { cause: error });
    }
    throw error;
  }
  const quantity = validateQuantity(item.quantity, `items.${index}.quantity`);
  return { sku: item.sku, name: product.name, price: product.price, quantity };
}

console.log(toOrderLine(formData.items[1], 1));

// Položky košíku. Prázdný košík je chyba políčka items.
function validateItems(items, fieldErrors) {
  if (!Array.isArray(items) || items.length === 0) {
    fieldErrors.items = 'Košík je prázdný';
    return [];
  }
  return items
    .map((item, index) => checkField(fieldErrors, () => toOrderLine(item, index)))
    .filter((line) => line !== undefined);
}

const itemErrors = {};
console.log(validateItems([{ sku: 'ETH-250', quantity: '2' }, { sku: 'GUA-250', quantity: '1' }], itemErrors), itemErrors);

// Cena objednávky v Kč: zboží se slevou (zaokrouhlené na koruny) plus doprava.
function orderTotal(lines, shippingPrice, discountPercent) {
  const goods = lines.reduce((sum, line) => sum + line.price * line.quantity, 0);
  return Math.round(goods * (1 - discountPercent / 100)) + shippingPrice;
}

console.log(orderTotal([{ price: 289, quantity: 2 }], 79, 10));

// Odeslání objednávky: buď { ok: true, order }, nebo { ok: false, fieldErrors }.
function submitOrder(data) {
  const fieldErrors = {};
  const contact = validateContact(data, fieldErrors);
  const address = validateAddress(data, fieldErrors);
  const shipping = checkField(fieldErrors, () => validateShipping(data.shipping));
  const discount = checkField(fieldErrors, () => validateCoupon(data.coupon));
  const lines = validateItems(data.items, fieldErrors);

  if (Object.keys(fieldErrors).length > 0) {
    return { ok: false, fieldErrors };
  }
  const total = orderTotal(lines, shipping.price, discount);
  return { ok: true, order: { ...contact, address, shipping: shipping.id, lines, total } };
}

console.log(submitOrder(formData));
