# --description--

Počet kusů přijde z formuláře jako text (`'2'`), z mobilní aplikace jako číslo (`2`). Pražírna posílá nejvýš 20 balení jednoho produktu.

**Úkol:** Napiš funkci `validateQuantity(value, field)`, která hodnotu převede na číslo přes `Number`. Když výsledek není celé číslo od 1 do 20, vyhodí `ValidationError` s políčkem `field` a zprávou `Zadej počet kusů od 1 do 20`. Jinak vrátí číslo. Pod ní vypiš počet kusů první položky z `formData` s políčkem `'items.0.quantity'`.

Jméno políčka dostává funkce v parametru, protože položek v košíku je víc: první má políčko `items.0.quantity`, druhá `items.1.quantity`.

Pozor na převod: `Number('')` je `0` a `Number('dva')` je `NaN`. Obojí musí kontrolou neprojít. Na jiných datech:

```js
const rating = Number(value);
if (!Number.isInteger(rating) || rating < 1 || rating > 5) {
  throw new ValidationError(field, 'Hodnocení je 1 až 5 hvězdiček');
}
```

# --hints--

`validateQuantity` vrátí číslo z textu i z čísla.

```js
assert.equal(validateQuantity('2', 'items.0.quantity'), 2, "validateQuantity('2', 'items.0.quantity') má vrátit číslo 2");
assert.equal(validateQuantity(20, 'items.1.quantity'), 20, "validateQuantity(20, 'items.1.quantity') má vrátit 20");
assert.equal(validateQuantity(' 1 ', 'items.0.quantity'), 1, "validateQuantity(' 1 ', 'items.0.quantity') má vrátit 1");
```

Hodnota, která není celé číslo od 1 do 20, vyhodí `ValidationError` s předaným políčkem.

```js
for (const value of ['0', '21', '2.5', 'dva', '', -1]) {
  let error;
  try {
    validateQuantity(value, 'items.3.quantity');
  } catch (caught) {
    error = caught;
  }
  assert.ok(error instanceof ValidationError, `validateQuantity(${JSON.stringify(value)}, 'items.3.quantity') má vyhodit ValidationError`);
  assert.equal(error.field, 'items.3.quantity', `chyba z validateQuantity(${JSON.stringify(value)}, 'items.3.quantity') má mít field 'items.3.quantity'`);
  assert.equal(error.message, 'Zadej počet kusů od 1 do 20', `chyba z validateQuantity(${JSON.stringify(value)}, …) má mít zprávu 'Zadej počet kusů od 1 do 20'`);
}
```

# --help--

## --tip--

Převod a kontrola na hranici — viz [Validace na hranici](see:js-chyby-ladeni/vlastni-chyby#validace-na-hranici).

## --tip--

Po `Number(value)` rozhodne `Number.isInteger` o celém čísle i o `NaN`. Prázdný text dá nulu a tu chytí dolní hranice.

// Katalog: kód produktu → název a cena za balení v Kč.
const catalog = {
  'ETH-250': { name: 'Etiopie Yirgacheffe 250 g', price: 289 },
  'COL-250': { name: 'Kolumbie Huila 250 g', price: 259 },
  'KEN-250': { name: 'Keňa Nyeri 250 g', price: 319 },
  'BRA-1000': { name: 'Brazílie Santos 1 kg', price: 749 },

// Způsoby dopravy a jejich cena v Kč.
const shippingOptions = [
  { id: 'zasilkovna', label: 'Zásilkovna', price: 79 },
  { id: 'ppl', label: 'PPL na adresu', price: 129 },
  { id: 'osobne', label: 'Osobní odběr v pražírně', price: 0 },
];

// Slevové kódy: kód → sleva v procentech.
const coupons = { KAVA10: 10, LETNA15: 15 };

// Pořadí políček ve formuláři a jejich popisky (použiješ na konci workshopu).
const fieldOrder = ['name', 'email', 'street', 'city', 'zip', 'shipping', 'coupon', 'items'];
const fieldLabels = {
  name: 'Jméno',
  email: 'E-mail',
  street: 'Ulice',
  city: 'Město',
  zip: 'PSČ',
  shipping: 'Doprava',
  coupon: 'Slevový kód',
  items: 'Košík',
  form: 'Objednávka',
};

// Data z formuláře objednávky tak, jak je poslal prohlížeč: všechno jsou texty.
const formData = {
  name: '  Jana   Nováková ',
  email: ' Jana.Novakova@Example.cz',
  street: 'Korunní 12',
  city: 'Praha',
  zip: '120 00',
  shipping: 'zasilkovna',
  coupon: '',
  items: [
    { sku: 'ETH-250', quantity: '2' },
    { sku: 'KEN-250', quantity: '1' },
  ],
};

// ----- Od kolegy -----

// Spustí kontrolu jednoho políčka. Když kontrola vyhodí chybu políčka,
// zapíše její zprávu do fieldErrors pod jméno políčka a vrátí undefined.
function checkField(fieldErrors, validate) {
  try {
    return validate();
  } catch (error) {
    if (error.field) {
      fieldErrors[error.field] = error.message;
    }
  }
}

// Sleva v procentech podle slevového kódu. Prázdný kód znamená žádnou slevu.
function validateCoupon(code) {
  if (code === undefined || code.trim() === '') {
    return 0;
  }
  const discount = coupons[code.trim().toUppercase()];
  if (discount === undefined) {
    throw new ValidationError('coupon', 'Tenhle slevový kód neznáme');
  }
  return discount;
}

// ----- Tvoje funkce -----

// Ořízne mezery na okrajích a víc mezer uvnitř nahradí jednou.
function normalizeSpaces(text) {
  return text.trim().replace(/\s+/g, ' ');
}

console.log(normalizeSpaces(formData.name));

class ValidationError extends Error {
  constructor(field, message, options) {
    super(message, options);
    this.name = 'ValidationError';
    this.field = field;
  }
}

// Produkt, který v katalogu není.
class NotFoundError extends Error {
  constructor(sku, options) {
    super(`Produkt ${sku} v katalogu není`, options);
    this.name = 'NotFoundError';
    this.sku = sku;
  }
}

console.log(`${new NotFoundError('GUA-250')}`);

// Povinné textové políčko: vrátí text bez zbytečných mezer, prázdné vyhodí ValidationError.
function requireText(value, field) {
  if (typeof value !== 'string' || value.trim() === '') {
    throw new ValidationError(field, 'Vyplň toto políčko');
  }
  return normalizeSpaces(value);
}

console.log(requireText(formData.street, 'street'));

// E-mail: povinný, ve tvaru jana@example.cz; vrátí ho malými písmeny.
function validateEmail(value) {
  const email = requireText(value, 'email').toLowerCase();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    throw new ValidationError('email', 'Zadej e-mail ve tvaru jana@example.cz');
  }
  return email;
}

console.log(validateEmail(formData.email));

// PSČ: pět číslic, mezery jsou dovolené; vrátí ho ve tvaru „120 00".
function validateZip(value) {
  const digits = requireText(value, 'zip').replaceAll(' ', '');
  if (!/^\d{5}$/.test(digits)) {
    throw new ValidationError('zip', 'PSČ má pět číslic, třeba 120 00');
  }
  return `${digits.slice(0, 3)} ${digits.slice(3)}`;
}

console.log(validateZip('12000'));
