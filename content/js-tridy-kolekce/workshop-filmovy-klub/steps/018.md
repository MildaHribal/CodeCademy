# --description--

Doporučovač je hotový, teď ho uvidíš. V kroku přibyly soubory `index.html` a `styles.css` se stránkou klubu. Na konci `index.html` je hotové ovládání: skript vytvoří `club` z pole `ratings`, naplní výběr členů z `club.members` a po každé změně výběru zavolá `renderRecommendations(club, member)`. Posluchače událostí se naučíš v sekci o DOM.

**Úkol:** Na konec `script.js` napiš funkci `renderRecommendations(club, member)`. Najde člena s nejpodobnějším vkusem, jeho jméno napíše do `#friend` a do seznamu `#recommendations` vypíše jednu položku `li` za každý film, který mu ten kamarád doporučuje, s názvem filmu jako textem. Když kamarád není (klub s jediným členem), do `#friend` napíše pomlčku `–` a seznam vyprázdní. Další zavolání seznam nahradí.

Na výpis pole do seznamu máš nástroje z nákupního seznamu i z bankovního výpisu. Doporučení jsou množina — z té nejdřív udělej pole. Až to bude fungovat, přepínej v náhledu členy a sleduj, jak se doporučení mění.

# --hints--

Po načtení stránky je v `#friend` nejpodobnější člen prvního člena ve výběru.

```js
const member = document.querySelector('#member').value;
const friend = club.mostSimilarTo(member);
assert.equal(document.querySelector('#friend').textContent.trim(), friend, `#friend má po načtení ukazovat „${friend}" — nejpodobnějšího člena pro ${member}`);
```

Po načtení stránky jsou v `#recommendations` filmy, které kamarád doporučuje.

```js
const member = document.querySelector('#member').value;
const expected = [...club.recommendFor(member, club.mostSimilarTo(member))];
const rows = [...document.querySelectorAll('#recommendations li')];
assert.equal(rows.length, expected.length, `v #recommendations má být ${expected.length} filmů`);
assert.deepEqual(rows.map((row) => row.textContent.trim()).sort(), [...expected].sort(), `#recommendations má obsahovat filmy ${expected.join(', ')}`);
```

Další zavolání s jiným klubem seznam i kamaráda nahradí; bez kamaráda je v `#friend` pomlčka a seznam je prázdný.

```js
const small = new FilmClub([
  { member: 'Eva', film: 'Pelíšky', stars: 5 },
  { member: 'Jan', film: 'Pelíšky', stars: 4 },
  { member: 'Jan', film: 'Kolja', stars: 4 },
]);
renderRecommendations(small, 'Eva');
assert.equal(document.querySelector('#friend').textContent.trim(), 'Jan', "po renderRecommendations(malý klub, 'Eva') má #friend ukazovat 'Jan'");
const rows = [...document.querySelectorAll('#recommendations li')];
assert.deepEqual(rows.map((row) => row.textContent.trim()), ['Kolja'], "po renderRecommendations(malý klub, 'Eva') má seznam obsahovat jen 'Kolja'");
const solo = new FilmClub([{ member: 'Eva', film: 'Pelíšky', stars: 5 }]);
renderRecommendations(solo, 'Eva');
assert.equal(document.querySelector('#friend').textContent.trim(), '–', "v klubu jen s Evou má #friend ukazovat pomlčku '–'");
assert.equal(document.querySelectorAll('#recommendations li').length, 0, 'v klubu jen s Evou má být seznam doporučení prázdný');
```

## --tip--

Výpis pole do seznamu jsi psal v [nákupním seznamu](see:js-pole/workshop-nakupni-seznam/032) i v bankovním výpisu.

## --tip--

Kamaráda ti dá `club.mostSimilarTo(member)`, doporučení `club.recommendFor(member, friend)`. Když je kamarád `null`, pracuj s prázdným polem filmů; jinak množinu rozprostři do pole, přes `map` z ní vyrob prvky `li` a předej je do `replaceChildren`.

const ratings = [
  { member: 'Eva', film: 'Pelíšky', stars: 5 },
  { member: 'Eva', film: 'Kolja', stars: 4 },
  { member: 'Eva', film: 'Vratné lahve', stars: 3 },
  { member: 'Eva', film: 'Samotáři', stars: 4 },
  { member: 'Jan', film: 'Pelíšky', stars: 4 },
  { member: 'Jan', film: 'Samotáři', stars: 5 },
  { member: 'Jan', film: 'Kulový blesk', stars: 5 },
  { member: 'Jan', film: 'Černí baroni', stars: 3 },
  { member: 'Bára', film: 'Kolja', stars: 5 },
  { member: 'Bára', film: 'Pelíšky', stars: 5 },
  { member: 'Bára', film: 'Obecná škola', stars: 4 },
  { member: 'Bára', film: 'Vratné lahve', stars: 4 },
  { member: 'Ondřej', film: 'Kulový blesk', stars: 4 },
  { member: 'Ondřej', film: 'Černí baroni', stars: 5 },
  { member: 'Ondřej', film: 'Samotáři', stars: 3 },
  { member: 'Ondřej', film: 'Pelíšky', stars: 3 },
];

// Názvy filmů ze všech hodnocení, v pořadí hodnocení.
function filmTitles(ratings) {
  return ratings.map((rating) => rating.film);
}

// Množina všech hodnocených filmů, každý jednou.
function uniqueFilms(ratings) {
  return new Set(filmTitles(ratings));
}

// Kolik členů který film hodnotilo.
function countViews(ratings) {
  const counts = new Map();
  for (const { film } of ratings) {
    counts.set(film, (counts.get(film) ?? 0) + 1);
  }
  return counts;
}

function addToGroup(groups, key, value) {
  if (!groups.has(key)) {
    groups.set(key, new Set());
  }
  groups.get(key).add(value);
}

class FilmClub {
  #ratings = [];
  #watched = new Map();
  #stars = new Map();

  constructor(ratings) {
    this.#ratings = [...ratings];
    for (const { member, film, stars } of ratings) {
      addToGroup(this.#watched, member, film);
      this.#stars.set(film, [...(this.#stars.get(film) ?? []), stars]);
    }
  }

  // Počet hodnocení v klubu.
  get size() {
    return this.#ratings.length;
  }

  // Jména členů v pořadí, v jakém poprvé hodnotili.
  get members() {
    return [...this.#watched.keys()];
  }

  // Všechny filmy, které kdo v klubu viděl.
  get allFilms() {
    return new Set(this.#ratings.map((rating) => rating.film));
  }

  // Kopie množiny filmů, které člen viděl; neznámý člen nic neviděl.
  watchedBy(member) {
    return new Set(this.#watched.get(member));
  }

  // Od kolegy: viděl člen tenhle film?
  hasSeen(member, film) {
    return this.#watched.get(member)?.has(film) ?? false;
  }

  // Filmy, které viděli oba členové.
  commonFilms(first, second) {
    return this.watchedBy(first).intersection(this.watchedBy(second));
  }

  // Filmy, které viděl kamarád a člen zatím ne.
  recommendFor(member, friend) {
    return this.watchedBy(friend).difference(this.watchedBy(member));
  }

  // Filmy, které viděli úplně všichni členové.
  seenByEveryone() {
    return this.members
      .map((member) => this.watchedBy(member))
      .reduce((common, films) => common.intersection(films), this.allFilms);
  }

  // Průměrné hodnocení filmu na jedno desetinné místo; neznámý film null.
  averageRating(film) {
    const stars = this.#stars.get(film);
    if (!stars) {
      return null;
    }
    const sum = stars.reduce((total, value) => total + value, 0);
    return Math.round((sum / stars.length) * 10) / 10;
  }

  // Nejlépe hodnocené filmy: od nejvyššího průměru, při shodě podle názvu.
  topFilms(count) {
    return [...this.allFilms]
      .map((film) => ({ film, average: this.averageRating(film) }))
      .toSorted((a, b) => b.average - a.average || a.film.localeCompare(b.film, 'cs'))
      .slice(0, count);
  }

  // Podobnost vkusu dvou členů: společné filmy / všechny jejich filmy, na dvě místa.
  similarity(first, second) {
    const a = this.watchedBy(first);
    const b = this.watchedBy(second);
    const all = a.union(b);
    if (all.size === 0) {
      return 0;
    }
    return Math.round((a.intersection(b).size / all.size) * 100) / 100;
  }

  // Od kolegy: filmy z klubu, které člen ještě neviděl.
  unseenBy(member) {
    return this.allFilms.difference(this.watchedBy(member));
  }

  // Člen s nejpodobnějším vkusem; při shodě ten, kdo hodnotil dřív.
  mostSimilarTo(member) {
    let best = null;
    let bestScore = -1;
    for (const other of this.members) {
      const score = this.similarity(member, other);
      if (other !== member && score > bestScore) {
        best = other;
        bestScore = score;
      }
    }
    return best;
  }
}

function renderRecommendations(club, member) {
  const friend = club.mostSimilarTo(member);
  document.querySelector('#friend').textContent = friend ?? '–';
  const films = friend ? [...club.recommendFor(member, friend)] : [];
  const items = films.map((film) => {
    const item = document.createElement('li');
