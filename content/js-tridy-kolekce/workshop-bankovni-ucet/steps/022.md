# --description--

Model účtu je hotový, teď ho uvidíš. V kroku přibyly soubory `index.html` a `styles.css` se stránkou výpisu z účtu. Náhled teď ukazuje stránku místo konzole.

Na konci `index.html` je hotové ovládání: skript vytvoří spořicí účet Evy Dvořákové, provede na něm pár pohybů a tlačítka **Vložit**, **Vybrat** a **Připsat úrok** volají tvoje metody. Chybu z účtu (třeba „Nedostatek prostředků") ukáže pod tlačítky. Po každé akci zavolá funkci `renderStatement(account)`, kterou napíšeš ty. Posluchače událostí se naučíš v sekci o DOM.

**Úkol:** Na konec `script.js` napiš funkci `renderStatement(account)`, která do prvku `#owner` napíše majitele a do `#balance` zůstatek ve tvaru `36543 Kč`.

Prvek najde `document.querySelector('#id')` a text do něj nastavíš přes `textContent`, stejně jako ve výpisu nákupního seznamu. Až to bude fungovat, zkus v náhledu vložit, vybrat víc, než je na účtu, nebo vybrat počtvrté.

# --hints--

Po načtení stránky je v `#owner` jméno majitele účtu.

```js
assert.equal(document.querySelector('#owner').textContent.trim(), account.owner, `#owner má po načtení ukazovat „${account.owner}" — napsal jsi renderStatement?`);
```

Po načtení stránky je v `#balance` zůstatek ve tvaru `36543 Kč`.

```js
assert.equal(document.querySelector('#balance').textContent.trim(), `${account.balance} Kč`, `#balance má po načtení ukazovat „${account.balance} Kč"`);
```

Když `renderStatement` zavoláš s jiným účtem, přepíše majitele i zůstatek.

```js
const other = BankAccount.open('Jan Novák', 750);
renderStatement(other);
assert.equal(document.querySelector('#owner').textContent.trim(), 'Jan Novák', "po renderStatement(účet Jana Nováka) má #owner ukazovat 'Jan Novák'");
assert.equal(document.querySelector('#balance').textContent.trim(), '750 Kč', "po renderStatement(účet s 750 Kč) má #balance ukazovat '750 Kč'");
```

// Bankovní účet. Všechny částky jsou v celých korunách.

// Vrátí funkci, která při každém zavolání vydá další id: TX-1, TX-2…
function createIdGenerator(prefix) {
  let count = 0;
  return () => {
    count += 1;
    return `${prefix}-${count}`;
  };
}

class BankAccount {
  #balance = 0;
  #transactions = [];
  #nextId = createIdGenerator('TX');

  // Založí účet a rovnou na něj vloží první částku.
  static open(owner, initialDeposit) {
    const account = new BankAccount(owner);
    account.deposit(initialDeposit);
    return account;
  }

  // Obnoví účet z uložených pohybů, třeba z databáze.
  static fromTransactions(owner, transactions) {
    const account = new BankAccount(owner);
    for (const { type, amount } of transactions) {
      if (type === 'deposit') {
        account.deposit(amount);
      } else {
        account.withdraw(amount);
      }
    }
    return account;
  }

  // Součet zůstatků všech účtů.
  static totalBalance(accounts) {
    return accounts.reduce((sum, account) => sum + account.balance, 0);
  }

  constructor(owner) {
    this.owner = owner;
  }

  get balance() {
    return this.#balance;
  }

  // Kopie historie: kdo ji dostane, nezmění pohyby uvnitř účtu.
  get transactions() {
    return [...this.#transactions];
  }

  // Kolik se na účet celkem vložilo.
  get totalDeposited() {
    return this.#transactions
      .filter((transaction) => transaction.type === 'deposit')
      .reduce((sum, transaction) => sum + transaction.amount, 0);
  }

  deposit(amount) {
    this.#assertPositive(amount);
    this.#balance += amount;
    this.#record('deposit', amount);
  }

  withdraw(amount) {
    this.#assertPositive(amount);
    if (amount > this.#balance) {
      throw new RangeError('Nedostatek prostředků');
    }
    this.#balance -= amount;
    this.#record('withdrawal', amount);
  }

  // Od kolegy: vloží víc částek najednou, třeba výplatu a vratku.
  depositAll(amounts) {
    amounts.forEach((amount) => {
      this.deposit(amount);
    });
  }

  // Popisek účtu, když ho vložíš do textu.
  toString() {
    return `${this.owner}: ${this.#balance} Kč`;
  }

  #assertPositive(amount) {
    if (!Number.isFinite(amount) || amount <= 0) {
      throw new RangeError('Částka musí být kladné číslo');
    }
  }

  #record(type, amount) {
    this.#transactions.push({ id: this.#nextId(), type, amount });
  }
}

class SavingsAccount extends BankAccount {
  #withdrawalsLeft = 3;

  constructor(owner, rate) {
    super(owner);
    this.rate = rate;
  }

  // Připíše úrok z aktuálního zůstatku, zaokrouhlený na celé koruny.
  addInterest() {
    const interest = Math.round(this.balance * this.rate);
    if (interest > 0) {
      this.deposit(interest);
    }
    return interest;
  }

  // Ze spořicího účtu jdou jen tři výběry.
  withdraw(amount) {
    if (this.#withdrawalsLeft === 0) {
      throw new Error('Limit výběrů vyčerpán');
    }
    super.withdraw(amount);
    this.#withdrawalsLeft -= 1;
  }
}

// Od kolegy: firemní účet s názvem firmy.
class BusinessAccount extends BankAccount {
  constructor(owner, company) {
    super(owner);
    this.company = company;
  }
}

<!DOCTYPE html>
<html lang="cs">
  <head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>Výpis z účtu</title>
    <link rel="stylesheet" href="styles.css">
  </head>
  <body>
    <main class="statement">
      <section class="card" aria-labelledby="owner">
        <p class="card__type" id="account-type">–</p>
        <h1 class="card__owner" id="owner">–</h1>
        <p class="card__label">Zůstatek</p>
        <p class="card__balance" id="balance">–</p>
      </section>

      <form class="actions" id="actions">
        <label class="actions__field">
          <span>Částka v Kč</span>
          <input id="amount" type="number" step="1" value="500">
        </label>
        <div class="actions__buttons">
          <button type="submit" value="deposit">Vložit</button>
          <button type="submit" value="withdraw">Vybrat</button>
          <button type="button" id="interest">Připsat úrok</button>
        </div>
        <p class="actions__message" id="message" role="status"></p>
      </form>

      <h2>Pohyby</h2>
      <ul class="transactions" id="transactions"></ul>
    </main>

    <script src="script.js"></script>
    <script>
      // Ovládání stránky je hotové. Posluchače událostí se naučíš v sekci o DOM.
      const account = new SavingsAccount('Eva Dvořáková', 0.02);
      account.depositAll([32000, 4500]);
      account.withdraw(1200);
      account.addInterest();

      const message = document.querySelector('#message');

      // Provede akci s účtem; chybu z účtu ukáže pod tlačítky.
      function run(action) {
        try {
          action();
          message.textContent = '';
        } catch (error) {
          message.textContent = error.message;
        }
        renderStatement(account);
      }

      document.querySelector('#actions').addEventListener('submit', (event) => {
        event.preventDefault();
        const amount = Number(document.querySelector('#amount').value);
        if (event.submitter.value === 'deposit') {
          run(() => account.deposit(amount));
        } else {
          run(() => account.withdraw(amount));
        }
      });

      document.querySelector('#interest').addEventListener('click', () => {
        run(() => account.addInterest());
      });

      renderStatement(account);
    </script>
  </body>
</html>
