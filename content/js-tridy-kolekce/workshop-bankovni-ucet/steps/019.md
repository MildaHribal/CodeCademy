# --description--

## Hlášení

Kolega na konec souboru přidal třídu `BusinessAccount` pro firemní účty. Založení prvního firemního účtu `new BusinessAccount('Jan Novák', 'Pekárna Novák s.r.o.')` ale skončí chybou:

`ReferenceError: Must call super constructor in derived class before accessing 'this' or returning from derived constructor`

Běžné a spořicí účty fungují.

## Úkol

Oprav konstruktor `BusinessAccount`, aby šel firemní účet založit s majitelem i názvem firmy. Změň co nejmenší kus kódu.

# --hints--

`new BusinessAccount('Jan Novák', 'Pekárna Novák s.r.o.')` vytvoří bankovní účet s majitelem a firmou.

```js
const business = new BusinessAccount('Jan Novák', 'Pekárna Novák s.r.o.');
assert.ok(business instanceof BankAccount, 'firemní účet má být instance BankAccount');
assert.equal(business.owner, 'Jan Novák', "new BusinessAccount('Jan Novák', 'Pekárna Novák s.r.o.').owner má být 'Jan Novák'");
assert.equal(business.company, 'Pekárna Novák s.r.o.', "new BusinessAccount('Jan Novák', 'Pekárna Novák s.r.o.').company má být 'Pekárna Novák s.r.o.'");
```

Firemní účet umí vklad a výběr.

```js
const business = new BusinessAccount('Jan Novák', 'Pekárna Novák s.r.o.');
business.deposit(25000);
business.withdraw(5000);
assert.equal(business.balance, 20000, 'firemní účet má po deposit(25000) a withdraw(5000) mít zůstatek 20000');
```

Spořicí účet funguje dál.

```js
const savings = new SavingsAccount('Eva Dvořáková', 0.02);
savings.deposit(10000);
assert.equal(savings.addInterest(), 200, 'SavingsAccount.addInterest() má dál vracet 200 z 10000 Kč');
```

## --file-- script.js

```js
// Bankovní účet. Všechny částky jsou v celých korunách.

// Vrátí funkci, která při každém zavolání vydá další id: TX-1, TX-2…
function createIdGenerator(prefix) {
  let count = 0;
  return () => {
    count += 1;
    return `${prefix}-${count}`;
  };
}

class BankAccount {
  #balance = 0;
  #transactions = [];
  #nextId = createIdGenerator('TX');

  // Založí účet a rovnou na něj vloží první částku.
  static open(owner, initialDeposit) {
    const account = new BankAccount(owner);
    account.deposit(initialDeposit);
    return account;
  }

  // Obnoví účet z uložených pohybů, třeba z databáze.
  static fromTransactions(owner, transactions) {
    const account = new BankAccount(owner);
    for (const { type, amount } of transactions) {
      if (type === 'deposit') {
        account.deposit(amount);
      } else {
        account.withdraw(amount);
      }
    }
    return account;
  }

  constructor(owner) {
    this.owner = owner;
  }

  get balance() {
    return this.#balance;
  }

  // Kopie historie: kdo ji dostane, nezmění pohyby uvnitř účtu.
  get transactions() {
    return [...this.#transactions];
  }

  // Kolik se na účet celkem vložilo.
  get totalDeposited() {
    return this.#transactions
      .filter((transaction) => transaction.type === 'deposit')
      .reduce((sum, transaction) => sum + transaction.amount, 0);
  }

  deposit(amount) {
    this.#assertPositive(amount);
    this.#balance += amount;
    this.#record('deposit', amount);
  }

  withdraw(amount) {
    this.#assertPositive(amount);
    if (amount > this.#balance) {
      throw new RangeError('Nedostatek prostředků');
    }
    this.#balance -= amount;
    this.#record('withdrawal', amount);
  }

  // Od kolegy: vloží víc částek najednou, třeba výplatu a vratku.
  depositAll(amounts) {
    amounts.forEach((amount) => {
      this.deposit(amount);
    });
  }

  #assertPositive(amount) {
    if (!Number.isFinite(amount) || amount <= 0) {
      throw new RangeError('Částka musí být kladné číslo');
    }
  }

  #record(type, amount) {
    this.#transactions.push({ id: this.#nextId(), type, amount });
  }
}

class SavingsAccount extends BankAccount {
  #withdrawalsLeft = 3;

  constructor(owner, rate) {
    super(owner);
    this.rate = rate;
  }

  // Připíše úrok z aktuálního zůstatku, zaokrouhlený na celé koruny.
  addInterest() {
    const interest = Math.round(this.balance * this.rate);
    if (interest > 0) {
      this.deposit(interest);
    }
    return interest;
  }

  // Ze spořicího účtu jdou jen tři výběry.
  withdraw(amount) {
    if (this.#withdrawalsLeft === 0) {
      throw new Error('Limit výběrů vyčerpán');
    }
    super.withdraw(amount);
    this.#withdrawalsLeft -= 1;
  }
}
--edit--

// Od kolegy: firemní účet s názvem firmy.
class BusinessAccount extends BankAccount {
  constructor(owner, company) {
    this.company = company;
    super(owner);
  }
}
```
