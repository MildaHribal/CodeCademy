# --description--

Ze spořicího účtu jdou jen tři výběry. Běžný účet žádný limit nemá.

**Úkol:** Přepiš v `SavingsAccount` metodu `withdraw(amount)`. Když už proběhly tři výběry, vyhodí `Error` se zprávou `'Limit výběrů vyčerpán'`. Jinak provede výběr **rodičovskou verzí** `withdraw` a teprve po úspěšném výběru si ho započítá. Počet zbývajících výběrů drž v soukromém poli spořicího účtu.

Záleží na pořadí: výběr, který rodič odmítne kvůli nedostatku peněz, se do limitu nepočítá.

# --hints--

Tři výběry projdou, čtvrtý vyhodí `Error` se zprávou `'Limit výběrů vyčerpán'` a zůstatek nezmění.

```js
const savings = new SavingsAccount('Eva Dvořáková', 0.02);
savings.deposit(1000);
savings.withdraw(100);
savings.withdraw(100);
savings.withdraw(100);
assert.throws(() => savings.withdraw(100), { message: 'Limit výběrů vyčerpán' }, "čtvrtý withdraw(100) ze spořicího účtu má vyhodit chybu 'Limit výběrů vyčerpán'");
assert.equal(savings.balance, 700, 'po třech výběrech po 100 a odmítnutém čtvrtém má být zůstatek 700');
```

Výběry ze spořicího účtu se dál zapisují do historie a kontrolují zůstatek.

```js
const savings = new SavingsAccount('Eva Dvořáková', 0.02);
savings.deposit(500);
savings.withdraw(200);
assert.deepEqual(savings.transactions.at(-1), { id: 'TX-2', type: 'withdrawal', amount: 200 }, 'výběr 200 se má zapsat jako pohyb TX-2 — voláš rodičovský withdraw?');
assert.throws(() => savings.withdraw(5000), { name: 'RangeError', message: 'Nedostatek prostředků' }, "withdraw(5000) z účtu s 300 Kč má vyhodit RangeError 'Nedostatek prostředků'");
```

Odmítnutý výběr kvůli nedostatku peněz se do limitu nepočítá.

```js
const savings = new SavingsAccount('Jan Novák', 0.02);
savings.deposit(1000);
try {
  savings.withdraw(5000);
} catch {
  // chyba je tu v pořádku
}
savings.withdraw(100);
savings.withdraw(100);
savings.withdraw(100);
assert.equal(savings.balance, 700, 'po odmítnutém withdraw(5000) mají projít ještě tři výběry — započítáváš výběr až po úspěchu?');
```

Běžný účet žádný limit nemá.

```js
const account = new BankAccount('Jan Novák');
account.deposit(1000);
for (let i = 0; i < 5; i += 1) {
  account.withdraw(100);
}
assert.equal(account.balance, 500, 'z běžného BankAccount má projít pět výběrů po 100');
```

# --help--

## --tip--

Přepsaná metoda může zavolat rodičovskou verzi přes `super` — viz [Dědičnost: `extends` a `super`](see:js-tridy-kolekce/tridy#dedicnost-extends-a-super).

## --tip--

Soukromé pole s počtem zbývajících výběrů patří do těla `SavingsAccount`. V nové `withdraw` nejdřív zkontroluj limit, pak zavolej `super.withdraw(amount)` a až za ním sniž počítadlo.

## --tip--

Na jiných datech: `ChildTicket` s přepsanou metodou `price()` by vracel `super.price() / 2`.

## --file-- script.js

```js
// Bankovní účet. Všechny částky jsou v celých korunách.

// Vrátí funkci, která při každém zavolání vydá další id: TX-1, TX-2…
function createIdGenerator(prefix) {
  let count = 0;
  return () => {
    count += 1;
    return `${prefix}-${count}`;
  };
}

class BankAccount {
  #balance = 0;
  #transactions = [];
  #nextId = createIdGenerator('TX');

  // Založí účet a rovnou na něj vloží první částku.
  static open(owner, initialDeposit) {
    const account = new BankAccount(owner);
    account.deposit(initialDeposit);
    return account;
  }

  // Obnoví účet z uložených pohybů, třeba z databáze.
  static fromTransactions(owner, transactions) {
    const account = new BankAccount(owner);
    for (const { type, amount } of transactions) {
      if (type === 'deposit') {
        account.deposit(amount);
      } else {
        account.withdraw(amount);
      }
    }
    return account;
  }

  constructor(owner) {
    this.owner = owner;
  }

  get balance() {
    return this.#balance;
  }

  // Kopie historie: kdo ji dostane, nezmění pohyby uvnitř účtu.
  get transactions() {
    return [...this.#transactions];
  }

  // Kolik se na účet celkem vložilo.
  get totalDeposited() {
    return this.#transactions
      .filter((transaction) => transaction.type === 'deposit')
      .reduce((sum, transaction) => sum + transaction.amount, 0);
  }

  deposit(amount) {
    this.#assertPositive(amount);
    this.#balance += amount;
    this.#record('deposit', amount);
  }

  withdraw(amount) {
    this.#assertPositive(amount);
    if (amount > this.#balance) {
      throw new RangeError('Nedostatek prostředků');
    }
    this.#balance -= amount;
    this.#record('withdrawal', amount);
  }

  // Od kolegy: vloží víc částek najednou, třeba výplatu a vratku.
  depositAll(amounts) {
    amounts.forEach((amount) => {
      this.deposit(amount);
    });
  }

  #assertPositive(amount) {
    if (!Number.isFinite(amount) || amount <= 0) {
      throw new RangeError('Částka musí být kladné číslo');
    }
  }

  #record(type, amount) {
    this.#transactions.push({ id: this.#nextId(), type, amount });
  }
}

class SavingsAccount extends BankAccount {
--edit--
```

```js
function createIdGenerator(prefix) {
  let count = 0;
  return () => {
    count += 1;

  };
}

class BankAccount {
  #balance = 0;
  #transactions = [];
  #nextId = createIdGenerator('TX');

  // Založí účet a rovnou na něj vloží první částku.
  static open(owner, initialDeposit) {
    const account = new BankAccount(owner);
    account.deposit(initialDeposit);
    return account;
  }

  // Obnoví účet z uložených pohybů, třeba z databáze.

    const account = new BankAccount(owner);
    for (const { type, amount } of transactions) {
      if (type === 'deposit') {
        account.deposit(amount);
      } else {
        account.withdraw(amount);
      }
    }
    return account;
  }

  constructor(owner) {
    this.owner = owner;
  }

  get balance() {
    return this.#balance;
  }

  // Kopie historie: kdo ji dostane, nezmění pohyby uvnitř účtu.
  get transactions() {
    return [...this.#transactions];
  }

  // Kolik se na účet celkem vložilo.
  get totalDeposited() {
    return this.#transactions
      .filter((transaction) => transaction.type === 'deposit')
      .reduce((sum, transaction) => sum + transaction.amount, 0);
  }

  deposit(amount) {
    this.#assertPositive(amount);
    this.#balance += amount;
    this.#record('deposit', amount);
  }

  withdraw(amount) {
    this.#assertPositive(amount);
    if (amount > this.#balance) {
      throw new RangeError('Nedostatek prostředků');
    }
    this.#balance -= amount;
    this.#record('withdrawal', amount);
  }

  // Od kolegy: vloží víc částek najednou, třeba výplatu a vratku.
  depositAll(amounts) {
    amounts.forEach((amount) => {
      this.deposit(amount);
    });
  }

  #assertPositive(amount) {
    if (!Number.isFinite(amount) || amount <= 0) {
      throw new RangeError('Částka musí být kladné číslo');
    }
  }

  #record(type, amount) {
    this.#transactions.push({ id: this.#nextId(), type, amount });
  }
}

class SavingsAccount extends BankAccount {
  #withdrawalsLeft = 3;

  constructor(owner, rate) {
    super(owner);
    this.rate = rate;
  }

  // Připíše úrok z aktuálního zůstatku, zaokrouhlený na celé koruny.
  addInterest() {
    const interest = Math.round(this.balance * this.rate);
    if (interest > 0) {
      this.deposit(interest);
    }
    return interest;
  }

  // Ze spořicího účtu jdou jen tři výběry.
  withdraw(amount) {
    if (this.#withdrawalsLeft === 0) {
      throw new Error('Limit výběrů vyčerpán');
    }
    super.withdraw(amount);
    this.#withdrawalsLeft -= 1;
  }
}
```
