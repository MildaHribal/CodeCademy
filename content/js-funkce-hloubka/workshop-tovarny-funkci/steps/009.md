# --description--

V souboru je `ticketPrice(tariff, km)`: cena jízdenky podle tarifu (`'adult'`, `'student'`, `'senior'`) a vzdálenosti. Studentská sekce webu volá funkci pořád s tarifem `'student'`. Hodila by se funkce, která tarif už **má předvyplněný** a čeká jen na kilometry. Tomu se říká [[částečná aplikace]] (*partial application*).

**Úkol:** Napiš funkci `partial(fn, ...presetArgs)`, která vrátí novou funkci. Když ji zavoláš s dalšími argumenty, zavolá `fn` nejdřív s předvyplněnými argumenty a za nimi s těmi novými, a výsledek vrátí.

```js
const studentPrice = partial(ticketPrice, 'student');
studentPrice(255); // totéž jako ticketPrice('student', 255)
```

Zbytkový parametr tu potřebuješ dvakrát: jednou v hlavičce `partial`, podruhé ve vrácené funkci. Obě pole pak rozprostřeš do volání `fn`. Předvyplnění ručně, bez `partial`, by na jiných datech vypadalo takhle:

```js
// 1. obecná funkce
const formatPrice = (currency, amount) => `${amount} ${currency}`;
// 2. předvyplněná měna, čeká se jen na částku
const inEuro = (amount) => formatPrice('EUR', amount);
```

Vypiš si studentskou cenu jízdenky Praha–Brno (255 km).

# --hints--

Vrácená funkce zavolá `fn` s předvyplněným tarifem a novými kilometry.

```js
const studentPrice = partial(ticketPrice, 'student');
assert.equal(typeof studentPrice, 'function', "partial(ticketPrice, 'student') má vrátit funkci");
assert.equal(studentPrice(100), 120, "partial(ticketPrice, 'student')(100) má vrátit 120, stejně jako ticketPrice('student', 100)");
```

Předvyplnit jde víc argumentů a zachová se jejich pořadí.

```js
const code = (line, number, letter) => `${line}-${number}-${letter}`;
assert.equal(partial(code, 'R', 7)('A'), 'R-7-A', "partial(code, 'R', 7)('A') má zavolat code('R', 7, 'A')");
```

Vrácenou funkci jde volat opakovaně a předvyplněné argumenty se nehromadí.

```js
const sum = (...numbers) => numbers.reduce((total, n) => total + n, 0);
const plusTen = partial(sum, 10);
assert.equal(plusTen(1), 11, 'partial(sum, 10)(1) má vrátit 11');
assert.equal(plusTen(2), 12, 'druhé zavolání partial(sum, 10)(2) má vrátit 12 — předvyplněné argumenty se nesmí měnit');
```

Bez předvyplněných argumentů se předají jen ty nové.

```js
assert.equal(partial(Math.max)(3, 8), 8, 'partial(Math.max)(3, 8) má vrátit 8');
```

// Stanice pro našeptávač, seřazené podle počtu cestujících.
const stations = [
  'Praha hl. n.',
  'Brno hl. n.',
  'Ostrava hl. n.',
  'Plzeň hl. n.',
  'Olomouc hl. n.',
  'Pardubice hl. n.',
  'Praha-Smíchov',
  'České Budějovice',
  'Hradec Králové hl. n.',
  'Ústí nad Labem hl. n.',
  'Liberec',
  'Zlín střed',
  'Česká Třebová',
  'Kolín',
  'Jihlava',
  'Cheb',
  'Chomutov',
  'Děčín hl. n.',
];

// Pomocné funkce pro pozdější kroky. Počítadla ukazují, kolikrát se opravdu spustily.
let timetableLoads = 0;
let travelCalls = 0;

// Načte jízdní řád (ve skutečné aplikaci by ho stahovala ze serveru).
function loadTimetable() {
  timetableLoads += 1;
  console.log('Načítám jízdní řád…');
  return { updated: '14. 9. 2026', trains: 1840 };
}

// Doba jízdy v minutách, zjednodušeně podle vzdálenosti od Prahy
// (ve skutečné aplikaci pomalý výpočet trasy).
const kmFromPrague = {
  'Praha hl. n.': 0,
  Kolín: 62,
  'Pardubice hl. n.': 104,
  'Česká Třebová': 164,
  'Olomouc hl. n.': 252,
  'Ostrava hl. n.': 356,
  'Plzeň hl. n.': 114,
  Cheb: 220,
};

function travelMinutes(from, to) {
  travelCalls += 1;
  const km = Math.abs(kmFromPrague[from] - kmFromPrague[to]);
  return Math.round((km / 96) * 60);
}

// Cena jízdenky v Kč: 1,60 Kč za kilometr a sleva podle tarifu.
function ticketPrice(tariff, km) {
  const discounts = { adult: 0, student: 0.25, senior: 0.5 };
  return Math.round(km * 1.6 * (1 - discounts[tariff]));
}

// Stanice, které obsahují dotaz, bez ohledu na velikost písmen.
function findStations(names, query) {
  const needle = query.toLowerCase();
  return names.filter((name) => name.toLowerCase().includes(needle));
}

console.log(findStations(stations, 'praha'));

// Nové pole stanic seřazené podle české abecedy.
function sortStations(names) {
  return names.toSorted((a, b) => a.localeCompare(b, 'cs'));
}

console.log(sortStations(findStations(stations, 'hl. n.')));

// Vyrobí funkci, která pozná, jestli název stanice obsahuje dotaz.
function matchesQuery(query) {
  const needle = query.toLowerCase();
  return (name) => name.toLowerCase().includes(needle);
}

console.log(stations.filter(matchesQuery('praha')));

// Počítadlo: každé zavolání přičte jedničku a vrátí nový stav.
function createCounter() {
  let count = 0;
  return () => {
    count += 1;
    return count;
  };
}

const countSearches = createCounter();
countSearches();
console.log('Hledání:', countSearches());

// Historie hledání: nejnovější dotaz první, nejvýš limit dotazů.
function createSearchHistory(limit) {
  let queries = [];
  return {
    add(query) {
      queries = [query, ...queries].slice(0, limit);
    },
    list() {
      return [...queries];
    },
  };
}

const recentSearches = createSearchHistory(3);
recentSearches.add('Brno');
recentSearches.add('Cheb');
console.log(recentSearches.list());

// Od kolegy: klávesové zkratky 1, 2, 3… pro oblíbené stanice.
// Každá zkratka je funkce, která vrátí svou stanici.
function createShortcuts(names) {
  const shortcuts = [];
  for (let i = 0; i < names.length; i++) {
    shortcuts.push(() => names[i]);
  }
  return shortcuts;
}

// Obal, který zavolá fn jen poprvé a potom vrací uložený výsledek.
function once(fn) {
  let called = false;
  let result;
  return (...args) => {
    if (!called) {
      called = true;
      result = fn(...args);
    }
    return result;
  };
}

const getTimetable = once(loadTimetable);
getTimetable();
console.log(getTimetable(), 'načteno:', timetableLoads);
