# --description--

Našeptávač teď hledá po každém písmenu. Se skutečným serverem by „Hradec Králové" poslalo čtrnáct požadavků.

**Úkol:** Uprav hledání tak, aby návrhy proběhly až **250 ms** poté, co uživatel přestal psát, a to pro text, který v poli je v tu chvíli. Hned po napsání se seznam ještě nemění.

Nástroj na to máš v souboru.

Vyzkoušej v náhledu: napiš rychle „olomouc" a sleduj, že se seznam objeví až po dopsání.

# --hints--

Hned po napsání je seznam návrhů ještě prázdný.

```js
const input = document.querySelector('#station');
await helpers.type(input, 'pra');
await helpers.wait(60);
assert.equal(document.querySelectorAll('#suggestions li').length, 0, "60 ms po napsání 'pra' se návrhy ještě nemají ukázat — hledání má počkat 250 ms");
```

Po chvíli klidu seznam ukazuje návrhy pro celý napsaný text.

```js
const input = document.querySelector('#station');
await helpers.type(input, 'ch');
const texts = () => [...document.querySelectorAll('#suggestions li')].map((item) => item.textContent);
await helpers.waitFor(() => texts().length > 0, 2000);
assert.deepEqual(texts(), suggest('ch'), "po dopsání 'ch' a chvíli klidu má seznam ukazovat suggest('ch')");
```

Když mezi písmeny uplyne méně než 250 ms, návrhy se ukážou až po posledním písmenu.

```js
const input = document.querySelector('#station');
const typeValue = (value) => {
  input.value = value;
  input.dispatchEvent(new InputEvent('input', { bubbles: true }));
};
typeValue('b');
await helpers.wait(200);
typeValue('br');
await helpers.wait(150);
assert.equal(document.querySelectorAll('#suggestions li').length, 0, "350 ms po 'b' a 150 ms po 'br' se návrhy ještě nemají ukázat — každé písmeno má odpočet 250 ms začít znovu");
```

# --help--

## --tip--

Jde o obal kolem hledání, [debounce](see:js-funkce-hloubka/workshop-tovarny-funkci/015) z tvé knihovny.

## --tip--

Obal vytvoř jednou, mimo posluchač, a posluchač bude volat jen jeho. Text pole mu předej jako argument.

## --tip--

Porovnej s pastí [Továrna zavolaná při každém použití](see:js-funkce-hloubka/closures#tovarna-zavolana-pri-kazdem-pouziti): obal vytvořený uvnitř posluchače by měl při každém písmenu nový časovač a nic by nerušil.

  'Praha hl. n.',
  'Brno hl. n.',

let timetableLoads = 0;
let travelCalls = 0;

// Načte jízdní řád (ve skutečné aplikaci by ho stahovala ze serveru).
function loadTimetable() {
  timetableLoads += 1;
  console.log('Načítám jízdní řád…');
  return { updated: '14. 9. 2026', trains: 1840 };
}

// Doba jízdy v minutách, zjednodušeně podle vzdálenosti od Prahy
// (ve skutečné aplikaci pomalý výpočet trasy).
const kmFromPrague = {
  'Praha hl. n.': 0,
  Kolín: 62,
  'Pardubice hl. n.': 104,
  'Česká Třebová': 164,
  'Olomouc hl. n.': 252,
  'Ostrava hl. n.': 356,
  'Plzeň hl. n.': 114,
  Cheb: 220,
};

function travelMinutes(from, to) {
  travelCalls += 1;
  const km = Math.abs(kmFromPrague[from] - kmFromPrague[to]);
  return Math.round((km / 96) * 60);
}

// Cena jízdenky v Kč: 1,60 Kč za kilometr a sleva podle tarifu.
function ticketPrice(tariff, km) {
  const discounts = { adult: 0, student: 0.25, senior: 0.5 };
  return Math.round(km * 1.6 * (1 - discounts[tariff]));
}

// Stanice, které obsahují dotaz, bez ohledu na velikost písmen.
function findStations(names, query) {
  const needle = query.toLowerCase();
  return names.filter((name) => name.toLowerCase().includes(needle));
}

console.log(findStations(stations, 'praha'));

// Nové pole stanic seřazené podle české abecedy.
function sortStations(names) {
  return names.toSorted((a, b) => a.localeCompare(b, 'cs'));
}

console.log(sortStations(findStations(stations, 'hl. n.')));

// Vyrobí funkci, která pozná, jestli název stanice obsahuje dotaz.
function matchesQuery(query) {
  const needle = query.toLowerCase();
  return (name) => name.toLowerCase().includes(needle);
}

console.log(stations.filter(matchesQuery('praha')));

// Počítadlo: každé zavolání přičte jedničku a vrátí nový stav.
function createCounter() {
  let count = 0;
  return () => {
    count += 1;
    return count;
  };
}

const countSearches = createCounter();
countSearches();
console.log('Hledání:', countSearches());

// Historie hledání: nejnovější dotaz první, nejvýš limit dotazů.
function createSearchHistory(limit) {
  let queries = [];
  return {
    add(query) {
      queries = [query, ...queries].slice(0, limit);
    },
    list() {
      return [...queries];
    },
  };
}

const recentSearches = createSearchHistory(3);
recentSearches.add('Brno');
recentSearches.add('Cheb');
console.log(recentSearches.list());

// Od kolegy: klávesové zkratky 1, 2, 3… pro oblíbené stanice.
// Každá zkratka je funkce, která vrátí svou stanici.
function createShortcuts(names) {
  const shortcuts = [];
  for (let i = 0; i < names.length; i++) {
    shortcuts.push(() => names[i]);
  }
  return shortcuts;
}

// Obal, který zavolá fn jen poprvé a potom vrací uložený výsledek.
function once(fn) {
  let called = false;
  let result;
  return (...args) => {
    if (!called) {
      called = true;
      result = fn(...args);
    }
    return result;
  };
}

const getTimetable = once(loadTimetable);
getTimetable();
console.log(getTimetable(), 'načteno:', timetableLoads);

// Částečná aplikace: předvyplní první argumenty funkce.
function partial(fn, ...presetArgs) {
  return (...laterArgs) => fn(...presetArgs, ...laterArgs);
}

const studentPrice = partial(ticketPrice, 'student');
console.log('Student Praha–Brno:', studentPrice(255), 'Kč');

// Čísla rezervací: prefix a pořadové číslo doplněné nulami na tři místa.
function createIdGenerator(prefix) {
  let lastNumber = 0;
  return () => {
    lastNumber += 1;
    return `${prefix}-${String(lastNumber).padStart(3, '0')}`;
  };
}

const nextReservation = createIdGenerator('R');
console.log(nextReservation(), nextReservation());

// Mezipaměť: pro stejné argumenty spočítá výsledek jen jednou.
function memoize(fn) {
  const cache = new Map();
  return (...args) => {
    const key = JSON.stringify(args);
    if (!cache.has(key)) {
      cache.set(key, fn(...args));
    }
    return cache.get(key);
  };
}

const minutesFromPrague = memoize((station) => travelMinutes('Praha hl. n.', station));
minutesFromPrague('Olomouc hl. n.');
console.log(minutesFromPrague('Olomouc hl. n.'), 'min, výpočtů:', travelCalls);

// Od kolegy: doba jízdy pro tabuli spojení, má se počítat jen jednou.
const travelCache = memoize(travelMinutes);

function cachedTravelMinutes(from, to) {
  return travelCache(from, to);
}

// Debounce: zavolá fn až po delay ms klidu, s argumenty posledního zavolání.
function debounce(fn, delay) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}

const logSearch = debounce((query) => console.log('Hledám:', query), 300);
logSearch('P');
logSearch('Pra');
logSearch('Praha');

// Od kolegy: zpožděné hledání pro pole Odkud a Kam.
function createDelayedSearch(search, delay) {
  let searchTimer;
  return (query) => {
    clearTimeout(searchTimer);
    searchTimer = setTimeout(() => search(query), delay);
  };
}

// Throttle: zavolá fn nejvýš jednou za interval ms, zavolání mezitím zahodí.
function throttle(fn, interval) {
  let lastRun = 0;
  return (...args) => {
    const now = Date.now();
    if (now - lastRun >= interval) {
      lastRun = now;
      fn(...args);
    }
  };
}

const refreshLog = throttle(() => console.log('Obnovuji odjezdy'), 1000);
refreshLog();
refreshLog();

// Obal, který fn zavolá nejvýš max-krát; další zavolání vrátí undefined.
function limitCalls(fn, max) {
  let calls = 0;
  return (...args) => {
    if (calls >= max) {
      return undefined;
    }
    calls += 1;
    return fn(...args);
  };
}

const freeSearch = limitCalls(findStations, 2);
console.log(freeSearch(stations, 'Brno'), freeSearch(stations, 'Cheb'), freeSearch(stations, 'Kolín'));

// Našeptávač: stanice podle dotazu, česky seřazené, nejvýš limit výsledků.
// Stejný dotaz se podruhé vezme z mezipaměti.
function createSuggester(names, limit = 5) {
  const sorted = sortStations(names);
  return memoize((query) => {
    const trimmed = query.trim();
    if (trimmed === '') {
      return [];
    }
    return sorted.filter(matchesQuery(trimmed)).slice(0, limit);
  });
}

const suggest = createSuggester(stations);
console.log(suggest('hl'));

// Vypíše návrhy do seznamu pod polem.
function renderSuggestions(names) {
  const list = document.querySelector('#suggestions');
  const items = names.map((name) => {
    const item = document.createElement('li');
    item.textContent = name;
    return item;
  });
  list.replaceChildren(...items);
}

const stationInput = document.querySelector('#station');

stationInput.addEventListener('input', () => {
  renderSuggestions(suggest(stationInput.value));
});

// Stejný dotaz se podruhé vezme z mezipaměti.
function createSuggester(names, limit = 5) {
  const sorted = sortStations(names);
  return memoize((query) => {
    const trimmed = query.trim();
    if (trimmed === '') {
      return [];
    }
    return sorted.filter(matchesQuery(trimmed)).slice(0, limit);
  });
}

// Vypíše návrhy do seznamu pod polem.
function renderSuggestions(names) {
  const list = document.querySelector('#suggestions');
  const items = names.map((name) => {
    const item = document.createElement('li');
    item.textContent = name;
    return item;
  });
  list.replaceChildren(...items);
}
