# --description--

V souboru je funkce `loadTimetable()`. Ve skutečné aplikaci by stahovala jízdní řád ze serveru, a to trvá. Jízdní řád přitom potřebuje několik míst stránky. Chceš ho načíst **jen jednou** a pak vracet to, co už máš.

**Úkol:** Napiš funkci `once(fn)`, která vrátí obalující funkci. Obal:

- při **prvním** zavolání zavolá `fn`, výsledek si uloží a vrátí ho,
- při každém dalším zavolání `fn` už nevolá a vrátí uložený výsledek.

Pozor na funkci, která nic nevrací: i ta se smí zavolat jen jednou. Nestačí proto kontrolovat, jestli už nějaký výsledek máš. Potřebuješ si pamatovat, **jestli už obal proběhl**.

Je to [[obalující funkce]] z lekce a pamatuje si stav jako počítadlo. Na jiných datech — funkce, která si pamatuje, jestli už uvítala:

```js
function createGreeter(name) {
  // 1. příznak v prostředí: už jsme uvítali?
  let greeted = false;
  // 2. první volání uvítá, další jen připomenou
  return () => {
    if (greeted) {
      return `Už jsme se viděli, ${name}.`;
    }
    greeted = true;
    return `Vítej, ${name}.`;
  };
}
```

Vyzkoušej to: `const getTimetable = once(loadTimetable);`, zavolej `getTimetable()` dvakrát a sleduj, kolikrát se v konzoli objeví „Načítám jízdní řád…" a co je v `timetableLoads`.

# --hints--

Obal zavolá `fn` jen při prvním zavolání a potom vrací tentýž výsledek.

```js
let calls = 0;
const load = once(() => {
  calls += 1;
  return { trains: 1840 };
});
assert.equal(calls, 0, 'once(fn) nemá fn volat hned při vytvoření obalu');
const first = load();
const second = load();
assert.equal(calls, 1, 'po dvou zavoláních obalu má fn proběhnout jen jednou');
assert.deepEqual(first, { trains: 1840 }, 'první zavolání obalu má vrátit výsledek fn');
assert.equal(second, first, 'druhé zavolání obalu má vrátit tentýž uložený výsledek');
```

I funkce, která nic nevrací, proběhne jen jednou.

```js
let calls = 0;
const notify = once(() => {
  calls += 1;
});
notify();
notify();
notify();
assert.equal(calls, 1, 'fn bez návratové hodnoty má po třech zavoláních obalu proběhnout jen jednou — pamatuješ si, že obal už proběhl?');
```

Dva obaly se navzájem neovlivňují.

```js
const loadTrains = once(() => 'vlaky');
const loadBuses = once(() => 'autobusy');
assert.equal(loadTrains(), 'vlaky', "once(() => 'vlaky')() má vrátit 'vlaky'");
assert.equal(loadBuses(), 'autobusy', "once(() => 'autobusy')() má vrátit 'autobusy' i poté, co proběhl jiný obal");
```

const stations = [
  'Praha hl. n.',
  'Brno hl. n.',
  'Ostrava hl. n.',

let timetableLoads = 0;

function loadTimetable() {
  timetableLoads += 1;
  console.log('Načítám jízdní řád…');

const kmFromPrague = {
  'Praha hl. n.': 0,
  Kolín: 62,
  'Pardubice hl. n.': 104,
  'Česká Třebová': 164,
  'Olomouc hl. n.': 252,
  'Ostrava hl. n.': 356,
  'Plzeň hl. n.': 114,
  Cheb: 220,
};

function travelMinutes(from, to) {
  travelCalls += 1;
  const km = Math.abs(kmFromPrague[from] - kmFromPrague[to]);
  return Math.round((km / 96) * 60);
}

// Cena jízdenky v Kč: 1,60 Kč za kilometr a sleva podle tarifu.
function ticketPrice(tariff, km) {
  const discounts = { adult: 0, student: 0.25, senior: 0.5 };
  return Math.round(km * 1.6 * (1 - discounts[tariff]));
}

// Stanice, které obsahují dotaz, bez ohledu na velikost písmen.
function findStations(names, query) {
  const needle = query.toLowerCase();
  return names.filter((name) => name.toLowerCase().includes(needle));
}

console.log(findStations(stations, 'praha'));

// Nové pole stanic seřazené podle české abecedy.
function sortStations(names) {
  return names.toSorted((a, b) => a.localeCompare(b, 'cs'));
}

console.log(sortStations(findStations(stations, 'hl. n.')));

// Vyrobí funkci, která pozná, jestli název stanice obsahuje dotaz.
function matchesQuery(query) {
  const needle = query.toLowerCase();
  return (name) => name.toLowerCase().includes(needle);
}

console.log(stations.filter(matchesQuery('praha')));

// Počítadlo: každé zavolání přičte jedničku a vrátí nový stav.
function createCounter() {
  let count = 0;
  return () => {
    count += 1;
    return count;
  };
}

const countSearches = createCounter();
countSearches();
console.log('Hledání:', countSearches());

// Historie hledání: nejnovější dotaz první, nejvýš limit dotazů.
function createSearchHistory(limit) {
  let queries = [];
  return {
    add(query) {
      queries = [query, ...queries].slice(0, limit);
    },
    list() {
      return [...queries];
    },
  };
}

const recentSearches = createSearchHistory(3);
recentSearches.add('Brno');
recentSearches.add('Cheb');
console.log(recentSearches.list());

// Od kolegy: klávesové zkratky 1, 2, 3… pro oblíbené stanice.
// Každá zkratka je funkce, která vrátí svou stanici.
function createShortcuts(names) {
  const shortcuts = [];
  for (let i = 0; i < names.length; i++) {
    shortcuts.push(() => names[i]);
  }
  return shortcuts;
}
