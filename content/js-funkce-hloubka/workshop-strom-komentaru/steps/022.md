# --description--

Nad diskusí je výběr „Řadit: nejoblíbenější". Seřadit jen první úroveň nestačí — i odpovědi pod každým komentářem mají jít od nejvíc srdíček.

**Úkol:** Napiš funkci `sortTree(nodes, compare)`, která vrátí nový strom seřazený porovnávací funkcí `compare` na **každé** úrovni. Vstupní strom se nezmění. Pod ni založ porovnávací funkci `byLikes`, která řadí od nejvíc srdíček.

Stránka se zatím nezmění, výběr zapojíš v dalším kroku. Vypiš si autory první úrovně stromu seřazeného podle srdíček.

# --hints--

`sortTree` seřadí uzly na všech úrovních.

```js
const tree = [
  { id: 1, likes: 2, replies: [{ id: 2, likes: 1, replies: [] }, { id: 3, likes: 5, replies: [{ id: 5, likes: 0, replies: [] }, { id: 6, likes: 4, replies: [] }] }] },
  { id: 4, likes: 9, replies: [] },
];
const result = sortTree(tree, (a, b) => b.likes - a.likes);
assert.deepEqual(result.map((node) => node.id), [4, 1], 'první úroveň má jít od nejvíc srdíček: 4, 1');
assert.deepEqual(result[1].replies.map((node) => node.id), [3, 2], 'odpovědi uzlu 1 mají jít 3, 2');
assert.deepEqual(result[1].replies[0].replies.map((node) => node.id), [6, 5], 'odpovědi uzlu 3 ve třetí úrovni mají jít 6, 5');
```

`sortTree` použije zadanou porovnávací funkci a vstupní strom nezmění.

```js
const tree = [{ id: 2, replies: [{ id: 9, replies: [] }, { id: 7, replies: [] }] }, { id: 1, replies: [] }];
const result = sortTree(tree, (a, b) => a.id - b.id);
assert.deepEqual(result.map((node) => node.id), [1, 2], 'sortTree(strom, podle id) má vrátit 1, 2');
assert.deepEqual(result[1].replies.map((node) => node.id), [7, 9], 'odpovědi mají být podle id 7, 9');
assert.deepEqual(tree.map((node) => node.id), [2, 1], 'sortTree nesmí přerovnat původní strom');
assert.deepEqual(tree[0].replies.map((node) => node.id), [9, 7], 'sortTree nesmí přerovnat odpovědi v původním stromu');
```

`byLikes` řadí od nejvíc srdíček.

```js
assert.ok(byLikes({ likes: 9 }, { likes: 2 }) < 0, 'byLikes({ likes: 9 }, { likes: 2 }) má vrátit záporné číslo — víc srdíček patří dopředu');
assert.ok(byLikes({ likes: 1 }, { likes: 4 }) > 0, 'byLikes({ likes: 1 }, { likes: 4 }) má vrátit kladné číslo');
```

## --tip--

Kombinuješ řazení bez mutace s rekurzí nad stromem — viz [Řazení a porovnávací funkce](see:js-pole/metody-pole-do-hloubky#razeni-a-porovnavaci-funkce).

## --tip--

Seřaď kopii pole uzlů a každý uzel nahraď kopií, jejíž `replies` jsou seřazené stejnou funkcí. Nezáleží, jestli nejdřív řadíš, nebo nejdřív zpracuješ odpovědi.

## --tip--

Na jiných datech: menu seřazené podle názvu na všech úrovních vrací kopie kategorií s podkategoriemi seřazenými toutéž funkcí.

// Smazaný komentář zůstane ve stromu kvůli odpovědím, jen bez autora a textu.
function maskDeleted(nodes) {
  return nodes.map((node) => {
    const replies = maskDeleted(node.replies);
    if (node.deleted) {
      return { ...node, author: null, text: '[smazaný komentář]', replies };
    }
    return { ...node, replies };
  });
}

// Smazané komentáře bez odpovědí zmizí úplně. Odpovědi se vyčistí dřív než jejich rodič.
function pruneDeleted(nodes) {
  return nodes
    .map((node) => ({ ...node, replies: pruneDeleted(node.replies) }))
    .filter((node) => !node.deleted || node.replies.length > 0);

function findComment(nodes, id) {
  for (const node of nodes) {
    if (node.id === id) {
      return node;
    }
    const found = findComment(node.replies, id);
    if (found) {
      return found;
    }
  }
  return undefined;
}

// Počet úrovní stromu podle nejhlubšího vlákna.
function maxDepth(nodes) {
  return nodes.reduce((deepest, node) => Math.max(deepest, 1 + maxDepth(node.replies)), 0);

// Strom s nejvýš levels úrovněmi; poslední úroveň si pamatuje, kolik odpovědí schovala.
function limitDepth(nodes, levels) {
  return nodes.map((node) => {
    if (levels === 1) {
      return { ...node, replies: [], hiddenReplies: countReplies(node) };
    }
    return { ...node, replies: limitDepth(node.replies, levels - 1), hiddenReplies: 0 };
  });
}

  return 1 + countAll(first?.replies ?? []) + countAll(rest);

function toLines(nodes, depth = 0) {
  return nodes.flatMap((node) => [
    `${'  '.repeat(depth)}${node.author}: ${node.text}`,
    ...toLines(node.replies, depth + 1),
  ]);

const pipe = (...fns) => (value) => fns.reduce((result, fn) => fn(result), value);

// Smazaný komentář zůstane ve stromu kvůli odpovědím, jen bez autora a textu.
function maskDeleted(nodes) {
  return nodes.map((node) => {
    const replies = maskDeleted(node.replies);
    if (node.deleted) {
      return { ...node, author: null, text: '[smazaný komentář]', replies };
    }
    return { ...node, replies };
  });
}

// Smazané komentáře bez odpovědí zmizí úplně. Odpovědi se vyčistí dřív než jejich rodič.
function pruneDeleted(nodes) {
  return nodes
    .map((node) => ({ ...node, replies: pruneDeleted(node.replies) }))
    .filter((node) => !node.deleted || node.replies.length > 0);
}

// Hotová diskuse z dat serveru.
const discussion = pipe(oldestFirst, buildTree, pruneDeleted, maskDeleted);

console.log(toLines(discussion(comments)).join('\n'));

// Najde uzel s daným id v libovolné hloubce stromu.
function findComment(nodes, id) {
  for (const node of nodes) {
    if (node.id === id) {
      return node;
    }
    const found = findComment(node.replies, id);
    if (found) {
      return found;
    }
  }
  return undefined;
}

console.log(findComment(discussion(comments), 8));

// Počet úrovní stromu podle nejhlubšího vlákna.
function maxDepth(nodes) {
  return nodes.reduce((deepest, node) => Math.max(deepest, 1 + maxDepth(node.replies)), 0);
}

console.log('Úrovní:', maxDepth(discussion(comments)));

// Strom s nejvýš levels úrovněmi; poslední úroveň si pamatuje, kolik odpovědí schovala.
function limitDepth(nodes, levels) {
  return nodes.map((node) => {
    if (levels === 1) {
      return { ...node, replies: [], hiddenReplies: countReplies(node) };
    }
    return { ...node, replies: limitDepth(node.replies, levels - 1), hiddenReplies: 0 };
  });
}

// Od kolegy: počet všech komentářů ve stromu pro nadpis diskuse.
function countAll(nodes) {
  if (nodes.length === 0) {
    return 0;
  }
  const [first, ...rest] = nodes;
  return 1 + countAll(first?.replies ?? []) + countAll(rest);
}

// Vytvoří seznam komentářů a do každého vnoří seznam jeho odpovědí.
function renderThread(nodes) {
  const list = document.createElement('ul');
  list.className = 'thread';
  for (const node of nodes) {
    const item = document.createElement('li');
    item.className = 'comment';

    const author = document.createElement('p');
    author.className = 'comment__author';
    author.textContent = node.author ?? '—';

    const text = document.createElement('p');
    text.className = 'comment__text';
    text.textContent = node.text;

    const likes = document.createElement('p');
    likes.className = 'comment__likes';
    likes.textContent = `♥ ${node.likes}`;

    item.append(author, text, likes);
    if (node.replies.length > 0) {
      const toggle = document.createElement('button');
      toggle.type = 'button';
      toggle.className = 'comment__toggle';
      toggle.setAttribute('aria-expanded', 'true');
      toggle.textContent = `Skrýt ${replyLabel(countReplies(node))}`;
      item.append(toggle, renderThread(node.replies));
    }
    list.append(item);
  }
  return list;
}

document.querySelector('#discussion').replaceChildren(renderThread(discussion(comments)));

// Od kolegy: tlačítka u vláken schovávají a ukazují odpovědi.
function connectToggles() {
  const buttons = document.querySelectorAll('#discussion .comment__toggle');
  for (let i = 0; i < buttons.length; i++) {
    buttons[i].addEventListener('click', () => {
      const replies = buttons[i].nextElementSibling;
      const expanded = buttons[i].getAttribute('aria-expanded') === 'true';
      replies.hidden = expanded;
      buttons[i].setAttribute('aria-expanded', String(!expanded));
      buttons[i].textContent = expanded
        ? buttons[i].textContent.replace('Skrýt', 'Zobrazit')
        : buttons[i].textContent.replace('Zobrazit', 'Skrýt');
    });
  }
}

connectToggles();
