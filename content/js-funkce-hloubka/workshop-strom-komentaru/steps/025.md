# --description--

Poslední krok. Na úzkém mobilu je čtvrtá úroveň odpovědí nečitelná.

**Úkol:** Diskuse na stránce ukáže nejvýš **tři** úrovně, v obou řazeních. Komentář, pod kterým se odpovědi schovaly, má místo nich tlačítko `button.comment__more` s textem `Pokračovat ve vlákně (1 odpověď)` — s počtem schovaných odpovědí ve správném tvaru. Tlačítko stačí vykreslit, přechod na celé vlákno je rozšíření.

Všechno potřebné už máš: `limitDepth`, `hiddenReplies` a `replyLabel`. Zbývá je zapojit do cesty dat na stránku.

Tím je diskuse hotová. Za sebou máš:

- strom z ploché tabulky přes rekurzi s výchozím parametrem,
- rekurze, které počítají, hledají, čistí, maskují, řadí a omezují hloubku — bez mutace vstupu,
- skládání úprav dat přes `pipe`,
- dvě kolegovy chyby: rekurzi bez základního případu a `var` v cyklu s posluchači,
- vykreslení stromu do vnořených seznamů s tlačítky a řazením.

Stejné stromy potkáš v menu e-shopu s podkategoriemi, ve složkách souborů, v organizační struktuře firmy i v komentářích pod videem.

## Udělej po svém

- **Vzhled vláken:** zvýrazni komentáře autora receptu (třeba barevným okrajem) a přidej plynulé schování odpovědí přes `transition`.
- **Odkaz na komentář:** každému `li` dej `id="komentar-8"` a po načtení zvýrazni komentář z adresy stránky pomocí `findComment`.
- **Pokračovat ve vlákně:** po kliknutí na tlačítko ukaž jen celé vlákno pod daným komentářem.
- **Jiné téma:** přeměň diskusi na organizační strukturu firmy, strom složek nebo menu kategorií — rekurzivní funkce zůstanou skoro stejné.

Příště samostatně: v labu Zpracování objednávek složíš pipeline z malých čistých funkcí sám, bez návodu.

  return Object.fromEntries(list.map((comment) => [comment.id, comment]));

function buildTree(list, parentId = null) {
  return list
    .filter((comment) => comment.parentId === parentId)
    .map((comment) => ({
      ...comment,
      replies: buildTree(list, comment.id),

function toLines(nodes, depth = 0) {
  return nodes.flatMap((node) => [
    `${'  '.repeat(depth)}${node.author}: ${node.text}`,
    ...toLines(node.replies, depth + 1),
  ]);
}

// Smazaný komentář zůstane ve stromu kvůli odpovědím, jen bez autora a textu.
function maskDeleted(nodes) {
  return nodes.map((node) => {
    const replies = maskDeleted(node.replies);
    if (node.deleted) {
      return { ...node, author: null, text: '[smazaný komentář]', replies };
    }
    return { ...node, replies };
  });
}

// Smazané komentáře bez odpovědí zmizí úplně. Odpovědi se vyčistí dřív než jejich rodič.
function pruneDeleted(nodes) {
  return nodes
    .map((node) => ({ ...node, replies: pruneDeleted(node.replies) }))
    .filter((node) => !node.deleted || node.replies.length > 0);
}

// Hotová diskuse z dat serveru.
const discussion = pipe(oldestFirst, buildTree, pruneDeleted, maskDeleted);

console.log(toLines(discussion(comments)).join('\n'));

// Najde uzel s daným id v libovolné hloubce stromu.
function findComment(nodes, id) {
  for (const node of nodes) {
    if (node.id === id) {
      return node;
    }
    const found = findComment(node.replies, id);
    if (found) {
      return found;
    }
  }
  return undefined;
}

console.log(findComment(discussion(comments), 8));

// Počet úrovní stromu podle nejhlubšího vlákna.
function maxDepth(nodes) {
  return nodes.reduce((deepest, node) => Math.max(deepest, 1 + maxDepth(node.replies)), 0);
}

console.log('Úrovní:', maxDepth(discussion(comments)));

// Strom s nejvýš levels úrovněmi; poslední úroveň si pamatuje, kolik odpovědí schovala.
function limitDepth(nodes, levels) {
  return nodes.map((node) => {
    if (levels === 1) {
      return { ...node, replies: [], hiddenReplies: countReplies(node) };
    }
    return { ...node, replies: limitDepth(node.replies, levels - 1), hiddenReplies: 0 };
  });
}

console.log(limitDepth(discussion(comments), 2));

// Od kolegy: počet všech komentářů ve stromu pro nadpis diskuse.
function countAll(nodes) {
  if (nodes.length === 0) {
    return 0;
  }
  const [first, ...rest] = nodes;
  return 1 + countAll(first?.replies ?? []) + countAll(rest);
}

console.log('Komentářů:', countAll(discussion(comments)));

// Počet odpovědí se správným tvarem slova.
const pluralRules = new Intl.PluralRules('cs');
const replyWords = { one: 'odpověď', few: 'odpovědi', other: 'odpovědí' };

function replyLabel(count) {

}

console.log(replyLabel(3));

// Vytvoří seznam komentářů a do každého vnoří seznam jeho odpovědí.
function renderThread(nodes) {
  const list = document.createElement('ul');
  list.className = 'thread';
  for (const node of nodes) {
    const item = document.createElement('li');
    item.className = 'comment';

    const author = document.createElement('p');
    author.className = 'comment__author';
    author.textContent = node.author ?? '—';

    const text = document.createElement('p');
    text.className = 'comment__text';
    text.textContent = node.text;

    const likes = document.createElement('p');
    likes.className = 'comment__likes';
    likes.textContent = `♥ ${node.likes}`;

    item.append(author, text, likes);
    if (node.replies.length > 0) {
      const toggle = document.createElement('button');
      toggle.type = 'button';
      toggle.className = 'comment__toggle';
      toggle.setAttribute('aria-expanded', 'true');
      toggle.textContent = `Skrýt ${replyLabel(countReplies(node))}`;
      item.append(toggle, renderThread(node.replies));
    }
    list.append(item);
  }
  return list;
}



// Od kolegy: tlačítka u vláken schovávají a ukazují odpovědi.
function connectToggles() {
  const buttons = document.querySelectorAll('#discussion .comment__toggle');
  for (let i = 0; i < buttons.length; i++) {
    buttons[i].addEventListener('click', () => {
      const replies = buttons[i].nextElementSibling;
      const expanded = buttons[i].getAttribute('aria-expanded') === 'true';
      replies.hidden = expanded;
      buttons[i].setAttribute('aria-expanded', String(!expanded));
      buttons[i].textContent = expanded
        ? buttons[i].textContent.replace('Skrýt', 'Zobrazit')
        : buttons[i].textContent.replace('Zobrazit', 'Skrýt');
    });
  }
}



// Nový strom seřazený porovnávací funkcí na každé úrovni.
function sortTree(nodes, compare) {
  return nodes
    .toSorted(compare)
    .map((node) => ({ ...node, replies: sortTree(node.replies, compare) }));
}

const byLikes = (a, b) => b.likes - a.likes;

console.log(sortTree(discussion(comments), byLikes).map((node) => node.author));

  document.querySelector('#discussion').replaceChildren(renderThread(sorted));
  connectToggles();
}

// Nová kopie seřazená od nejstaršího komentáře. Časy ve formátu ISO jdou porovnat jako text.
function oldestFirst(list) {
  return list.toSorted((a, b) => a.createdAt.localeCompare(b.createdAt));
}

console.log(oldestFirst(comments).at(-1).author);

// Přímé odpovědi na komentář s daným id; null vrátí komentáře přímo pod receptem.
function repliesTo(list, parentId) {
  return list.filter((comment) => comment.parentId === parentId);
}

console.log(repliesTo(comments, null).map((comment) => comment.author));

// Hloubka komentáře: přímo pod receptem 0, odpověď 1, odpověď na odpověď 2…
function depthOf(comment, index) {
  if (comment.parentId === null) {
    return 0;
  }
  return 1 + depthOf(index[comment.parentId], index);
}

console.log(depthOf(byId[8], byId));

// Autoři vlákna od komentáře pod receptem až po zadaný komentář.
function threadAuthors(comment, index) {
  if (comment.parentId === null) {
    return [comment.author];
  }
  return [...threadAuthors(index[comment.parentId], index), comment.author];
}

console.log(threadAuthors(byId[8], byId).join(' → '));

function buildTree(list, parentId = null) {
  return list
    .filter((comment) => comment.parentId === parentId)
    .map((comment) => ({
      ...comment,
      replies: buildTree(list, comment.id),
    }));
}

// Strom diskuse v pořadí serveru: uzel = kopie komentáře a v replies strom jeho odpovědí.
const thread = buildTree(comments);

// Počet všech odpovědí pod uzlem, včetně odpovědí na odpovědi.
function countReplies(node) {
  return node.replies.reduce((sum, reply) => sum + 1 + countReplies(reply), 0);
}

console.log(`${thread[0].author}: ${countReplies(thread[0])}`);

// Odsazený výpis: každá úroveň odpovědí o dvě mezery dál.
function toLines(nodes, depth = 0) {
  return nodes.flatMap((node) => [
    `${'  '.repeat(depth)}${node.author}: ${node.text}`,
    ...toLines(node.replies, depth + 1),
  ]);
}

console.log(toLines(thread).join('\n'));

// Složí funkce zleva doprava: výstup jedné je vstupem další.
const pipe = (...fns) => (value) => fns.reduce((result, fn) => fn(result), value);

// Strom přímo z dat serveru: nejdřív seřadit od nejstaršího, pak postavit strom.
const threadFromApi = pipe(oldestFirst, buildTree);

console.log(toLines(threadFromApi(comments)).join('\n'));

// Smazaný komentář zůstane ve stromu kvůli odpovědím, jen bez autora a textu.
function maskDeleted(nodes) {
  return nodes.map((node) => {
    const replies = maskDeleted(node.replies);
    if (node.deleted) {
      return { ...node, author: null, text: '[smazaný komentář]', replies };
    }
    return { ...node, replies };
  });
}

// Smazané komentáře bez odpovědí zmizí úplně. Odpovědi se vyčistí dřív než jejich rodič.
function pruneDeleted(nodes) {
  return nodes
    .map((node) => ({ ...node, replies: pruneDeleted(node.replies) }))
    .filter((node) => !node.deleted || node.replies.length > 0);
}

// Hotová diskuse z dat serveru.
const discussion = pipe(oldestFirst, buildTree, pruneDeleted, maskDeleted);

console.log(toLines(discussion(comments)).join('\n'));

// Najde uzel s daným id v libovolné hloubce stromu.
function findComment(nodes, id) {
  for (const node of nodes) {
    if (node.id === id) {
      return node;
    }
    const found = findComment(node.replies, id);
    if (found) {
      return found;
    }
  }
  return undefined;
}

console.log(findComment(discussion(comments), 8));

// Počet úrovní stromu podle nejhlubšího vlákna.
function maxDepth(nodes) {
  return nodes.reduce((deepest, node) => Math.max(deepest, 1 + maxDepth(node.replies)), 0);
}

console.log('Úrovní:', maxDepth(discussion(comments)));

// Strom s nejvýš levels úrovněmi; poslední úroveň si pamatuje, kolik odpovědí schovala.
function limitDepth(nodes, levels) {
  return nodes.map((node) => {
    if (levels === 1) {
      return { ...node, replies: [], hiddenReplies: countReplies(node) };
    }
    return { ...node, replies: limitDepth(node.replies, levels - 1), hiddenReplies: 0 };
  });
}

console.log(limitDepth(discussion(comments), 2));

// Od kolegy: počet všech komentářů ve stromu pro nadpis diskuse.
function countAll(nodes) {
  if (nodes.length === 0) {
    return 0;
  }
  const [first, ...rest] = nodes;
  return 1 + countAll(first?.replies ?? []) + countAll(rest);
}

console.log('Komentářů:', countAll(discussion(comments)));

const pluralRules = new Intl.PluralRules('cs');
const replyWords = { one: 'odpověď', few: 'odpovědi', other: 'odpovědí' };

function replyLabel(count) {
  return `${count} ${replyWords[pluralRules.select(count)]}`;
}

// Vytvoří seznam komentářů a do každého vnoří seznam jeho odpovědí.
function renderThread(nodes) {
  const list = document.createElement('ul');
  list.className = 'thread';
  for (const node of nodes) {
    const item = document.createElement('li');
    item.className = 'comment';

    const author = document.createElement('p');
    author.className = 'comment__author';
    author.textContent = node.author ?? '—';

    const text = document.createElement('p');
    text.className = 'comment__text';
    text.textContent = node.text;

    const likes = document.createElement('p');
    likes.className = 'comment__likes';
    likes.textContent = `♥ ${node.likes}`;

    item.append(author, text, likes);
    if (node.hiddenReplies > 0) {
      const more = document.createElement('button');
      more.type = 'button';
      more.className = 'comment__more';
      more.textContent = `Pokračovat ve vlákně (${replyLabel(node.hiddenReplies)})`;
      item.append(more);
    }
    if (node.replies.length > 0) {
      const toggle = document.createElement('button');
      toggle.type = 'button';
      toggle.className = 'comment__toggle';
      toggle.setAttribute('aria-expanded', 'true');
      toggle.textContent = `Skrýt ${replyLabel(countReplies(node))}`;
      item.append(toggle, renderThread(node.replies));
    }
    list.append(item);
  }
  return list;
}



// Od kolegy: tlačítka u vláken schovávají a ukazují odpovědi.
function connectToggles() {
  const buttons = document.querySelectorAll('#discussion .comment__toggle');
  for (let i = 0; i < buttons.length; i++) {
    buttons[i].addEventListener('click', () => {
      const replies = buttons[i].nextElementSibling;
      const expanded = buttons[i].getAttribute('aria-expanded') === 'true';
      replies.hidden = expanded;
      buttons[i].setAttribute('aria-expanded', String(!expanded));
      buttons[i].textContent = expanded
        ? buttons[i].textContent.replace('Skrýt', 'Zobrazit')
        : buttons[i].textContent.replace('Zobrazit', 'Skrýt');
    });
  }
}



// Nový strom seřazený porovnávací funkcí na každé úrovni.
function sortTree(nodes, compare) {
  return nodes
    .toSorted(compare)
    .map((node) => ({ ...node, replies: sortTree(node.replies, compare) }));
}
