# --description--

## Hlášení

Kolega zapojil tlačítka u vláken funkcí `connectToggles()` na konci souboru. Kliknutí má odpovědi schovat, druhé kliknutí je zase ukázat. Po kliknutí na **kterékoli** tlačítko se ale nestane nic a konzole hlásí:

`TypeError: Cannot read properties of undefined (reading 'nextElementSibling')`

Kolega tvrdí, že `document.querySelectorAll('.comment__toggle')[0].nextElementSibling` seznam odpovědí najde, a to je pravda.

## Úkol

Oprav `connectToggles` tak, aby každé tlačítko schovávalo a ukazovalo odpovědi svého komentáře. Změň co nejmenší kus kódu.

# --help--

## --tip--

Chyba odpovídá pasti [`var` v cyklu se `setTimeout`](see:js-funkce-hloubka/closures#var-v-cyklu-se-settimeout) — posluchač je taky funkce, která se spustí až po cyklu.

## --tip--

Jakou hodnotu má `i` ve chvíli, kdy uživatel klikne? Kolik proměnných `i` v cyklu existuje a co je `buttons[i]` pro takové číslo?

## --tip--

Ověř hypotézu: na první řádek posluchače dočasně připiš `console.log(i, buttons.length)` a klikni na libovolné tlačítko. Obě čísla jsou stejná, a proto `buttons[i]` nic nenajde. Hledej deklaraci, která v hlavičce cyklu vytvoří pro každé kolo novou proměnnou.

# --explain--

Vysvětli vlastními slovy, proč všechna tlačítka hlásila chybu, i když cyklus proběhl pro každé z nich.

## --model--

Posluchač se nespustí v cyklu, ale až po kliknutí, kdy cyklus dávno doběhl. `var` vytvoří pro celou funkci jedinou proměnnou `i` a všechny posluchače si přes closure pamatují právě ji, ne hodnotu z jejich kola. Po cyklu je v `i` délka seznamu tlačítek, takže `buttons[i]` je `undefined` a čtení `nextElementSibling` spadne. S `let` v hlavičce cyklu vznikne v každém kole nová proměnná a každý posluchač má svoje číslo.

## --checklist--

- Posluchač se spustí až po kliknutí, kdy cyklus doběhl.
- Posluchač si pamatuje proměnnou `i`, ne její hodnotu.
- `var` vytvoří jedinou proměnnou pro všechna kola.
- Po cyklu je `i` rovno délce pole, takže `buttons[i]` je `undefined`.
- `let` v hlavičce cyklu vytvoří pro každé kolo novou proměnnou.

function maskDeleted(nodes) {
  return nodes.map((node) => {
    const replies = maskDeleted(node.replies);
    if (node.deleted) {
      return { ...node, author: null, text: '[smazaný komentář]', replies };

    return { ...node, replies };
  });

function findComment(nodes, id) {
  for (const node of nodes) {
    if (node.id === id) {
      return node;
    }
    const found = findComment(node.replies, id);
    if (found) {
      return found;

// Strom s nejvýš levels úrovněmi; poslední úroveň si pamatuje, kolik odpovědí schovala.
function limitDepth(nodes, levels) {
  return nodes.map((node) => {
    if (levels === 1) {
      return { ...node, replies: [], hiddenReplies: countReplies(node) };
    }
    return { ...node, replies: limitDepth(node.replies, levels - 1), hiddenReplies: 0 };
  });
}

// Od kolegy: počet všech komentářů ve stromu pro nadpis diskuse.
function countAll(nodes) {

    return 0;

// Vytvoří seznam komentářů a do každého vnoří seznam jeho odpovědí.
function renderThread(nodes) {
  const list = document.createElement('ul');
  list.className = 'thread';
  for (const node of nodes) {
    const item = document.createElement('li');
    item.className = 'comment';

    const author = document.createElement('p');
    author.className = 'comment__author';
    author.textContent = node.author ?? '—';

    const text = document.createElement('p');
    text.className = 'comment__text';
    text.textContent = node.text;

    const likes = document.createElement('p');
    likes.className = 'comment__likes';
    likes.textContent = `♥ ${node.likes}`;

    item.append(author, text, likes);
    if (node.replies.length > 0) {
      const toggle = document.createElement('button');
      toggle.type = 'button';
      toggle.className = 'comment__toggle';
      toggle.setAttribute('aria-expanded', 'true');
      toggle.textContent = `Skrýt ${replyLabel(countReplies(node))}`;
      item.append(toggle, renderThread(node.replies));
    }
    list.append(item);
  }
  return list;
}

document.querySelector('#discussion').replaceChildren(renderThread(discussion(comments)));

// Od kolegy: tlačítka u vláken schovávají a ukazují odpovědi.
function connectToggles() {
  const buttons = document.querySelectorAll('#discussion .comment__toggle');
  for (var i = 0; i < buttons.length; i++) {
    buttons[i].addEventListener('click', () => {
      const replies = buttons[i].nextElementSibling;
      const expanded = buttons[i].getAttribute('aria-expanded') === 'true';
      replies.hidden = expanded;
      buttons[i].setAttribute('aria-expanded', String(!expanded));
      buttons[i].textContent = expanded
        ? buttons[i].textContent.replace('Skrýt', 'Zobrazit')
        : buttons[i].textContent.replace('Zobrazit', 'Skrýt');
    });
  }
}

// Strom diskuse v pořadí serveru: uzel = kopie komentáře a v replies strom jeho odpovědí.

// Počet všech odpovědí pod uzlem, včetně odpovědí na odpovědi.
function countReplies(node) {
  return node.replies.reduce((sum, reply) => sum + 1 + countReplies(reply), 0);
}

// Odsazený výpis: každá úroveň odpovědí o dvě mezery dál.
function toLines(nodes, depth = 0) {
  return nodes.flatMap((node) => [
    `${'  '.repeat(depth)}${node.author}: ${node.text}`,
    ...toLines(node.replies, depth + 1),
  ]);
}

console.log(toLines(thread).join('\n'));

// Složí funkce zleva doprava: výstup jedné je vstupem další.
const pipe = (...fns) => (value) => fns.reduce((result, fn) => fn(result), value);

// Strom přímo z dat serveru: nejdřív seřadit od nejstaršího, pak postavit strom.
const threadFromApi = pipe(oldestFirst, buildTree);

// Smazaný komentář zůstane ve stromu kvůli odpovědím, jen bez autora a textu.
function maskDeleted(nodes) {
  return nodes.map((node) => {
    const replies = maskDeleted(node.replies);
    if (node.deleted) {
      return { ...node, author: null, text: '[smazaný komentář]', replies };
    }
    return { ...node, replies };
  });
}

// Smazané komentáře bez odpovědí zmizí úplně. Odpovědi se vyčistí dřív než jejich rodič.
function pruneDeleted(nodes) {
  return nodes
    .map((node) => ({ ...node, replies: pruneDeleted(node.replies) }))
    .filter((node) => !node.deleted || node.replies.length > 0);
}

// Hotová diskuse z dat serveru.
const discussion = pipe(oldestFirst, buildTree, pruneDeleted, maskDeleted);

console.log(toLines(discussion(comments)).join('\n'));

// Najde uzel s daným id v libovolné hloubce stromu.
function findComment(nodes, id) {
  for (const node of nodes) {
    if (node.id === id) {
      return node;
    }
    const found = findComment(node.replies, id);
    if (found) {
      return found;
    }
  }
  return undefined;
}

console.log(findComment(discussion(comments), 8));

// Počet úrovní stromu podle nejhlubšího vlákna.
function maxDepth(nodes) {
  return nodes.reduce((deepest, node) => Math.max(deepest, 1 + maxDepth(node.replies)), 0);
}

console.log('Úrovní:', maxDepth(discussion(comments)));

// Strom s nejvýš levels úrovněmi; poslední úroveň si pamatuje, kolik odpovědí schovala.
function limitDepth(nodes, levels) {
  return nodes.map((node) => {
    if (levels === 1) {
      return { ...node, replies: [], hiddenReplies: countReplies(node) };
    }
    return { ...node, replies: limitDepth(node.replies, levels - 1), hiddenReplies: 0 };
  });
}

console.log(limitDepth(discussion(comments), 2));

// Od kolegy: počet všech komentářů ve stromu pro nadpis diskuse.
function countAll(nodes) {
  if (nodes.length === 0) {
    return 0;
  }
  const [first, ...rest] = nodes;
  return 1 + countAll(first?.replies ?? []) + countAll(rest);
}

console.log('Komentářů:', countAll(discussion(comments)));

// Vytvoří seznam komentářů a do každého vnoří seznam jeho odpovědí.
function renderThread(nodes) {
  const list = document.createElement('ul');
  list.className = 'thread';
  for (const node of nodes) {
    const item = document.createElement('li');
    item.className = 'comment';

    const author = document.createElement('p');
    author.className = 'comment__author';
    author.textContent = node.author ?? '—';

    const text = document.createElement('p');
    text.className = 'comment__text';
    text.textContent = node.text;

    const likes = document.createElement('p');
    likes.className = 'comment__likes';
    likes.textContent = `♥ ${node.likes}`;

    item.append(author, text, likes);
    if (node.replies.length > 0) {
      const toggle = document.createElement('button');
      toggle.type = 'button';
      toggle.className = 'comment__toggle';
      toggle.setAttribute('aria-expanded', 'true');
      toggle.textContent = `Skrýt ${replyLabel(countReplies(node))}`;
      item.append(toggle, renderThread(node.replies));
    }
    list.append(item);
  }
  return list;
}

document.querySelector('#discussion').replaceChildren(renderThread(discussion(comments)));

// Od kolegy: tlačítka u vláken schovávají a ukazují odpovědi.
function connectToggles() {
  const buttons = document.querySelectorAll('#discussion .comment__toggle');
  for (let i = 0; i < buttons.length; i++) {
    buttons[i].addEventListener('click', () => {
      const replies = buttons[i].nextElementSibling;
      const expanded = buttons[i].getAttribute('aria-expanded') === 'true';
      replies.hidden = expanded;
      buttons[i].setAttribute('aria-expanded', String(!expanded));
      buttons[i].textContent = expanded
        ? buttons[i].textContent.replace('Skrýt', 'Zobrazit')
        : buttons[i].textContent.replace('Zobrazit', 'Skrýt');
    });
  }
}
