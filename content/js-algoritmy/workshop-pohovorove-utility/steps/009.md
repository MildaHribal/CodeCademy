# --description--

Máš dvě stě obrázků ke zpracování. Pustit je všechny naráz znamená dvě stě souběžných
požadavků — prohlížeč je stejně zařadí do fronty, server se může zalknout a první
chyba shodí celek. Pustit je po jednom je zase zbytečně pomalé.

**Úkol:** Napiš `async function mapLimit(items, limit, fn)`. Zpracuje všechny položky
funkcí `fn(item, index)` tak, aby **najednou běželo nejvýš `limit`** volání, a vrátí
pole výsledků **v pořadí vstupu** (ne v pořadí dokončení).

Jak na to, ti tentokrát neporadím. Rozmysli si, co musí být pravda: jakmile se jedno
volání dokončí, má hned začít další, dokud nějaké zbývají.

# --hints--

Výsledky jsou v pořadí vstupu.

```js
const vysledky = await mapLimit([1, 2, 3, 4, 5], 2, async (n) => {
  await helpers.wait((6 - n) * 10);
  return n * 10;
});
assert.deepEqual(vysledky, [10, 20, 30, 40, 50], 'Pořadí výsledků se řídí vstupem, ne dokončením');
```

Nikdy neběží víc volání, než je limit.

```js
let bezi = 0;
let vrchol = 0;
await mapLimit([...Array(9).keys()], 3, async () => {
  bezi += 1;
  vrchol = Math.max(vrchol, bezi);
  await helpers.wait(20);
  bezi -= 1;
});
assert.equal(vrchol, 3, `Souběžně mělo běžet nejvýš 3 volání, běželo ${vrchol}`);
```

Souběh se opravdu využije.

```js
const zacatek = Date.now();
await mapLimit([...Array(8).keys()], 4, async () => { await helpers.wait(40); });
const trvani = Date.now() - zacatek;
assert.ok(trvani < 200, `Osm úloh po 40 ms se čtyřmi souběžně má trvat kolem 80 ms, trvalo ${trvani} ms`);
assert.ok(trvani >= 60, 'Všechny úlohy se mají opravdu spustit a dokončit');
```

`fn` dostane i index a prázdný vstup projde.

```js
const indexy = await mapLimit(['a', 'b', 'c'], 2, async (hodnota, index) => `${index}:${hodnota}`);
assert.deepEqual(indexy, ['0:a', '1:b', '2:c'], 'fn má dostat položku i její index');
assert.deepEqual(await mapLimit([], 3, async () => 1), [], 'Prázdný vstup dá prázdné pole');
```

Limit větší než počet položek nevadí.

```js
const vysledky = await mapLimit([1, 2], 10, async (n) => n + 1);
assert.deepEqual(vysledky, [2, 3], 'Limit větší než počet položek nesmí nic rozbít');
```

# --help--

## --tip--

Zkus to popsat slovy dřív, než začneš psát: „spustím nejvýš `limit` pracovníků a každý
si bere další volnou položku, dokud nějaká zbývá."

## --tip--

Výsledky ukládej na index vstupu do předem připraveného pole — pak se o pořadí nemusíš
starat vůbec.

## --tip--

Sdílený „ukazatel na další položku" je obyčejná proměnná v uzávěru. Každý pracovník si
z ní vezme číslo a hned ho posune, takže si dva pracovníci nikdy nevezmou totéž.

// utils.js — malá knihovna pomocných funkcí pro tým.
// Každá z nich se zároveň ptá na pohovorech: nejde o to je znát nazpaměť,
// ale umět je na místě odvodit a říct, kdy se hodí která.

// Ukázková data, na kterých si funkce zkoušíš.
const objednavky = [
  { id: 'o1', mesto: 'Brno', castka: 1290, stav: 'nova' },
  { id: 'o2', mesto: 'Praha', castka: 890, stav: 'odeslana' },
  { id: 'o3', mesto: 'Brno', castka: 2450, stav: 'odeslana' },
  { id: 'o4', mesto: 'Ostrava', castka: 640, stav: 'nova' },
  { id: 'o5', mesto: 'Praha', castka: 1790, stav: 'nova' },
];

/**
 * Zavolá `fn` až po `delay` ms bez dalšího volání.
 * Vrácená funkce si v uzávěru pamatuje běžící časovač.
 */
function debounce(fn, delay) {
  let cekani;
  return (...args) => {
    clearTimeout(cekani);
    cekani = setTimeout(() => fn(...args), delay);
  };
}

/**
 * Pustí `fn` nejvýš jednou za `limit` ms. První volání projde hned.
 * Debounce čeká na klid, throttle drží tempo.
 */
function throttle(fn, limit) {
  let naposledy = 0;
  return (...args) => {
    const ted = Date.now();
    if (ted - naposledy < limit) return;
    naposledy = ted;
    fn(...args);
  };
}

/**
 * Zavolá `fn` jen při prvním volání, další volání vrací uložený výsledek.
 * Příznak je oddělený od výsledku — jinak by funkce vracející 0 běžela pořád.
 */
function once(fn) {
  let hotovo = false;
  let vysledek;
  return (...args) => {
    if (!hotovo) {
      vysledek = fn(...args);
      hotovo = true;
    }
    return vysledek;
  };
}

/**
 * Uloží výsledky `fn` podle klíče z `keyFn`.
 * Map + has se ptá na existenci záznamu, ne na jeho pravdivost.
 */
function memoize(fn, keyFn = (prvni) => prvni) {
  const cache = new Map();
  return (...args) => {
    const klic = keyFn(...args);
    if (!cache.has(klic)) cache.set(klic, fn(...args));
    return cache.get(klic);
  };
}

/**
 * Rozdělí položky do skupin podle klíče.
 * Map, ne objekt: klíč „constructor" ani „__proto__" pak nic nerozbije.
 */
function groupBy(items, keyFn) {
  const skupiny = new Map();
  for (const item of items) {
    const klic = keyFn(item);
    if (!skupiny.has(klic)) skupiny.set(klic, []);
    skupiny.get(klic).push(item);
  }
  return skupiny;
}

/**
 * Hluboká kopie, která zvládne i cykly.
 * Kopie se do paměti ukládá PŘED kopírováním vlastností — jinak se rekurze zacyklí.
 */
function deepClone(hodnota, videne = new WeakMap()) {
  if (hodnota === null || typeof hodnota !== 'object') return hodnota;
  if (hodnota instanceof Date) return new Date(hodnota);
  if (videne.has(hodnota)) return videne.get(hodnota);

  if (Array.isArray(hodnota)) {
    const kopie = [];
    videne.set(hodnota, kopie);
    for (const polozka of hodnota) kopie.push(deepClone(polozka, videne));
    return kopie;
  }

  const kopie = {};
  videne.set(hodnota, kopie);
  for (const [klic, vnorena] of Object.entries(hodnota)) {
    kopie[klic] = deepClone(vnorena, videne);
  }
  return kopie;
}

/** Počká zadaný počet milisekund. */
function pockej(ms) {
  return new Promise((hotovo) => setTimeout(hotovo, ms));
}

/**
 * Zkusí `fn` až `pokusy`krát, mezi pokusy počká `prodleva` ms.
 * Po vyčerpání pokusů vyhodí poslední chybu — jinak volající neví, na čem to stálo.
 */
async function retry(fn, { pokusy = 3, prodleva = 100 } = {}) {
  let posledni;
  for (let pokus = 1; pokus <= pokusy; pokus++) {
    try {
      return await fn();
    } catch (chyba) {
      posledni = chyba;
      if (pokus < pokusy) await pockej(prodleva);
    }
  }
  throw posledni;
}

/**
 * Zpracuje všechny položky s nejvýš `limit` souběžnými voláními.
 * Výsledky se ukládají na index vstupu, takže pořadí odpovídá vstupu.
 */
async function mapLimit(items, limit, fn) {
  const vysledky = new Array(items.length);
  let dalsi = 0;

  async function pracovnik() {
    while (dalsi < items.length) {
      const index = dalsi;
      dalsi += 1;
      vysledky[index] = await fn(items[index], index);
    }
  }

  const pocet = Math.min(limit, items.length);
  await Promise.all(Array.from({ length: pocet }, () => pracovnik()));
  return vysledky;
}
