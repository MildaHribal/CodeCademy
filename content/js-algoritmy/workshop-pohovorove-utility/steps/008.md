# --description--

Požadavek na API neselže proto, že je kód špatně — selže proto, že zrovna vypadla
wifi. Na to se nepíše `if`, na to se píše opakování.

**Úkol:** Napiš `async function retry(fn, moznosti)`. Zavolá `fn()` a když vyhodí
chybu (nebo se její promise odmítne), zkusí to znovu. Nastavení:

- `pokusy` — kolikrát se celkem smí volat, výchozí `3`,
- `prodleva` — kolik milisekund počkat mezi pokusy, výchozí `100`.

Když projde některý pokus, vrať jeho výsledek. Když selžou všechny, **vyhoď poslední
chybu** — volající musí vidět, na čem to skončilo.

# --hints--

První úspěch se vrátí a dál se nezkouší.

```js
let volani = 0;
const vysledek = await retry(async () => { volani += 1; return 'ok'; });
assert.equal(vysledek, 'ok', 'Úspěšné volání má vrátit svůj výsledek');
assert.equal(volani, 1, 'Když první pokus projde, druhý se nekoná');
```

Po dvou selháních projde třetí pokus.

```js
let volani = 0;
const vysledek = await retry(async () => {
  volani += 1;
  if (volani < 3) throw new Error('síť');
  return 'konečně';
}, { pokusy: 4, prodleva: 10 });
assert.equal(vysledek, 'konečně', 'Má se vrátit výsledek prvního úspěšného pokusu');
assert.equal(volani, 3, 'Volání mají být tři: dvě selhání a jeden úspěch');
```

Když selžou všechny pokusy, vyhodí se poslední chyba.

```js
let volani = 0;
let chycena = null;
try {
  await retry(async () => { volani += 1; throw new Error(`pokus ${volani}`); }, { pokusy: 3, prodleva: 5 });
} catch (chyba) {
  chycena = chyba;
}
assert.ok(chycena instanceof Error, 'Po vyčerpání pokusů má retry chybu vyhodit');
assert.equal(chycena.message, 'pokus 3', 'Vyhodit se má poslední chyba, ne ta první');
assert.equal(volani, 3, 'Pokusy mají být přesně tři');
```

Mezi pokusy se opravdu čeká.

```js
const zacatek = Date.now();
let volani = 0;
try {
  await retry(async () => { volani += 1; throw new Error('ne'); }, { pokusy: 3, prodleva: 40 });
} catch {}
const trvani = Date.now() - zacatek;
assert.ok(trvani >= 70, `Mezi třemi pokusy se má dvakrát čekat 40 ms, celé to trvalo ${trvani} ms`);
```

# --help--

## --tip--

Uvnitř `async` funkce se čeká přes `await new Promise((hotovo) => setTimeout(hotovo, ms))`.
Pomocnou funkci na to si klidně vyrob vedle.

## --tip--

Cyklus potřebuje dvě věci: pamatovat si poslední chybu a po posledním pokusu nečekat
zbytečně. Podmínka „je tohle poslední kolo?" se dá napsat porovnáním čísla pokusu
s počtem pokusů.

## --tip--

V produkci se prodleva obvykle zdvojnásobuje (*exponential backoff*) a přidává se
náhodný rozptyl, aby všichni klienti nezkoušeli znovu naráz. Zmínit to na pohovoru
uděláš dobrý dojem.

// utils.js — malá knihovna pomocných funkcí pro tým.
// Každá z nich se zároveň ptá na pohovorech: nejde o to je znát nazpaměť,
// ale umět je na místě odvodit a říct, kdy se hodí která.

// Ukázková data, na kterých si funkce zkoušíš.
const objednavky = [
  { id: 'o1', mesto: 'Brno', castka: 1290, stav: 'nova' },
  { id: 'o2', mesto: 'Praha', castka: 890, stav: 'odeslana' },
  { id: 'o3', mesto: 'Brno', castka: 2450, stav: 'odeslana' },
  { id: 'o4', mesto: 'Ostrava', castka: 640, stav: 'nova' },
  { id: 'o5', mesto: 'Praha', castka: 1790, stav: 'nova' },
];

/**
 * Zavolá `fn` až po `delay` ms bez dalšího volání.
 * Vrácená funkce si v uzávěru pamatuje běžící časovač.
 */
function debounce(fn, delay) {
  let cekani;
  return (...args) => {
    clearTimeout(cekani);
    cekani = setTimeout(() => fn(...args), delay);
  };
}

/**
 * Pustí `fn` nejvýš jednou za `limit` ms. První volání projde hned.
 * Debounce čeká na klid, throttle drží tempo.
 */
function throttle(fn, limit) {
  let naposledy = 0;
  return (...args) => {
    const ted = Date.now();
    if (ted - naposledy < limit) return;
    naposledy = ted;
    fn(...args);
  };
}

/**
 * Zavolá `fn` jen při prvním volání, další volání vrací uložený výsledek.
 * Příznak je oddělený od výsledku — jinak by funkce vracející 0 běžela pořád.
 */
function once(fn) {
  let hotovo = false;
  let vysledek;
  return (...args) => {
    if (!hotovo) {
      vysledek = fn(...args);
      hotovo = true;
    }
    return vysledek;
  };
}

/**
 * Uloží výsledky `fn` podle klíče z `keyFn`.
 * Map + has se ptá na existenci záznamu, ne na jeho pravdivost.
 */
function memoize(fn, keyFn = (prvni) => prvni) {
  const cache = new Map();
  return (...args) => {
    const klic = keyFn(...args);
    if (!cache.has(klic)) cache.set(klic, fn(...args));
    return cache.get(klic);
  };
}

/**
 * Rozdělí položky do skupin podle klíče.
 * Map, ne objekt: klíč „constructor" ani „__proto__" pak nic nerozbije.
 */
function groupBy(items, keyFn) {
  const skupiny = new Map();
  for (const item of items) {
    const klic = keyFn(item);
    if (!skupiny.has(klic)) skupiny.set(klic, []);
    skupiny.get(klic).push(item);
  }
  return skupiny;
}

/**
 * Hluboká kopie, která zvládne i cykly.
 * Kopie se do paměti ukládá PŘED kopírováním vlastností — jinak se rekurze zacyklí.
 */
function deepClone(hodnota, videne = new WeakMap()) {
  if (hodnota === null || typeof hodnota !== 'object') return hodnota;
  if (hodnota instanceof Date) return new Date(hodnota);
  if (videne.has(hodnota)) return videne.get(hodnota);

  if (Array.isArray(hodnota)) {
    const kopie = [];
    videne.set(hodnota, kopie);
    for (const polozka of hodnota) kopie.push(deepClone(polozka, videne));
    return kopie;
  }

  const kopie = {};
  videne.set(hodnota, kopie);
  for (const [klic, vnorena] of Object.entries(hodnota)) {
    kopie[klic] = deepClone(vnorena, videne);
  }
  return kopie;
}

/** Počká zadaný počet milisekund. */
function pockej(ms) {
  return new Promise((hotovo) => setTimeout(hotovo, ms));
}

/**
 * Zkusí `fn` až `pokusy`krát, mezi pokusy počká `prodleva` ms.
 * Po vyčerpání pokusů vyhodí poslední chybu — jinak volající neví, na čem to stálo.
 */
async function retry(fn, { pokusy = 3, prodleva = 100 } = {}) {
  let posledni;
  for (let pokus = 1; pokus <= pokusy; pokus++) {
    try {
      return await fn();
    } catch (chyba) {
      posledni = chyba;
      if (pokus < pokusy) await pockej(prodleva);
    }
  }
  throw posledni;
}
