# --description--

Hluboká kopie má jednu díru: objekt, který odkazuje sám na sebe. Uzel stromu se
odkazem na rodiče, dva propojené záznamy, DOM-like struktura — a `deepClone` se zacyklí
až do `RangeError: Maximum call stack size exceeded`.

**Úkol:** Uprav `deepClone` tak, aby zvládla i cykly. Každý objekt, který jsi už
zkopíroval, si zapamatuj spolu s jeho kopií; když na něj narazíš podruhé, vrať tu
uloženou kopii místo nové.

Paměť si předej druhým parametrem s výchozí hodnotou, ať se volání zvenčí nezmění:
`deepClone(hodnota)` musí dál fungovat.

# --hints--

Cyklus přes sebe sama nespadne a odkaz zůstane cyklem.

```js
const uzel = { nazev: 'kořen' };
uzel.sam = uzel;
const kopie = deepClone(uzel);
assert.equal(kopie.nazev, 'kořen', 'Kopie má mít stejný obsah');
assert.notEqual(kopie, uzel, 'Kopie musí být nový objekt');
assert.equal(kopie.sam, kopie, 'Odkaz na sebe sama má v kopii ukazovat na kopii');
```

Vzájemné odkazy zůstanou vzájemné.

```js
const rodic = { nazev: 'rodič', deti: [] };
const dite = { nazev: 'dítě', rodic };
rodic.deti.push(dite);
const kopie = deepClone(rodic);
assert.equal(kopie.deti[0].nazev, 'dítě', 'Struktura má zůstat zachovaná');
assert.equal(kopie.deti[0].rodic, kopie, 'Dítě v kopii má ukazovat na kopii rodiče, ne na originál');
assert.notEqual(kopie.deti[0], dite, 'Dítě musí být nová instance');
```

Stejný objekt na dvou místech zůstane jednou kopií.

```js
const adresa = { ulice: 'Lipová', psc: '60200' };
const original = { fakturacni: adresa, dorucovaci: adresa };
const kopie = deepClone(original);
assert.equal(kopie.fakturacni, kopie.dorucovaci, 'Sdílený objekt má v kopii zůstat sdílený');
assert.notEqual(kopie.fakturacni, adresa, 'Ale musí to být nová instance');
```

Všechno z předchozího kroku platí dál.

```js
assert.equal(deepClone(7), 7, 'Primitivní hodnoty se vrací beze změny');
const d = deepClone({ kdy: new Date('2026-03-01T10:00:00Z'), polozky: [{ id: 1 }] });
assert.ok(d.kdy instanceof Date, 'Date má zůstat datem');
assert.ok(Array.isArray(d.polozky), 'Pole má zůstat polem');
assert.equal(deepClone(objednavky)[2].id, 'o3', 'Kopie ukázkových dat má dál sedět');
```

# --help--

## --tip--

Klíčem paměti je **objekt sám**, ne nějaký jeho identifikátor. Na to potřebuješ
strukturu, která umí objekt jako klíč.

## --tip--

Pořadí uvnitř je důležité: kopii musíš do paměti uložit **dřív**, než začneš kopírovat
její vlastnosti. Jinak se rekurze do paměti nikdy nepodívá včas.

## --tip--

`WeakMap` se hodí líp než `Map`: nedrží klíče naživu, takže zkopírované objekty může
uklidit garbage collector — viz
[WeakMap](see:js-tridy-kolekce/map-a-set#weakmap-data-k-objektu-dokud-objekt-zije).

  { id: 'o4', mesto: 'Ostrava', castka: 640, stav: 'nova' },
  { id: 'o5', mesto: 'Praha', castka: 1790, stav: 'nova' },

/**
 * Zavolá `fn` až po `delay` ms bez dalšího volání.
 * Vrácená funkce si v uzávěru pamatuje běžící časovač.
 */
function debounce(fn, delay) {
  let cekani;
  return (...args) => {
    clearTimeout(cekani);
    cekani = setTimeout(() => fn(...args), delay);
  };
}

/**
 * Pustí `fn` nejvýš jednou za `limit` ms. První volání projde hned.
 * Debounce čeká na klid, throttle drží tempo.
 */
function throttle(fn, limit) {
  let naposledy = 0;
  return (...args) => {
    const ted = Date.now();
    if (ted - naposledy < limit) return;
    naposledy = ted;
    fn(...args);
  };
}

/**
 * Zavolá `fn` jen při prvním volání, další volání vrací uložený výsledek.
 * Příznak je oddělený od výsledku — jinak by funkce vracející 0 běžela pořád.
 */
function once(fn) {
  let hotovo = false;
  let vysledek;
  return (...args) => {
    if (!hotovo) {
      vysledek = fn(...args);
      hotovo = true;
    }
    return vysledek;
  };
}

/**
 * Uloží výsledky `fn` podle klíče z `keyFn`.
 * Map + has se ptá na existenci záznamu, ne na jeho pravdivost.
 */
function memoize(fn, keyFn = (prvni) => prvni) {
  const cache = new Map();
  return (...args) => {
    const klic = keyFn(...args);
    if (!cache.has(klic)) cache.set(klic, fn(...args));
    return cache.get(klic);
  };
}

/**
 * Rozdělí položky do skupin podle klíče.
 * Map, ne objekt: klíč „constructor" ani „__proto__" pak nic nerozbije.
 */
function groupBy(items, keyFn) {
  const skupiny = new Map();
  for (const item of items) {
    const klic = keyFn(item);
    if (!skupiny.has(klic)) skupiny.set(klic, []);
    skupiny.get(klic).push(item);
  }
  return skupiny;
}

function deepClone(hodnota) {
  if (hodnota === null || typeof hodnota !== 'object') return hodnota;
  if (hodnota instanceof Date) return new Date(hodnota);
  if (Array.isArray(hodnota)) return hodnota.map((polozka) => deepClone(polozka));
  const kopie = {};
  for (const [klic, vnorena] of Object.entries(hodnota)) {
    kopie[klic] = deepClone(vnorena);
  }
  return kopie;
}

    const klic = keyFn(item);
    if (!skupiny.has(klic)) skupiny.set(klic, []);
    skupiny.get(klic).push(item);
  }
  return skupiny;

/**
 * Hluboká kopie, která zvládne i cykly.
 * Kopie se do paměti ukládá PŘED kopírováním vlastností — jinak se rekurze zacyklí.
 */
function deepClone(hodnota, videne = new WeakMap()) {
  if (hodnota === null || typeof hodnota !== 'object') return hodnota;
  if (hodnota instanceof Date) return new Date(hodnota);
  if (videne.has(hodnota)) return videne.get(hodnota);

  if (Array.isArray(hodnota)) {
    const kopie = [];
    videne.set(hodnota, kopie);
    for (const polozka of hodnota) kopie.push(deepClone(polozka, videne));
    return kopie;
  }

  const kopie = {};
  videne.set(hodnota, kopie);
  for (const [klic, vnorena] of Object.entries(hodnota)) {
    kopie[klic] = deepClone(vnorena, videne);
  }
  return kopie;
}
