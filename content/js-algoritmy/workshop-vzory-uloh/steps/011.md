# --description--

Do přehledu chybí poslední údaj: který jediný záznam měl nejdelší čas na stránce. Šéf z něj pozná, jestli lidé někde čtou, nebo jestli se jen ztratili.

**Úkol:** Napiš funkci `longestVisit(records)`, která vrátí **celý záznam** s nejvyšší hodnotou `seconds`. U prázdného pole vrátí `null` a vstupní pole nechá beze změny.

Nástroj si vyber sám — projde každé rozumné řešení. Než začneš, napiš si tři příklady včetně prázdného pole a ujasni si, co má funkce v každém z nich vrátit.

# --hints--

`longestVisit` vrátí záznam s nejdelším časem.

```js
const longest = longestVisit(views);
assert.equal(longest.seconds, 205, 'longestVisit(views).seconds má být 205 — nejdelší návštěva v datech');
assert.equal(longest.path, '/produkt/caj', 'longestVisit(views).path má být /produkt/caj');
```

Pro prázdné pole vrátí `longestVisit` hodnotu `null`.

```js
assert.equal(longestVisit([]), null, 'longestVisit([]) má vrátit null, ne undefined');
```

Vrací se původní objekt ze vstupu, ne jeho kopie.

```js
const records = [
  { visitorId: 'a', path: '/x', seconds: 10 },
  { visitorId: 'b', path: '/y', seconds: 90 },
];
assert.equal(longestVisit(records), records[1], 'longestVisit() má vrátit tentýž objekt, který je ve vstupním poli');
```

Vstupní pole zůstane beze změny.

```js
const records = [
  { visitorId: 'a', path: '/x', seconds: 10 },
  { visitorId: 'b', path: '/y', seconds: 90 },
];
longestVisit(records);
assert.equal(records[0].seconds, 10, 'longestVisit() nesmí vstupní data přerovnat ani změnit');
assert.equal(records[1].seconds, 90, 'longestVisit() nesmí vstupní data přerovnat ani změnit');
```

# --help--

## --tip--

Hledání maxima je jeden průchod, při kterém si pamatuješ dosud nejlepší nález — stejný tvar jako u hledání nejnavštěvovanější stránky.

## --tip--

Pozor na první krok: dokud nemáš žádný nález, není s čím porovnávat. Vyřeš to buď výchozí hodnotou, nebo podmínkou na prázdný nález.

## --tip--

Řazení je taky cesta, ale musí zůstat u kopie. Kdo seřadí vstupní pole, rozbije data všem ostatním funkcím v přehledu.

// Záznamy o zobrazení stránek z malého e-shopu s čajem a kávou.
// Jeden řádek = jedno zobrazení, seconds = čas strávený na stránce.
const views = [
  { visitorId: 'u7', path: '/kosik', seconds: 42 },
  { visitorId: 'u3', path: '/produkt/kava', seconds: 118 },
  { visitorId: 'u7', path: '/produkt/kava', seconds: 35 },
  { visitorId: 'u9', path: '/kosik', seconds: 12 },
  { visitorId: 'u3', path: '/kontakt', seconds: 64 },
  { visitorId: 'u7', path: '/kosik', seconds: 27 },
  { visitorId: 'u1', path: '/produkt/caj', seconds: 205 },
];

// Tržby za posledních sedm dní, v korunách.
const dailySales = [12400, 9800, 15300, 7200, 18900, 11100, 16600];

// Ceník, vzestupně seřazený. Dárkové poukázky se skládají ze dvou položek.
const sortedPrices = [49, 129, 249, 349, 499, 899, 1290];

// Kolik různých návštěvníků je v záznamech.
function uniqueVisitorCount(records) {
  const ids = new Set();
  for (const record of records) {
    ids.add(record.visitorId);
  }
  return ids.size;
}

// Součet všech vteřin strávených na webu.
function totalSeconds(records) {
  return records.reduce((total, record) => total + record.seconds, 0);
}

// Kolikrát se která stránka zobrazila. Klíč = cesta, hodnota = počet.
function viewsByPath(records) {
  const counts = new Map();
  for (const record of records) {
    counts.set(record.path, (counts.get(record.path) ?? 0) + 1);
  }
  return counts;
}

function topPath(counts) {
  let bestPath = null;
  let bestCount = 0;
  for (const [path, count] of counts) {
    if (count > bestCount) {
      bestPath = path;
      bestCount = count;
    }
  }
  return bestPath;
}

// Cesty, které se v záznamech objevily právě jednou.
function onlyOnce(counts) {
  const paths = [];
  for (const [path, count] of counts) {
    if (count === 1) {
      paths.push(path);
    }
  }
  return paths;
}

// Kolik stránek si který návštěvník prohlédl. Napsal kolega z marketingu.
function viewsByVisitor(records) {
  const counts = {};
  for (const record of records) {
    counts[record.visitorId] = (counts[record.visitorId] ?? 0) + 1;
  }
  return counts;
}

// Zobrazení od návštěvníků, kteří jsou ve věrnostním programu.
function vipViews(records, vipIds) {
  const vip = new Set(vipIds);
  return records.filter((record) => vip.has(record.visitorId));
}

function bestWindow(sales, size) {
  if (sales.length < size) return 0;
  let sum = 0;
  for (let i = 0; i < size; i++) sum += sales[i];
  let best = sum;
  for (let i = size; i < sales.length; i++) {
    sum = sum + sales[i] - sales[i - size];
    if (sum > best) best = sum;
  }
  return best;
}

// Index dne, kterým začíná nejlepší okno délky size. -1, když se okno nevejde.
function bestWindowStart(sales, size) {
  if (sales.length < size) return -1;
  let sum = 0;
  for (let i = 0; i < size; i++) sum += sales[i];
  let best = sum;
  let bestStart = 0;
  for (let i = size; i < sales.length; i++) {
    sum = sum + sales[i] - sales[i - size];
    if (sum > best) {
      best = sum;
      bestStart = i - size + 1;
    }
  }
  return bestStart;
}

// Existuje ve vzestupně seřazeném poli dvojice různých položek se součtem target?
function hasPairWithSum(sortedValues, target) {
  let left = 0;
  let right = sortedValues.length - 1;
  while (left < right) {
    const sum = sortedValues[left] + sortedValues[right];
    if (sum === target) return true;
    if (sum < target) left++;
    else right--;
  }
  return false;
}
