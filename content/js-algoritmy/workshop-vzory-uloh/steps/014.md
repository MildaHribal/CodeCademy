# --description--

Přehled je hotový po částech, ale nikdo si ho nebude skládat ručně. Jedna funkce z něj udělá objekt, který se dá poslat dál — do šablony, do API i do testu.

**Úkol:** Napiš funkci `trafficSummary(records, sales)`, která vrátí objekt s klíči `visitors`, `views`, `seconds`, `topPath` a `bestThreeDays`. Každou hodnotu vezmi z funkce, kterou už máš hotovou — nic nepočítej znovu.

`views` je prostý počet záznamů, `bestThreeDays` je nejlepší třídenní součet tržeb. Na prázdných datech musí funkce projít taky: nesmí spadnout a `topPath` v ní bude `null`.

# --hints--

`trafficSummary` vrátí kompletní shrnutí.

```js
assert.deepEqual(trafficSummary(views, dailySales), {
  visitors: 4,
  views: 7,
  seconds: 503,
  topPath: '/kosik',
  bestThreeDays: 46600,
}, 'trafficSummary(views, dailySales) má vrátit shrnutí se čtyřmi návštěvníky, sedmi zobrazeními a nejlepší trojicí 46600');
```

Na prázdných datech shrnutí nespadne.

```js
assert.deepEqual(trafficSummary([], []), {
  visitors: 0,
  views: 0,
  seconds: 0,
  topPath: null,
  bestThreeDays: 0,
}, 'trafficSummary([], []) má vrátit samé nuly a topPath null');
```

Shrnutí nesahá na vstupní data.

```js
const before = views.length;
trafficSummary(views, dailySales);
assert.equal(views.length, before, 'trafficSummary() nesmí měnit vstupní záznamy');
assert.equal(dailySales[0], 12400, 'trafficSummary() nesmí měnit pole tržeb');
```

# --help--

## --tip--

Žádný nový vzor tu není — jde jen o to zavolat hotové funkce a výsledky poskládat do objektu.

## --tip--

Nejčastější stránku nedostaneš ze záznamů přímo. Nejdřív potřebuješ mapu počtů, teprve z ní vítěze.

# --explain--

Kdy se vyplatí složit shrnutí z pěti hotových funkcí a kdy je naopak lepší spočítat všechno jedním průchodem?

## --model--

Skládání z hotových funkcí se vyplatí skoro vždycky: každá funkce se dá otestovat zvlášť, jméno říká, co počítá, a změna jedné se nedotkne ostatních. Cena je v tom, že se data projdou tolikrát, kolik je funkcí — pět průchodů místo jednoho. To je pořád lineární, takže na běžných datech je to jedno. Jeden společný průchod dává smysl až tehdy, když je dat opravdu hodně a měření ukáže, že tohle místo brzdí.

## --checklist--

- Oddělené funkce se snáz testují a čtou.
- Cenou je několik průchodů místo jednoho, pořád ale lineárně.
- Sloučit do jednoho průchodu se vyplatí až podle měření na velkých datech.

// Jeden řádek = jedno zobrazení, seconds = čas strávený na stránce.
const views = [

  { visitorId: 'u7', path: '/kosik', seconds: 27 },
  { visitorId: 'u1', path: '/produkt/caj', seconds: 205 },
];

// Tržby za posledních sedm dní, v korunách.
const dailySales = [12400, 9800, 15300, 7200, 18900, 11100, 16600];

// Ceník, vzestupně seřazený. Dárkové poukázky se skládají ze dvou položek.
const sortedPrices = [49, 129, 249, 349, 499, 899, 1290];

// Kolik různých návštěvníků je v záznamech.
function uniqueVisitorCount(records) {
  const ids = new Set();
  for (const record of records) {
    ids.add(record.visitorId);
  }
  return ids.size;
}

// Součet všech vteřin strávených na webu.
function totalSeconds(records) {
  return records.reduce((total, record) => total + record.seconds, 0);
}

// Kolikrát se která stránka zobrazila. Klíč = cesta, hodnota = počet.
function viewsByPath(records) {
  const counts = new Map();
  for (const record of records) {
    counts.set(record.path, (counts.get(record.path) ?? 0) + 1);
  }
  return counts;
}

function topPath(counts) {
  let bestPath = null;
  let bestCount = 0;
  for (const [path, count] of counts) {
    if (count > bestCount) {
      bestPath = path;
      bestCount = count;
    }
  }
  return bestPath;
}

// Cesty, které se v záznamech objevily právě jednou.
function onlyOnce(counts) {
  const paths = [];
  for (const [path, count] of counts) {
    if (count === 1) {
      paths.push(path);
    }
  }
  return paths;
}

// Kolik stránek si který návštěvník prohlédl. Napsal kolega z marketingu.
function viewsByVisitor(records) {
  const counts = {};
  for (const record of records) {
    counts[record.visitorId] = (counts[record.visitorId] ?? 0) + 1;
  }
  return counts;
}

// Zobrazení od návštěvníků, kteří jsou ve věrnostním programu.
function vipViews(records, vipIds) {
  const vip = new Set(vipIds);
  return records.filter((record) => vip.has(record.visitorId));
}

function bestWindow(sales, size) {
  if (sales.length < size) return 0;
  let sum = 0;
  for (let i = 0; i < size; i++) sum += sales[i];
  let best = sum;
  for (let i = size; i < sales.length; i++) {
    sum = sum + sales[i] - sales[i - size];
    if (sum > best) best = sum;
  }
  return best;
}

// Index dne, kterým začíná nejlepší okno délky size. -1, když se okno nevejde.
function bestWindowStart(sales, size) {
  if (sales.length < size) return -1;
  let sum = 0;
  for (let i = 0; i < size; i++) sum += sales[i];
  let best = sum;
  let bestStart = 0;
  for (let i = size; i < sales.length; i++) {
    sum = sum + sales[i] - sales[i - size];
    if (sum > best) {
      best = sum;
      bestStart = i - size + 1;
    }
  }
  return bestStart;
}

// Existuje ve vzestupně seřazeném poli dvojice různých položek se součtem target?
function hasPairWithSum(sortedValues, target) {
  let left = 0;
  let right = sortedValues.length - 1;
  while (left < right) {
    const sum = sortedValues[left] + sortedValues[right];
    if (sum === target) return true;
    if (sum < target) left++;
    else right--;
  }
  return false;
}

// Záznam s nejdelším časem na stránce, u prázdných dat null.
function longestVisit(records) {
  let best = null;
  for (const record of records) {
    if (best === null || record.seconds > best.seconds) {
      best = record;
    }
  }
  return best;
}

// Různé cesty ze záznamů, seřazené podle české abecedy.
function uniquePathsSorted(records) {
  const paths = new Set();
  for (const record of records) {
    paths.add(record.path);
  }
  return [...paths].sort((a, b) => a.localeCompare(b, 'cs'));
}

// Dvojice hodnot se součtem target, od menší k větší. Vstup je vzestupně seřazený.
function pairForVoucher(sortedValues, target) {
  let left = 0;
  let right = sortedValues.length - 1;
  while (left < right) {
    const sum = sortedValues[left] + sortedValues[right];
    if (sum === target) return [sortedValues[left], sortedValues[right]];
    if (sum < target) left++;
    else right--;
  }
  return null;
}
