# --description--

Hláška v košíku má být konkrétní: ne „dvojice existuje", ale „za 598 Kč si můžeš vzít položku za 249 a za 349 Kč". Potřebuješ tedy samotné hodnoty, ne jen odpověď ano/ne.

**Úkol:** Napiš funkci `pairForVoucher(sortedValues, target)`, která vrátí pole dvou hodnot se součtem `target`, od menší k větší. Když taková dvojice neexistuje, vrátí `null`.

Vzor už znáš z kroku s poukázkou, teď ho napíšeš potřetí a bez opory. Rozdíl je jen v tom, co se vrací ve chvíli, kdy součet sedne.

# --hints--

`pairForVoucher` vrátí dvojici z prostředka ceníku.

```js
assert.deepEqual(pairForVoucher(sortedPrices, 598), [249, 349], 'pairForVoucher(sortedPrices, 598) má vrátit [249, 349]');
```

Menší hodnota je ve výsledku první.

```js
assert.deepEqual(pairForVoucher(sortedPrices, 1339), [49, 1290], 'pairForVoucher(sortedPrices, 1339) má vrátit [49, 1290] — nejdřív menší cena');
```

Když dvojice neexistuje, vrátí `pairForVoucher` hodnotu `null`.

```js
assert.equal(pairForVoucher(sortedPrices, 100), null, 'pairForVoucher(sortedPrices, 100) má vrátit null');
assert.equal(pairForVoucher([], 10), null, 'pairForVoucher([], 10) má vrátit null');
```

Funguje i na jiném seřazeném ceníku.

```js
assert.deepEqual(pairForVoucher([100, 200, 300, 400], 500), [100, 400], 'pairForVoucher([100, 200, 300, 400], 500) má vrátit [100, 400] — dvojici od krajů');
```

# --help--

## --tip--

Je to tentýž postup jako u kontroly, jestli dvojice existuje — viz [Nejdřív hrubá síla, pak zlepšení](see:js-algoritmy/postup-reseni#nejdriv-hruba-sila-pak-zlepseni).

## --tip--

Změna je jediná: ve chvíli shody nevracíš odpověď ano, ale obě hodnoty, na které ukazatele právě ukazují.

    ids.add(record.visitorId);

function totalSeconds(records) {

}

function viewsByPath(records) {
  const counts = new Map();

    counts.set(record.path, (counts.get(record.path) ?? 0) + 1);
  }
  return counts;
}

function topPath(counts) {
  let bestPath = null;
  let bestCount = 0;
  for (const [path, count] of counts) {
    if (count > bestCount) {
      bestPath = path;
      bestCount = count;
    }
  }
  return bestPath;
}

function onlyOnce(counts) {
  const paths = [];
  for (const [path, count] of counts) {
    if (count === 1) {
      paths.push(path);
    }
  }
  return paths;
}

// Kolik stránek si který návštěvník prohlédl. Napsal kolega z marketingu.
function viewsByVisitor(records) {
  const counts = {};
  for (const record of records) {
    counts[record.visitorId] = (counts[record.visitorId] ?? 0) + 1;
  }
  return counts;
}

// Zobrazení od návštěvníků, kteří jsou ve věrnostním programu.
function vipViews(records, vipIds) {
  const vip = new Set(vipIds);
  return records.filter((record) => vip.has(record.visitorId));
}

function bestWindow(sales, size) {
  if (sales.length < size) return 0;
  let sum = 0;
  for (let i = 0; i < size; i++) sum += sales[i];
  let best = sum;
  for (let i = size; i < sales.length; i++) {
    sum = sum + sales[i] - sales[i - size];
    if (sum > best) best = sum;
  }
  return best;
}

// Index dne, kterým začíná nejlepší okno délky size. -1, když se okno nevejde.
function bestWindowStart(sales, size) {
  if (sales.length < size) return -1;
  let sum = 0;
  for (let i = 0; i < size; i++) sum += sales[i];
  let best = sum;
  let bestStart = 0;
  for (let i = size; i < sales.length; i++) {
    sum = sum + sales[i] - sales[i - size];
    if (sum > best) {
      best = sum;
      bestStart = i - size + 1;
    }
  }
  return bestStart;
}

// Existuje ve vzestupně seřazeném poli dvojice různých položek se součtem target?
function hasPairWithSum(sortedValues, target) {
  let left = 0;
  let right = sortedValues.length - 1;
  while (left < right) {
    const sum = sortedValues[left] + sortedValues[right];
    if (sum === target) return true;
    if (sum < target) left++;
    else right--;
  }
  return false;
}

// Záznam s nejdelším časem na stránce, u prázdných dat null.
function longestVisit(records) {
  let best = null;
  for (const record of records) {
    if (best === null || record.seconds > best.seconds) {
      best = record;
    }
  }
  return best;
}

// Různé cesty ze záznamů, seřazené podle české abecedy.
function uniquePathsSorted(records) {
  const paths = new Set();
  for (const record of records) {
    paths.add(record.path);
  }
  return [...paths].sort((a, b) => a.localeCompare(b, 'cs'));
