# --description--

Zákazník neví, že na části lampy jde kliknout. Pomůže, když se část pod myší jemně zvýrazní — stejně jako tlačítko při najetí.

Jak zvýraznění uděláš, je na tobě. Materiál má několik vlastností, které vzhled mění, a část můžeš i jinak odlišit. Jen hlídej, aby se po odjetí myši vrátil přesně původní vzhled — i barva stínidla, kterou si zákazník mezitím vybral.

**Úkol:** Když myš jede nad canvasem a ukazuje na část lampy, tahle část se viditelně odliší. Když myš ukazuje jinam (na jinou část nebo do prázdna), předchozí část se vrátí do původního vzhledu.

# --hints--

Najetí myší na stínidlo změní jeho vzhled.

```js
const { camera, lamp } = await helpers.importFile('main.js');
const canvas = document.querySelector('.viewer__canvas');
const look = (mesh) => { let count = 0; mesh.traverse(() => count++); return JSON.stringify([mesh.material.uuid, mesh.material.color?.getHex(), mesh.material.emissive?.getHex(), mesh.material.emissiveIntensity, mesh.material.opacity, mesh.material.wireframe, mesh.scale.toArray(), mesh.visible, count]); };
const moveTo = (object, x, y, z) => {
  lamp.updateWorldMatrix(true, true);
  const point = object.localToWorld(object.position.clone().set(x, y, z)).project(camera);
  const rect = canvas.getBoundingClientRect();
  canvas.dispatchEvent(new PointerEvent('pointermove', { clientX: rect.left + ((point.x + 1) / 2) * rect.width, clientY: rect.top + ((1 - point.y) / 2) * rect.height, bubbles: true, pointerId: 1, isPrimary: true }));
};
const shade = lamp?.getObjectByName('shade');
assert.ok(shade, 'v lampě má být stínidlo');
const original = look(shade);
moveTo(shade, 0, 0.5, 0);
await helpers.waitFor(() => look(shade) !== original).catch(() => {});
assert.notEqual(look(shade), original, 'když myš ukazuje na stínidlo, má se jeho vzhled změnit');
```

Po odjetí myši do prázdna se stínidlo vrátí do původního vzhledu.

assert.ok(shade, 'v lampě má být stínidlo');
await helpers.click(document.querySelector('[data-color="#c2553d"]'));

const point = shade.localToWorld(shade.position.clone().set(0, 0.5, 0)).project(camera);

await helpers.waitFor(() => look(shade) !== original).catch(() => {});
assert.notEqual(look(shade), original, 'když myš ukazuje na stínidlo, má se jeho vzhled změnit');
move(rect.left + 10, rect.top + 10);

Přejezd ze stínidla na podstavu zvýrazní podstavu a stínidlo vrátí.

```js
const { camera, lamp } = await helpers.importFile('main.js');
const canvas = document.querySelector('.viewer__canvas');
const look = (mesh) => { let count = 0; mesh.traverse(() => count++); return JSON.stringify([mesh.material.uuid, mesh.material.color?.getHex(), mesh.material.emissive?.getHex(), mesh.material.emissiveIntensity, mesh.material.opacity, mesh.material.wireframe, mesh.scale.toArray(), mesh.visible, count]); };
const moveTo = (object, x, y, z) => {
  lamp.updateWorldMatrix(true, true);
  const point = object.localToWorld(object.position.clone().set(x, y, z)).project(camera);
  const rect = canvas.getBoundingClientRect();
  canvas.dispatchEvent(new PointerEvent('pointermove', { clientX: rect.left + ((point.x + 1) / 2) * rect.width, clientY: rect.top + ((1 - point.y) / 2) * rect.height, bubbles: true, pointerId: 1, isPrimary: true }));
};
const shade = lamp?.getObjectByName('shade');
const base = lamp?.getObjectByName('base');
assert.ok(shade && base, 'v lampě má být stínidlo i podstava');
const shadeOriginal = look(shade);
const baseOriginal = look(base);
moveTo(shade, 0, 0.5, 0);
await helpers.waitFor(() => look(shade) !== shadeOriginal).catch(() => {});
moveTo(base, 0, 0.1, 0.6);
await helpers.waitFor(() => look(base) !== baseOriginal && look(shade) === shadeOriginal).catch(() => {});
assert.notEqual(look(base), baseOriginal, 'když myš přejede na podstavu, má se zvýraznit podstava');
assert.equal(look(shade), shadeOriginal, 'když myš přejede ze stínidla na podstavu, má se stínidlo vrátit do původního vzhledu');
```

# --help--

## --tip--

Jak zjistit, na co myš ukazuje, už máš v obsluze kliknutí z minulého kroku. Vlastnosti materiálu jsou v části [Mesh: geometrie a materiál](see:web-3d-efekty/threejs-zaklady#mesh-geometrie-a-material) a v dokumentaci `MeshStandardMaterial`.

## --tip--

Posluchač `pointermove` na canvasu zjistí zasaženou část stejně jako kliknutí. Pamatuj si v proměnné mimo posluchač, která část je právě zvýrazněná: když se liší od nové, starou vrať a novou zvýrazni. Vlastnost, která na barvu nesahá, ušetří starosti s vybranou barvou.

## --tip--

Podcíle bez konkrétní vlastnosti:

```text
// 1. zjisti část pod myší (nebo nic)
// 2. když je to pořád stejná část, skonči
// 3. předchozí části vrať vzhled
// 4. novou část zvýrazni a zapamatuj si ji
```

```js
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';
import { RoomEnvironment } from 'three/addons/environments/RoomEnvironment.js';

const canvas = document.querySelector('.viewer__canvas');
const viewer = document.querySelector('.viewer');

export const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true, preserveDrawingBuffer: true });
export const scene = new THREE.Scene();
export const camera = new THREE.PerspectiveCamera(35, 1, 0.1, 100);
camera.position.set(4, 3, 7);
camera.lookAt(0, 0, 0);

// Prostředí, které odráží kov
scene.environment = new THREE.PMREMGenerator(renderer).fromScene(new RoomEnvironment()).texture;

// Světla
scene.add(new THREE.AmbientLight('#ffffff', 0.4));
const sun = new THREE.DirectionalLight('#ffffff', 1.5);
sun.position.set(3, 6, 4);
scene.add(sun);

// Lampa
export const lamp = new THREE.Group();
lamp.position.set(-0.4, -1.4, 0);
scene.add(lamp);

const base = new THREE.Mesh(
  new THREE.CylinderGeometry(0.9, 1, 0.2, 48),
  new THREE.MeshStandardMaterial({ color: '#3a3632', roughness: 0.35, metalness: 1 }),
);
base.name = 'base';
base.userData.label = 'Ocelová podstava';
base.position.y = 0.1;
lamp.add(base);

const armPivot = new THREE.Group();
armPivot.name = 'armPivot';
armPivot.position.y = 0.2;
armPivot.rotation.z = -0.35;
lamp.add(armPivot);

const arm = new THREE.Mesh(
  new THREE.CylinderGeometry(0.06, 0.06, 2.4, 16),
  new THREE.MeshStandardMaterial({ color: '#3a3632', roughness: 0.35, metalness: 1 }),
);
arm.name = 'arm';
arm.userData.label = 'Ohýbané rameno';
arm.position.y = 1.2;
armPivot.add(arm);

const shade = new THREE.Mesh(
  new THREE.SphereGeometry(0.7, 48, 24, 0, Math.PI * 2, 0, Math.PI / 2),
  new THREE.MeshStandardMaterial({ color: '#e0a526', roughness: 0.6, side: THREE.DoubleSide }),
);
shade.name = 'shade';
shade.userData.label = 'Hliníkové stínidlo';
shade.position.y = 2.4;
armPivot.add(shade);

const bulb = new THREE.PointLight('#ffd89b', 0, 6);
bulb.position.y = 2.25;
armPivot.add(bulb);

// Otáčení myší
export const controls = new OrbitControls(camera, canvas);
controls.enableDamping = true;
controls.enablePan = false;
controls.minDistance = 4;
controls.maxDistance = 12;
controls.maxPolarAngle = Math.PI / 2;

// Barva stínidla
const swatches = document.querySelectorAll('.swatch');
for (const swatch of swatches) {
  swatch.addEventListener('click', () => {
    shade.material.color.set(swatch.dataset.color);
    for (const other of swatches) {
      other.setAttribute('aria-pressed', String(other === swatch));
    }
  });
}

// Vypínač
const lightSwitch = document.querySelector('.switch');
lightSwitch.addEventListener('click', () => {
  const isOn = lightSwitch.getAttribute('aria-pressed') !== 'true';
  bulb.intensity = isOn ? 4 : 0;
  lightSwitch.setAttribute('aria-pressed', String(isOn));
});

const raycaster = new THREE.Raycaster();
const pointer = new THREE.Vector2();
const label = document.querySelector('.viewer__label');
canvas.addEventListener('click', (event) => {
  const rect = canvas.getBoundingClientRect();
  pointer.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
  pointer.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;
  raycaster.setFromCamera(pointer, camera);
  const [hit] = raycaster.intersectObjects(lamp.children, true);
  label.textContent = hit ? hit.object.userData.label : '';
});

// Velikost plátna podle prohlížeče (sjednotil Tomáš)
function fitToViewer() {
  const { width, height } = viewer.getBoundingClientRect();
  renderer.setSize(Math.round(width), Math.round(height), false);
  camera.aspect = width / height;
  camera.updateProjectionMatrix();
}
new ResizeObserver(fitToViewer).observe(viewer);

renderer.setAnimationLoop(() => {
  controls.update();
  renderer.render(scene, camera);
});
```
