# --description--

Lidem s poruchou rovnováhy nebo migrénou může otáčející se planeta a kamera, která plave za myší, způsobit nevolnost. V systému si proto zapínají omezený pohyb a stránka to pozná přes media dotaz `prefers-reduced-motion: reduce`.

S omezeným pohybem má zůstat scéna vidět, jen se nic nehýbe samo: planeta se neotáčí, kamera nejezdí za myší ani při scrollu. Uživatel může nastavení přepnout i během návštěvy, proto stav nečti jednou při načtení, ale v každém snímku z objektu, který vrací `matchMedia`.

**Úkol:** Když je zapnutý omezený pohyb, v render smyčce se planeta neotáčí a kamera se nepohne za myší ani při scrollu. Scéna se dál vykresluje. Bez omezeného pohybu funguje všechno jako dřív.

# --hints--

S omezeným pohybem se planeta neotáčí, i když se scéna dál kreslí.

```js
const { planet, renderer } = await helpers.importFile('scene.js');
assert.ok(planet, 'scene.js má exportovat planet');
const matches = Object.getOwnPropertyDescriptor(MediaQueryList.prototype, 'matches');
Object.defineProperty(MediaQueryList.prototype, 'matches', { configurable: true, get() { return /prefers-reduced-motion:\s*reduce/.test(this.media) || matches.get.call(this); } });
await helpers.wait(100);
const start = planet.rotation.y;
const frames = renderer.info.render.frame;
await helpers.wait(500);
Object.defineProperty(MediaQueryList.prototype, 'matches', matches);
assert.ok(renderer.info.render.frame > frames, 'scéna se má s omezeným pohybem dál vykreslovat');
assert.ok(Math.abs(planet.rotation.y - start) < 0.001, 's omezeným pohybem se planeta nemá otáčet — čti matches z matchMedia v každém snímku');
```

S omezeným pohybem kamera nejede za myší.

```js
const { camera } = await helpers.importFile('scene.js');
const matches = Object.getOwnPropertyDescriptor(MediaQueryList.prototype, 'matches');
Object.defineProperty(MediaQueryList.prototype, 'matches', { configurable: true, get() { return /prefers-reduced-motion:\s*reduce/.test(this.media) || matches.get.call(this); } });
await helpers.wait(100);
const start = camera.position.clone();
document.body.dispatchEvent(new PointerEvent('pointermove', { clientX: window.innerWidth - 1, clientY: 0, bubbles: true }));
window.scrollTo(0, document.querySelector('.hero').offsetHeight / 2);
await helpers.wait(500);
Object.defineProperty(MediaQueryList.prototype, 'matches', matches);
assert.ok(camera.position.distanceTo(start) < 0.001, `s omezeným pohybem se kamera nemá hýbat za myší ani při scrollu, posunula se o ${camera.position.distanceTo(start).toFixed(2)}`);
```

Bez omezeného pohybu se planeta otáčí a kamera jede za myší.

```js
const { planet, camera } = await helpers.importFile('scene.js');
assert.ok(planet, 'scene.js má exportovat planet');
const start = planet.rotation.y;
document.body.dispatchEvent(new PointerEvent('pointermove', { clientX: window.innerWidth - 1, clientY: window.innerHeight / 2, bubbles: true }));
await helpers.waitFor(() => planet.rotation.y - start > 0.02 && camera.position.x > 0.5, 3000).catch(() => {});
assert.ok(planet.rotation.y - start > 0.02, 'bez omezeného pohybu se má planeta otáčet');
assert.ok(camera.position.x > 0.5, 'bez omezeného pohybu má kamera jet za myší');
```

# --help--

## --tip--

Ukázka je v části [Bez WebGL a s omezeným pohybem](see:web-3d-efekty/interakce-a-castice#bez-webgl-a-s-omezenym-pohybem), proč na tom záleží, víš z [lekce o animacích](see:css-animace/transition-transform#omezeny-pohyb-prefers-reduced-motion).

## --tip--

Jednou nahoře si ulož výsledek `window.matchMedia('(prefers-reduced-motion: reduce)')`. Ve `tick` obal otáčení planety a všechny posuny kamery podmínkou na jeho `matches`. `timer.update`, `lookAt` a vykreslení nech mimo podmínku.

## --tip--

Stejná podmínka u automatického posouvání karuselu: v každém kroku `if (!reducedMotion.matches) showNextSlide()`, jinak karusel stojí a uživatel přepíná šipkami.

## --file-- scene.js

```js
import * as THREE from 'three';

const hero = document.querySelector('.hero');
const canvas = document.querySelector('.hero__canvas');

export const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true, preserveDrawingBuffer: true });
export const scene = new THREE.Scene();
export const camera = new THREE.PerspectiveCamera(50, 1, 0.1, 200);
camera.position.set(0, 0, 7);

// Slunce
scene.add(new THREE.AmbientLight('#ffffff', 0.15));
const sun = new THREE.DirectionalLight('#fff4e0', 3);
sun.position.set(-5, 3, 4);
scene.add(sun);

// Pruhy na povrchu planety nakreslené do 2D canvasu
function createBandsTexture() {
  const bands = document.createElement('canvas');
  bands.width = 512;
  bands.height = 256;
  const context = bands.getContext('2d');
  const colors = ['#e8d3a6', '#d9b77e', '#c49a5c', '#e3c796', '#b88a4f', '#d6b27a'];
  let y = 0;
  let index = 0;
  while (y < bands.height) {
    const bandHeight = 6 + ((index * 7) % 5) * 6;
    context.fillStyle = colors[index % colors.length];
    context.fillRect(0, y, bands.width, bandHeight);
    y += bandHeight;
    index += 1;
  }
  const texture = new THREE.CanvasTexture(bands);
  texture.colorSpace = THREE.SRGBColorSpace;
  return texture;
}

// Saturn: planeta a prstenec ve skloněné skupině
export const saturn = new THREE.Group();
saturn.position.set(2.4, 0.4, 0);
saturn.rotation.set(0.45, 0, 0.3);
scene.add(saturn);

export const planet = new THREE.Mesh(
  new THREE.SphereGeometry(1.1, 64, 32),
  new THREE.MeshStandardMaterial({ map: createBandsTexture(), roughness: 0.9 }),
);
planet.name = 'planet';
saturn.add(planet);

const ring = new THREE.Mesh(
  new THREE.RingGeometry(1.5, 2.5, 128),
  new THREE.MeshStandardMaterial({ color: '#cbb48a', roughness: 1, side: THREE.DoubleSide, transparent: true, opacity: 0.75 }),
);
ring.name = 'ring';
ring.rotation.x = -Math.PI / 2;
saturn.add(ring);
const starPositions = new Float32Array(4000 * 3);
for (let i = 0; i < 4000; i++) {
  const star = new THREE.Vector3().randomDirection();
  star.multiplyScalar(THREE.MathUtils.randFloat(20, 60));
  star.toArray(starPositions, i * 3);
}
const starGeometry = new THREE.BufferGeometry();
starGeometry.setAttribute('position', new THREE.BufferAttribute(starPositions, 3));
const stars = new THREE.Points(
  starGeometry,
  new THREE.PointsMaterial({ size: 0.25, color: '#dbeafe', transparent: true, blending: THREE.AdditiveBlending, depthWrite: false }),
);
scene.add(stars);

// Poloha myši od -1 do 1
export const pointer = { x: 0, y: 0 };
window.addEventListener('pointermove', (event) => {
  pointer.x = (event.clientX / window.innerWidth) * 2 - 1;
  pointer.y = -(event.clientY / window.innerHeight) * 2 + 1;
});

const timer = new THREE.Timer();
timer.connect(document);

function tick(time) {
  timer.update(time);
  const delta = timer.getDelta();

  planet.rotation.y += 0.15 * delta;
  camera.position.x += (pointer.x * 1.5 - camera.position.x) * 0.05;
  camera.position.y += (pointer.y - camera.position.y) * 0.05;
  camera.position.z = 7 + Math.min(window.scrollY / hero.offsetHeight, 1) * 5;
  camera.lookAt(0, 0, 0);
  renderer.render(scene, camera);
}

// Kreslit, jen když je hero sekce vidět
const visibility = new IntersectionObserver(([entry]) => {
  renderer.setAnimationLoop(entry.isIntersecting ? tick : null);
});
visibility.observe(hero);

// Úklid při odchodu ze stránky
window.addEventListener('pagehide', () => {
  visibility.disconnect();
  renderer.setAnimationLoop(null);
  scene.traverse((object) => {
    object.geometry?.dispose();
    object.material?.map?.dispose();
    object.material?.dispose();
  });
  renderer.dispose();
});

// Velikost plátna podle hero sekce
function resize() {
  const width = hero.clientWidth;
  const height = hero.clientHeight;
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.setSize(width, height, false);
  camera.aspect = width / height;
  camera.updateProjectionMatrix();
  starGeometry.setDrawRange(0, width < 600 ? 1500 : Infinity);
}
new ResizeObserver(resize).observe(hero);
```
