# --description--

## Hlášení

Kolega Marek upravil otáčení planety („s deltou mi to na notebooku občas cuklo, pevný přírůstek je plynulejší"). U něj na notebooku se 60Hz displejem se Saturn točí stejně jako dřív. Ředitel planetária má ale herní monitor se 144 Hz a píše: „Planeta se na úvodní stránce točí jako káča, dvakrát rychleji než na prezentaci."

## Úkol

Vrať otáčení planety na rychlost 0.15 radiánu za sekundu nezávislou na počtu snímků. Zbytek scény nech beze změny.

## Mezistav

Po opravě máš hotovou planetu s prstencem a hvězdné pole. Klidně si dej pauzu — další část přidá pohyb za myší a scrollem.

# --hints--

Planeta se otáčí rychlostí asi 0.15 radiánu za sekundu.

```js
const { planet } = await helpers.importFile('scene.js');
assert.ok(planet, 'scene.js má exportovat planet');
const start = planet.rotation.y;
const startTime = performance.now();
await helpers.wait(800);
const speed = (planet.rotation.y - start) / ((performance.now() - startTime) / 1000);
assert.ok(speed > 0.1 && speed < 0.2, `planeta se má otáčet rychlostí 0.15 rad/s, točí se ${speed.toFixed(3)} rad/s`);
```

Při polovičním počtu snímků za sekundu se planeta otáčí stejně rychle.

```js
const { planet } = await helpers.importFile('scene.js');
assert.ok(planet, 'scene.js má exportovat planet');
const measure = async () => {
  const start = planet.rotation.y;
  const startTime = performance.now();
  await helpers.wait(600);
  return (planet.rotation.y - start) / ((performance.now() - startTime) / 1000);
};
await helpers.wait(400);
const before = await measure();
const original = window.requestAnimationFrame;
window.requestAnimationFrame = (callback) => original.call(window, () => original.call(window, callback));
await helpers.wait(100);
const slow = await measure();
window.requestAnimationFrame = original;
await helpers.wait(100);
const normal = Math.max(before, await measure());
assert.ok(normal > 0, 'planeta se má otáčet');
assert.ok(slow / normal > 0.7, `při polovičním počtu snímků se planeta točí jen ${Math.round((slow / normal) * 100)} % rychlosti — na rychlejším monitoru se tedy točí rychleji`);
```

Hvězdy a prstenec zůstaly ve scéně.

```js
const { scene, saturn } = await helpers.importFile('scene.js');
let points = 0;
scene?.traverse((object) => { if (object.isPoints) points += 1; });
assert.equal(points, 1, 'hvězdné pole má ve scéně zůstat');
assert.ok(saturn?.getObjectByName('ring'), 'prstenec má ve skupině saturn zůstat');
```

# --help--

## --tip--

Příznak „na rychlejším monitoru rychleji" je past z části [Typické chyby a pasti](see:web-3d-efekty/threejs-zaklady#typicke-chyby-a-pasti).

## --tip--

Podívej se na řádek, který mění `planet.rotation.y`, a porovnej ho s tím, co ve funkci `tick` dává `timer`.

## --tip--

Stejná oprava u větrníku: místo `blades.rotation.z += 0.03` rychlost za sekundu krát čas snímku, `blades.rotation.z += 1.8 * delta`.

```js
import * as THREE from 'three';

const hero = document.querySelector('.hero');
const canvas = document.querySelector('.hero__canvas');

export const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true, preserveDrawingBuffer: true });
export const scene = new THREE.Scene();
export const camera = new THREE.PerspectiveCamera(50, 1, 0.1, 200);
camera.position.set(0, 0, 7);

// Slunce
scene.add(new THREE.AmbientLight('#ffffff', 0.15));
const sun = new THREE.DirectionalLight('#fff4e0', 3);
sun.position.set(-5, 3, 4);
scene.add(sun);

// Pruhy na povrchu planety nakreslené do 2D canvasu
function createBandsTexture() {
  const bands = document.createElement('canvas');
  bands.width = 512;
  bands.height = 256;
  const context = bands.getContext('2d');
  const colors = ['#e8d3a6', '#d9b77e', '#c49a5c', '#e3c796', '#b88a4f', '#d6b27a'];
  let y = 0;
  let index = 0;
  while (y < bands.height) {
    const bandHeight = 6 + ((index * 7) % 5) * 6;
    context.fillStyle = colors[index % colors.length];
    context.fillRect(0, y, bands.width, bandHeight);
    y += bandHeight;
    index += 1;
  }
  const texture = new THREE.CanvasTexture(bands);
  texture.colorSpace = THREE.SRGBColorSpace;
  return texture;
}

// Saturn: planeta a prstenec ve skloněné skupině
export const saturn = new THREE.Group();
saturn.position.set(2.4, 0.4, 0);
saturn.rotation.set(0.45, 0, 0.3);
scene.add(saturn);

export const planet = new THREE.Mesh(
  new THREE.SphereGeometry(1.1, 64, 32),
  new THREE.MeshStandardMaterial({ map: createBandsTexture(), roughness: 0.9 }),
);
planet.name = 'planet';
saturn.add(planet);

const ring = new THREE.Mesh(
  new THREE.RingGeometry(1.5, 2.5, 128),
  new THREE.MeshStandardMaterial({ color: '#cbb48a', roughness: 1, side: THREE.DoubleSide, transparent: true, opacity: 0.75 }),
);
ring.name = 'ring';
ring.rotation.x = -Math.PI / 2;
saturn.add(ring);
const starPositions = new Float32Array(4000 * 3);
for (let i = 0; i < 4000; i++) {
  const star = new THREE.Vector3().randomDirection();
  star.multiplyScalar(THREE.MathUtils.randFloat(20, 60));
  star.toArray(starPositions, i * 3);
}
const starGeometry = new THREE.BufferGeometry();
starGeometry.setAttribute('position', new THREE.BufferAttribute(starPositions, 3));
const stars = new THREE.Points(
  starGeometry,
  new THREE.PointsMaterial({ size: 0.25, color: '#dbeafe', transparent: true, blending: THREE.AdditiveBlending, depthWrite: false }),
);
scene.add(stars);

const timer = new THREE.Timer();

function tick(time) {
  timer.update(time);
  const delta = timer.getDelta();

  planet.rotation.y += 0.0025; // pevný přírůstek, plynulejší než s deltou
  renderer.render(scene, camera);
}
renderer.setAnimationLoop(tick);

// Velikost plátna podle hero sekce
function resize() {
  const width = hero.clientWidth;
  const height = hero.clientHeight;
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.setSize(width, height, false);
  camera.aspect = width / height;
  camera.updateProjectionMatrix();
}
new ResizeObserver(resize).observe(hero);
```
