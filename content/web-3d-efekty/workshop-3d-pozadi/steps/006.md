# --description--

Saturn má pruhy mraků. Obrázek k nim stahovat nemusíš — nakreslíš je do obyčejného 2D `<canvas>` v paměti a Three.js z něj udělá texturu. `THREE.CanvasTexture(canvas)` převezme cokoli, co do canvasu nakreslíš (pruhy, text, graf), a materiál ho přes vlastnost `map` obtočí kolem koule.

Jedna past: barvy z 2D canvasu jsou v barevném prostoru sRGB, jako všechno na webu. Three.js to u textury neví, dokud mu to neřekneš, a pruhy pak vyjdou vybledlé. Proto `texture.colorSpace = THREE.SRGBColorSpace`.

Na jiných datech, textura s nápisem pro 3D ceduli:

```js
function createSignTexture(text) {
  const sign = document.createElement('canvas');
  sign.width = 512;
  sign.height = 128;
  const context = sign.getContext('2d');
  context.fillStyle = '#14532d';
  context.fillRect(0, 0, sign.width, sign.height);
  context.fillStyle = '#ffffff';
  context.font = '64px system-ui';
  context.fillText(text, 24, 88);

  const texture = new THREE.CanvasTexture(sign);
  texture.colorSpace = THREE.SRGBColorSpace;
  return texture;
}
```

**Úkol:** Napiš funkci `createBandsTexture()`, která do canvasu 512 × 256 nakreslí vodorovné pruhy v odstínech písku a hnědé (barvy i výšky pruhů si zvol, stačí 5–6 barev) a vrátí z něj `CanvasTexture` v prostoru sRGB. Materiál planety místo barvy dostane `map` z této funkce (`roughness` 0.9 zůstává).

# --hints--

Materiál planety má texturu z canvasu (`CanvasTexture`) v barevném prostoru sRGB.

```js
const THREE = await import('three');
const { planet } = await helpers.importFile('scene.js');
const map = planet?.material?.map;
assert.ok(map?.isCanvasTexture, 'materiál planety má mít map = new THREE.CanvasTexture(…)');
assert.equal(map.colorSpace, THREE.SRGBColorSpace, 'textura má mít colorSpace = THREE.SRGBColorSpace, jinak budou pruhy vybledlé');
```

Textura je 512 × 256 bodů a má víc barevných pruhů nad sebou.

```js
const { planet } = await helpers.importFile('scene.js');
const image = planet?.material?.map?.image;
assert.ok(image, 'materiál planety má mít texturu z canvasu');
assert.deepEqual([image.width, image.height], [512, 256], 'canvas textury má mít rozměry 512 × 256');
const context = image.getContext('2d');
const colors = new Set();
for (let y = 2; y < image.height; y += 4) {
  colors.add(context.getImageData(256, y, 1, 1).data.slice(0, 3).join(','));
}
assert.ok(colors.size >= 3, `v textuře mají být aspoň 3 barevné pruhy nad sebou, jsou ${colors.size}`);
```

);

planet.position.set(2.4, 0.4, 0);
scene.add(planet);

function tick() {
  renderer.render(scene, camera);
}
renderer.setAnimationLoop(tick);

// Velikost plátna podle hero sekce
function resize() {
  const width = hero.clientWidth;
  const height = hero.clientHeight;
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.setSize(width, height, false);
  camera.aspect = width / height;
  camera.updateProjectionMatrix();
}
new ResizeObserver(resize).observe(hero);

## --file-- scene.js

```js
import * as THREE from 'three';

const hero = document.querySelector('.hero');
const canvas = document.querySelector('.hero__canvas');

export const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true, preserveDrawingBuffer: true });
export const scene = new THREE.Scene();
export const camera = new THREE.PerspectiveCamera(50, 1, 0.1, 200);
camera.position.set(0, 0, 7);

// Slunce
scene.add(new THREE.AmbientLight('#ffffff', 0.15));
const sun = new THREE.DirectionalLight('#fff4e0', 3);
sun.position.set(-5, 3, 4);
scene.add(sun);

// Pruhy na povrchu planety nakreslené do 2D canvasu
function createBandsTexture() {
  const bands = document.createElement('canvas');
  bands.width = 512;
  bands.height = 256;
  const context = bands.getContext('2d');
  const colors = ['#e8d3a6', '#d9b77e', '#c49a5c', '#e3c796', '#b88a4f', '#d6b27a'];
  let y = 0;
  let index = 0;
  while (y < bands.height) {
    const bandHeight = 6 + ((index * 7) % 5) * 6;
    context.fillStyle = colors[index % colors.length];
    context.fillRect(0, y, bands.width, bandHeight);
    y += bandHeight;
    index += 1;
  }
  const texture = new THREE.CanvasTexture(bands);
  texture.colorSpace = THREE.SRGBColorSpace;
  return texture;
}

export const planet = new THREE.Mesh(
  new THREE.SphereGeometry(1.1, 64, 32),
  new THREE.MeshStandardMaterial({ map: createBandsTexture(), roughness: 0.9 }),
);
planet.name = 'planet';
planet.position.set(2.4, 0.4, 0);
scene.add(planet);

function tick() {
  renderer.render(scene, camera);
}
renderer.setAnimationLoop(tick);

// Velikost plátna podle hero sekce
function resize() {
  const width = hero.clientWidth;
  const height = hero.clientHeight;
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.setSize(width, height, false);
  camera.aspect = width / height;
  camera.updateProjectionMatrix();
}
new ResizeObserver(resize).observe(hero);
```
