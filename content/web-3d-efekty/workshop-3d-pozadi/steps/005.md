# --description--

Teď první viditelná věc: Saturn. Zatím hladká koule v barvě písku, texturu přidáš v dalším kroku.

Ve vesmíru svítí jen Slunce, jedno silné světlo z jednoho směru. Odvrácená strana planety má být skoro tmavá, jen s nádechem, aby nesplynula s černou oblohou. Proto velmi slabé světlo odevšad.

Na jiných datech, Měsíc nad krajinou:

```js
scene.add(new THREE.AmbientLight('#ffffff', 0.05));
const moonlight = new THREE.DirectionalLight('#c7d2fe', 1.2);
moonlight.position.set(4, 6, -2);
scene.add(moonlight);
```

**Úkol:** Za kameru přidej:

1. `AmbientLight` bílé barvy s intenzitou 0.15,
2. `DirectionalLight` barvy `#fff4e0` s intenzitou 3 svítící z `(-5, 3, 4)` — zleva shora, ze strany textu,
3. vyexportovanou `planet` — mesh s geometrií `SphereGeometry(1.1, 64, 32)`, materiálem `MeshStandardMaterial` barvy `#d9b77e` s `roughness` 0.9, `name` `'planet'`, na pozici `(2.4, 0.4, 0)` vpravo od textu, přidaný do scény.

# --hints--

`scene.js` exportuje `planet`: kouli o poloměru 1.1 s `MeshStandardMaterial` na `(2.4, 0.4, 0)` ve scéně.

```js
const { scene, planet } = await helpers.importFile('scene.js');
assert.ok(planet?.isMesh, 'scene.js má exportovat planet (mesh)');
assert.equal(planet.name, 'planet', 'planeta má mít name = "planet"');
assert.equal(planet.geometry.type, 'SphereGeometry', 'planeta má mít geometrii SphereGeometry');
assert.equal(planet.geometry.parameters.radius, 1.1, 'planeta má mít poloměr 1.1');
assert.ok(planet.material?.isMeshStandardMaterial, 'planeta má mít MeshStandardMaterial');
let root = planet;
while (root.parent) root = root.parent;
assert.equal(root, scene, 'planeta má být přidaná do scény');
assert.deepEqual(planet.getWorldPosition(planet.position.clone()).toArray().map((n) => Math.round(n * 100) / 100), [2.4, 0.4, 0], 'planeta má stát na (2.4, 0.4, 0)');
```

Ve scéně svítí slabé `AmbientLight` (nejvýš 0.3) a silné `DirectionalLight` z `(-5, 3, 4)`.

```js
const { scene } = await helpers.importFile('scene.js');
const ambient = [];
const suns = [];
scene?.traverse((object) => {
  if (object.isAmbientLight) ambient.push(object);
  if (object.isDirectionalLight) suns.push(object);
});
assert.ok(ambient.some((light) => light.intensity > 0 && light.intensity <= 0.3), 've scéně má být AmbientLight se slabou intenzitou (0.15)');
assert.ok(suns.some((sun) => sun.intensity >= 1 && sun.position.equals(sun.position.clone().set(-5, 3, 4))), 've scéně má být DirectionalLight s intenzitou 3 na pozici (-5, 3, 4)');
```

```js
import * as THREE from 'three';

const hero = document.querySelector('.hero');
const canvas = document.querySelector('.hero__canvas');

export const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true, preserveDrawingBuffer: true });
export const scene = new THREE.Scene();
export const camera = new THREE.PerspectiveCamera(50, 1, 0.1, 200);
camera.position.set(0, 0, 7);

function tick() {
  renderer.render(scene, camera);
}
renderer.setAnimationLoop(tick);

// Velikost plátna podle hero sekce
function resize() {
  const width = hero.clientWidth;
  const height = hero.clientHeight;
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.setSize(width, height, false);
  camera.aspect = width / height;
  camera.updateProjectionMatrix();
}
new ResizeObserver(resize).observe(hero);
```

## --file-- scene.js

```js
import * as THREE from 'three';

const hero = document.querySelector('.hero');
const canvas = document.querySelector('.hero__canvas');

export const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true, preserveDrawingBuffer: true });
export const scene = new THREE.Scene();
export const camera = new THREE.PerspectiveCamera(50, 1, 0.1, 200);
camera.position.set(0, 0, 7);

// Slunce
scene.add(new THREE.AmbientLight('#ffffff', 0.15));
const sun = new THREE.DirectionalLight('#fff4e0', 3);
sun.position.set(-5, 3, 4);
scene.add(sun);

export const planet = new THREE.Mesh(
  new THREE.SphereGeometry(1.1, 64, 32),
  new THREE.MeshStandardMaterial({ color: '#d9b77e', roughness: 0.9 }),
);
planet.name = 'planet';
planet.position.set(2.4, 0.4, 0);
scene.add(planet);

function tick() {
  renderer.render(scene, camera);
}
renderer.setAnimationLoop(tick);

// Velikost plátna podle hero sekce
function resize() {
  const width = hero.clientWidth;
  const height = hero.clientHeight;
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.setSize(width, height, false);
  camera.aspect = width / height;
  camera.updateProjectionMatrix();
}
new ResizeObserver(resize).observe(hero);
```
