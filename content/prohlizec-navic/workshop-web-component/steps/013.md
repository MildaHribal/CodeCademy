# --description--

Komponenty hlásí změnu, ale nikdo je neposlouchá. Teď to napravíš na straně stránky: jediný posluchač na seznamu `#filmy` zachytí hlášení od všech čtyř karet a do hlavičky napíše průměr.

Seřaď řádky tak, aby po každé změně hodnocení vznikl v `#prumer` text ve tvaru `Průměr tvých hodnocení: 4.0 z 5`. Posluchač si hodnoty přečte rovnou z komponent — každá je nabízí jako vlastnost.

Odsazení nastavíš klávesou Tab, do mezery doplň chybějící kus. Dva řádky jsou navíc a do řešení nepatří.

# --hints--

Po změně hodnocení se v hlavičce objeví průměr.

```js
const prvek = document.querySelector('#filmy .film[data-id="bratri"] hodnoceni-hvezdy');
await helpers.click(prvek.shadowRoot.querySelectorAll('button')[3]);
await helpers.flush();
const text = document.querySelector('#prumer').textContent;
assert.match(text, /Průměr tvých hodnocení/, 'v #prumer má být text „Průměr tvých hodnocení: … z 5"');
assert.match(text, /4\.0/, 'z hodnot 5, 4, 3 a 4 vyjde průměr 4.0 (na jedno desetinné místo)');
```

Každá další změna průměr přepočítá.

```js
const vlny = document.querySelector('#filmy .film[data-id="vlny"] hodnoceni-hvezdy');
await helpers.click(vlny.shadowRoot.querySelectorAll('button')[0]);
await helpers.flush();
assert.match(document.querySelector('#prumer').textContent, /2\.5/, 'po snížení Vln na 1 vyjde z hodnot 1, 4, 3 a 2 průměr 2.5');
const banik = document.querySelector('#filmy .film[data-id="banik"] hodnoceni-hvezdy');
await helpers.click(banik.shadowRoot.querySelectorAll('button')[4]);
await helpers.flush();
assert.match(document.querySelector('#prumer').textContent, /2\.8/, 'z hodnot 1, 5, 3 a 2 vyjde 2.8');
```

Dokud uživatel nic nezmění, zůstane v hlavičce původní text.

```js
assert.match(document.querySelector('#prumer').textContent, /Zatím/, 'před první změnou se nic přepočítávat nemá');
```

// Filmy, které redakce tenhle měsíc sleduje. Pole je jen ke čtení.
const filmy = [
  { id: 'vlny', nazev: 'Vlny', rok: 2024, rezie: 'Jiří Mádl', hodnoceni: 5, hlasy: 412 },
  { id: 'banik', nazev: 'Baník', rok: 2024, rezie: 'Ondřej Provazník', hodnoceni: 4, hlasy: 187 },
  { id: 'zivot-k-sezrani', nazev: 'Život k sežrání', rok: 2024, rezie: 'Kristina Dufková', hodnoceni: 3, hlasy: 96 },
  { id: 'bratri', nazev: 'Bratři', rok: 2023, rezie: 'Tomáš Mašín', hodnoceni: 2, hlasy: 254 },
];

// Prvky stránky, se kterými budeš pracovat.
const seznam = document.querySelector('#filmy');
const prumer = document.querySelector('#prumer');
const formular = document.querySelector('#hodnoceni');
const vystup = document.querySelector('#vystup');

// Formulář nic neodesílá, jen vypíše, co by na server poslal.
formular.addEventListener('submit', (udalost) => {
  udalost.preventDefault();
  const radky = [...new FormData(formular)].map(([klic, hodnota]) => klic + ': ' + hodnota);
  vystup.textContent = radky.join('\n');
});

// Hvězdičkové hodnocení jako vlastní HTML značka.
class HodnoceniHvezdy extends HTMLElement {
  static observedAttributes = ['hodnota'];

  constructor() {
    super();
    this.attachShadow({ mode: 'open' });
    // Posluchač patří dovnitř stínu: tady se cíl události ještě nepřepisuje.
    this.shadowRoot.addEventListener('click', (udalost) => this.naKlik(udalost));
  }

  get hodnota() {
    return Number(this.getAttribute('hodnota')) || 0;
  }

  set hodnota(nova) {
    this.setAttribute('hodnota', Math.min(5, Math.max(0, Number(nova) || 0)));
  }

  vykresli() {
    // 1. Zjisti, kolik hvězd má svítit
    const hodnota = this.hodnota;
    // 2. Slož kostru stínu i s jeho vlastními styly
    this.shadowRoot.innerHTML = `
      <style>
        :host {
          display: inline-flex;
          align-items: center;
          gap: 0.4rem;
          line-height: 1;
        }
        .hvezdy { display: flex; gap: 0.1rem; }
        button {
          padding: 0 0.05rem;
          border: 0;
          border-radius: 0.25rem;
          background: none;
          color: #64748b;
          font: inherit;
          font-size: 1.35rem;
          line-height: 1;
          cursor: pointer;
          transition: scale 0.15s ease, color 0.15s ease;
        }
        button.plna { color: #f59e0b; }
        button:hover { scale: 1.2; }
        button:focus-visible { outline: 3px solid #a5b4fc; outline-offset: 2px; }
        .popis { color: #94a3b8; font-size: 0.78rem; }
      </style>
      <div class="hvezdy"></div>
      <span class="popis"><slot>Tvoje hodnocení</slot></span>
    `;
    // 3. Doplň pět hvězd, plné podle hodnoty
    const hvezdy = this.shadowRoot.querySelector('.hvezdy');
    for (const cislo of [1, 2, 3, 4, 5]) {
      const tlacitko = document.createElement('button');
      tlacitko.type = 'button';
      tlacitko.dataset.hodnota = cislo;
      tlacitko.textContent = cislo <= hodnota ? '★' : '☆';
      tlacitko.classList.toggle('plna', cislo <= hodnota);
      hvezdy.append(tlacitko);
    }
  }

  connectedCallback() {
    this.vykresli();
  }

  attributeChangedCallback(jmeno, stara, nova) {
    if (stara !== nova) this.vykresli();
  }

  naKlik(udalost) {
    const tlacitko = udalost.target.closest('button');
    if (!tlacitko) return;
    this.hodnota = tlacitko.dataset.hodnota;
    this.ohlasZmenu();
  }

  ohlasZmenu() {
    this.dispatchEvent(new CustomEvent('zmena', {
      detail: { film: this.getAttribute('name'), hodnota: this.hodnota },
      bubbles: true,
      composed: true,
    }));
  }
}

customElements.define('hodnoceni-hvezdy', HodnoceniHvezdy);

// Karta jednoho filmu ze šablony #sablona-filmu.
function vyrobKartu(film) {
  const karta = document.querySelector('#sablona-filmu').content.firstElementChild.cloneNode(true);
  karta.dataset.id = film.id;
  karta.querySelector('.film__nazev').textContent = film.nazev;
  karta.querySelector('.film__rok').textContent = film.rok;
  karta.querySelector('.film__rezie').textContent = 'Režie: ' + film.rezie;
  const hvezdy = karta.querySelector('hodnoceni-hvezdy');
  hvezdy.setAttribute('hodnota', film.hodnoceni);
  hvezdy.setAttribute('name', film.id);
  hvezdy.textContent = film.hlasy + ' hlasů';
  return karta;
}
