Testy máš, teď funkce. V `src/promo/promo-codes.ts` je připravený typ `PromoCode`
a import `isOnOrBefore` ze `src/lib/dates.ts`. Ten pomocník v projektu už je
a porovnává přesně to, co potřebuješ: den „stejný nebo dřív". Napsat si vlastní
porovnání dat by prošlo testy, ale v review by ses dozvěděl, že projekt na to
funkci má.

Seřaď řádky `findPromoCode` tak, aby prošly tvoje testy z minulého kroku.
Postup: znormalizuj vstup, najdi kód, neznámý odmítni, prošlý odmítni, jinak ho
vrať. Odsazení nastavíš klávesou Tab. Dva řádky jsou navíc.

```ts
export function findPromoCode(codes: PromoCode[], input: string, day: string): PromoCode | null {
  const normalized = input.trim().toUpperCase();
  const promo = codes.find((item) => item.code === normalized);
  if (!promo) return null;
  if (promo.validUntil !== null && !isOnOrBefore(day, promo.validUntil)) return null;
  return promo;
}
```

```md
# #42 Slevové kódy při rezervaci

**Nahlásila:** Petra Dvořáková (marketing)
**Štítky:** feature, backend, frontend

Na jarní akci rozdáváme letáky se slevovými kódy. Zákazník zadá kód ve formuláři

## Co má platit

- Kódy jsou v `data/promo-codes.json` (dodal marketing). Kód má procento slevy
  a nepovinné `validUntil` — poslední den platnosti, včetně.
- Na velikosti písmen a mezerách okolo nezáleží: ` jaro20 ` je totéž co `JARO20`.

  neplatí.“ (kód chyby `INVALID_PROMO_CODE`).
- Rezervace si pamatuje kód, jen když se sleva z něj opravdu použila (kvůli
  vyúčtování akce).
- Ve formuláři už pole pro kód je (připravil ho grafik), jen se zatím nikam neposílá.
```

```ts
import type { IncomingMessage, ServerResponse } from 'node:http';
import { HttpError } from './http/http-error.ts';
import { sendJson } from './http/json.ts';
import { routes } from './routes.ts';

export async function handleRequest(req: IncomingMessage, res: ServerResponse): Promise<void> {
  const url = new URL(req.url ?? '/', 'http://localhost');

  try {
    for (const route of routes) {
      const match = route.path.exec(url.pathname);
      if (match && route.method === req.method) {
        const result = await route.handle({ req, params: match.groups ?? {} });
        sendJson(res, result.status, result.body);
        return;
      }
    }
    throw new HttpError(404, 'NOT_FOUND', 'Tahle adresa neexistuje.');
  } catch (error) {
    if (error instanceof HttpError) {
      sendJson(res, error.status, { error: { code: error.code, message: error.message } });
      return;
    }
    console.error(error);
    sendJson(res, 500, { error: { code: 'INTERNAL_ERROR', message: 'Na serveru se něco pokazilo.' } });
  }
}
```

```ts
import type { IncomingMessage, ServerResponse } from 'node:http';
import { HttpError } from './http-error.ts';

export function sendJson(res: ServerResponse, status: number, body: unknown): void {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(body));
}

export async function readJsonBody(req: IncomingMessage): Promise<unknown> {
  let text = '';
  for await (const chunk of req) text += chunk;
  if (text === '') return {};

  try {
    return JSON.parse(text);
  } catch {
    throw new HttpError(400, 'INVALID_JSON', 'Tělo požadavku není platný JSON.');
  }
}
```

```ts
import type { RouteContext, RouteResult } from '../routes.ts';
import type { Price } from '../pricing/pricing.ts';
import type { BookingInput } from './booking-input.ts';
import { HttpError } from '../http/http-error.ts';
import { readJsonBody } from '../http/json.ts';
import { loadCourts } from '../courts/court-repository.ts';
import { loadMembers } from '../members/member-repository.ts';
import { isMember } from '../members/members.ts';
import { calculatePrice } from '../pricing/pricing.ts';
import { parseBookingInput } from './booking-input.ts';
import { findBooking, saveBooking } from './booking-store.ts';

async function priceBooking(input: BookingInput): Promise<Price> {
  const courts = await loadCourts();
  const court = courts.find((item) => item.id === input.courtId);
  if (!court) throw new HttpError(404, 'COURT_NOT_FOUND', 'Takový kurt neexistuje.');

  const members = await loadMembers();
  return calculatePrice(court, input.startHour, input.hours, isMember(members, input.email));
}

export async function quoteBookingRoute({ req }: RouteContext): Promise<RouteResult> {
  const input = parseBookingInput(await readJsonBody(req));
  return { status: 200, body: { price: await priceBooking(input) } };
}

export async function createBookingRoute({ req }: RouteContext): Promise<RouteResult> {
  const input = parseBookingInput(await readJsonBody(req));
  const booking = saveBooking(input, await priceBooking(input));
  return { status: 201, body: booking };
}

export async function getBookingRoute({ params }: RouteContext): Promise<RouteResult> {
  const booking = findBooking(params.id);
  if (!booking) throw new HttpError(404, 'BOOKING_NOT_FOUND', 'Rezervace neexistuje.');
  return { status: 200, body: booking };
}
```

```ts
import { HttpError } from '../http/http-error.ts';
import { isValidDay } from '../lib/dates.ts';

export type BookingInput = {
  courtId: string;
  email: string;
  date: string;
  startHour: number;
  hours: number;
};

const OPENING_HOUR = 7;
const CLOSING_HOUR = 22;

function invalid(message: string): never {
  throw new HttpError(400, 'INVALID_INPUT', message);
}

export function parseBookingInput(body: unknown): BookingInput {
  if (typeof body !== 'object' || body === null) invalid('Pošli rezervaci jako JSON objekt.');
  const data = body as Record<string, unknown>;

  if (typeof data.courtId !== 'string' || data.courtId === '') invalid('Chybí kurt (courtId).');
  if (typeof data.email !== 'string' || !data.email.includes('@')) invalid('E-mail není platný.');
  if (typeof data.date !== 'string' || !isValidDay(data.date)) invalid('Datum musí být ve tvaru RRRR-MM-DD.');

  const { startHour, hours } = data;
  if (typeof startHour !== 'number' || !Number.isInteger(startHour)) invalid('Začátek musí být celá hodina.');
  if (typeof hours !== 'number' || !Number.isInteger(hours) || hours < 1 || hours > 3) {
    invalid('Rezervovat jde 1 až 3 hodiny.');
  }
  if (startHour < OPENING_HOUR || startHour + hours > CLOSING_HOUR) {
    invalid('Areál je otevřený od 7 do 22 hodin.');
  }

  return {
    courtId: data.courtId,
    email: data.email.trim().toLowerCase(),
    date: data.date,
    startHour,
    hours,
  };
}
```

## --file-- src/pricing/pricing.ts

```ts
import type { Court } from '../courts/court-repository.ts';
import { percentOf } from '../lib/money.ts';

export type DiscountReason = 'member' | null;

export type Price = {
  base: number;
  discount: number;
  total: number;
  discountReason: DiscountReason;
};

export const EVENING_FROM_HOUR = 17;
export const MEMBER_DISCOUNT_PERCENT = 10;

export function hourRate(court: Court, hour: number): number {
  return hour > EVENING_FROM_HOUR ? court.eveningRate : court.hourlyRate;
}

export function calculatePrice(court: Court, startHour: number, hours: number, member: boolean): Price {
  let base = 0;
  for (let hour = startHour; hour < startHour + hours; hour++) {
    base += hourRate(court, hour);
  }

  const discount = member ? percentOf(base, MEMBER_DISCOUNT_PERCENT) : 0;
  return { base, discount, total: base - discount, discountReason: member ? 'member' : null };
}
```

```ts
const DAY_FORMAT = /^\d{4}-\d{2}-\d{2}$/;

/** Je `value` skutečný den ve tvaru RRRR-MM-DD? (`2026-02-30` není.) */
export function isValidDay(value: string): boolean {
  if (!DAY_FORMAT.test(value)) return false;
  const date = new Date(`${value}T00:00:00Z`);
  return !Number.isNaN(date.getTime()) && date.toISOString().startsWith(value);
}

/** Je `day` týž den jako `lastDay`, nebo dřív? Oba ve tvaru RRRR-MM-DD. */
export function isOnOrBefore(day: string, lastDay: string): boolean {
  return day <= lastDay;
}
```

## --file-- frontend/main.js

```js
import { postJson } from './api.js';
import { buildBookingRequest } from './booking-request.js';
import { formatCzk, priceLines } from './price-summary.js';

const form = document.querySelector('#booking-form');
const summary = document.querySelector('#price-summary');
const message = document.querySelector('#form-message');

function renderPrice(price) {
  const rows = priceLines(price).map(({ label, amount }) => {
    const row = document.createElement('div');
    const term = document.createElement('dt');
    const value = document.createElement('dd');
    term.textContent = label;
    value.textContent = formatCzk(amount);
    row.append(term, value);
    return row;
  });
  summary.replaceChildren(...rows);
}

async function send(path) {
  message.textContent = '';
  const values = Object.fromEntries(new FormData(form));
  try {
    return await postJson(path, buildBookingRequest(values));
  } catch (error) {
    message.textContent = error.message;
    return null;
  }
}

form.addEventListener('change', async () => {
  const data = await send('/api/bookings/quote');
  if (data) renderPrice(data.price);
});

form.addEventListener('submit', async (event) => {
  event.preventDefault();
  const booking = await send('/api/bookings');
  if (booking) message.textContent = `Rezervace potvrzená, číslo ${booking.id.slice(0, 8)}.`;
});
```

```js
export async function postJson(path, body) {
  const response = await fetch(path, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  });
  const data = await response.json();
  if (!response.ok) throw new Error(data.error?.message ?? 'Požadavek se nepovedl.');
  return data;
}
```

## --file-- frontend/price-summary.js

```js
const czk = new Intl.NumberFormat('cs-CZ', { style: 'currency', currency: 'CZK' });

/** Haléře → „1 234,50 Kč“. */
export function formatCzk(halere) {
  return czk.format(halere / 100);
}

/** Řádky rozpisu ceny pod formulářem. */
export function priceLines(price) {
  const lines = [{ label: 'Základní cena', amount: price.base }];
  if (price.discountReason === 'member') {
    lines.push({ label: 'Členská sleva', amount: -price.discount });
  }
  lines.push({ label: 'Celkem', amount: price.total });
  return lines;
}
```
