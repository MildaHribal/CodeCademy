# --description--

Každý štítek dovednosti má křížek. Kliknutí dovednost z profilu odebere.

**Úkol:** Napiš funkci `removeSkill(profile, skill)`, která vrátí nový profil s novým polem `skills` bez zadané dovednosti; ostatní dovednosti zůstanou v původním pořadí. Když profil dovednost nemá, vrátí **tentýž** profil. Původní pole se nesmí změnit.

Metody pole na vybírání položek přijdou až v sekci Pole v JavaScriptu. Tady ti stačí cyklus a nové pole.

# --hints--

`removeSkill` vrátí profil bez zadané dovednosti.

```js
assert.deepEqual(removeSkill(savedProfile, 'CSS').skills, ['HTML', 'JavaScript'], "removeSkill(savedProfile, 'CSS').skills má být ['HTML', 'JavaScript']");
```

Původní pole dovedností se nezmění.

```js
const result = removeSkill(savedProfile, 'HTML');
assert.deepEqual(savedProfile.skills, ['HTML', 'CSS', 'JavaScript'], "removeSkill nesmí odebrat 'HTML' z pole uloženého profilu");
assert.notEqual(result.skills, savedProfile.skills, 'výsledek removeSkill má mít nové pole skills');
```

Dovednost, kterou profil nemá, nic nezmění a vrátí se tentýž profil.

```js
assert.equal(removeSkill(savedProfile, 'PHP'), savedProfile, "removeSkill(savedProfile, 'PHP') má vrátit tentýž profil — PHP v něm není");
```

# --help--

## --tip--

Pole se mění stejně jako objekt, novým polem — viz [Úprava bez mutace: nový objekt](see:js-objekty/reference-a-mutace#uprava-bez-mutace-novy-objekt).

## --tip--

Guard clause jako u `addSkill`. Pak si založ prázdné pole, v cyklu `for…of` do něj dávej dovednosti kromě té odebírané a vrať nový profil s tím polem. Do nového pole zapisovat smíš, patří jen tobě.

## --tip--

Podcíle na jiných datech — odebrání štítku z příspěvku:

```js
// 1. štítek v příspěvku není → vrať příspěvek
// 2. nové prázdné pole, do něj všechny štítky kromě odebíraného
// 3. vrať nový příspěvek s novým polem
```

// Uložený profil vývojářky na pracovním portálu, jak ho vrací API.
const savedProfile = {
  id: 481,
  username: 'jana-dev',
  fullName: 'Jana Dvořáková',
  headline: 'Frontend vývojářka',
  email: 'jana.dvorakova@example.cz',
  location: { city: 'Brno', remote: true },
  links: { github: 'github.com/jana-dev', linkedin: null, website: 'jana-dev.example.cz' },
  privacy: { showEmail: false, showCity: true },
  skills: ['HTML', 'CSS', 'JavaScript'],
};

// Od kolegy: profil s novým nastavením soukromí, třeba updatePrivacy(savedProfile, { showEmail: true }).
function updatePrivacy(profile, changes) {
  const privacy = { ...profile.privacy };
  for (const [key, value] of Object.entries(changes)) {
    privacy[key] = value;
  }
  return { ...profile, privacy };
}

// Od kolegy: profil pro náhled firmám, bez odkazů, které nejsou vyplněné (delete smaže vlastnost).
function withoutEmptyLinks(profile) {
  const copy = { ...profile, links: { ...profile.links } };
  for (const [network, url] of Object.entries(copy.links)) {
    if (url === null) {
      delete copy.links[network];
    }
  }
  return copy;
}

// Jméno na profilu, bez celého jména přezdívka se zavináčem.
function displayName(profile) {
  return profile.fullName ?? `@${profile.username}`;
}

// Profil z tvaru, který posílá starší API (snake_case), do tvaru aplikace.
function fromApi(apiUser) {
  const { user_name: username, full_name: fullName = null, job_title: headline = 'Vývojář' } = apiUser;
  return { username, fullName, headline };
}

// Nový profil s jiným popisem, původní zůstane beze změny.
function withHeadline(profile, headline) {
  return { ...profile, headline };
}

// Profil se změnami z formuláře; id a username změnit nejde.
function updateProfile(profile, changes) {
  return { ...profile, ...changes, id: profile.id, username: profile.username };
}

function createProfile({
  username,
  fullName = null,
  headline = 'Vývojář',
  city = null,
  remote = false,
} = {}) {
  return { username, fullName, headline, location: { city, remote }, skills: [] };
}

// Tři odkazy profilu, chybějící mají null.
function normalizeLinks({ github = null, linkedin = null, website = null } = {}) {
  return { github, linkedin, website };
}

// Nový profil s jiným městem, původní i jeho location zůstanou.
function withCity(profile, city) {
  return { ...profile, location: { ...profile.location, city } };
}

// Nový profil s odkazem pro zadanou síť.
function setLink(profile, network, url) {
  return { ...profile, links: { ...profile.links, [network]: url } };
}

// Nový profil s dovedností navíc; dovednost, která už v profilu je, se nepřidá.
function addSkill(profile, skill) {
  if (profile.skills.includes(skill)) {
    return profile;
  }
  return { ...profile, skills: [...profile.skills, skill] };
}

// Veřejný profil podle nastavení soukromí.
function publicProfile(profile) {
  const { email, privacy, ...visible } = profile;
  let result = visible;
  if (privacy.showEmail) {
    result = { ...result, email };
  }
  if (!privacy.showCity) {
    result = { ...result, location: { ...result.location, city: null } };
  }
  return result;
}

// Jména vlastností prvního patra, které se v konceptu liší od uloženého profilu.
function changedFields(saved, draft) {
  const changed = [];
  for (const key of Object.keys(saved)) {
    if (saved[key] !== draft[key]) {
      changed.push(key);
    }
  }
  return changed;
}

// Údaj je vyplněný, když není null, undefined ani prázdný text.
function isFilled(value) {
  return value !== null && value !== undefined && value !== '';
}

// Procento vyplněných údajů profilu (6 údajů), zaokrouhlené.
function completeness(profile) {
  const values = [
    profile.fullName,
    profile.headline,
    profile.location?.city,
    profile.links?.github,
    profile.links?.linkedin,
  ];
  let filled = 0;
  for (const value of values) {
    if (isFilled(value)) {
      filled += 1;
    }
  }
  if ((profile.skills?.length ?? 0) > 0) {
    filled += 1;
  }
  return Math.round((filled / 6) * 100);
}

// Nový profil s otočeným nastavením soukromí.
function togglePrivacy(profile, setting) {
  return { ...profile, privacy: { ...profile.privacy, [setting]: !profile.privacy[setting] } };
}
