# --description--

Nastavení soukromí má ve veřejném profilu něco říct: kdo zapne `showEmail`, chce e-mail ukázat, a kdo vypne `showCity`, nechce prozradit město.

**Úkol:** Uprav `publicProfile` tak, aby:

- e-mail ve výsledku byl, když má profil `privacy.showEmail: true`,
- `location.city` byl `null`, když má profil `privacy.showCity: false` (`remote` zůstane),
- `privacy` ve výsledku nebyl nikdy a uložený profil ani jeho `location` se nezměnily.

Proměnné, které ti destrukturalizace vytáhla, se ti hodí. Na město použij postup z kroku `withCity`, tentokrát nad výsledkem.

# --hints--

S `showEmail: true` obsahuje veřejný profil e-mail.

```js
const profile = { ...savedProfile, privacy: { showEmail: true, showCity: true } };
const result = publicProfile(profile);
assert.equal(result.email, 'jana.dvorakova@example.cz', 'publicProfile pro profil se showEmail: true má obsahovat e-mail');
assert.equal(Object.hasOwn(result, 'privacy'), false, 'publicProfile nemá nikdy obsahovat privacy');
```

Se `showEmail: false` e-mail ve veřejném profilu není.

```js
assert.equal(Object.hasOwn(publicProfile(savedProfile), 'email'), false, 'publicProfile(savedProfile) nemá obsahovat e-mail — showEmail je false');
```

Se `showCity: false` je město `null` a uložený profil zůstane beze změny.

```js
const profile = { ...savedProfile, privacy: { showEmail: false, showCity: false } };
const result = publicProfile(profile);
assert.deepEqual(result.location, { city: null, remote: true }, 'publicProfile pro showCity: false má mít location { city: null, remote: true }');
assert.equal(profile.location.city, 'Brno', "publicProfile nesmí změnit město v původním profilu — má zůstat 'Brno'");
assert.equal(publicProfile(savedProfile).location.city, 'Brno', "publicProfile(savedProfile) má město ukázat — showCity je true");
```

# --help--

## --tip--

Kombinuješ rest z minulého kroku a úpravu bez mutace — viz [Úprava bez mutace: nový objekt](see:js-objekty/reference-a-mutace#uprava-bez-mutace-novy-objekt).

## --tip--

Výsledek si drž v proměnné `let`. Podle `privacy.showEmail` do ní přiřaď nový objekt s e-mailem, podle `privacy.showCity` nový objekt s novým `location`. Do proměnné přiřazuješ, do objektů nezapisuješ.

## --tip--

Podcíle na jiných datech — recept, u kterého autor skryl kalorie:

```js
// 1. vytáhni settings a zbytek receptu
// 2. když settings.showAuthor, přidej do výsledku author
// 3. když není settings.showCalories, nutrition dostane nový objekt s calories: null
```

const savedProfile = {
  id: 481,
  username: 'jana-dev',
  fullName: 'Jana Dvořáková',
  headline: 'Frontend vývojářka',
  email: 'jana.dvorakova@example.cz',
  location: { city: 'Brno', remote: true },
  links: { github: 'github.com/jana-dev', linkedin: null, website: 'jana-dev.example.cz' },
  privacy: { showEmail: false, showCity: true },
  skills: ['HTML', 'CSS', 'JavaScript'],
};

// Od kolegy: profil s novým nastavením soukromí, třeba updatePrivacy(savedProfile, { showEmail: true }).
function updatePrivacy(profile, changes) {
  const privacy = { ...profile.privacy };
  for (const [key, value] of Object.entries(changes)) {
    privacy[key] = value;
  }
  return { ...profile, privacy };
}

// Od kolegy: profil pro náhled firmám, bez odkazů, které nejsou vyplněné (delete smaže vlastnost).
function withoutEmptyLinks(profile) {
  const copy = { ...profile };
  for (const [network, url] of Object.entries(copy.links)) {
    if (url === null) {
      delete copy.links[network];
    }
  }
  return copy;
}

// Jméno na profilu, bez celého jména přezdívka se zavináčem.
function displayName(profile) {
  return profile.fullName ?? `@${profile.username}`;
}

// Profil z tvaru, který posílá starší API (snake_case), do tvaru aplikace.
function fromApi(apiUser) {
  const { user_name: username, full_name: fullName = null, job_title: headline = 'Vývojář' } = apiUser;
  return { username, fullName, headline };
}

// Nový profil s jiným popisem, původní zůstane beze změny.
function withHeadline(profile, headline) {
  return { ...profile, headline };
}

// Profil se změnami z formuláře; id a username změnit nejde.
function updateProfile(profile, changes) {
  return { ...profile, ...changes, id: profile.id, username: profile.username };
}

function createProfile({
  username,
  fullName = null,
  headline = 'Vývojář',
  city = null,
  remote = false,
} = {}) {
  return { username, fullName, headline, location: { city, remote }, skills: [] };
}

// Tři odkazy profilu, chybějící mají null.
function normalizeLinks({ github = null, linkedin = null, website = null } = {}) {
  return { github, linkedin, website };
}

// Nový profil s jiným městem, původní i jeho location zůstanou.
function withCity(profile, city) {
  return { ...profile, location: { ...profile.location, city } };
}

// Nový profil s odkazem pro zadanou síť.
function setLink(profile, network, url) {
  return { ...profile, links: { ...profile.links, [network]: url } };
}

// Nový profil s dovedností navíc; dovednost, která už v profilu je, se nepřidá.
function addSkill(profile, skill) {
  if (profile.skills.includes(skill)) {
    return profile;
  }
  return { ...profile, skills: [...profile.skills, skill] };
}

function publicProfile(profile) {
  const { email, privacy, ...visible } = profile;
  return visible;
}

// Veřejný profil podle nastavení soukromí.
function publicProfile(profile) {
  const { email, privacy, ...visible } = profile;
  let result = visible;
  if (privacy.showEmail) {
    result = { ...result, email };
  }
  if (!privacy.showCity) {
    result = { ...result, location: { ...result.location, city: null } };

  return result;
