# --description--

Nastavení jde přenést do jiného zařízení: uživatel vloží text, který si dřív vyexportoval. Vložit ale může cokoli.

**Úkol:** Napiš funkci `importSettings(text)`, která z textu vrátí úplné nastavení (sloučené s výchozím). Když text není platný JSON, **nebo** v něm není objekt, vrátí `null`, aby panel mohl ukázat „Tohle není soubor s nastavením".

- Platný JSON jsou i texty `'42'`, `'"dark"'` nebo `'null'` — a objekt to není ani jeden.
- `typeof null` je `'object'`, takže samotné `typeof` nestačí.

Bezpečné čtení píšeš stejně jako v `readSaved`, jen s jinou náhradní hodnotou a kontrolou navíc.

# --hints--

S platným objektem vrátí `importSettings` úplné nastavení.

```js
const result = importSettings('{"theme":"dark","downloads":{"quality":"nízká"}}');
assert.equal(result.theme, 'dark', "importSettings('{\"theme\":\"dark\", …}').theme má být 'dark'");
assert.deepEqual(result.downloads, { wifiOnly: true, quality: 'nízká', keepEpisodes: 5 }, 'importSettings má skupinu downloads doplnit výchozími hodnotami');
```

S neplatným JSON vrátí `importSettings` hodnotu `null`.

```js
assert.equal(importSettings('{theme: dark}'), null, "importSettings('{theme: dark}') má vrátit null — to není platný JSON");
assert.equal(importSettings(''), null, "importSettings('') má vrátit null");
```

S JSON, ve kterém není objekt, vrátí `importSettings` hodnotu `null`.

```js
assert.equal(importSettings('42'), null, "importSettings('42') má vrátit null — číslo není nastavení");
assert.equal(importSettings('"dark"'), null, "importSettings('\"dark\"') má vrátit null — text není nastavení");
assert.equal(importSettings('null'), null, "importSettings('null') má vrátit null — pozor, typeof null je 'object'");
```

# --help--

## --tip--

Stejný tvar jako `readSaved` — viz [Bezpečné načtení JSON](see:js-objekty/kopie-a-json#bezpecne-nacteni-json).

## --tip--

Výsledek `JSON.parse` si v `try` ulož do proměnné a v `catch` vrať `null`. Za blokem zkontroluj `typeof` a zvlášť `null`, a teprve pak slučuj s výchozím nastavením.

## --tip--

Podcíle na jiných datech — import seznamu úkolů:

```js
// 1. zkus text naparsovat, při chybě vrať null
// 2. když výsledek není objekt (nebo je null), vrať null
// 3. jinak vrať sloučení s výchozím seznamem
```

function syncedLabel(settings) {

    return 'Zatím nesynchronizováno';
  }
  return `Synchronizováno ${settings.lastSynced.toLocaleDateString('cs-CZ')}`;
}

// Nové nastavení s jiným tématem.
function withTheme(settings, theme) {
  return { ...settings, theme };

    ...saved,

  };
}

  localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
}

function readSaved() {

  if (text === null) {
    return {};
  }
  try {
    return JSON.parse(text);
  } catch {
    return {};
  }
}

  return mergeSettings(DEFAULT_SETTINGS, readSaved());

  lastSynced: null,

  nightOwl: { theme: 'dark', notifications: { newEpisodes: false, weeklySummary: false } },
};

function applyPreset(presetName) {
  return Object.assign(DEFAULT_SETTINGS, PRESETS[presetName]);
}

function mergeSettings(defaults, saved) {
  return {
    ...defaults,
    ...saved,
    playback: { ...defaults.playback, ...saved.playback },
    downloads: { ...defaults.downloads, ...saved.downloads },
    notifications: { ...defaults.notifications, ...saved.notifications },
  };
}

// Uloží nastavení do localStorage jako JSON.
function saveSettings(settings) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
}

function readSaved() {
  const text = localStorage.getItem(STORAGE_KEY);
  if (text === null) {
    return {};
  }
  try {
    return JSON.parse(text);
  } catch {
    return {};
  }
}

// Úplné nastavení při startu: uložené hodnoty doplněné výchozími.
function loadSettings() {
  return mergeSettings(DEFAULT_SETTINGS, readSaved());
}

// Nastavení z vloženého textu, nebo null, když to nastavení není.
function importSettings(text) {
  let parsed;
  try {
    parsed = JSON.parse(text);
  } catch {
    return null;
  }
  if (typeof parsed !== 'object' || parsed === null) {
    return null;
  }
  return mergeSettings(DEFAULT_SETTINGS, parsed);
}
