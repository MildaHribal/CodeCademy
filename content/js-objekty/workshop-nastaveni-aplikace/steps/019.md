# --description--

Úplně dole v panelu je tlačítko **Obnovit vše**. Smaže uložené nastavení a panel ukáže výchozí hodnoty.

**Úkol:** Napiš funkci `resetAll()`, která smaže záznam pod `STORAGE_KEY` z `localStorage` (metoda `removeItem` s klíčem) a vrátí úplné výchozí nastavení.

- Vrácené nastavení má stejné hodnoty jako `DEFAULT_SETTINGS`, ale nesmí to být tentýž objekt ani sdílet jeho skupiny — panel s ním bude dál pracovat jako s konceptem.
- Hotové funkce, které tohle umí, už máš.

# --hints--

Po `resetAll` v úložišti nic není.

```js
saveSettings(mergeSettings(DEFAULT_SETTINGS, { theme: 'dark' }));
resetAll();
assert.equal(localStorage.getItem(STORAGE_KEY), null, "po resetAll() nemá být v localStorage nic pod klíčem 'poslech:settings'");
```

`resetAll` vrátí výchozí hodnoty v nových objektech.

```js
saveSettings(mergeSettings(DEFAULT_SETTINGS, { playback: { speed: 2 } }));
const result = resetAll();
assert.deepEqual(result, DEFAULT_SETTINGS, 'resetAll() má vrátit výchozí hodnoty, i když bylo uložené něco jiného');
assert.notEqual(result, DEFAULT_SETTINGS, 'resetAll() nemá vracet samotný DEFAULT_SETTINGS');
assert.notEqual(result.playback, DEFAULT_SETTINGS.playback, 'resetAll() nemá sdílet skupinu playback s DEFAULT_SETTINGS');
```

# --help--

## --tip--

Kterou kopii na čerstvé výchozí nastavení, najdeš v [Kterou kopii vybrat](see:js-objekty/kopie-a-json#kterou-kopii-vybrat).

## --tip--

Nejdřív `localStorage.removeItem` s klíčem nastavení. Pak vrať výsledek funkce, která při prázdném úložišti sloučí výchozí nastavení do nových objektů — napsal jsi ji v pátém kroku.

## --tip--

Podcíle na jiných datech — vyprázdnění košíku:

```js
// 1. smaž košík z úložiště
// 2. vrať čerstvý prázdný košík z funkce, která ho načítá
```

```js
// Výchozí nastavení přehrávače podcastů. Uživatel si mění jen část, zbytek se bere odsud.
const DEFAULT_SETTINGS = {
  theme: 'system',
  lastSynced: null,
  playback: { speed: 1, skipSeconds: 15, autoplay: true },
  downloads: { wifiOnly: true, quality: 'vysoká', keepEpisodes: 5 },
  notifications: { newEpisodes: true, weeklySummary: false },
};

// Klíč, pod kterým se nastavení ukládá do localStorage.
const STORAGE_KEY = 'poslech:settings';

// Od kolegy: předvolby, které jde zapnout jedním tlačítkem.
const PRESETS = {
  dataSaver: { downloads: { wifiOnly: true, quality: 'nízká', keepEpisodes: 2 } },
  nightOwl: { theme: 'dark', notifications: { newEpisodes: false, weeklySummary: false } },
};

// Od kolegy: nastavení se zapnutou předvolbou, třeba applyPreset('dataSaver').
function applyPreset(presetName) {
  return Object.assign({}, DEFAULT_SETTINGS, PRESETS[presetName]);
}

// Od kolegy: text do patičky panelu, třeba 'Synchronizováno 14. 9. 2026'.
function syncedLabel(settings) {
  if (settings.lastSynced === null) {
    return 'Zatím nesynchronizováno';
  }
  return `Synchronizováno ${new Date(settings.lastSynced).toLocaleDateString('cs-CZ')}`;
}

// Nové nastavení s jiným tématem.
function withTheme(settings, theme) {
  return { ...settings, theme };
}

// Úplné nastavení: uložené hodnoty přes výchozí, každá skupina sloučená zvlášť.
function mergeSettings(defaults, saved) {
  return {
    ...defaults,
    ...saved,
    playback: { ...defaults.playback, ...saved.playback },
    downloads: { ...defaults.downloads, ...saved.downloads },
    notifications: { ...defaults.notifications, ...saved.notifications },
  };
}

// Uloží do localStorage jako JSON jen to, čím se nastavení liší od výchozího.
function saveSettings(settings) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(changedSettings(DEFAULT_SETTINGS, settings)));
}

function readSaved() {
  const text = localStorage.getItem(STORAGE_KEY);
  if (text === null) {
    return {};
  }
  try {
    return JSON.parse(text);
  } catch {
    return {};
  }
}

// Úplné nastavení při startu: uložené hodnoty doplněné výchozími.
function loadSettings() {
  return mergeSettings(DEFAULT_SETTINGS, readSaved());
}

// Nastavení z vloženého textu, nebo null, když to nastavení není.
function importSettings(text) {
  let parsed;
  try {
    parsed = JSON.parse(text);
  } catch {
    return null;
  }
  if (typeof parsed !== 'object' || parsed === null) {
    return null;
  }
  return mergeSettings(DEFAULT_SETTINGS, parsed);
}

// JSON k exportu: bez data synchronizace, odsazený pro čtení.
function exportSettings(settings) {
  const { lastSynced, ...exportable } = settings;
  return JSON.stringify(exportable, null, 2);
}

// Nastavení ze starého plochého formátu (verze 1) do formátu se skupinami.
function migrateV1(saved) {
  const { speed, wifiOnly, ...rest } = saved;
  const playback = {};
  if (speed !== undefined) {
    playback.speed = speed;
  }
  const downloads = {};
  if (wifiOnly !== undefined) {
    downloads.wifiOnly = wifiOnly;
  }
  return { ...rest, playback, downloads };
}

// Nové nastavení s datem synchronizace, bez data aktuální čas.
function markSynced(settings, now = new Date()) {
  return { ...settings, lastSynced: now };
}

// Koncept nastavení: hluboká kopie, do které jde zapisovat.
function draftFrom(settings) {
  return structuredClone(settings);
}

// Krátký JSON ke sdílení: jen téma a přehrávání.
function shareableSettings(settings) {
  return JSON.stringify({ theme: settings.theme, playback: settings.playback });
}

// Liší se koncept od uloženého nastavení? Pořadí klíčů nerozhoduje.
function hasUnsavedChanges(saved, draft) {
  const savedText = JSON.stringify(mergeSettings(DEFAULT_SETTINGS, saved));
  const draftText = JSON.stringify(mergeSettings(DEFAULT_SETTINGS, draft));
  return savedText !== draftText;
}

// Jen hodnoty, kterými se nastavení liší od výchozího.
function changedSettings(defaults, settings) {
  const changes = {};
  for (const [key, defaultValue] of Object.entries(defaults)) {
    const value = settings[key];
    if (typeof defaultValue === 'object' && defaultValue !== null) {
      const groupChanges = {};
      for (const [name, defaultGroupValue] of Object.entries(defaultValue)) {
        if (value[name] !== defaultGroupValue) {
          groupChanges[name] = value[name];
        }
      }
      if (Object.keys(groupChanges).length > 0) {
        changes[key] = groupChanges;
      }
    } else if (value !== defaultValue) {
      changes[key] = value;
    }
  }
  return changes;
}

// Nové nastavení s otočenou notifikací.
function toggleNotification(settings, name) {
  return { ...settings, notifications: { ...settings.notifications, [name]: !settings.notifications[name] } };
}

// Nové nastavení, ve kterém má jedna skupina výchozí hodnoty.
function resetGroup(settings, group) {
  return { ...settings, [group]: { ...DEFAULT_SETTINGS[group] } };
}
```
