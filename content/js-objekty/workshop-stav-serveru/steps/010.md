# --description--

## Hlášení

Štítky s pravidly se na kartě ukazují správně, jenže místo hodnot na nich svítí `undefined`: „obtížnost: undefined", „whitelist: undefined". Hodnoty čte kolegova funkce `ruleValue(server, ruleName)` z horní části souboru. V konzoli žádná chyba není.

## Úkol

Oprav `ruleValue` tak, aby vracela hodnotu pravidla, jehož jméno dostane v parametru `ruleName`. Změň co nejmenší kus kódu.

# --hints--

`ruleValue(kostkov, 'difficulty')` vrátí `'normal'`.

```js
assert.equal(ruleValue(kostkov, 'difficulty'), 'normal', "ruleValue(kostkov, 'difficulty') má vrátit 'normal'");
```

`ruleValue` vrátí i hodnotu `false` a pro chybějící pravidlo `undefined`.

```js
assert.equal(ruleValue(kostkov, 'pvp'), false, "ruleValue(kostkov, 'pvp') má vrátit false");
assert.equal(ruleValue(kostkov, 'fly'), undefined, "ruleValue(kostkov, 'fly') má vrátit undefined — takové pravidlo server nemá");
```

Data serveru a ostatní funkce fungují dál.

```js
assert.equal(kostkov.rules.whitelist, true, 'kostkov.rules.whitelist má zůstat true');
assert.equal(serverTitle(kostkov), 'Kostkov Survival (26.1)', "serverTitle(kostkov) má dál vracet 'Kostkov Survival (26.1)'");
assert.equal(hasRule(kostkov, 'pvp'), true, "hasRule(kostkov, 'pvp') má dál vracet true");
```

# --help--

## --tip--

Chyba odpovídá pasti [Tečka místo závorek](see:js-objekty/objekty#tecka-misto-zavorek).

## --tip--

Podívej se, jaký klíč hledá výraz za `return`. Jméno pravidla je v parametru, ne přímo v kódu.

## --tip--

Na jiných datech: `settings.options[optionName]` přečte volbu, jejíž jméno je v proměnné `optionName`.

// ping = odezva v milisekundách podle regionu, ze kterého se hráč připojuje.
const kostkov = {

function ruleValue(server, ruleName) {
  return server.rules.ruleName;

function playersText(online, max) {
  return `${online} z ${max} hráčů`;
}

function serverTitle(server) {
  return `${server.name} (${server.version})`;
}

function playersLabel(server) {
  return playersText(server.players.online, server.players.max);
}

// Odezva v ms pro zadaný region, třeba pingFor(kostkov, 'us').
function pingFor(server, region) {
  return server.ping[region];
}

// Jedno měření odezvy jako objekt, třeba { asia: 231 }.
function pingEntry(region, ms) {
  return { [region]: ms };
}

// Od kolegy: jedno měření pro graf odezvy, třeba measurement('eu', 24).
const measurement = (region, ms) => ({ region, ms });

// Hranice odezvy a jejich slovní hodnocení.
const pingQuality = {
  goodLimit: 60,
  okLimit: 150,
  label(ms) {
    if (ms <= pingQuality.goodLimit) {
      return 'výborný';
    }
    if (ms <= pingQuality.okLimit) {
      return 'dobrý';
    }
    return 'slabý';
  },
};

// Jméno majitele, nebo 'bez majitele'.
function ownerName(server) {
  return server.owner?.name ?? 'bez majitele';
}

// Uvádí server pravidlo se zadaným jménem (i s hodnotou false)?
function hasRule(server, rule) {
  return Object.hasOwn(server.rules, rule);
}
