# --description--

Karta doporučí hráči region s nejnižší odezvou.

**Úkol:** Napiš funkci `bestRegion(server)`, která vrátí **jméno** regionu s nejnižší odezvou, třeba `'eu'`.

- Když mají dva regiony stejnou nejnižší odezvu, vrať ten, který je v objektu dřív.
- Server bez měření vrátí `undefined`.
- Objekt `ping` se nesmí změnit.

Nástroj si vyber sám. Rozumných cest je víc a všechno, co potřebuješ, už znáš.

# --hints--

`bestRegion(kostkov)` vrátí `'eu'`.

```js
assert.equal(bestRegion(kostkov), 'eu', "bestRegion(kostkov) má vrátit 'eu' — 24 ms je nejméně");
```

`bestRegion` najde nejnižší odezvu i tehdy, když není na prvním ani posledním místě.

```js
assert.equal(bestRegion({ ping: { us: 140, sa: 95, asia: 210 } }), 'sa', "bestRegion({ ping: { us: 140, sa: 95, asia: 210 } }) má vrátit 'sa'");
```

Při shodě vrátí `bestRegion` region, který je v objektu dřív.

```js
assert.equal(bestRegion({ ping: { us: 80, eu: 40, oc: 40 } }), 'eu', "bestRegion({ ping: { us: 80, eu: 40, oc: 40 } }) má vrátit 'eu' — eu i oc mají 40 ms a eu je dřív");
```

Pro server bez měření vrátí `bestRegion` hodnotu `undefined` a `ping` zůstane beze změny.

```js
assert.equal(bestRegion({ ping: {} }), undefined, 'bestRegion({ ping: {} }) má vrátit undefined');
const lobby = { ping: { eu: 30, us: 10 } };
bestRegion(lobby);
assert.deepEqual(lobby.ping, { eu: 30, us: 10 }, 'bestRegion nesmí změnit původní objekt ping');
```

# --help--

## --tip--

Potřebuješ klíč i hodnotu zároveň — porovnej, co vracejí funkce v [Procházení objektu](see:js-objekty/objekty#prochazeni-objektu-object-keys-values-a-entries).

## --tip--

Pamatuj si v proměnných nad cyklem dosud nejlepší region a jeho odezvu. Přepiš je jen tehdy, když ještě žádný nemáš, nebo je další odezva **ostře** nižší — tak při shodě vyhraje dřívější region.

## --tip--

Stejný postup na jiných datech — nejlevnější dopravce, rozepsaný na podcíle:

```js
// 1. zatím žádný nejlevnější dopravce a žádná cena
// 2. pro každou dvojici [dopravce, cena]: když ještě nic nemáš, nebo je cena ostře nižší, zapamatuj si oba údaje
// 3. po cyklu vrať zapamatovaného dopravce (bez dopravců zůstane undefined)
```

// ping = odezva v milisekundách podle regionu, ze kterého se hráč připojuje.
const kostkov = {

function ruleValue(server, ruleName) {
  return server.rules[ruleName];
}

// Text s počtem hráčů, třeba '37 z 100 hráčů'.
function playersText(online, max) {
  return `${online} z ${max} hráčů`;
}

// Název a verze, třeba 'Kostkov Survival (26.1)'.
function serverTitle(server) {
  return `${server.name} (${server.version})`;
}

// Text o hráčích serveru, třeba '37 z 100 hráčů'.
function playersLabel(server) {
  return playersText(server.players.online, server.players.max);
}

// Odezva v ms pro zadaný region, třeba pingFor(kostkov, 'us').
function pingFor(server, region) {
  return server.ping[region];
}

// Jedno měření odezvy jako objekt, třeba { asia: 231 }.
function pingEntry(region, ms) {
  return { [region]: ms };
}

// Hranice odezvy a jejich slovní hodnocení.
const pingQuality = {
  goodLimit: 60,
  okLimit: 150,
  label(ms) {
    if (ms <= pingQuality.goodLimit) {
      return 'výborný';
    }
    if (ms <= pingQuality.okLimit) {
      return 'dobrý';
    }
    return 'slabý';
  },
};

// Jméno majitele, nebo 'bez majitele'.
function ownerName(server) {
  return server.owner?.name ?? 'bez majitele';
}

// Uvádí server pravidlo se zadaným jménem (i s hodnotou false)?
function hasRule(server, rule) {
  return Object.hasOwn(server.rules, rule);
}

// Z kolika regionů se měří odezva.
function regionCount(server) {
  return Object.keys(server.ping).length;
}

// Průměrná odezva zaokrouhlená na celé ms, bez měření null.
function averagePing(server) {
  const values = Object.values(server.ping);
  if (values.length === 0) {
    return null;
  }
  let total = 0;
  for (const ms of values) {
    total += ms;
  }
  return Math.round(total / values.length);
}

// Popisky odezvy s jednotkou, třeba { eu: '24 ms' }.
function pingLabels(server) {
  const pairs = [];
  for (const [region, ms] of Object.entries(server.ping)) {
    pairs.push([region, `${ms} ms`]);
  }
  return Object.fromEntries(pairs);
}

// Slovní hodnocení odezvy pro každý region.
function qualityByRegion(server) {
  const pairs = [];
  for (const [region, ms] of Object.entries(server.ping)) {
    pairs.push([region, pingQuality.label(ms)]);
  }
  return Object.fromEntries(pairs);
}

function qualityByRegion(server) {
  const pairs = [];
  for (const [region, ms] of Object.entries(server.ping)) {
    pairs.push([region, pingQuality.label(ms)]);
  }
  return Object.fromEntries(pairs);
}

// Region s nejnižší odezvou; při shodě ten dřívější, bez měření undefined.
function bestRegion(server) {
  let best;
  let bestPing;
  for (const [region, ms] of Object.entries(server.ping)) {
    if (best === undefined || ms < bestPing) {
      best = region;
      bestPing = ms;
    }
  }
  return best;
}
