# --description--

Štítky s pravidly jsou pro oko. Čtečka obrazovky z nich ale přečte jen útržky, a proto seznam štítků dostane popisek jednou větou: `pvp: ne · whitelist: ano · difficulty: normal`.

**Úkol:** Napiš funkci `rulesSummary(server)`, která vrátí pravidla z `server.rules` v pořadí, v jakém jsou v objektu, ve tvaru `jméno: hodnota` a oddělená mezerou, tečkou `·` a mezerou.

- Hodnota `true` se napíše jako `ano`, `false` jako `ne`, ostatní hodnoty tak, jak jsou.
- Server bez pravidel (`rules: {}`) vrátí `'bez pravidel'`.

# --hints--

`rulesSummary(kostkov)` vrátí `'pvp: ne · whitelist: ano · difficulty: normal'`.

```js
assert.equal(rulesSummary(kostkov), 'pvp: ne · whitelist: ano · difficulty: normal', "rulesSummary(kostkov) má vrátit 'pvp: ne · whitelist: ano · difficulty: normal'");
```

`rulesSummary` pracuje s pravidly serveru z parametru a čísla nechá čísly.

```js
const creative = { rules: { maxHeight: 320, pvp: true, spawnProtection: 0 } };
assert.equal(rulesSummary(creative), 'maxHeight: 320 · pvp: ano · spawnProtection: 0', "rulesSummary pro { maxHeight: 320, pvp: true, spawnProtection: 0 } má vrátit 'maxHeight: 320 · pvp: ano · spawnProtection: 0'");
```

Server bez pravidel vrátí `'bez pravidel'`.

```js
assert.equal(rulesSummary({ rules: {} }), 'bez pravidel', "rulesSummary({ rules: {} }) má vrátit 'bez pravidel'");
```

# --help--

## --tip--

Potřebuješ jméno i hodnotu každého pravidla — viz [Procházení objektu](see:js-objekty/objekty#prochazeni-objektu-object-keys-values-a-entries).

## --tip--

Texty `jméno: hodnota` si při procházení dvojic sbírej do pole a na konci je spoj metodou `join`, kterou znáš z textů. Prázdný objekt pravidel ohlídej zvlášť. Pozor, `0` není `false`: porovnávej přesně s `true` a `false`.

## --tip--

Podcíle na jiných datech — vybavení bytu v inzerátu:

```js
// 1. prázdné pole textů
// 2. pro každou dvojici [vybavení, hodnota] přidej text, true jako 'ano' a false jako 'ne'
// 3. prázdné pole → 'bez vybavení', jinak texty spojené ' · '
```

// Stav herního serveru, jak ho vrací monitorovací služba.
// ping = odezva v milisekundách podle regionu, ze kterého se hráč připojuje.
const kostkov = {
  name: 'Kostkov Survival',
  address: 'mc.kostkov.cz',
  version: '26.1',
  online: true,
  motd: 'Nová mapa od pátku!',
  players: { online: 37, max: 100 },
  ping: { eu: 24, us: 131, asia: 248 },
  rules: { pvp: false, whitelist: true, difficulty: 'normal' },
  owner: null,
};

// Od kolegy: hodnota jednoho pravidla, třeba ruleValue(kostkov, 'difficulty').
function ruleValue(server, ruleName) {
  return server.rules[ruleName];
}

// Text s počtem hráčů, třeba '37 z 100 hráčů'.
function playersText(online, max) {
  return `${online} z ${max} hráčů`;
}

// Název a verze, třeba 'Kostkov Survival (26.1)'.
function serverTitle(server) {
  return `${server.name} (${server.version})`;
}

// Text o hráčích serveru, třeba '37 z 100 hráčů'.
function playersLabel(server) {
  return playersText(server.players.online, server.players.max);
}

// Odezva v ms pro zadaný region, třeba pingFor(kostkov, 'us').
function pingFor(server, region) {
  return server.ping[region];
}

// Jedno měření odezvy jako objekt, třeba { asia: 231 }.
function pingEntry(region, ms) {
  return { [region]: ms };
}

// Od kolegy: jedno měření pro graf odezvy, třeba measurement('eu', 24).
const measurement = (region, ms) => ({ region, ms });

// Hranice odezvy a jejich slovní hodnocení.
const pingQuality = {
  goodLimit: 60,
  okLimit: 150,
  label(ms) {
    if (ms <= pingQuality.goodLimit) {
      return 'výborný';
    }
    if (ms <= pingQuality.okLimit) {
      return 'dobrý';
    }
    return 'slabý';
  },
};

// Jméno majitele, nebo 'bez majitele'.
function ownerName(server) {
  return server.owner?.name ?? 'bez majitele';
}

// Uvádí server pravidlo se zadaným jménem (i s hodnotou false)?
function hasRule(server, rule) {
  return Object.hasOwn(server.rules, rule);
}

// Z kolika regionů se měří odezva.
function regionCount(server) {
  return Object.keys(server.ping).length;
}

// Průměrná odezva zaokrouhlená na celé ms, bez měření null.
function averagePing(server) {
  const values = Object.values(server.ping);
  if (values.length === 0) {
    return null;
  }
  let total = 0;
  for (const ms of values) {
    total += ms;
  }
  return Math.round(total / values.length);
}

// Popisky odezvy s jednotkou, třeba { eu: '24 ms' }.
function pingLabels(server) {
  const pairs = [];
  for (const [region, ms] of Object.entries(server.ping)) {
    pairs.push([region, `${ms} ms`]);
  }
  return Object.fromEntries(pairs);
}

// Slovní hodnocení odezvy pro každý region.
function qualityByRegion(server) {
  const pairs = [];
  for (const [region, ms] of Object.entries(server.ping)) {
    pairs.push([region, pingQuality.label(ms)]);
  }
  return Object.fromEntries(pairs);
}

// Region s nejnižší odezvou; při shodě ten dřívější, bez měření undefined.
function bestRegion(server) {
  let best;
  let bestPing;
  for (const [region, ms] of Object.entries(server.ping)) {
    if (best === undefined || ms < bestPing) {
      best = region;
      bestPing = ms;
    }
  }
  return best;
}

// Adresa pro připojení i s portem, bez portu výchozí 25565.
function joinAddress({ address, port = 25565 }) {
  return `${address}:${port}`;
}

// Od kolegy: pruh se zprávou dne nad kartou, třeba motdBanner(kostkov).
function motdBanner({ motd = 'Dnes bez zprávy' } = {}) {
  return `Zpráva dne: ${motd}`;
}

// Řádek stavu do záhlaví, třeba 'Kostkov Survival · online · 37 z 100 hráčů'.
function statusLine({ name, online, players }) {
  if (!online) {
    return `${name} · offline`;
  }
  return `${name} · online · ${playersText(players.online, players.max)}`;
}

function rulesSummary(server) {
