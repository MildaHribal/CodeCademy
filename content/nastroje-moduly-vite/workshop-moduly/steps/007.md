# --description--

Složka `orders/` má dva soubory a brzy přibudou další. Kdyby si `app.js` bral každou
funkci přímo z konkrétního souboru, musel by vědět, jak je složka uvnitř rozdělená —
a každé přestěhování funkce by rozbilo importy. Proto složka dostane [[barrel soubor]]
`orders/index.js`: nic nepočítá, jen pustí ven funkce, které smí používat zbytek pokladny.

Zápis `export { … } from './soubor.js'` vezme exporty jiného modulu a rovnou je
vyexportuje dál. Krátký seznam jmen se vejde na jeden řádek, dlouhý se rozepíše
po jednom jménu na řádek — obojí znamená totéž:

```js
export { signIn, signOut } from './session.js';
```

**Úkol:** seřaď řádky `orders/index.js` tak, aby vystavoval `createOrder`, `addItem`
a `orderTotal` z `order.js` a `renderReceipt` z `receipt.js`. Odsazení nastavíš
klávesou Tab. Dva řádky jsou navíc. `app.js` zatím importuje postaru — na nový vstup
ho přepojíš v dalším kroku.

# --hints--

`orders/index.js` vystavuje `createOrder`, `addItem`, `orderTotal` a `renderReceipt`.

```js
const orders = await helpers.importFile('orders/index.js');
for (const name of ['createOrder', 'addItem', 'orderTotal', 'renderReceipt']) {
  assert.equal(typeof orders[name], 'function', `orders/index.js má vystavovat funkci ${name}`);
}
```

Funkce z `orders/index.js` spolu fungují: objednávka se 2× croissantem má účtenku se součtem 130 Kč.

```js
const sameLine = (a) => a.normalize('NFC').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();
const orders = await helpers.importFile('orders/index.js');
const order = orders.addItem(orders.createOrder(4), 'croissant', 2);
assert.equal(orders.orderTotal(order), 130, "orderTotal po addItem(createOrder(4), 'croissant', 2) má vrátit 130");
const last = orders.renderReceipt(order).split('\n').map(sameLine).at(-1);
assert.equal(last, 'Celkem 130 Kč', 'renderReceipt má pro 2× croissant skončit „Celkem 130 Kč"');
```

## --file-- orders/order.js

```js
// Objednávky: založení, položky a součty.
import { findItem } from '../menu.js';

export function createOrder(table) {
  return { table, lines: [] };
}

export function addItem(order, id, quantity = 1) {
  const item = findItem(id);
  const line = { name: item.name, price: item.price, quantity };
  return { ...order, lines: [...order.lines, line] };
}

export function lineTotal(line) {
  return line.price * line.quantity;
}

export function orderTotal(order) {
  return order.lines.reduce((sum, line) => sum + lineTotal(line), 0);
}
```

## --file-- orders/receipt.js

```js
// Tisk účtenky. Přestěhoval sem Tomáš z app.js.
import { formatPrice } from '../money.js';
import { format as formatDate } from '../dates.js';
import { lineTotal, orderTotal } from './order.js';

const RECEIPT_WIDTH = 32;

function padLine(left, right) {
  return left.padEnd(RECEIPT_WIDTH - right.length) + right;
}

export function renderReceipt(order) {
  const rows = order.lines.map((line) =>
    padLine(`${line.quantity}× ${line.name}`, formatPrice(lineTotal(line))),
  );
  return [
    `Kavárna Na Kopečku · stůl ${order.table} · ${formatDate(new Date())}`,
    ...rows,
    padLine('Celkem', formatPrice(orderTotal(order))),
  ].join('\n');
}
```
