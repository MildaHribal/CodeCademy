# --description--

Ceny chce měnit vedoucí kavárny, a ta do JavaScriptu sahat nebude. Proto ceník
přestěhoval do souboru `menu.json` — data jsou tam stejná jako v poli v `menu.js`.
Díky tomu, že je pole schované v modulu a ven jde jen `findItem`, se tahle změna
dotkne jediného souboru.

**Úkol:** uprav `menu.js` tak, aby data bral z `menu.json` jako [[ES modul|modul]]
s daty JSON. Pole napsané přímo v `menu.js` zmizí. `findItem` funguje dál stejně
a nic jiného v pokladně se nezmění. Když vedoucí změní cenu v `menu.json`, pokladna ji
po dalším spuštění použije.

# --hints--

`findItem('lemonade')` z `menu.js` vrátí limonádu za 69 Kč a `menu.js` už ceník sám neobsahuje.

```js
const { findItem } = await helpers.importFile('menu.js');
assert.equal(findItem('lemonade')?.price, 69, "findItem('lemonade') má vrátit položku s cenou 69");
assert.equal(findItem('espresso')?.name, 'Espresso', "findItem('espresso') má vrátit položku s name 'Espresso'");
assert.doesNotMatch(helpers.stripComments(files['menu.js'], 'js'), /Flat white|Cheesecake/, 'menu.js už nemá obsahovat položky ceníku — data patří do menu.json');
```

Pokladna tiskne účtenky jako dřív.

```js
const sameLine = (a) => a.normalize('NFC').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();
const result = await helpers.run('node app.js');
assert.equal(result.code, 0, `node app.js má doběhnout bez chyby:\n${result.stderr}`);
const lines = result.stdout.split('\n').map(sameLine);
assert.ok(lines.includes('Celkem 215 Kč') && lines.includes('Celkem 385 Kč') && lines.includes('Věrnostní body: 4'), `Účtenky mají vypadat jako dřív. Vypsalo se:\n${result.stdout}`);
```

Po změně ceny croissantu v `menu.json` na 70 Kč účtenka stolu 3 ukáže novou cenu.

```js
const sameLine = (a) => a.normalize('NFC').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();
const menu = JSON.parse(files['menu.json']);
const changed = menu.map((item) => (item.id === 'croissant' ? { ...item, price: 70 } : item));
const fs = await import('node:fs/promises');
await fs.writeFile(`${helpers.dir}/menu.json`, JSON.stringify(changed));
const result = await helpers.run('node app.js');
const lines = result.stdout.split('\n').map(sameLine);
await fs.writeFile(`${helpers.dir}/menu.json`, files['menu.json']);
assert.ok(lines.includes('1× Croissant 70 Kč'), `Po změně menu.json na croissant za 70 Kč má účtenka ukázat „1× Croissant 70 Kč". Vypsalo se:\n${result.stdout}${result.stderr}`);
```

## --tip--

Jak se importuje soubor JSON a co je jeho výchozí export, popisuje část
[Import JSON](see:nastroje-moduly-vite/es-moduly#import-json).

## --tip--

Mění se jen začátek `menu.js`: místo deklarace pole přijde import ze sousedního
souboru `menu.json`. Import potřebuje atribut s typem a jméno, pod kterým data
použije `findItem`.

## --tip--

Stejný vzor jinde: modul s překlady má `import czech from './cs.json'` s atributem
typu JSON a jeho funkce `t(key)` pak čte `czech[key]`.

## --file-- orders/order.js

```js
// Objednávky: založení, položky a součty.
import { findItem } from '../menu.js';
import { nextOrderNumber } from './numbering.js';

export function createOrder(table) {
  return { number: nextOrderNumber(), table, lines: [] };
}

export function addItem(order, id, quantity = 1) {
  const item = findItem(id);
  const line = { name: item.name, price: item.price, quantity };
  return { ...order, lines: [...order.lines, line] };
}

export function lineTotal(line) {
  return line.price * line.quantity;
}

export function orderTotal(order) {
  return order.lines.reduce((sum, line) => sum + lineTotal(line), 0);
}

export function itemCount(order) {
  return order.lines.reduce((sum, line) => sum + line.quantity, 0);
}
```

## --file-- orders/receipt.js

```js
// Tisk účtenky. Přestěhoval sem Tomáš z app.js.
import { formatPrice } from '../money.js';
import { format as formatDate } from '../dates.js';
import { lineTotal, orderTotal } from './order.js';

const RECEIPT_WIDTH = 32;

function padLine(left, right) {
  return left.padEnd(RECEIPT_WIDTH - right.length) + right;
}

export function renderReceipt(order) {
  const rows = order.lines.map((line) =>
    padLine(`${line.quantity}× ${line.name}`, formatPrice(lineTotal(line))),
  );
  return [
    `Kavárna Na Kopečku · stůl ${order.table} · ${formatDate(new Date())}`,
    ...rows,
    padLine('Celkem', formatPrice(orderTotal(order))),
  ].join('\n');
}
```

import { orderTotal } from '../orders/index.js';
import { formatPrice } from '../money.js';

export function renderDailyReport(orders) {
  const revenue = orders.reduce((sum, order) => sum + orderTotal(order), 0);
