# --description--

Každá účtenka musí mít pořadové číslo, které se za den neopakuje. Číslování bude mít
vlastní modul `orders/numbering.js` se **stavem**: proměnnou s posledním vydaným
číslem a funkcí, která vydá další.

**Úkol:**

1. V `orders/numbering.js` exportuj proměnnou `lastNumber`, která začíná na `0`,
   a funkci `nextOrderNumber()`. Ta zvýší `lastNumber` o jedna a nové číslo vrátí.
2. V `orders/order.js` ji použij: `createOrder` přidá objednávce vlastnost `number`
   s dalším číslem v řadě. První objednávka má číslo 1, druhá 2.

K proměnné, kterou funkce mění, se hodí `let`; u `const` by přičtení spadlo. Stejný
tvar má modul, který počítá přihlášky na závod:

```js
export let registeredCount = 0;

export function register(runner) {
  registeredCount += 1;
  return { ...runner, startNumber: registeredCount };
}
```

Na účtenkách se zatím nic nezmění. Čísla využiješ v dalším kroku.

# --hints--

`nextOrderNumber()` vrací čísla 1, 2, … a `lastNumber` z `orders/numbering.js` ukazuje poslední vydané.

```js
const numbering = await helpers.importFile('orders/numbering.js');
assert.equal(numbering.lastNumber, 0, 'lastNumber má na začátku být 0');
assert.equal(typeof numbering.nextOrderNumber, 'function', 'orders/numbering.js má exportovat funkci nextOrderNumber');
assert.equal(numbering.nextOrderNumber(), 1, 'první volání nextOrderNumber() má vrátit 1');
assert.equal(numbering.nextOrderNumber(), 2, 'druhé volání nextOrderNumber() má vrátit 2');
assert.equal(numbering.lastNumber, 2, 'po dvou voláních nextOrderNumber() má lastNumber být 2');
```

`createOrder` z `orders/order.js` dává objednávkám čísla po sobě: první 1, druhá 2.

```js
const order = await helpers.importFile('orders/order.js');
const first = order.createOrder(3);
const second = order.createOrder(5);
assert.equal(first.number, 1, 'createOrder(3) jako první objednávka má mít number 1');
assert.equal(second.number, 2, 'createOrder(5) jako druhá objednávka má mít number 2');
assert.equal(second.table, 5, 'createOrder(5) má dál nastavit table 5');
assert.deepEqual(second.lines, [], 'nová objednávka má dál mít prázdné lines');
```

Účtenky se tisknou jako dřív.

```js
const sameLine = (a) => a.normalize('NFC').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();
const result = await helpers.run('node app.js');
assert.equal(result.code, 0, `node app.js má doběhnout bez chyby:\n${result.stderr}`);
const lines = result.stdout.split('\n').map(sameLine);
assert.ok(lines.includes('Celkem 215 Kč') && lines.includes('Studentská sleva: -39 Kč'), `Účtenky mají vypadat jako dřív. Vypsalo se:\n${result.stdout}`);
```

# --help--

## --tip--

Modul může držet stav v proměnné na nejvyšší úrovni. Jak se na ni dívají ostatní
soubory, popisuje část [Živé vazby](see:nastroje-moduly-vite/es-moduly#zive-vazby).

## --tip--

V `orders/numbering.js` potřebuješ dvě exportované věci: proměnnou deklarovanou tak,
aby šla měnit, a funkci, která ji zvýší a vrátí. V `orders/order.js` přibude import
ze sousedního souboru a nová vlastnost v objektu, který vrací `createOrder`.

## --tip--

Stejný vzor: `export let ticketCount = 0;` a `export function nextTicket() { … }`,
kterou pak použije `createTicket` jako `{ id: nextTicket(), title }`.

```js
// Objednávky: založení, položky a součty.
import { findItem } from '../menu.js';

export function createOrder(table) {
  return { table, lines: [] };
}

export function addItem(order, id, quantity = 1) {
  const item = findItem(id);
  const line = { name: item.name, price: item.price, quantity };
  return { ...order, lines: [...order.lines, line] };
}

export function lineTotal(line) {
  return line.price * line.quantity;
}

export function orderTotal(order) {
  return order.lines.reduce((sum, line) => sum + lineTotal(line), 0);
}

export function itemCount(order) {
  return order.lines.reduce((sum, line) => sum + line.quantity, 0);
}
```

## --file-- orders/receipt.js

```js
// Tisk účtenky. Přestěhoval sem Tomáš z app.js.
import { formatPrice } from '../money.js';
import { format as formatDate } from '../dates.js';
import { lineTotal, orderTotal } from './order.js';

const RECEIPT_WIDTH = 32;

function padLine(left, right) {
  return left.padEnd(RECEIPT_WIDTH - right.length) + right;
}

export function renderReceipt(order) {
  const rows = order.lines.map((line) =>
    padLine(`${line.quantity}× ${line.name}`, formatPrice(lineTotal(line))),
  );
  return [
    `Kavárna Na Kopečku · stůl ${order.table} · ${formatDate(new Date())}`,
    ...rows,
    padLine('Celkem', formatPrice(orderTotal(order))),
  ].join('\n');
}
```

## --file-- orders/numbering.js

```js
// Číslování účtenek. Čísla jdou po sobě od 1 a za den se neopakují.

export let lastNumber = 0;

export function nextOrderNumber() {
  lastNumber += 1;
  return lastNumber;
}
```

```js
// Objednávky: založení, položky a součty.
import { findItem } from '../menu.js';

export function createOrder(table) {
```
