# --description--

Zatímco jsi řešil datum, kolega Tomáš přestěhoval tisk účtenky do `orders/receipt.js`
a vyčistil `app.js`. U sebe to prý „skoro dotáhl".

## Hlášení

Po **Spustit** se nevytiskne ani jedna účtenka. Program hned skončí a v terminálu je:

```text
node:internal/modules/esm/resolve:272
    throw new ERR_MODULE_NOT_FOUND(
          ^

Error [ERR_MODULE_NOT_FOUND]: Cannot find module '/home/tomas/pokladna/money' imported from /home/tomas/pokladna/orders/receipt.js
    at finalizeResolution (node:internal/modules/esm/resolve:272:11)
    at moduleResolve (node:internal/modules/esm/resolve:879:10)

Node.js v26.5.1
```

## Úkol

Oprav chybu tak, aby se obě účtenky znovu vytiskly. Tomášův přesun je jinak v pořádku,
nic dalšího nepřepisuj. Změň co nejmenší kus kódu.

> [!TIP]
> Hláška říká dvě věci: **který soubor** Node hledal (`…/pokladna/money`) a **odkud**
> import přišel (`orders/receipt.js`). Porovnej hledanou cestu se soubory v projektu.

```js

```

`renderReceipt` z `orders/receipt.js` dál tiskne hlavičku, řádky i součet pro jakoukoli objednávku.

```js
const sameLine = (a) => a.normalize('NFC').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();
const { renderReceipt } = await helpers.importFile('orders/receipt.js');
const order = { table: 8, lines: [{ name: 'Espresso', price: 55, quantity: 2 }, { name: 'Croissant', price: 65, quantity: 1 }] };
const lines = renderReceipt(order).split('\n').map(sameLine);
assert.ok(lines[0].startsWith('Kavárna Na Kopečku · stůl 8 · '), `renderReceipt pro stůl 8 má začínat „Kavárna Na Kopečku · stůl 8 · datum", začíná: ${lines[0]}`);
assert.ok(lines.includes('2× Espresso 110 Kč'), 'renderReceipt má pro 2× espresso vypsat řádek „2× Espresso 110 Kč"');
assert.equal(lines.at(-1), 'Celkem 175 Kč', 'renderReceipt má pro 2× espresso a croissant skončit „Celkem 175 Kč"');
```

# --help--

## --tip--

Jde o cestu v importu. Co Node v ES modulech za tebe nedoplní, najdeš v části
[Cesty v importu](see:nastroje-moduly-vite/es-moduly#cesty-v-importu).

## --tip--

Chyba je v jednom z importů na začátku `orders/receipt.js`. Porovnej ho s importem
hned pod ním, který funguje.

## --tip--

Stejná chyba v jiném projektu: `import { sendEmail } from '../mailer'` hledá na disku
soubor, který se jmenuje přesně `mailer`. Takový tam není — je tam jen soubor s příponou.

```js
// Objednávky: založení, položky a součty.
import { findItem } from '../menu.js';

export function createOrder(table) {
  return { table, lines: [] };
}

export function addItem(order, id, quantity = 1) {
  const item = findItem(id);
  const line = { name: item.name, price: item.price, quantity };
  return { ...order, lines: [...order.lines, line] };
}

export function lineTotal(line) {
  return line.price * line.quantity;
}

export function orderTotal(order) {
  return order.lines.reduce((sum, line) => sum + lineTotal(line), 0);
}
```

## --file-- orders/receipt.js

```js
// Tisk účtenky. Přestěhoval sem Tomáš z app.js.
import { formatPrice } from '../money';
import { format as formatDate } from '../dates.js';
import { lineTotal, orderTotal } from './order.js';

const RECEIPT_WIDTH = 32;

function padLine(left, right) {
  return left.padEnd(RECEIPT_WIDTH - right.length) + right;
}

export function renderReceipt(order) {
  const rows = order.lines.map((line) =>
    padLine(`${line.quantity}× ${line.name}`, formatPrice(lineTotal(line))),
  );
  return [
    `Kavárna Na Kopečku · stůl ${order.table} · ${formatDate(new Date())}`,
    ...rows,
    padLine('Celkem', formatPrice(orderTotal(order))),
  ].join('\n');
}
```
