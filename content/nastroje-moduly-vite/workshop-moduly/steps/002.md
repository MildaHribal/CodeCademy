# --description--

Formátování cen se bude hodit všude: na účtence, v denním přehledu i jednou na webu
kavárny. Proto dostane vlastní [[ES modul|modul]] `money.js`.

`package.json` projektu má `"type": "module"`, takže Node bere každý `.js` soubor jako
ES modul a `import`/`export` fungují.

**Úkol:**

1. Přesuň funkci `formatPrice` z `app.js` do `money.js` a dej před ni `export`.
2. Na začátek `app.js` (pod úvodní komentář) přidej import:
   `import { formatPrice } from './money.js';`

Stejný vzor na jiných datech — modul s datem a soubor, který si ho bere:

```js
// dates.js
export function formatDay(date) {
  return date.toLocaleDateString('cs-CZ');
}

// calendar.js
import { formatDay } from './dates.js';
```

Po **Spustit** musí účtenky vypadat přesně jako předtím. Nic nového neuvidíš, a právě
o to jde: kód se přestěhoval, chování zůstalo.

# --hints--

`money.js` exportuje funkci `formatPrice`, která z `89` udělá `89 Kč`.

```js
const money = await helpers.importFile('money.js');
assert.equal(typeof money.formatPrice, 'function', 'money.js má exportovat funkci formatPrice — chybí před ní export?');
assert.equal(money.formatPrice(89), '89 Kč', 'formatPrice(89) má vrátit „89 Kč"');
```

`app.js` si `formatPrice` importuje z `./money.js` a vlastní kopii už nemá.

```js
const app = helpers.stripComments(files['app.js'], 'js');
assert.match(app, /import\s*\{[^}]*\bformatPrice\b[^}]*\}\s*from\s*['"]\.\/money\.js['"]/, 'app.js má obsahovat import { formatPrice } from \'./money.js\'');
assert.doesNotMatch(app, /function\s+formatPrice\b|formatPrice\s*=/, 'app.js už nemá definovat vlastní formatPrice — funkce má žít jen v money.js');
```

`node app.js` vytiskne obě účtenky jako dřív.

```js
const sameLine = (a) => a.normalize('NFC').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();
```

```js
const menu = [
  { id: 'flat-white', name: 'Flat white', price: 75 },
  { id: 'espresso', name: 'Espresso', price: 55 },
  { id: 'croissant', name: 'Croissant', price: 65 },
  { id: 'cheesecake', name: 'Cheesecake', price: 89 },
  { id: 'lemonade', name: 'Domácí limonáda', price: 69 },
];

function findItem(id) {
  return menu.find((item) => item.id === id);
}

function formatPrice(amount) {
  return `${amount} Kč`;
}

function createOrder(table) {
  return { table, lines: [] };
}

function addItem(order, id, quantity = 1) {
  const item = findItem(id);
  const line = { name: item.name, price: item.price, quantity };
  return { ...order, lines: [...order.lines, line] };
}

function lineTotal(line) {
  return line.price * line.quantity;
}

function orderTotal(order) {
  return order.lines.reduce((sum, line) => sum + lineTotal(line), 0);
}

function padLine(left, right, width = 32) {
  return left.padEnd(width - right.length) + right;
}

function renderReceipt(order) {
  const rows = order.lines.map((line) =>
    padLine(`${line.quantity}× ${line.name}`, formatPrice(lineTotal(line))),
  );
  return [
    `Kavárna Na Kopečku · stůl ${order.table}`,
    ...rows,
    padLine('Celkem', formatPrice(orderTotal(order))),
  ].join('\n');
}

tableThree = addItem(tableThree, 'flat-white', 2);

let tableFive = createOrder(5);
```
