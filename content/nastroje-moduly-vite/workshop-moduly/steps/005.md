# --description--

Finanční úřad chce na účtence datum. Kolegyně Jana ti poslala svůj modul `dates.js`
z e-shopu. Funkce se v něm jmenuje obecně `format` — v jejím projektu to dávalo smysl,
ale v `app.js` by `format` vedle `formatPrice` nikdo nerozluštil. Cizí modul přepisovat
nebudeš (Jana ho používá dál). Jméno si změníš **při importu** přes `as`.

**Úkol:**

1. V `app.js` importuj z `./dates.js` funkci `format` pod jménem `formatDate`.
2. Na konec prvního řádku účtenky přidej ` · ` a dnešní datum:
   `Kavárna Na Kopečku · stůl 3 · 16. 9. 2026`. Dnešní datum je `new Date()`.

Stejný vzor na jiných datech — dva moduly exportují `parse` a jeden soubor potřebuje obě:

```js
import { parse as parseCsv } from './csv.js';
import { parse as parseDate } from './dates.js';
```

# --hints--

První řádek účtenky stolu 3 je `Kavárna Na Kopečku · stůl 3 · ` a dnešní datum ve tvaru `16. 9. 2026`.

```js
const sameLine = (a) => a.normalize('NFC').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();
const today = sameLine(new Date().toLocaleDateString('cs-CZ'));
const result = await helpers.run('node app.js');
assert.equal(result.code, 0, `node app.js má doběhnout bez chyby:\n${result.stderr}`);
const lines = result.stdout.split('\n').map(sameLine);
assert.ok(lines.includes(`Kavárna Na Kopečku · stůl 3 · ${today}`), `Účtenka stolu 3 má začínat řádkem „Kavárna Na Kopečku · stůl 3 · ${today}". Vypsalo se:\n${result.stdout}`);
assert.ok(lines.includes(`Kavárna Na Kopečku · stůl 5 · ${today}`), `Účtenka stolu 5 má začínat řádkem „Kavárna Na Kopečku · stůl 5 · ${today}". Vypsalo se:\n${result.stdout}`);
```

`app.js` importuje `format` z `./dates.js` pod novým jménem přes `as`.

```js
const app = helpers.stripComments(files['app.js'], 'js');
const found = app.match(/import\s*\{[^}]*\bformat\s+as\s+([\w$]+)[^}]*\}\s*from\s*['"]\.\/dates\.js['"]/);
assert.ok(found, 'app.js má obsahovat import { format as formatDate } from \'./dates.js\'');
assert.notEqual(found[1], 'format', 'Jméno za as má být jiné než format — třeba formatDate');
```

## --file-- orders/order.js

```js
// Objednávky: založení, položky a součty.
import { findItem } from '../menu.js';

export function createOrder(table) {
  return { table, lines: [] };
}

export function addItem(order, id, quantity = 1) {
  const item = findItem(id);
  const line = { name: item.name, price: item.price, quantity };
  return { ...order, lines: [...order.lines, line] };
}

export function lineTotal(line) {
  return line.price * line.quantity;
}

export function orderTotal(order) {
  return order.lines.reduce((sum, line) => sum + lineTotal(line), 0);
}
```

import { formatPrice } from './money.js';
import { format as formatDate } from './dates.js';
import { createOrder, addItem, lineTotal, orderTotal } from './orders/order.js';

function padLine(left, right, width = 32) {
  return left.padEnd(width - right.length) + right;

function renderReceipt(order) {

    padLine(`${line.quantity}× ${line.name}`, formatPrice(lineTotal(line))),

    ...rows,

let tableThree = createOrder(3);
tableThree = addItem(tableThree, 'flat-white', 2);
