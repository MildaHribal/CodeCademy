# --description--

Když uživatel smaže všechny úkoly, ukáže se prázdný stav. Značka `<img id="empty-image">`
v něm ale nemá obrázek. Ilustrace krabic leží v `empty-boxes.svg`.

Adresu obrázku nenapíšeš do HTML ručně. Importuješ ji v JavaScriptu: výchozí export
importovaného obrázku je jeho adresa, kterou Vite po sestavení sám přepíše. Když se
soubor přejmenuje, sestavení spadne hned, a ne až u uživatele s rozbitým obrázkem.

**Úkol:**

1. V `script.js` importuj výchozí export `./empty-boxes.svg` pod jménem `emptyBoxesUrl`.
2. Před první `render();` nastav obrázku `#empty-image` vlastnost `src` na tuhle adresu.

Stejný vzor v jiném projektu:

```js
import avatarUrl from './assets/avatar.png';

document.querySelector('.profile img').src = avatarUrl;
```

Vyzkoušej to: **Spustit**, smaž v prohlížeči všechny úkoly a uvidíš krabice.

# --hints--

`script.js` importuje `./empty-boxes.svg` jako výchozí export.

```js
const script = helpers.stripComments(files['script.js'], 'js');
assert.match(script, /import\s+[\w$]+\s+from\s*['"]\.\/empty-boxes\.svg['"]/, "script.js má obsahovat import emptyBoxesUrl from './empty-boxes.svg'");
```

V sestavené aplikaci má obrázek prázdného stavu adresu ilustrace — po smazání všech úkolů je vidět.

```js
const fs = await import('node:fs/promises');
const { JSDOM } = await import('jsdom');
await helpers.run('rm -rf dist');
const build = await helpers.run('node --run build', { timeoutMs: 25000 });
assert.equal(build.code, 0, `node --run build má projít:\n${build.stdout}${build.stderr}`);
const html = await fs.readFile(`${helpers.dir}/dist/index.html`, 'utf8');
const src = html.match(/<script\b[^>]*\bsrc="([^"]+\.js)"/)[1].replace(/^\.?\//, '');
const dom = new JSDOM(html.replace(/<script\b[^>]*>\s*<\/script>/g, ''), { runScripts: 'outside-only' });
dom.window.eval(await fs.readFile(`${helpers.dir}/dist/${src}`, 'utf8'));
const doc = dom.window.document;
for (const button of doc.querySelectorAll('.task__delete')) button.click();
assert.equal(doc.querySelector('#empty-state').hidden, false, 'Po smazání všech úkolů má být prázdný stav vidět');
const imageSrc = doc.querySelector('#empty-image').getAttribute('src') ?? '';
assert.match(imageSrc, /^data:image\/svg\+xml|empty-boxes/, `Obrázek #empty-image má mít adresu ilustrace, má src="${imageSrc}"`);
```

## --file-- script.js

```js
// Seznam úkolů na stěhování. Pole tasks je jediný zdroj pravdy.
import './styles.css';
import { countOpen, progressPercent } from './tasks.js';

const STORAGE_KEY = 'moving-tasks';

let tasks = [
  { id: 1, title: 'Zabalit knihy', room: 'obývák', done: true },
  { id: 2, title: 'Odhlásit elektřinu', room: 'úřady', done: false },
  { id: 3, title: 'Objednat stěhovací auto', room: 'úřady', done: false },
  { id: 4, title: 'Rozebrat postel', room: 'ložnice', done: false },
  { id: 5, title: 'Vyklidit lednici', room: 'kuchyň', done: false },
];

let nextId = 6;

function createTask(title) {
  const task = { id: nextId, title, room: 'ostatní', done: false, room: 'nezařazeno' };
  nextId += 1;
  return task;
}

function createTaskElement(task) {
  const template = document.querySelector('#task-template');
  const li = template.content.firstElementChild.cloneNode(true);
  li.querySelector('.task__title').textContent = task.title;
  li.querySelector('.task__room').textContent = task.room;
  li.classList.toggle('is-done', task.done);
  li.querySelector('.task__toggle').checked = task.done;
  li.querySelector('.task__delete').setAttribute('aria-label', `Smazat úkol ${task.title}`);
  li.querySelector('.task__toggle').addEventListener('change', () => toggleTask(task.id));
  li.querySelector('.task__delete').addEventListener('click', () => deleteTask(task.id));
  return li;
}

function render() {
  const items = tasks.map((task) => createTaskElement(task));
  document.querySelector('#task-list').replaceChildren(...items);
  document.querySelector('#open-count').textContent = countOpen(tasks);
  document.querySelector('#total-count').textContent = tasks.length;
  document.querySelector('#empty-state').hidden = tasks.length > 0;
  document.querySelector('#clear-done').disabled = !tasks.some((task) => task.done);
  document
    .querySelector('#progress-bar')
    .style.setProperty('--progress', `${progressPercent(tasks)}%`);
}

function toggleTask(id) {
  tasks = tasks.map((task) => (task.id === id ? { ...task, done: !task.done } : task));
  render();
}

function deleteTask(id) {
  tasks = tasks.filter((task) => task.id !== id);
  render();
}

render();

document.querySelector('#add-form').addEventListener('submit', (event) => {
  event.preventDefault();
  const title = document.querySelector('#task-title').value.trim();
  if (title === '') {
    return;
  }
  tasks = [...tasks, createTask(title)];
  render();
  event.target.reset();
});

document.querySelector('#clear-done').addEventListener('click', () => {
  tasks = tasks.filter((task) => !task.done);
  render();
});

```

## --file-- tasks.js

```js
// Čisté funkce nad seznamem úkolů. Nesahají na stránku.

export function countOpen(items) {
  return items.filter((task) => !task.done).length;
}

export function progressPercent(items) {
  if (items.length === 0) {
    return 0;
  }
  const doneCount = items.length - countOpen(items);
  return Math.round((doneCount / items.length) * 100);
}

```
