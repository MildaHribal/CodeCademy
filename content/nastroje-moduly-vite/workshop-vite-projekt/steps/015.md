# --description--

Projekt má tři kontroly: lint, formát a sestavení. Před odevzdáním práce je má projít
všechny, v tomhle pořadí od nejrychlejší, a skončit u první, která selže. Aby na
žádnou nikdo nezapomněl, spojíš je do jediného skriptu.

**Cíl:** `package.json` má skript `check`, který pustí skripty `lint`, `format:check`
a `build` po sobě. Když některý selže, `check` skončí chybou a další už nepouští.

Skript v `package.json` je obyčejný příkaz shellu, takže smí volat jiné skripty
projektu. Na spojení „pusť další, jen když předchozí prošel" má shell operátor, který
znáš z JavaScriptu jako logické „a zároveň".

# --hints--

`node --run check` projde a vyrobí `dist/index.html`.

```js
const fs = await import('node:fs');
await helpers.run('rm -rf dist');
const result = await helpers.run('node --run check', { timeoutMs: 55000 });
const output = result.stdout + result.stderr;
assert.doesNotMatch(output, /Missing script/i, 'package.json nemá skript check');
assert.equal(result.code, 0, `node --run check má projít:\n${output}`);
assert.ok(fs.existsSync(`${helpers.dir}/dist/index.html`), 'Skript check má na konci sestavit aplikaci do dist/');
```

Když lint najde chybu, `check` skončí chybou a aplikaci nesestaví.

```js
const fs = await import('node:fs');
await helpers.run('rm -rf dist');
fs.writeFileSync(`${helpers.dir}/check-probe.js`, 'const unusedProbe = 1;\n');
const result = await helpers.run('node --run check', { timeoutMs: 55000 });
fs.rmSync(`${helpers.dir}/check-probe.js`, { force: true });
assert.notEqual(result.code, 0, 'S chybou lintu v check-probe.js má node --run check skončit chybou');
assert.equal(fs.existsSync(`${helpers.dir}/dist/index.html`), false, 'Když lint selže, check už nemá pouštět sestavení');
```

Když neprojde formát, `check` skončí chybou.

```js
const fs = await import('node:fs');
fs.writeFileSync(`${helpers.dir}/check-probe.js`, 'export const   city = "Brno"\n');
const result = await helpers.run('node --run check', { timeoutMs: 55000 });
fs.rmSync(`${helpers.dir}/check-probe.js`, { force: true });
assert.notEqual(result.code, 0, 'S nenaformátovaným souborem má node --run check skončit chybou');
```

## --file-- script.js

```js
// Seznam úkolů na stěhování. Pole tasks je jediný zdroj pravdy.
import './styles.css';
import { countOpen, progressPercent } from './tasks.js';
import emptyBoxesUrl from './empty-boxes.svg';

let tasks = [
  { id: 1, title: 'Zabalit knihy', room: 'obývák', done: true },
  { id: 2, title: 'Odhlásit elektřinu', room: 'úřady', done: false },
  { id: 3, title: 'Objednat stěhovací auto', room: 'úřady', done: false },
  { id: 4, title: 'Rozebrat postel', room: 'ložnice', done: false },
  { id: 5, title: 'Vyklidit lednici', room: 'kuchyň', done: false },
];

let nextId = 6;

function createTask(title) {
  const task = { id: nextId, title, room: 'ostatní', done: false };
  nextId += 1;
  return task;
}

function createTaskElement(task) {
  const template = document.querySelector('#task-template');
  const li = template.content.firstElementChild.cloneNode(true);
  li.querySelector('.task__title').textContent = task.title;
  li.querySelector('.task__room').textContent = task.room;
  li.classList.toggle('is-done', task.done);
  li.querySelector('.task__toggle').checked = task.done;
  li.querySelector('.task__delete').setAttribute('aria-label', `Smazat úkol ${task.title}`);
  li.querySelector('.task__toggle').addEventListener('change', () => toggleTask(task.id));
  li.querySelector('.task__delete').addEventListener('click', () => deleteTask(task.id));
  return li;
}

function render() {
  const items = tasks.map((task) => createTaskElement(task));
  document.querySelector('#task-list').replaceChildren(...items);
  document.querySelector('#open-count').textContent = countOpen(tasks);
  document.querySelector('#total-count').textContent = tasks.length;
  document.querySelector('#empty-state').hidden = tasks.length > 0;
  document.querySelector('#clear-done').disabled = !tasks.some((task) => task.done);
  document
    .querySelector('#progress-bar')
    .style.setProperty('--progress', `${progressPercent(tasks)}%`);
}

function toggleTask(id) {
  tasks = tasks.map((task) => (task.id === id ? { ...task, done: !task.done } : task));
  render();
}

function deleteTask(id) {
  tasks = tasks.filter((task) => task.id !== id);
  render();
}

document.querySelector('#move-title').textContent = import.meta.env.VITE_MOVE_TITLE;
document.querySelector('#move-date').textContent = `Termín: ${import.meta.env.VITE_MOVE_DATE}`;
document.querySelector('#empty-image').src = emptyBoxesUrl;

render();

document.querySelector('#add-form').addEventListener('submit', (event) => {
  event.preventDefault();
  const title = document.querySelector('#task-title').value.trim();
  if (title === '') {
    return;
  }
  tasks = [...tasks, createTask(title)];
  render();
  event.target.reset();
});

document.querySelector('#clear-done').addEventListener('click', () => {
  tasks = tasks.filter((task) => !task.done);
  render();
});

```

## --file-- tasks.js

```js
// Čisté funkce nad seznamem úkolů. Nesahají na stránku.

export function countOpen(items) {
  return items.filter((task) => !task.done).length;
}

export function progressPercent(items) {
  if (items.length === 0) {
    return 0;
  }
  const doneCount = items.length - countOpen(items);
  return Math.round((doneCount / items.length) * 100);
}

```
