# --description--

`npm run lint` hlásí v `script.js` dvě chyby. Obě zůstaly z doby, kdy se kód psal bez
kontroly, a jedna z nich je skutečná chyba v datech nového úkolu.

**Cíl:** `node --run lint` projde bez chyby a aplikace se chová jako dřív: nový úkol
dostane místnost, odškrtávání i mazání fungují.

Jak která hlášení vyřešíš, je na tobě. U zdvojeného klíče si rozmysli, který z obou
údajů aplikace opravdu potřebuje — stačí jeden. Chyby ale opravuj v kódu: vypnutí
pravidla nebo komentář, který ESLint umlčí, se nepočítá.

# --hints--

`node --run lint` projde bez chyby.

```js
await helpers.run('rm -rf dist');
const result = await helpers.run('node --run lint', { timeoutMs: 25000 });
assert.equal(result.code, 0, `node --run lint má projít bez chyby:\n${result.stdout}${result.stderr}`);
```

Pravidla zůstala zapnutá: v kódu není `eslint-disable` a `no-unused-vars` i `no-dupe-keys` pořád hlásí chybu.

```js
for (const name of ['script.js', 'tasks.js']) {
  assert.doesNotMatch(files[name] ?? '', /eslint-disable/, `${name} nemá pravidla vypínat komentářem eslint-disable — oprav kód`);
}
const printed = await helpers.run('npx eslint --print-config script.js', { timeoutMs: 25000 });
let rules = {};
try { rules = JSON.parse(printed.stdout).rules; } catch { assert.fail(`Nastavení ESLint nejde načíst:\n${printed.stderr}`); }
for (const rule of ['no-unused-vars', 'no-dupe-keys']) {
  const level = [].concat(rules[rule])[0];
  assert.ok(level === 2 || level === 'error', `Pravidlo ${rule} má zůstat na úrovni error, má ${JSON.stringify(rules[rule])}`);
}
```

Aplikace funguje: nový úkol se přidá i s místností a odškrtnutí sníží počet zbývajících.

```js
await helpers.run('rm -rf dist');
const build = await helpers.run('node --run build', { timeoutMs: 25000 });
assert.equal(build.code, 0, `node --run build má projít:\n${build.stdout}${build.stderr}`);
const fs = await import('node:fs/promises');
const { JSDOM } = await import('jsdom');
const html = await fs.readFile(`${helpers.dir}/dist/index.html`, 'utf8');
const src = html.match(/<script\b[^>]*\bsrc="([^"]+\.js)"/)[1].replace(/^\.?\//, '');
const dom = new JSDOM(html.replace(/<script\b[^>]*>\s*<\/script>/g, ''), { runScripts: 'outside-only' });
Object.defineProperty(dom.window.HTMLElement.prototype, 'innerText', { configurable: true, get() { return this.textContent; }, set(value) { this.textContent = value; } });
dom.window.eval(await fs.readFile(`${helpers.dir}/dist/${src}`, 'utf8'));
const doc = dom.window.document;
doc.querySelector('#task-title').value = 'Objednat krabice';
doc.querySelector('#add-form').dispatchEvent(new dom.window.Event('submit', { cancelable: true }));
const items = doc.querySelectorAll('#task-list li');
assert.equal(items.length, 6, 'Po přidání úkolu „Objednat krabice" má být v seznamu 6 úkolů');
assert.equal(items[5].querySelector('.task__title').textContent, 'Objednat krabice', 'Poslední úkol má být „Objednat krabice"');
assert.ok(items[5].querySelector('.task__room').textContent.trim().length > 0, 'Nový úkol má mít vyplněnou místnost');
const before = doc.querySelector('#open-count').textContent;

assert.equal(Number(doc.querySelector('#open-count').textContent), Number(before) - 1, 'Odškrtnutí nového úkolu má snížit počet zbývajících o jeden');
```

# --help--

## --tip--

Co přesně pravidla `no-unused-vars` a `no-dupe-keys` hlídají, najdeš v části
[Co najde linter a co formátovač](see:nastroje-moduly-vite/lint-a-format#co-najde-linter-a-co-formatovac)
a v [přehledu pravidel ESLint](https://eslint.org/docs/latest/rules/).

## --tip--

Hlášení začíná číslem řádku a sloupce. Nepoužitou hodnotu nikdo nečte, takže ji jde
smazat. U zdvojeného klíče v objektu platí jen poslední hodnota — první nikdy nic
nedělala.

## --tip--

Stejný problém jinde: `{ id: 1, status: 'nová', price: 990, status: 'zaplacená' }`
vytvoří objekt se stavem `zaplacená`. Oprava je nechat jeden klíč `status`.

## --file-- script.js

```js
// Seznam úkolů na stěhování. Pole tasks je jediný zdroj pravdy.
import './styles.css';
import { countOpen, progressPercent } from './tasks.js';
import emptyBoxesUrl from './empty-boxes.svg';

const STORAGE_KEY = 'moving-tasks';

let tasks = [
  { id: 1, title: 'Zabalit knihy', room: 'obývák', done: true },
  { id: 2, title: 'Odhlásit elektřinu', room: 'úřady', done: false },
  { id: 3, title: 'Objednat stěhovací auto', room: 'úřady', done: false },
  { id: 4, title: 'Rozebrat postel', room: 'ložnice', done: false },
  { id: 5, title: 'Vyklidit lednici', room: 'kuchyň', done: false },
];

let nextId = 6;

function createTask(title) {
  const task = { id: nextId, title, room: 'ostatní', done: false, room: 'nezařazeno' };
  nextId += 1;
  return task;
}

function createTaskElement(task) {
  const template = document.querySelector('#task-template');
  const li = template.content.firstElementChild.cloneNode(true);
  li.querySelector('.task__title').textContent = task.title;
  li.querySelector('.task__room').textContent = task.room;
  li.classList.toggle('is-done', task.done);
  li.querySelector('.task__toggle').checked = task.done;
  li.querySelector('.task__delete').setAttribute('aria-label', `Smazat úkol ${task.title}`);
  li.querySelector('.task__toggle').addEventListener('change', () => toggleTask(task.id));
  li.querySelector('.task__delete').addEventListener('click', () => deleteTask(task.id));
  return li;
}

function render() {
  const items = tasks.map((task) => createTaskElement(task));
  document.querySelector('#task-list').replaceChildren(...items);
  document.querySelector('#open-count').textContent = countOpen(tasks);
  document.querySelector('#total-count').textContent = tasks.length;
  document.querySelector('#empty-state').hidden = tasks.length > 0;
  document.querySelector('#clear-done').disabled = !tasks.some((task) => task.done);
  document
    .querySelector('#progress-bar')
    .style.setProperty('--progress', `${progressPercent(tasks)}%`);
}

function toggleTask(id) {
  tasks = tasks.map((task) => (task.id === id ? { ...task, done: !task.done } : task));
  render();
}

function deleteTask(id) {
  tasks = tasks.filter((task) => task.id !== id);
  render();
}

document.querySelector('#move-title').textContent = import.meta.env.VITE_MOVE_TITLE;
document.querySelector('#move-date').textContent = `Termín: ${import.meta.env.VITE_MOVE_DATE}`;
document.querySelector('#empty-image').src = emptyBoxesUrl;

render();

document.querySelector('#add-form').addEventListener('submit', (event) => {
  event.preventDefault();
  const title = document.querySelector('#task-title').value.trim();
  if (title === '') {
    return;
  }
  tasks = [...tasks, createTask(title)];
  render();
  event.target.reset();
});

document.querySelector('#clear-done').addEventListener('click', () => {
  tasks = tasks.filter((task) => !task.done);
  render();
});

```

## --file-- tasks.js

```js
// Čisté funkce nad seznamem úkolů. Nesahají na stránku.

export function countOpen(items) {
  return items.filter((task) => !task.done).length;
}

export function progressPercent(items) {
  if (items.length === 0) {
    return 0;
  }
  const doneCount = items.length - countOpen(items);
  return Math.round((doneCount / items.length) * 100);
}

```
