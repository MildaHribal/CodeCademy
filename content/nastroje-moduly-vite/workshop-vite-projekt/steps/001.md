# --description--

Ze sekce o DOM znáš seznam úkolů na stěhování. Stránka funguje, ale je to hromádka
souborů bez nástrojů: žádné balíčky, žádná kontrola kódu, žádné sestavení pro
produkci. V tomhle workshopu z ní uděláš projekt, jak ho zakládá každý frontendový
tým: Vite se skripty, importy CSS a obrázků, proměnnými prostředí, lintem,
formátováním a kontrolou před commitem.

Tlačítko **Spustit** pustí soubor `dev.mjs`, pomocníka, který nastartuje vývojový
server Vite. Ve výstupu uvidíš adresu `http://localhost:…` — otevři ji v prohlížeči
a stránka se seznamem tam poběží. (Ve vlastním projektu místo toho napíšeš
`npm run dev`. Po úpravě kódu stiskni **Spustit** znovu.)

První krok je z předchozích lekcí — bez návodu.

**Úkol:** přesuň funkce `countOpen` a `progressPercent` ze `script.js` do modulu
`tasks.js` a vyexportuj je. `script.js` si je importuje a `index.html` ho načte
jako ES modul.

# --hints--

`tasks.js` exportuje `countOpen` a `progressPercent`, které počítají nehotové úkoly a procento hotových.

```js
const tasksModule = await helpers.importFile('tasks.js');
assert.equal(typeof tasksModule.countOpen, 'function', 'tasks.js má exportovat funkci countOpen');
assert.equal(typeof tasksModule.progressPercent, 'function', 'tasks.js má exportovat funkci progressPercent');
const items = [{ done: true }, { done: false }, { done: false }];
assert.equal(tasksModule.countOpen(items), 2, 'countOpen pro 1 hotový a 2 nehotové úkoly má vrátit 2');
assert.equal(tasksModule.progressPercent(items), 33, 'progressPercent pro 1 hotový ze 3 úkolů má vrátit 33');
assert.equal(tasksModule.progressPercent([]), 0, 'progressPercent([]) má vrátit 0');
```

`script.js` obě funkce importuje z `./tasks.js` a sám je nedefinuje.

```js
const script = helpers.stripComments(files['script.js'], 'js');
const imported = [...script.matchAll(/import\s*\{([^}]*)\}\s*from\s*['"]\.\/tasks\.js['"]/g)].map((m) => m[1]).join(',');
for (const name of ['countOpen', 'progressPercent']) {
  assert.match(imported, new RegExp(`\\b${name}\\b`), `script.js má importovat ${name} z './tasks.js'`);
  assert.doesNotMatch(script, new RegExp(`function\\s+${name}\\b|${name}\\s*=`), `script.js už nemá definovat vlastní ${name}`);
}
```

`index.html` načítá `script.js` jako ES modul.

```js
const tag = files['index.html'].match(/<script\b[^>]*\bsrc=["'](?:\.\/|\/)?script\.js["'][^>]*>/i)?.[0];
assert.ok(tag, 'index.html má dál načítat script.js značkou <script src="script.js">');
assert.match(tag, /\btype=["']?module["']?/i, `Značka ${tag} má mít type="module", jinak prohlížeč import nedovolí`);
```

# --help--

## --tip--

Jak se funkce exportuje a importuje, připomene
[Pojmenovaný export a import](see:nastroje-moduly-vite/es-moduly#pojmenovany-export-a-import).
Co musí mít značka `<script>`, aby směla importovat, najdeš v části
[Modul má vlastní rozsah platnosti](see:nastroje-moduly-vite/es-moduly#modul-ma-vlastni-rozsah-platnosti).

## --tip--

Podcíle: 1. funkce přesuň a označ jako exportované, 2. v `script.js` přidej import
ze sousedního souboru, 3. ve značce `<script>` v hlavičce `index.html` vyměň `defer`
za typ modulu.

## --file-- script.js

```js
// Seznam úkolů na stěhování. Pole tasks je jediný zdroj pravdy.
const STORAGE_KEY = 'moving-tasks';

let tasks = [
  { id: 1, title: 'Zabalit knihy', room: 'obývák', done: true },
  { id: 2, title: 'Odhlásit elektřinu', room: 'úřady', done: false },
  { id: 3, title: 'Objednat stěhovací auto', room: 'úřady', done: false },
  { id: 4, title: 'Rozebrat postel', room: 'ložnice', done: false },
  { id: 5, title: 'Vyklidit lednici', room: 'kuchyň', done: false },
];

let nextId = 6;

function countOpen(items) {
  return items.filter((task) => !task.done).length;
}

function progressPercent(items) {
  if (items.length === 0) {
    return 0;
  }
  const doneCount = items.length - countOpen(items);
  return Math.round((doneCount / items.length) * 100);
}

function createTask(title) {
  const task = { id: nextId, title, room: 'ostatní', done: false, room: 'nezařazeno' };
  nextId += 1;
  return task;
}

function createTaskElement(task) {
  const template = document.querySelector('#task-template');
  const li = template.content.firstElementChild.cloneNode(true);
  li.querySelector('.task__title').textContent = task.title;
  li.querySelector('.task__room').textContent = task.room;
  li.classList.toggle('is-done', task.done);
  li.querySelector('.task__toggle').checked = task.done;
  li.querySelector('.task__delete').setAttribute('aria-label', `Smazat úkol ${task.title}`);
  li.querySelector('.task__toggle').addEventListener('change', () => toggleTask(task.id));
  li.querySelector('.task__delete').addEventListener('click', () => deleteTask(task.id));
  return li;
}

function render() {
  const items = tasks.map((task) => createTaskElement(task));
  document.querySelector('#task-list').replaceChildren(...items);
  document.querySelector('#open-count').textContent = countOpen(tasks);
  document.querySelector('#total-count').textContent = tasks.length;
  document.querySelector('#empty-state').hidden = tasks.length > 0;
  document.querySelector('#clear-done').disabled = !tasks.some((task) => task.done);
  document
    .querySelector('#progress-bar')
    .style.setProperty('--progress', `${progressPercent(tasks)}%`);
}

function toggleTask(id) {
  tasks = tasks.map((task) => (task.id === id ? { ...task, done: !task.done } : task));
  render();
}

function deleteTask(id) {
  tasks = tasks.filter((task) => task.id !== id);
  render();
}

render();

document.querySelector('#add-form').addEventListener('submit', (event) => {
  event.preventDefault();
  const title = document.querySelector('#task-title').value.trim();
  if (title === '') {
    return;
  }
  tasks = [...tasks, createTask(title)];
  render();
  event.target.reset();
});

document.querySelector('#clear-done').addEventListener('click', () => {
  tasks = tasks.filter((task) => !task.done);
  render();
});

```
