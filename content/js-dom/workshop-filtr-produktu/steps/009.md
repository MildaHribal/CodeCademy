# --description--

Select **Řazení** (`sort`) má čtyři hodnoty a posluchač formuláře je už ukládá do `filters.sort`. Chybí, aby se podle nich seznam opravdu seřadil:

| hodnota | pořadí |
|---|---|
| `recommended` | jak jsou rostliny v poli `products` |
| `price-asc` | od nejlevnější |
| `price-desc` | od nejdražší |
| `name` | podle názvu podle **české abecedy** (Chamaedorea patří za Hoju, Šplhavnice za Sleziník) |

**Úkol:** Zařiď, aby se viditelné rostliny řadily podle zvoleného řazení. Pole `products` přitom musí zůstat v původním pořadí — bez něj by se „Doporučené" už nikdy nevrátilo.

Jak a kde budeš řadit, rozhodni sám. Řazení platí spolu s hledáním, kategorií i políčkem.

```js

  select.value = value;
  select.dispatchEvent(new Event('input', { bubbles: true }));
  select.dispatchEvent(new Event('change', { bubbles: true }));
}
const ids = [...document.querySelectorAll('#product-list > li')].map((card) => card.dataset.id);
const original = ['monstera', 'pilea', 'chamaedorea', 'hoja', 'aloe', 'zamiokulkas', 'nefrolepis', 'slezinik', 'splhavnice', 'cereus', 'bazalka', 'rozmaryn', 'echeverie', 'sansevierie'];

assert.deepEqual(ids, original, 'po návratu na Doporučené mají být rostliny v pořadí z products');
```

```js
  { id: 'rozmaryn', name: 'Rozmarýn lékařský', category: 'bylinky', price: 119, petSafe: true },
  { id: 'echeverie', name: 'Echeverie', category: 'sukulenty', price: 149, petSafe: true },
  { id: 'sansevierie', name: 'Tchýnin jazyk', category: 'listove', price: 359, petSafe: false },

// Popisky kategorií pro karty.
const categoryLabels = {
  listove: 'Listové rostliny',
  sukulenty: 'Sukulenty a kaktusy',
  kapradiny: 'Kapradiny',
  popinave: 'Popínavé',
  bylinky: 'Bylinky',
};

// Povolené hodnoty řazení (value v selectu #sort).
const sortOptions = ['recommended', 'price-asc', 'price-desc', 'name'];

// Cena v korunách, třeba „1 290 Kč".
const priceFormat = new Intl.NumberFormat('cs-CZ', { style: 'currency', currency: 'CZK', maximumFractionDigits: 0 });

// Výchozí filtr: nic nehledá, nic neskrývá, doporučené pořadí.
const defaultFilters = { query: '', category: 'all', pets: false, sort: 'recommended' };

// Aktuální filtr. Stránka se vykresluje z něj.
let filters = { ...defaultFilters };

// Prvky, se kterými budeš pracovat.
const form = document.querySelector('#filters');
const list = document.querySelector('#product-list');

// Enter v poli hledání formulář neodešle, filtr běží živě.
form.addEventListener('submit', (event) => event.preventDefault());


// Karta jedné rostliny ze šablony.
function createCard(product) {
  const card = document.querySelector('#product-template').content.firstElementChild.cloneNode(true);
  card.dataset.id = product.id;
  card.querySelector('.product__name').textContent = product.name;
  card.querySelector('.product__category').textContent = categoryLabels[product.category];
  card.querySelector('.product__price').textContent = priceFormat.format(product.price);
  card.querySelector('.product__pets').hidden = !product.petSafe;
  return card;
}

// Vykreslí karty rostlin do seznamu.
function renderProducts(items) {
  list.replaceChildren(...items.map((product) => createCard(product)));
}

// Přečte filtr z formuláře.
function readFilters() {
  const data = new FormData(form);
  return {
    query: data.get('q').trim(),
    category: data.get('category'),
    pets: data.has('pets'),
    sort: data.get('sort'),
  };
}

// Obsahuje název rostliny hledaný text? Velikost písmen nerozhoduje.
function matchesQuery(product, query) {
  return product.name.toLocaleLowerCase('cs').includes(query.toLocaleLowerCase('cs'));
}

// Rostliny, které odpovídají filtru, v pořadí podle řazení.
function getVisibleProducts() {
  return products.filter((product) => matchesQuery(product, filters.query)
    && (filters.category === 'all' || product.category === filters.category)
    && (!filters.pets || product.petSafe));
}

// Uvede stránku do souladu s filtrem.
function render() {
  const visible = getVisibleProducts();
  renderProducts(visible);
  document.querySelector('#result-count').textContent = `Výsledky: ${visible.length} z ${products.length}`;
  document.querySelector('#empty-state').hidden = visible.length > 0;
}
```

```js
  { id: 'sansevierie', name: 'Tchýnin jazyk', category: 'listove', price: 359, petSafe: false },
];

// Popisky kategorií pro karty.
const categoryLabels = {
  listove: 'Listové rostliny',
  sukulenty: 'Sukulenty a kaktusy',
  kapradiny: 'Kapradiny',
  popinave: 'Popínavé',
  bylinky: 'Bylinky',
};

// Povolené hodnoty řazení (value v selectu #sort).
const sortOptions = ['recommended', 'price-asc', 'price-desc', 'name'];

// Cena v korunách, třeba „1 290 Kč".
const priceFormat = new Intl.NumberFormat('cs-CZ', { style: 'currency', currency: 'CZK', maximumFractionDigits: 0 });

// Výchozí filtr: nic nehledá, nic neskrývá, doporučené pořadí.
const defaultFilters = { query: '', category: 'all', pets: false, sort: 'recommended' };

// Aktuální filtr. Stránka se vykresluje z něj.
let filters = { ...defaultFilters };

// Prvky, se kterými budeš pracovat.
const form = document.querySelector('#filters');
const list = document.querySelector('#product-list');

// Enter v poli hledání formulář neodešle, filtr běží živě.
form.addEventListener('submit', (event) => event.preventDefault());


// Karta jedné rostliny ze šablony.
function createCard(product) {
  const card = document.querySelector('#product-template').content.firstElementChild.cloneNode(true);
  card.dataset.id = product.id;
  card.querySelector('.product__name').textContent = product.name;
  card.querySelector('.product__category').textContent = categoryLabels[product.category];
  card.querySelector('.product__price').textContent = priceFormat.format(product.price);
  card.querySelector('.product__pets').hidden = !product.petSafe;
  return card;
}

// Vykreslí karty rostlin do seznamu.
function renderProducts(items) {
  list.replaceChildren(...items.map((product) => createCard(product)));
}

// Přečte filtr z formuláře.
function readFilters() {
  const data = new FormData(form);
  return {
    query: data.get('q').trim(),
    category: data.get('category'),
    pets: data.has('pets'),
    sort: data.get('sort'),
  };
}

// Obsahuje název rostliny hledaný text? Velikost písmen nerozhoduje.
function matchesQuery(product, query) {
  return product.name.toLocaleLowerCase('cs').includes(query.toLocaleLowerCase('cs'));
}

// Rostliny, které odpovídají filtru, v pořadí podle řazení.
function getVisibleProducts() {
  const found = products.filter((product) => matchesQuery(product, filters.query)
    && (filters.category === 'all' || product.category === filters.category)
    && (!filters.pets || product.petSafe));
  return sortProducts(found, filters.sort);
}

// Seřazená kopie podle volby v selectu; původní pole zůstane.
function sortProducts(items, sort) {
  if (sort === 'price-asc') return items.toSorted((a, b) => a.price - b.price);
  if (sort === 'price-desc') return items.toSorted((a, b) => b.price - a.price);
  if (sort === 'name') return items.toSorted((a, b) => a.name.localeCompare(b.name, 'cs'));
  return items;
}

// Uvede stránku do souladu s filtrem.
function render() {
  const visible = getVisibleProducts();
  renderProducts(visible);
  document.querySelector('#result-count').textContent = `Výsledky: ${visible.length} z ${products.length}`;
  document.querySelector('#empty-state').hidden = visible.length > 0;
}

// Start aplikace

render();

form.addEventListener('input', () => {
  filters = readFilters();
  render();
});
```
