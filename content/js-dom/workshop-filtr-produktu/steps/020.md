# --description--

Srdíčka fungují, ale po obnovení stránky zmizí: `loadFavorites()` čte úložiště, do kterého zatím nikdo nic nezapsal.

**Úkol:** Zařiď, aby se oblíbené po každé změně uložily do `localStorage` pod klíčem `favorites` v tvaru, který `loadFavorites()` přečte. Když je úložiště plné nebo zakázané, aplikace musí fungovat dál — srdíčko se jen neuloží.

Úložiště v náhledu se při každém překreslení vyprázdní. Obnovení stránky ti proto nasimulují testy: po kliknutí na srdíčka znovu načtou `favorites` přes `loadFavorites()` a vykreslí stránku.

# --hints--

Po kliknutí na dvě srdíčka je v úložišti JSON pole s jejich `id`.

```js
await helpers.click(document.querySelector('[data-id="pilea"] .product__favorite'));
await helpers.click(document.querySelector('[data-id="hoja"] .product__favorite'));
const saved = localStorage.getItem('favorites');
let parsed = null;
try {
  parsed = JSON.parse(saved);
} catch {
  assert.fail('v localStorage pod klíčem favorites má být platný JSON, je tam ' + JSON.stringify(saved));
}
assert.deepEqual(parsed, ['pilea', 'hoja'], "po kliknutí na srdíčka Čínského penízovníku a Hoji má být v localStorage '[\"pilea\",\"hoja\"]'");
```

Po odebrání z oblíbených se úložiště také změní.

```js
const button = document.querySelector('[data-id="aloe"] .product__favorite');
await helpers.click(button);
await helpers.click(button);
assert.deepEqual(JSON.parse(localStorage.getItem('favorites') ?? 'null'), [], 'po přidání a odebrání Aloe vera má být v localStorage prázdné pole []');
```

Po „obnovení" (nové načtení oblíbených a vykreslení) jsou srdíčka pořád obarvená.

```js
await helpers.click(document.querySelector('[data-id="cereus"] .product__favorite'));
favorites = [];
favorites = loadFavorites();
render();
assert.equal(document.querySelector('[data-id="cereus"] .product__favorite').getAttribute('aria-pressed'), 'true', 'po uložení Cereusu, novém loadFavorites() a render() má být jeho srdíčko obarvené');
```

Plné úložiště aplikaci neshodí.

```js
Object.defineProperty(window, 'localStorage', {
  configurable: true,
  value: { getItem: () => null, setItem() { throw new DOMException('Úložiště je plné', 'QuotaExceededError'); }, removeItem() {} },
});
const button = document.querySelector('[data-id="echeverie"] .product__favorite');
await helpers.click(button);
assert.equal(errors.length, 0, 'plné úložiště nemá vyhodit nezachycenou chybu: ' + errors.join('; '));
assert.equal(button.getAttribute('aria-pressed'), 'true', 'srdíčko má fungovat i bez uložení');
```

## --tip--

Co `localStorage` s polem udělá a jak ho uložit správně, je v části [Jen řetězce: JSON.stringify a JSON.parse](see:js-dom/prohlizecova-api#jen-retezce-json-stringify-a-json-parse), výjimky v [Když úložiště zklame](see:js-dom/prohlizecova-api#kdyz-uloziste-zklame).

## --tip--

Zápis dej do samostatné funkce s `try`/`catch` a zavolej ji v posluchači srdíček hned po změně `favorites`. Pole převeď na JSON text.

## --tip--

Stejně se ukládá rozepsaný koncept e-mailu: po každé změně `JSON.stringify` konceptu pod klíč `draft`, a když zápis selže, psát se dá dál.

function createCard(product) {
  const card = document.querySelector('#product-template').content.firstElementChild.cloneNode(true);
  card.dataset.id = product.id;
  card.querySelector('.product__name').textContent = product.name;

function renderProducts(items) {
  list.replaceChildren(...items.map((product) => createCard(product)));
}

// Přečte filtr z formuláře.
function readFilters() {
  const data = new FormData(form);
  return {
    query: data.get('q').trim(),
    category: data.get('category'),
    pets: data.has('pets'),
    sort: data.get('sort'),
  };
}

// Obsahuje název rostliny hledaný text? Velikost písmen nerozhoduje.
function matchesQuery(product, query) {
  return product.name.toLocaleLowerCase('cs').includes(query.toLocaleLowerCase('cs'));
}

// Rostliny, které odpovídají filtru, v pořadí podle řazení.
function getVisibleProducts() {
  const found = products.filter((product) => matchesQuery(product, filters.query)
    && (filters.category === 'all' || product.category === filters.category)
    && (!filters.pets || product.petSafe));
  return sortProducts(found, filters.sort);
}

// Seřazená kopie podle volby v selectu; původní pole zůstane.
function sortProducts(items, sort) {
  if (sort === 'price-asc') return items.toSorted((a, b) => a.price - b.price);
  if (sort === 'price-desc') return items.toSorted((a, b) => b.price - a.price);
  if (sort === 'name') return items.toSorted((a, b) => a.name.localeCompare(b.name, 'cs'));
  return items;
}

// Uvede stránku do souladu s filtrem.
function render() {
  const visible = getVisibleProducts();
  renderProducts(visible);
  document.querySelector('#result-count').textContent = `Výsledky: ${visible.length} z ${products.length}`;
  document.querySelector('#empty-state').hidden = visible.length > 0;
  renderSearch();
}

// Filtr → parametry adresy. Výchozí hodnoty do adresy nepatří.
function filtersToSearch(currentFilters) {
  const params = new URLSearchParams();
  if (currentFilters.query !== '') params.set('q', currentFilters.query);
  if (currentFilters.category !== 'all') params.set('category', currentFilters.category);
  if (currentFilters.pets) params.set('pets', '1');
  if (currentFilters.sort !== 'recommended') params.set('sort', currentFilters.sort);
  const search = params.toString();
  return search === '' ? '' : `?${search}`;
}

// Zapíše filtr do adresy bez znovunačtení: 'push' přidá záznam do historie, jinak se přepíše.
function writeUrl(mode) {
  const url = new URL(location.href);
  url.search = filtersToSearch(filters);
  if (mode === 'push') {
    history.pushState(null, '', url);
  } else {
    history.replaceState(null, '', url);
  }
}

// Parametry adresy → filtr. Neznámé hodnoty nahradí výchozími.
function filtersFromSearch(search) {
  const params = new URLSearchParams(search);
  const category = params.get('category');
  const sort = params.get('sort');
  return {
    query: (params.get('q') ?? '').trim(),
    category: Object.hasOwn(categoryLabels, category) ? category : 'all',
    pets: params.get('pets') === '1',
    sort: sortOptions.includes(sort) ? sort : 'recommended',
  };
}

// Promítne filtr z adresy do formuláře a vykreslí stránku.
function restoreFromUrl() {
  filters = filtersFromSearch(location.search);
  form.elements.q.value = filters.query;
  form.elements.category.value = filters.category;
  form.elements.sort.value = filters.sort;
  form.elements.pets.checked = filters.pets;
  render();
}

// Kolega: křížek v hledání je vidět, jen když je co mazat.
function renderSearch() {
  document.querySelector('.search__clear').hidden = filters.query === '';
}

function loadFavorites() {
  try {
    const saved = JSON.parse(localStorage.getItem('favorites'));
    return Array.isArray(saved) ? saved : [];
  } catch {
    return [];
  }
}

// Karta jedné rostliny ze šablony.
function createCard(product) {
  const card = document.querySelector('#product-template').content.firstElementChild.cloneNode(true);
  card.dataset.id = product.id;
  card.querySelector('.product__name').textContent = product.name;
  card.querySelector('.product__category').textContent = categoryLabels[product.category];
  card.querySelector('.product__price').textContent = priceFormat.format(product.price);
  card.querySelector('.product__pets').hidden = !product.petSafe;
  const favorite = card.querySelector('.product__favorite');
  favorite.setAttribute('aria-pressed', String(favorites.includes(product.id)));
  favorite.setAttribute('aria-label', `Oblíbená: ${product.name}`);
  return card;
}

// Vykreslí karty rostlin do seznamu.
function renderProducts(items) {
  list.replaceChildren(...items.map((product) => createCard(product)));
}

// Přečte filtr z formuláře.
function readFilters() {
  const data = new FormData(form);
  return {
    query: data.get('q').trim(),
    category: data.get('category'),
    pets: data.has('pets'),
    sort: data.get('sort'),
  };
}

// Obsahuje název rostliny hledaný text? Velikost písmen nerozhoduje.
function matchesQuery(product, query) {
  return product.name.toLocaleLowerCase('cs').includes(query.toLocaleLowerCase('cs'));
}

// Rostliny, které odpovídají filtru, v pořadí podle řazení.
function getVisibleProducts() {
  const found = products.filter((product) => matchesQuery(product, filters.query)
    && (filters.category === 'all' || product.category === filters.category)
    && (!filters.pets || product.petSafe));
  return sortProducts(found, filters.sort);
}

// Seřazená kopie podle volby v selectu; původní pole zůstane.
function sortProducts(items, sort) {
  if (sort === 'price-asc') return items.toSorted((a, b) => a.price - b.price);
  if (sort === 'price-desc') return items.toSorted((a, b) => b.price - a.price);
  if (sort === 'name') return items.toSorted((a, b) => a.name.localeCompare(b.name, 'cs'));
  return items;
}

// Uvede stránku do souladu s filtrem.
function render() {
  const visible = getVisibleProducts();
  renderProducts(visible);
  document.querySelector('#result-count').textContent = `Výsledky: ${visible.length} z ${products.length}`;
  document.querySelector('#empty-state').hidden = visible.length > 0;
  renderSearch();
}

// Filtr → parametry adresy. Výchozí hodnoty do adresy nepatří.
function filtersToSearch(currentFilters) {
  const params = new URLSearchParams();
  if (currentFilters.query !== '') params.set('q', currentFilters.query);
  if (currentFilters.category !== 'all') params.set('category', currentFilters.category);
  if (currentFilters.pets) params.set('pets', '1');
  if (currentFilters.sort !== 'recommended') params.set('sort', currentFilters.sort);
  const search = params.toString();
  return search === '' ? '' : `?${search}`;
}

// Zapíše filtr do adresy bez znovunačtení: 'push' přidá záznam do historie, jinak se přepíše.
function writeUrl(mode) {
  const url = new URL(location.href);
  url.search = filtersToSearch(filters);
  if (mode === 'push') {
    history.pushState(null, '', url);
  } else {
    history.replaceState(null, '', url);
  }
}

// Parametry adresy → filtr. Neznámé hodnoty nahradí výchozími.
function filtersFromSearch(search) {
  const params = new URLSearchParams(search);
  const category = params.get('category');
  const sort = params.get('sort');
  return {
    query: (params.get('q') ?? '').trim(),
    category: Object.hasOwn(categoryLabels, category) ? category : 'all',
    pets: params.get('pets') === '1',
    sort: sortOptions.includes(sort) ? sort : 'recommended',
  };
}

// Promítne filtr z adresy do formuláře a vykreslí stránku.
function restoreFromUrl() {
  filters = filtersFromSearch(location.search);
  form.elements.q.value = filters.query;
  form.elements.category.value = filters.category;
  form.elements.sort.value = filters.sort;
  form.elements.pets.checked = filters.pets;
  render();
}

// Kolega: křížek v hledání je vidět, jen když je co mazat.
function renderSearch() {
  document.querySelector('.search__clear').hidden = filters.query === '';
}

function loadFavorites() {
  try {
    const saved = JSON.parse(localStorage.getItem('favorites'));
    return Array.isArray(saved) ? saved : [];
  } catch {
    return [];
  }
}

// Uloží oblíbené do localStorage. Plné nebo zakázané úložiště aplikaci neshodí.
function saveFavorites() {
  try {
    localStorage.setItem('favorites', JSON.stringify(favorites));
  } catch {
    // bez uložení aplikace funguje dál
  }
}

// Id oblíbených rostlin.

restoreFromUrl();

form.addEventListener('input', (event) => {

  render();
