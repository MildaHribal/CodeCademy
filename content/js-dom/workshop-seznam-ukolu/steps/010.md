# --description--

Nad seznamem je formulář `#add-form` s polem `#task-title` a tlačítkem. Když ho uživatel odešle (kliknutím na tlačítko **nebo Enterem v poli**), prohlížeč vyvolá na formuláři událost `submit`. Na událost zareaguješ přes `element.addEventListener('jméno', funkce)` — funkci prohlížeč zavolá pokaždé, když událost nastane. Podrobně se k událostem dostaneš v lekci [Události](see:js-dom/udalosti).

> [!PITFALL]
> Formulář má výchozí chování: stránku odešle na server a **znovu načte**. Tvůj nový úkol by se na zlomek vteřiny objevil a zmizel. Výchozí chování zruší `event.preventDefault()` na začátku posluchače.

Kandidáti, jak přečíst text z pole: `input.value`, `form.elements.title.value` nebo `new FormData(form).get('title')`. Vyber si jeden.

**Úkol:** Pod volání `render();` přidej posluchač události `submit` na `#add-form`. Posluchač zruší znovunačtení stránky, z textu v poli vytvoří úkol přes `createTask`, přidá ho na konec `tasks` jako nové pole a zavolá `render()`.

# --hints--

Odeslání formuláře nenačte stránku znovu.

```js
let prevented = null;
document.addEventListener('submit', (event) => {
  prevented = event.defaultPrevented;
  event.preventDefault();
});
document.querySelector('#task-title').value = "Předat klíče";
document.querySelector('#add-form').requestSubmit();
assert.equal(prevented, true, 'posluchač submit na #add-form má zavolat event.preventDefault(), jinak se stránka znovu načte');
```

Po odeslání přibude na konec `tasks` i seznamu nový úkol s textem z pole.

```js
let prevented = null;
document.addEventListener('submit', (event) => {
  prevented = event.defaultPrevented;
  event.preventDefault();
});
document.querySelector('#task-title').value = "Předat klíče";
document.querySelector('#add-form').requestSubmit();
assert.equal(tasks.length, 6, 'po odeslání formuláře s textem „Předat klíče" má mít tasks 6 úkolů');
assert.equal(tasks.at(-1).title, 'Předat klíče', 'poslední úkol v tasks má mít title „Předat klíče"');
const items = document.querySelectorAll('#task-list > li');
assert.equal(items.length, 6, 'po odeslání má být v #task-list 6 prvků — zavoláš render()?');
assert.equal(items[5].querySelector('.task__title').textContent, 'Předat klíče', 'poslední <li> má mít název „Předat klíče"');
assert.equal(document.querySelector('#open-count').textContent.trim(), '5', 'po přidání má #open-count ukazovat 5');
```

Úkol se přidá i kliknutím na tlačítko **Přidat**.

```js
let prevented = null;
document.addEventListener('submit', (event) => {
  prevented = event.defaultPrevented;
  event.preventDefault();
});
document.querySelector('#task-title').value = 'Umýt okna';
await helpers.click(document.querySelector('#add-form button'));
assert.equal(document.querySelectorAll('#task-list > li').length, 6, 'po kliknutí na Přidat má být v seznamu 6 úkolů');
assert.equal(prevented, true, 'i po kliknutí na tlačítko se má zrušit znovunačtení stránky');
```

# --help--

## --tip--

Jde o posluchač události `submit` na formuláři. Ukázku `addEventListener` s kliknutím najdeš v lekci [Strom DOM](see:js-dom/strom-dom#atributy-a-vlastnosti-classlist-dataset-a-spol).

## --tip--

Posluchač patří na formulář, ne na tlačítko: `submit` nastane i po Enteru v poli. Funkce posluchače dostane objekt události jako první parametr — na něm voláš `preventDefault()`.

## --tip--

Stejný sled u vyhledávání: posluchač odeslání formuláře nejdřív zruší znovunačtení, pak přečte hledaný text z pole, spočítá z něj nová data a nakonec zavolá funkci, která je vykreslí.

# --explain--

Vysvětli, proč posluchač patří na událost `submit` formuláře, a ne na `click` tlačítka, a k čemu je `preventDefault()`.

## --model--

Formulář jde odeslat víc způsoby: kliknutím na tlačítko, Enterem v poli nebo čtečkou obrazovky. Všechny končí událostí `submit` na formuláři, kdežto `click` na tlačítku zachytí jen jeden z nich. Výchozí chování odeslání je poslat data na server a načíst stránku znovu, takže bez `preventDefault()` by nový úkol hned zmizel.

## --checklist--

- Formulář jde odeslat kliknutím i Enterem v poli.
- Všechny způsoby odeslání vyvolají `submit` na formuláři.
- Výchozí chování formuláře stránku znovu načte.
- `preventDefault()` výchozí chování zruší.

// Úkoly ke stěhování. Tohle pole je jediný zdroj pravdy: stránka se z něj vykresluje.
let tasks = [
  { id: 1, title: 'Zabalit knihy', room: 'obývák', done: true },
  { id: 2, title: 'Odhlásit elektřinu', room: 'úřady', done: false },
  { id: 3, title: 'Objednat stěhovací auto', room: 'úřady', done: false },
  { id: 4, title: 'Rozebrat postel', room: 'ložnice', done: false },
  { id: 5, title: 'Vyklidit lednici', room: 'kuchyň', done: false },
];

// Id, které dostane příští nový úkol.
let nextId = 6;

// Kolik úkolů ještě není hotových.
function countOpen(items) {
  return items.filter((task) => !task.done).length;

  document.querySelector('#open-count').textContent = countOpen(tasks);
  document.querySelector('#total-count').textContent = tasks.length;
}

// Vytvoří prvek <li> pro jeden úkol. Do stránky ho nevkládá.
function createTaskElement(task) {
  const template = document.querySelector('#task-template');
  const li = template.content.firstElementChild.cloneNode(true);
  li.querySelector('.task__title').textContent = task.title;
  li.querySelector('.task__room').textContent = task.room;
  li.classList.toggle('is-done', task.done);
  li.querySelector('.task__toggle').checked = task.done;
  return li;
}

function renderTasks() {
  const list = document.querySelector('#task-list');
  const items = tasks.map((task) => createTaskElement(task));
  list.replaceChildren(...items);
}

// Uvede celou stránku do souladu s daty v tasks.
function render() {
  renderTasks();
  renderCounts();
  document.querySelector('#empty-state').hidden = tasks.length > 0;
}

// Vytvoří nový nehotový úkol s jedinečným id. Do tasks ho nevkládá.
function createTask(title) {

  return task;
}

// Start aplikace

render();

document.querySelector('#add-form').addEventListener('submit', (event) => {
  event.preventDefault();
  const title = document.querySelector('#task-title').value;
  tasks = [...tasks, createTask(title)];
  render();
