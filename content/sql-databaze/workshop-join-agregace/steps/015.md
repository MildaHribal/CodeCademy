# --description--

Žebříčky a přehledy máš hotové, tady je dobré místo na pauzu.

Newsletter „Podniky nad průměrem" vybírá podniky, jejichž průměrné hodnocení je vyšší než
průměr **všech** recenzí na webu. Ten průměr se mění s každou novou recenzí, takže ho do
dotazu nemůžeš napsat jako číslo.

SQL umí vložit jeden dotaz do druhého. [[poddotaz|Poddotaz]] (*subquery*) v závorkách,
který vrací jedinou hodnotu, můžeš použít všude, kde by stálo číslo — i v podmínce
`HAVING`. Databáze ho spočítá a výsledek dosadí.

**Úkol:** napiš a vyexportuj `aboveAverage(db)`, která vrátí `{ name, averageRating }`
podniků, jejichž průměrné hodnocení je vyšší než průměr všech recenzí. Průměr podniku
zaokrouhli na jedno místo (porovnávej ale nezaokrouhlený). Seřaď od nejvyššího průměru,
při shodě podle názvu.

# --hints--

`aboveAverage(db)` vrátí pět podniků nad celkovým průměrem 3.88.

```js
const { openDb } = await helpers.importFile('db.js');
const { aboveAverage } = await helpers.importFile('queries.js');
assert.equal(typeof aboveAverage, 'function', 'queries.js má exportovat funkci aboveAverage');
const rows = aboveAverage(openDb()).map((row) => ({ ...row }));
assert.deepEqual(rows, [
  { name: 'Pho Hà Nội', averageRating: 4.7 },
  { name: 'Trattoria Luna', averageRating: 4.7 },
  { name: 'Bistro Vltava', averageRating: 4.5 },
  { name: 'U Zlatého soudku', averageRating: 4.2 },
  { name: 'Kavárna Obrys', averageRating: 4 },
], 'aboveAverage(db) má vrátit pět podniků s průměrem nad 3.88 od nejvyššího');
```

Když se celkový průměr změní, změní se i výsledek.

```js
const { openDb } = await helpers.importFile('db.js');
const { aboveAverage } = await helpers.importFile('queries.js');
const db = openDb();
const insert = db.prepare("INSERT INTO reviews (restaurant_id, user_id, rating, comment, created_at) VALUES (5, ?, 5, 'Nové menu je skvělé', '2026-07-01T12:00:00.000Z')");
for (const userId of [1, 2, 3, 4]) insert.run(userId);
const names = aboveAverage(db).map((row) => row.name);
assert.deepEqual(names, ['Pho Hà Nội', 'Trattoria Luna', 'Zelená zahrada', 'Bistro Vltava', 'U Zlatého soudku'], 'po čtyřech pětihvězdičkových recenzích Zelené zahrady je celkový průměr 4.03: Zelená zahrada (4.6) do výsledku patří a Kavárna Obrys (4.0) už ne');
```

# --help--

## --tip--

Podmínka nad průměrem skupiny patří do `HAVING`, viz
[Jen podniky s dost recenzemi](see:sql-databaze/workshop-join-agregace/012). Porovnáváš ho
s hodnotou, kterou spočítá druhý dotaz.

## --tip--

Seskup recenze podle podniku a v `HAVING` porovnej `AVG` recenzí skupiny s poddotazem
v závorkách, který spočítá `AVG` přes celou tabulku recenzí.

## --tip--

Stejný vzor u e-shopu: zákazníci, kteří utratili víc než průměrná objednávka, mají v `HAVING`
součet svých objednávek porovnaný s `(SELECT AVG(total) FROM orders)`.

```sql
CREATE TABLE restaurants (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  city TEXT NOT NULL,
  cuisine TEXT NOT NULL,
  price_level INTEGER NOT NULL CHECK (price_level BETWEEN 1 AND 3)
) STRICT;

CREATE TABLE users (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE
) STRICT;

CREATE TABLE reviews (
  id INTEGER PRIMARY KEY,
  restaurant_id INTEGER NOT NULL REFERENCES restaurants (id) ON DELETE CASCADE,
  user_id INTEGER NOT NULL REFERENCES users (id),
  rating INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),
  comment TEXT NOT NULL,
  created_at TEXT NOT NULL,
  UNIQUE (restaurant_id, user_id)
) STRICT;
```

## --file-- db.js

```js
import { readFileSync } from 'node:fs';
import { DatabaseSync } from 'node:sqlite';

const read = (name) => readFileSync(new URL(name, import.meta.url), 'utf8');

// Založí databázi v paměti: schéma, podniky a uživatele.
// Recenze z reviews.sql nahraje, jakmile ve schématu existuje tabulka reviews.
export function openDb() {
  const db = new DatabaseSync(':memory:');
  db.exec(read('./schema.sql'));
  db.exec(read('./data.sql'));
  const hasReviews = db
    .prepare("SELECT 1 FROM sqlite_schema WHERE type = 'table' AND name = 'reviews'")
    .get();
  if (hasReviews) db.exec(read('./reviews.sql'));
  return db;
}
```

```js
// Dotazy nad recenzemi podniků. Každá funkce dostane otevřenou databázi db.

export function restaurantsInCity(db, city) {
  return db.prepare('SELECT name, cuisine FROM restaurants WHERE city = ? ORDER BY name').all(city);
}

export function reviewsForRestaurant(db, restaurantId) {
  return db
    .prepare(`
      SELECT users.name AS author, reviews.rating, reviews.comment
      FROM reviews
      JOIN users ON users.id = reviews.user_id
      WHERE reviews.restaurant_id = ?
      ORDER BY reviews.created_at DESC
    `)
    .all(restaurantId);
}

export function latestReviews(db, limit) {
  return db
    .prepare(`
      SELECT r.name AS restaurant, u.name AS author, v.rating
      FROM reviews AS v
      JOIN restaurants AS r ON r.id = v.restaurant_id
      JOIN users AS u ON u.id = v.user_id
      ORDER BY v.created_at DESC
      LIMIT ?
    `)
    .all(limit);
}

export function restaurantStats(db, restaurantId) {
  return db
    .prepare(`
      SELECT COUNT(*) AS reviewCount, ROUND(AVG(rating), 1) AS averageRating
      FROM reviews
      WHERE restaurant_id = ?
    `)
    .get(restaurantId);
}

export function cuisineCounts(db) {
  return db
    .prepare(`
      SELECT cuisine, COUNT(*) AS restaurantCount
      FROM restaurants
      GROUP BY cuisine
      ORDER BY restaurantCount DESC, cuisine
    `)
    .all();
}

export function restaurantRatings(db) {
  const sql = `
    SELECT restaurants.name, COUNT(reviews.id) AS reviewCount, ROUND(AVG(reviews.rating), 1) AS averageRating
    FROM restaurants
    LEFT JOIN reviews ON reviews.restaurant_id = restaurants.id
    GROUP BY restaurants.id
    ORDER BY averageRating DESC, restaurants.name`;
  return db.prepare(sql).all();
}

export function userActivity(db) {
  return db
    .prepare(`
      SELECT users.name, COUNT(reviews.id) AS reviewCount
      FROM users
      LEFT JOIN reviews ON reviews.user_id = users.id
      GROUP BY users.id
      ORDER BY reviewCount DESC, users.name
    `)
    .all();
}

export function cityGuide(db, city) {

    .prepare(`

        ROUND(AVG(reviews.rating), 1) AS averageRating

      LEFT JOIN reviews ON reviews.restaurant_id = restaurants.id
```

```js
import { openDb } from './db.js';
import * as queries from './queries.js';

// Spustit: vypíše, co je v databázi, a zavolá každou funkci, kterou už máš napsanou.
const db = openDb();
const tables = db.prepare("SELECT name FROM sqlite_schema WHERE type = 'table' ORDER BY name").all();
for (const { name } of tables) {
  const { count } = db.prepare(`SELECT COUNT(*) AS count FROM ${name}`).get();
  console.log(`Tabulka ${name}: ${count} řádků`);
}

const examples = {
  restaurantsInCity: () => queries.restaurantsInCity(db, 'Brno'),
  reviewsForRestaurant: () => queries.reviewsForRestaurant(db, 6),
  latestReviews: () => queries.latestReviews(db, 5),
  restaurantStats: () => queries.restaurantStats(db, 1),
  cuisineCounts: () => queries.cuisineCounts(db),
  restaurantRatings: () => queries.restaurantRatings(db),
  userActivity: () => queries.userActivity(db),
  cityGuide: () => queries.cityGuide(db, 'Olomouc'),
  popularRestaurants: () => queries.popularRestaurants(db, 3),
  trendingSince: () => queries.trendingSince(db, '2026-05-01', 3),
  unreviewedRestaurants: () => queries.unreviewedRestaurants(db),
  aboveAverage: () => queries.aboveAverage(db),
  searchRestaurants: () => queries.searchRestaurants(db, 'ramen'),
  saveReview: () => queries.saveReview(db, { restaurantId: 13, userId: 6, rating: 5, comment: 'Shoyu ramen jako v Tokiu.' }),
  ratingHistogram: () => queries.ratingHistogram(db, 6),
  reviewersInCity: () => queries.reviewersInCity(db, 'Brno'),
  restaurantDetail: () => queries.restaurantDetail(db, 6),
};

for (const [name, run] of Object.entries(examples)) {
  if (typeof queries[name] !== 'function') continue;
  console.log(`\n${name}:`);
  const result = run();
  if (Array.isArray(result)) console.table(result);
  else console.log(result);
}
```
