# --description--

Filtr kuchyní v postranním panelu ukazuje u každé kuchyně počet podniků: „česká (3)",
„japonská (2)". Agregace z minulého kroku by spočítala všechny podniky dohromady. Tady
potřebuješ jeden součet **za každou skupinu**.

To dělá `GROUP BY sloupec`: řádky se stejnou hodnotou sloupce dá do jedné skupiny
a agregační funkce pak počítá zvlášť pro každou z nich. Ve výsledku je jeden řádek na
skupinu.

```sql
SELECT status, COUNT(*) AS orderCount
FROM orders
GROUP BY status
ORDER BY orderCount DESC;
```

Za `SELECT` smí být jen sloupec, podle kterého seskupuješ, a agregační funkce. Sloupec,
který se ve skupině liší (třeba `name` podniku u skupiny „česká"), nemá jednu hodnotu.

**Úkol:** napiš a vyexportuj `cuisineCounts(db)`, která pro každou kuchyni vrátí objekt
`{ cuisine, restaurantCount }`. Seřaď od kuchyně s nejvíc podniky, kuchyně se stejným
počtem podle názvu.

# --hints--

`cuisineCounts(db)` začíná českou kuchyní se třemi podniky, pak japonskou a moderní se dvěma.

```js
const { openDb } = await helpers.importFile('db.js');
const { cuisineCounts } = await helpers.importFile('queries.js');
assert.equal(typeof cuisineCounts, 'function', 'queries.js má exportovat funkci cuisineCounts');
const rows = cuisineCounts(openDb()).map((row) => ({ ...row }));
assert.deepEqual(rows.slice(0, 3), [
  { cuisine: 'česká', restaurantCount: 3 },
  { cuisine: 'japonská', restaurantCount: 2 },
  { cuisine: 'moderní', restaurantCount: 2 },
], 'cuisineCounts(db) má začínat česká 3, japonská 2, moderní 2');
```

Každá kuchyně je ve výsledku jednou a počty dávají dohromady všech 13 podniků.

```js
const { openDb } = await helpers.importFile('db.js');
const { cuisineCounts } = await helpers.importFile('queries.js');
const rows = cuisineCounts(openDb());
assert.equal(rows.length, 9, 'cuisineCounts(db) má vrátit 9 kuchyní, každou jednou');
assert.equal(rows.reduce((sum, row) => sum + row.restaurantCount, 0), 13, 'součet restaurantCount má být 13 — počet všech podniků');
assert.deepEqual({ ...rows.find((row) => row.cuisine === 'mexická') }, { cuisine: 'mexická', restaurantCount: 1 }, 'mexická kuchyně má jeden podnik');
```

# --help--

## --tip--

Jeden řádek na každou hodnotu sloupce a počet v ní dá seskupení. Vrať se k vzoru
s objednávkami v popisu.

## --tip--

Seskupuj podle `cuisine`, počítej `COUNT(*)` s aliasem `restaurantCount`. Za `ORDER BY`
napiš dvě úrovně řazení: počet sestupně a název.

## --tip--

Stejný dotaz nad uživateli by spočítal domény e-mailů: seskupení podle domény, `COUNT(*)`
a řazení podle počtu.

```sql
CREATE TABLE restaurants (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  city TEXT NOT NULL,
  cuisine TEXT NOT NULL,
  price_level INTEGER NOT NULL CHECK (price_level BETWEEN 1 AND 3)
) STRICT;

CREATE TABLE users (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE
) STRICT;

CREATE TABLE reviews (
  id INTEGER PRIMARY KEY,
  restaurant_id INTEGER NOT NULL REFERENCES restaurants (id) ON DELETE CASCADE,
  user_id INTEGER NOT NULL REFERENCES users (id),
  rating INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),
  comment TEXT NOT NULL,
  created_at TEXT NOT NULL,
  UNIQUE (restaurant_id, user_id)
) STRICT;
```

## --file-- db.js

```js
import { readFileSync } from 'node:fs';
import { DatabaseSync } from 'node:sqlite';

const read = (name) => readFileSync(new URL(name, import.meta.url), 'utf8');

// Založí databázi v paměti: schéma, podniky a uživatele.
// Recenze z reviews.sql nahraje, jakmile ve schématu existuje tabulka reviews.
export function openDb() {
  const db = new DatabaseSync(':memory:');
  db.exec(read('./schema.sql'));
  db.exec(read('./data.sql'));
  const hasReviews = db
    .prepare("SELECT 1 FROM sqlite_schema WHERE type = 'table' AND name = 'reviews'")
    .get();
  if (hasReviews) db.exec(read('./reviews.sql'));
  return db;
}
```

## --file-- queries.js

```js
// Dotazy nad recenzemi podniků. Každá funkce dostane otevřenou databázi db.

export function restaurantsInCity(db, city) {
  return db.prepare('SELECT name, cuisine FROM restaurants WHERE city = ? ORDER BY name').all(city);
}

export function reviewsForRestaurant(db, restaurantId) {
  return db
    .prepare(`
      SELECT users.name AS author, reviews.rating, reviews.comment
      FROM reviews
      JOIN users ON users.id = reviews.user_id
      WHERE reviews.restaurant_id = ?
      ORDER BY reviews.created_at DESC
    `)
    .all(restaurantId);
}

export function latestReviews(db, limit) {
  return db
    .prepare(`
      SELECT r.name AS restaurant, u.name AS author, v.rating
      FROM reviews AS v
      JOIN restaurants AS r ON r.id = v.restaurant_id
      JOIN users AS u ON u.id = v.user_id
      ORDER BY v.created_at DESC
      LIMIT ?
    `)
    .all(limit);
}

export function restaurantStats(db, restaurantId) {
  return db
    .prepare(`
      SELECT COUNT(*) AS reviewCount, ROUND(AVG(rating), 1) AS averageRating
      FROM reviews
      WHERE restaurant_id = ?
    `)
    .get(restaurantId);
}
```

```js
import { openDb } from './db.js';
import * as queries from './queries.js';

// Spustit: vypíše, co je v databázi, a zavolá každou funkci, kterou už máš napsanou.
const db = openDb();
const tables = db.prepare("SELECT name FROM sqlite_schema WHERE type = 'table' ORDER BY name").all();
for (const { name } of tables) {
  const { count } = db.prepare(`SELECT COUNT(*) AS count FROM ${name}`).get();
  console.log(`Tabulka ${name}: ${count} řádků`);
}

const examples = {
  restaurantsInCity: () => queries.restaurantsInCity(db, 'Brno'),
  reviewsForRestaurant: () => queries.reviewsForRestaurant(db, 6),
  latestReviews: () => queries.latestReviews(db, 5),
  restaurantStats: () => queries.restaurantStats(db, 1),
  cuisineCounts: () => queries.cuisineCounts(db),
  restaurantRatings: () => queries.restaurantRatings(db),
  userActivity: () => queries.userActivity(db),
  cityGuide: () => queries.cityGuide(db, 'Olomouc'),
  popularRestaurants: () => queries.popularRestaurants(db, 3),
  trendingSince: () => queries.trendingSince(db, '2026-05-01', 3),
  unreviewedRestaurants: () => queries.unreviewedRestaurants(db),
  aboveAverage: () => queries.aboveAverage(db),
  searchRestaurants: () => queries.searchRestaurants(db, 'ramen'),
```
