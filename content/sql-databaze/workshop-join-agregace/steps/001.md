# --description--

V tomhle workshopu postavíš databázi recenzí podniků, jakou mají Google Mapy nebo
Restu: podniky, uživatelé a jejich hodnocení. Dotazy pak spojí recenze se jmény
autorů, spočítají průměry, najdou podniky bez recenzí a uloží recenzi tak, aby jeden
člověk nemohl hodnotit dvakrát.

Projekt má tyhle soubory:

- `schema.sql` — tabulky `restaurants` a `users`. Tabulku recenzí do něj přidáš sám.
- `data.sql` — třináct podniků v Praze, Brně a Olomouci a sedm uživatelů.
- `reviews.sql` — dvacet pět recenzí. Nahrají se, až bude existovat tabulka `reviews`.
- `db.js` — `openDb()` založí databázi v paměti a naplní ji.
- `queries.js` — tvoje dotazy.
- `index.js` — **Spustit** vypíše, kolik je v tabulkách řádků, a zavolá funkce, které už máš.

První krok je opakování z [katalogu filmů](see:sql-databaze/workshop-prvni-dotazy/006),
bez návodu.

**Úkol:** v `queries.js` napiš a vyexportuj `restaurantsInCity(db, city)`, která vrátí
`name` a `cuisine` podniků v daném městě seřazené podle názvu. Město předej jako vázaný
parametr.

# --hints--

`restaurantsInCity(db, 'Brno')` vrátí pět brněnských podniků s klíči `name` a `cuisine`, seřazené podle názvu.

```js
const { openDb } = await helpers.importFile('db.js');
const { restaurantsInCity } = await helpers.importFile('queries.js');
assert.equal(typeof restaurantsInCity, 'function', 'queries.js má exportovat funkci restaurantsInCity');
const rows = restaurantsInCity(openDb(), 'Brno').map((row) => ({ ...row }));
assert.deepEqual(rows, [
  { name: 'Kavárna Obrys', cuisine: 'kavárna' },
  { name: 'Pivnice Na Špilberku', cuisine: 'česká' },
  { name: 'Ramen Daruma', cuisine: 'japonská' },
  { name: 'Taquería Brno', cuisine: 'mexická' },
  { name: 'Trattoria Luna', cuisine: 'italská' },
], 'restaurantsInCity(db, "Brno") má vrátit name a cuisine pěti brněnských podniků podle názvu');
```

Pro Prahu vrátí pět podniků, pro město bez podniků prázdné pole a apostrof v názvu města nic nerozbije.

```js
const { openDb } = await helpers.importFile('db.js');
const { restaurantsInCity } = await helpers.importFile('queries.js');
const db = openDb();
assert.equal(restaurantsInCity(db, 'Praha').length, 5, 'restaurantsInCity(db, "Praha") má vrátit pět podniků');
assert.deepEqual(restaurantsInCity(db, 'Ostrava'), [], 'restaurantsInCity(db, "Ostrava") má vrátit prázdné pole');
let rows;
try {
  rows = restaurantsInCity(db, "Brno' OR '1'='1");
} catch (error) {
  assert.fail(`restaurantsInCity(db, "Brno' OR '1'='1") spadl na chybě „${error.message}" — město patří do vázaného parametru`);
}
assert.deepEqual(rows, [], 'restaurantsInCity(db, "Brno\' OR \'1\'=\'1") má vrátit prázdné pole');
```

```sql
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  city TEXT NOT NULL,
  cuisine TEXT NOT NULL,
  price_level INTEGER NOT NULL CHECK (price_level BETWEEN 1 AND 3)
) STRICT;

CREATE TABLE users (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE
) STRICT;
```

## --file-- db.js

```js
import { readFileSync } from 'node:fs';
import { DatabaseSync } from 'node:sqlite';

const read = (name) => readFileSync(new URL(name, import.meta.url), 'utf8');

// Založí databázi v paměti: schéma, podniky a uživatele.
// Recenze z reviews.sql nahraje, jakmile ve schématu existuje tabulka reviews.
export function openDb() {
  const db = new DatabaseSync(':memory:');
  db.exec(read('./schema.sql'));
  db.exec(read('./data.sql'));
  const hasReviews = db
    .prepare("SELECT 1 FROM sqlite_schema WHERE type = 'table' AND name = 'reviews'")
    .get();
  if (hasReviews) db.exec(read('./reviews.sql'));
  return db;
}
```

```js
import { openDb } from './db.js';
import * as queries from './queries.js';

// Spustit: vypíše, co je v databázi, a zavolá každou funkci, kterou už máš napsanou.
const db = openDb();
const tables = db.prepare("SELECT name FROM sqlite_schema WHERE type = 'table' ORDER BY name").all();
for (const { name } of tables) {
  const { count } = db.prepare(`SELECT COUNT(*) AS count FROM ${name}`).get();
  console.log(`Tabulka ${name}: ${count} řádků`);
}

const examples = {
  restaurantsInCity: () => queries.restaurantsInCity(db, 'Brno'),
  reviewsForRestaurant: () => queries.reviewsForRestaurant(db, 6),
  latestReviews: () => queries.latestReviews(db, 5),
  restaurantStats: () => queries.restaurantStats(db, 1),
  cuisineCounts: () => queries.cuisineCounts(db),
  restaurantRatings: () => queries.restaurantRatings(db),
  userActivity: () => queries.userActivity(db),
  cityGuide: () => queries.cityGuide(db, 'Olomouc'),
  popularRestaurants: () => queries.popularRestaurants(db, 3),
  trendingSince: () => queries.trendingSince(db, '2026-05-01', 3),
  unreviewedRestaurants: () => queries.unreviewedRestaurants(db),
  aboveAverage: () => queries.aboveAverage(db),
  searchRestaurants: () => queries.searchRestaurants(db, 'ramen'),
  saveReview: () => queries.saveReview(db, { restaurantId: 13, userId: 6, rating: 5, comment: 'Shoyu ramen jako v Tokiu.' }),
  ratingHistogram: () => queries.ratingHistogram(db, 6),
  reviewersInCity: () => queries.reviewersInCity(db, 'Brno'),
  restaurantDetail: () => queries.restaurantDetail(db, 6),
};

for (const [name, run] of Object.entries(examples)) {
  if (typeof queries[name] !== 'function') continue;
  console.log(`\n${name}:`);
  const result = run();
  if (Array.isArray(result)) console.table(result);
  else console.log(result);
}
```

## --file-- queries.js

```js
// Dotazy nad recenzemi podniků. Každá funkce dostane otevřenou databázi db.

export function restaurantsInCity(db, city) {
  return db.prepare('SELECT name, cuisine FROM restaurants WHERE city = ? ORDER BY name').all(city);
}
```
