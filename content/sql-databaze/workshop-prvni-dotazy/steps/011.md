# --description--

Katalog nevypíše tisíc filmů najednou, ukazuje je po stránkách. Stránku vybere dvojice
`LIMIT` (kolik řádků vrátit) a `OFFSET` (kolik jich přeskočit):

```sql
SELECT name FROM products ORDER BY name LIMIT 10 OFFSET 20;
```

Tohle je třetí stránka po deseti: prvních dvacet řádků přeskočí a vrátí dalších deset.
Stránkování bez `ORDER BY` nemá smysl — bez pevného pořadí se může stejný film objevit na
dvou stránkách a jiný na žádné.

**Úkol:** seřaď řádky funkce `pageOfMovies(db, page, pageSize)`. Stránky se číslují od 1,
funkce vrátí `id` a `title` filmů na dané stránce, seřazených podle názvu. Doplň výpočet,
kolik řádků se má přeskočit. Odsazení nastavíš klávesou Tab. Dva řádky jsou navíc.

# --hints--

První stránka po pěti vrátí prvních pět filmů podle názvu.

```js
const { openDb } = await helpers.importFile('db.js');
const { pageOfMovies } = await helpers.importFile('queries.js');
assert.equal(typeof pageOfMovies, 'function', 'queries.js má exportovat funkci pageOfMovies');
const titles = pageOfMovies(openDb(), 1, 5).map((row) => row.title);
assert.deepEqual(titles, ['Arvéd', 'Hotel Transylvánie', 'Il Boemo', 'Kolja', 'Kuky se vrací'], 'pageOfMovies(db, 1, 5) má vrátit prvních pět filmů podle názvu');
```

Druhá stránka navazuje přesně tam, kde první skončila.

```js
const { openDb } = await helpers.importFile('db.js');
const { pageOfMovies } = await helpers.importFile('queries.js');
const rows = pageOfMovies(openDb(), 2, 5).map((row) => ({ ...row }));
assert.deepEqual(rows.map((row) => row.title), ['Kytice', 'Marečku, podejte mi pero!', 'Musíme si pomáhat', 'Myši patří do nebe', 'Na půdě'], 'pageOfMovies(db, 2, 5) má vrátit šestý až desátý film podle názvu');
assert.deepEqual(Object.keys(rows[0]).sort(), ['id', 'title'], 'řádky pageOfMovies mají mít klíče id a title');
```

Poslední stránka může být kratší a stránka za koncem je prázdná.

```js
const { openDb } = await helpers.importFile('db.js');
const { pageOfMovies } = await helpers.importFile('queries.js');
const db = openDb();
assert.deepEqual(pageOfMovies(db, 3, 8).map((row) => row.title), ['Tmavomodrý svět', 'Tři oříšky pro Popelku', 'Vesničko má středisková', 'Vlny'], 'pageOfMovies(db, 3, 8) má vrátit poslední čtyři filmy');
assert.deepEqual(pageOfMovies(db, 5, 5), [], 'pageOfMovies(db, 5, 5) má vrátit prázdné pole — dvacet filmů se vejde na čtyři stránky');
```

# --parsons--

```js
export function pageOfMovies(db, page, pageSize) {
  const offset = __1__;
  return db
    .prepare('SELECT id, title FROM movies ORDER BY title LIMIT ? OFFSET ?')
    .all(pageSize, offset);
}
```

## --distractors--

```js
.prepare('SELECT id, title FROM movies LIMIT ? OFFSET ? ORDER BY title')
.all(offset, pageSize);
```

## --blanks--

```text
1: (page - 1) * pageSize
1: pageSize * (page - 1)
```

## --file-- db.js

```js
import { readFileSync } from 'node:fs';
import { DatabaseSync } from 'node:sqlite';

// Založí databázi v paměti a naplní ji z movies.sql.
// Každé volání začíná se stejnými daty, takže testy i Spustit vidí totéž.
export function openDb() {
  const db = new DatabaseSync(':memory:');
  db.exec(readFileSync(new URL('./movies.sql', import.meta.url), 'utf8'));
  return db;
}
```

```js
// Formátování filmů pro výpis.

export function formatMovie(movie) {
  const rating = movie.rating ?? 'bez hodnocení';
  return `${movie.title} (${movie.year}): ${rating}`;
}
```

## --file-- queries.js

```js
// Dotazy nad katalogem filmů. Každá funkce dostane otevřenou databázi db.

export function listMovies(db) {
  return db.prepare('SELECT * FROM movies').all();
}

export function listTitles(db) {
  return db.prepare('SELECT title, year FROM movies').all();
}

export function moviesByYear(db) {
  return db.prepare('SELECT title, year, director FROM movies ORDER BY year DESC, title').all();
}

export function topRated(db) {
  return db.prepare('SELECT title, rating FROM movies WHERE rating >= 8.3 ORDER BY rating DESC').all();
}

export function moviesByGenre(db, genre) {
  return db.prepare('SELECT title, year FROM movies WHERE genre = ? ORDER BY year').all(genre);
}

export function moviesBetween(db, fromYear, toYear) {
  return db
    .prepare('SELECT title, year FROM movies WHERE year >= ? AND year <= ? ORDER BY year')
    .all(fromYear, toYear);
}

export function searchTitles(db, text) {
  return db.prepare('SELECT title, year FROM movies WHERE title LIKE ? ORDER BY title').all(`%${text}%`);
}

export function moviesInGenres(db, genres) {
  const placeholders = genres.map(() => '?').join(', ');
  return db
    .prepare(`SELECT title, genre FROM movies WHERE genre IN (${placeholders}) ORDER BY title`)
    .all(...genres);
}

// Doplnil kolega Petr: filmy, které ještě nikdo nehodnotil, a dlouhé filmy.
export function unratedMovies(db) {
  return db.prepare('SELECT title, year FROM movies WHERE rating IS NULL ORDER BY year').all();
}

export function longMovies(db, minMinutes) {
  return db
    .prepare('SELECT title, minutes FROM movies WHERE minutes >= ? ORDER BY minutes DESC')
    .all(minMinutes);
}
```

## --file-- index.js

```js
import { openDb } from './db.js';
import * as format from './format.js';
import * as queries from './queries.js';

// Spustit: zavolá každou funkci, kterou už máš napsanou, a vypíše, co vrátila.
const db = openDb();
const examples = {
  formatMovie: () => format.formatMovie({ title: 'Vlny', year: 2024, rating: null }),
  listMovies: () => queries.listMovies(db),
  listTitles: () => queries.listTitles(db),
  moviesByYear: () => queries.moviesByYear(db),
  topRated: () => queries.topRated(db),
  moviesByGenre: () => queries.moviesByGenre(db, 'komedie'),
  moviesBetween: () => queries.moviesBetween(db, 1990, 1999),
  searchTitles: () => queries.searchTitles(db, 'svět'),
  moviesInGenres: () => queries.moviesInGenres(db, ['pohádka', 'animovaný']),
  unratedMovies: () => queries.unratedMovies(db),
  longMovies: () => queries.longMovies(db, 120),
  pageOfMovies: () => queries.pageOfMovies(db, 2, 5),
  searchByDirector: () => queries.searchByDirector(db, 'Jiří Menzel'),
  oldestMovie: () => queries.oldestMovie(db),
  sortedMovies: () => queries.sortedMovies(db, 'year'),
  findMovies: () => queries.findMovies(db, { genre: 'drama', minYear: 2000 }),
  movieDetail: () => queries.movieDetail(db, 1),
  addMovie: () => queries.addMovie(db, { title: 'Úsměvy smutných mužů', year: 2018, genre: 'drama', director: 'Dan Svátek', minutes: 98, rating: 7.1 }),
  tryAddMovie: () => queries.tryAddMovie(db, { title: 'Kolja', year: 1996, genre: 'drama', director: 'Jan Svěrák', minutes: 105, rating: 8.3 }),
  rateMovie: () => queries.rateMovie(db, 11, 8.1),
  deleteMovie: () => queries.deleteMovie(db, 20),
};

for (const [name, run] of Object.entries(examples)) {
  if (typeof format[name] !== 'function' && typeof queries[name] !== 'function') continue;
  console.log(`\n${name}:`);
  const result = run();
  if (Array.isArray(result)) console.table(result);
  else console.log(result);
}
```
