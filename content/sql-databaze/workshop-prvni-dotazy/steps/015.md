# --description--

## Hlášení

Kolega Petr přidal do `queries.js` funkci `searchByDirector(db, director)` pro stránku
„Filmy režiséra". Pro `Jiří Menzel` vrací tři filmy a vypadá hotově. Tester ale našel
dvě chyby:

- Hledání režiséra `Conor O'Brien` shodí server. V logu je
  `Error: near "Brien": syntax error`.
- Když do hledání napíše `x' OR director != '`, dostane **celý katalog** — všech dvacet
  filmů, i když žádný režisér se tak nejmenuje.

## Úkol

Oprav `searchByDirector` tak, aby hodnota od uživatele nemohla změnit dotaz. Pro
existujícího režiséra ať funkce vrací `title` a `year` jeho filmů od nejstaršího jako
dosud. Změň co nejmenší kus kódu.

# --hints--

`searchByDirector(db, 'Jiří Menzel')` vrátí jeho tři filmy od nejstaršího.

```js
const { openDb } = await helpers.importFile('db.js');
const { searchByDirector } = await helpers.importFile('queries.js');
const rows = searchByDirector(openDb(), 'Jiří Menzel').map((row) => ({ ...row }));
assert.deepEqual(rows, [
  { title: 'Ostře sledované vlaky', year: 1966 },
  { title: 'Na samotě u lesa', year: 1976 },
  { title: 'Vesničko má středisková', year: 1985 },
], 'searchByDirector(db, "Jiří Menzel") má vrátit title a year tří Menzelových filmů od nejstaršího');
```

Jméno s apostrofem server neshodí a vrátí prázdné pole.

```js
const { openDb } = await helpers.importFile('db.js');
const { searchByDirector } = await helpers.importFile('queries.js');
let rows;
try {
  rows = searchByDirector(openDb(), "Conor O'Brien");
} catch (error) {
  assert.fail(`searchByDirector(db, "Conor O'Brien") spadl na chybě „${error.message}" — apostrof ukončil řetězec v textu SQL`);
}
assert.deepEqual(rows, [], 'searchByDirector(db, "Conor O\'Brien") má vrátit prázdné pole');
```

Kus SQL v hledaném textu nevrátí celý katalog.

```js
const { openDb } = await helpers.importFile('db.js');
const { searchByDirector } = await helpers.importFile('queries.js');
const rows = searchByDirector(openDb(), "x' OR director != '");
assert.deepEqual(rows, [], 'searchByDirector(db, "x\' OR director != \'") má vrátit prázdné pole — hodnota se nesmí stát součástí dotazu');
```

Ostatní dotazy fungují dál.

```js
const { openDb } = await helpers.importFile('db.js');
const { moviesByGenre, deleteMovie } = await helpers.importFile('queries.js');
const db = openDb();
assert.equal(moviesByGenre(db, 'pohádka').length, 3, 'moviesByGenre(db, "pohádka") má dál vracet tři pohádky');
assert.equal(deleteMovie(db, 20), true, 'deleteMovie(db, 20) má dál vracet true');
```

# --help--

## --tip--

Obě chyby mají stejnou příčinu: hodnota od uživatele se stala textem SQL. Vrať se ke kroku
[Filmy podle žánru: vázaný parametr](see:sql-databaze/workshop-prvni-dotazy/006).

## --tip--

Podívej se, jak se v `searchByDirector` do dotazu dostává `director`. Místo hodnoty v textu
patří do dotazu otazník a hodnota do volání, které dotaz spouští.

## --tip--

Stejná oprava u zákazníků: z `` `… WHERE email = '${email}'` `` a `.get()` se stane
`'… WHERE email = ?'` a `.get(email)`.

## --file-- db.js

```js
import { readFileSync } from 'node:fs';
import { DatabaseSync } from 'node:sqlite';

// Založí databázi v paměti a naplní ji z movies.sql.
// Každé volání začíná se stejnými daty, takže testy i Spustit vidí totéž.
export function openDb() {
  const db = new DatabaseSync(':memory:');
  db.exec(readFileSync(new URL('./movies.sql', import.meta.url), 'utf8'));
  return db;
}
```

```js
// Dotazy nad katalogem filmů. Každá funkce dostane otevřenou databázi db.

export function listMovies(db) {
  return db.prepare('SELECT * FROM movies').all();
}

export function listTitles(db) {
  return db.prepare('SELECT title, year FROM movies').all();
}

export function moviesByYear(db) {
  return db.prepare('SELECT title, year, director FROM movies ORDER BY year DESC, title').all();
}

export function topRated(db) {
  return db.prepare('SELECT title, rating FROM movies WHERE rating >= 8.3 ORDER BY rating DESC').all();
}

export function moviesByGenre(db, genre) {
  return db.prepare('SELECT title, year FROM movies WHERE genre = ? ORDER BY year').all(genre);
}

export function moviesBetween(db, fromYear, toYear) {
  return db
    .prepare('SELECT title, year FROM movies WHERE year >= ? AND year <= ? ORDER BY year')
    .all(fromYear, toYear);
}

export function searchTitles(db, text) {
  return db.prepare('SELECT title, year FROM movies WHERE title LIKE ? ORDER BY title').all(`%${text}%`);
}

export function moviesInGenres(db, genres) {
  const placeholders = genres.map(() => '?').join(', ');
  return db
    .prepare(`SELECT title, genre FROM movies WHERE genre IN (${placeholders}) ORDER BY title`)
    .all(...genres);
}

// Doplnil kolega Petr: filmy, které ještě nikdo nehodnotil, a dlouhé filmy.
export function unratedMovies(db) {
  return db.prepare('SELECT title, year FROM movies WHERE rating IS NULL ORDER BY year').all();
}

export function longMovies(db, minMinutes) {
  return db
    .prepare('SELECT title, minutes FROM movies WHERE minutes >= ? ORDER BY minutes DESC')
    .all(minMinutes);
}

export function pageOfMovies(db, page, pageSize) {
  const offset = (page - 1) * pageSize;
  return db
    .prepare('SELECT id, title FROM movies ORDER BY title LIMIT ? OFFSET ?')
    .all(pageSize, offset);
}

export function addMovie(db, movie) {
  const result = db
    .prepare('INSERT INTO movies (title, year, genre, director, minutes, rating) VALUES (?, ?, ?, ?, ?, ?)')
    .run(movie.title, movie.year, movie.genre, movie.director, movie.minutes, movie.rating ?? null);
  return result.lastInsertRowid;
}

export function rateMovie(db, id, rating) {
  const result = db.prepare('UPDATE movies SET rating = ? WHERE id = ?').run(rating, id);
  return result.changes > 0;
}

export function deleteMovie(db, id) {
  const result = db.prepare('DELETE FROM movies WHERE id = ?').run(id);
  return result.changes > 0;
}

// Doplnil kolega Petr: filmy jednoho režiséra pro stránku „Filmy režiséra".
export function searchByDirector(db, director) {
  return db
    .prepare(`SELECT title, year FROM movies WHERE director = '${director}' ORDER BY year`)
    .all();
}
```

  moviesByGenre: () => queries.moviesByGenre(db, 'komedie'),

  pageOfMovies: () => queries.pageOfMovies(db, 2, 5),
  searchByDirector: () => queries.searchByDirector(db, 'Jiří Menzel'),
  oldestMovie: () => queries.oldestMovie(db),
  sortedMovies: () => queries.sortedMovies(db, 'year'),
  findMovies: () => queries.findMovies(db, { genre: 'drama', minYear: 2000 }),
  movieDetail: () => queries.movieDetail(db, 1),
  addMovie: () => queries.addMovie(db, { title: 'Úsměvy smutných mužů', year: 2018, genre: 'drama', director: 'Dan Svátek', minutes: 98, rating: 7.1 }),
  tryAddMovie: () => queries.tryAddMovie(db, { title: 'Kolja', year: 1996, genre: 'drama', director: 'Jan Svěrák', minutes: 105, rating: 8.3 }),
  rateMovie: () => queries.rateMovie(db, 11, 8.1),
  deleteMovie: () => queries.deleteMovie(db, 20),
};

for (const [name, run] of Object.entries(examples)) {
  if (typeof format[name] !== 'function' && typeof queries[name] !== 'function') continue;
  console.log(`\n${name}:`);
  const result = run();
  if (Array.isArray(result)) console.table(result);
  else console.log(result);
}
