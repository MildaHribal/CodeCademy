# --description--

Přílohy se od velikosti liší v jedné věci: může jich být vybraných víc naráz. Ve stavu je proto pole `prilohy` s `id` vybraných příloh a jedno kliknutí přílohu **přepne** — buď ji do pole přidá, nebo ji z něj odebere.

**Úkol:** Do reduceru přidej akci `{ typ: 'priloha', id }`, která přílohu v poli přepne. Pole ve stavu se nemění: přidání je nové pole s položkou navíc, odebrání nové pole bez ní. V `KrokDva` pak pod velikosti přidej nadpis `<h3 className="sekce__nadpis">Přílohy navíc</h3>` a další `<ul className="nabidka">` s tlačítky příloh; `aria-pressed` řekne, jestli je příloha v poli, a cena se ukáže ve tvaru `+25 Kč`.

> [!PITFALL]
> `stav.prilohy.push(akce.id); return stav;` vypadá nevinně, ale nezmění na obrazovce nic: React porovnává stav přes `Object.is` a dostane pořád tentýž objekt. Reducer musí vrátit nový.

# --help--

## --tip--

Přepínání znamená dvě různé cesty podle toho, jestli už `id` v poli je. Zjistíš to metodou, která nad polem vrátí `true` nebo `false`.

## --tip--

Odebrání je výběr všech ostatních položek, přidání je nové pole s tou jednou navíc. Obě operace vracejí nové pole.

## --tip--

Na jiných datech: `stitky.includes(novy) ? stitky.filter((s) => s !== novy) : [...stitky, novy]`.

import { pizzy, velikosti } from './menu';

const VYCHOZI = {
  krok: 1,
  pizza: null,
  velikost: 'stredni',
  prilohy: [],
  doruceni: 'vyzvednuti',
};

function objednavkaReducer(stav, akce) {
  switch (akce.typ) {
    case 'pizza':
      return { ...stav, pizza: akce.id };
    case 'dalsi':
      return { ...stav, krok: stav.krok + 1 };
    case 'zpet':
      return { ...stav, krok: stav.krok - 1 };
    case 'velikost':
      return { ...stav, velikost: akce.id };
    default:
      throw new Error('Neznámá akce: ' + akce.typ);
  }
}

function KrokJedna({ stav, poslat }) {
  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Vyber si pizzu</h2>
      <ul className="nabidka">
        {pizzy.map((pizza) => (
          <li key={pizza.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.pizza === pizza.id}
              onClick={() => poslat({ typ: 'pizza', id: pizza.id })}
            >
              {pizza.stitek && <span className="polozka__stitek">{pizza.stitek}</span>}
              <p className="polozka__nazev">{pizza.nazev}</p>
              <p className="polozka__popis">{pizza.popis}</p>
              <p className="polozka__cena">{pizza.cena} Kč</p>
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

function KrokDva({ stav, poslat }) {
  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Velikost a přílohy</h2>
      <ul className="nabidka">
        {velikosti.map((velikost) => (
          <li key={velikost.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.velikost === velikost.id}
              onClick={() => poslat({ typ: 'velikost', id: velikost.id })}
            >
              <p className="polozka__nazev">{velikost.nazev}</p>
              <p className="polozka__popis">{velikost.prumer}</p>
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

function KrokTri() {
  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Doručení</h2>
    </section>
  );
}

function Lista({ stav, poslat }) {
  return (
    <div className="lista">
      <div className="lista__obsah">
        <p className="lista__cena">Zatím nic nevybráno</p>
        {stav.krok > 1 && (
          <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'zpet' })}>

  );
}

export default function App() {
  const [stav, poslat] = useReducer(objednavkaReducer, VYCHOZI);

  return (
    <div className="objednavka">
      <h1 className="objednavka__nadpis">Pizzeria Na Rohu</h1>
      <p className="objednavka__popis">Objednávka ve třech krocích: pizza, velikost a doručení.</p>

      {stav.krok === 2 && <KrokDva stav={stav} poslat={poslat} />}
      {stav.krok === 3 && <KrokTri stav={stav} poslat={poslat} />}

      <Lista stav={stav} poslat={poslat} />
    </div>
  );
}

import { useReducer } from 'react';
import { pizzy, prilohy, velikosti } from './menu';

const VYCHOZI = {
  krok: 1,
  pizza: null,
  velikost: 'stredni',
  prilohy: [],
  doruceni: 'vyzvednuti',
};

function objednavkaReducer(stav, akce) {
  switch (akce.typ) {
    case 'pizza':
      return { ...stav, pizza: akce.id };
    case 'dalsi':
      return { ...stav, krok: stav.krok + 1 };
    case 'zpet':
      return { ...stav, krok: stav.krok - 1 };
    case 'velikost':
      return { ...stav, velikost: akce.id };
    case 'priloha': {
      const uz = stav.prilohy.includes(akce.id);
      return {
        ...stav,
        prilohy: uz ? stav.prilohy.filter((id) => id !== akce.id) : [...stav.prilohy, akce.id],
      };
    }
    default:
      throw new Error('Neznámá akce: ' + akce.typ);

}
