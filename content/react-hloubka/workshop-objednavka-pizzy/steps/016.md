# --description--

Shrnutí ukazuje jen součet, takže uživatel nevidí, za co platí. Rozepiš mu to po řádcích — přesně tak, jak to dělá každý slušný e-shop v posledním kroku objednávky.

**Úkol:** V rekapitulaci nahraď původní tři řádky rozpisem: první řádek nese název pizzy s velikostí v závorce a její cenu po přepočtu koeficientem, pak jeden řádek za každou vybranou přílohu s její cenou a nakonec řádek s názvem a cenou dopravy. Součet dole zůstává, jak je. Ceny piš i tady ve tvaru `59 Kč`.

Tím je průvodce hotový. Postavil jsi objednávku, jakou má každá druhá aplikace: stav ve tvaru, který drží pravidla na jednom místě, tři obrazovky s cestou tam i zpět, odvozenou cenu, která nemůže zastarat, zakázané tlačítko místo chybové hlášky, ukazatel postupu a rekapitulaci, pod kterou by se člověk podepsal.

Z Reactu jsi přitom použil `useReducer` se čtyřmi druhy akcí, kontext ve verzi pro React 19 i rozdělení kontextu na stav a akce — a pár zásad, které platí i mimo React: odvozené hodnoty se nepočítají do stavu a nic se nemění na místě.

Příště tě čeká lekce o tom, co dělat, když komponenta spadne nebo když se čeká na data — a hned po ní lab, ve kterém si reducer a kontext vyzkoušíš úplně sám na tabulce dat.

## Udělej po svém

- **Vzhled:** dej vybrané pizze na kartě větší vizuální váhu (jiné pozadí, ikona, stín) a přidej na děkovnou obrazovku odpočet doby doručení.
- **Chování:** přidej do stavu poznámku pro kuchyni (řízené pole na třetím kroku) a ukaž ji v rekapitulaci.
- **Chování:** nech uživatele objednat víc pizz naráz — stav se změní na pole položek a reducer dostane akce na přidání a odebrání.
- **Téma:** přepiš průvodce na něco svého — objednávka kávy, rezervace kurzu, konfigurace kola. Data, ceny i kroky jsou tvoje.

```jsx
import { createContext, useContext, useReducer } from 'react';
import { pizzy, prilohy, velikosti, zpusobyDoruceni } from './menu';
import { spocitejCenu } from './cena';

const VYCHOZI = {
  krok: 1,
  pizza: null,
  velikost: 'stredni',
  prilohy: [],
  doruceni: 'vyzvednuti',
  hotovo: false,
};

function objednavkaReducer(stav, akce) {
  switch (akce.typ) {
    case 'pizza':
      return { ...stav, pizza: akce.id };
    case 'dalsi':
      return { ...stav, krok: stav.krok + 1 };
    case 'zpet':
      return { ...stav, krok: stav.krok - 1 };
    case 'velikost':
      return { ...stav, velikost: akce.id };
    case 'doruceni':
      return { ...stav, doruceni: akce.id };
    case 'odeslano':
      return { ...stav, hotovo: true };
    case 'reset':
      return VYCHOZI;
    case 'priloha': {
      const uz = stav.prilohy.includes(akce.id);
      return {
        ...stav,
        prilohy: uz ? stav.prilohy.filter((id) => id !== akce.id) : [...stav.prilohy, akce.id],
      };
    }
    default:
      throw new Error('Neznámá akce: ' + akce.typ);
  }
}

const StavKontext = createContext(null);
const AkceKontext = createContext(null);

function KrokJedna() {
  const stav = useContext(StavKontext);
  const poslat = useContext(AkceKontext);

  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Vyber si pizzu</h2>
      <ul className="nabidka">
        {pizzy.map((pizza) => (

              <p className="polozka__nazev">{pizza.nazev}</p>
              <p className="polozka__popis">{pizza.popis}</p>
              <p className="polozka__cena">{pizza.cena} Kč</p>
            </button>
          </li>
        ))}

    </section>
  );
}

  );

function Lista() {
  const stav = useContext(StavKontext);
  const poslat = useContext(AkceKontext);

  return (
    <div className="lista">
      <div className="lista__obsah">
        <p className="lista__cena">
          {stav.pizza ? `Celkem ${spocitejCenu(stav)} Kč` : 'Zatím nic nevybráno'}

            Zpět
          </button>
        )}
        {stav.krok < 3 && (
          <button
            type="button"
            className="tlacitko"
            disabled={stav.pizza === null}
            onClick={() => poslat({ typ: 'dalsi' })}
          >
            Pokračovat
          </button>
        )}
        {stav.krok === 3 && (
          <button type="button" className="tlacitko" onClick={() => poslat({ typ: 'odeslano' })}>
            Objednat
          </button>
        )}
        <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'reset' })}>
          Začít znovu
        </button>
      </div>
    </div>
  );
}

export default function App() {
  const [stav, poslat] = useReducer(objednavkaReducer, VYCHOZI);

  return (

      <AkceKontext value={poslat}>
        <div className="objednavka">
          <h1 className="objednavka__nadpis">Pizzeria Na Rohu</h1>

              {stav.krok === 1 && <KrokJedna />}
              {stav.krok === 2 && <KrokDva />}
              {stav.krok === 3 && <KrokTri />}

              <Lista />
            </>
          )}
        </div>
      </AkceKontext>
    </StavKontext>
  );
}
```

```js
import { pizzy, prilohy, velikosti, zpusobyDoruceni } from './menu';

/**
 * Spočítá celkovou cenu objednávky: pizzu přepočtenou na vybranou velikost,
 * přílohy a dopravu. Dokud není vybraná pizza, vrací nulu.
 */
export function spocitejCenu(stav) {
const pizza = pizzy.find((p) => p.id === stav.pizza);
if (!pizza) return 0;
const velikost = velikosti.find((v) => v.id === stav.velikost);
const doprava = zpusobyDoruceni.find((z) => z.id === stav.doruceni);
const priplatky = stav.prilohy.reduce((soucet, id) => soucet + prilohy.find((p) => p.id === id).cena, 0);
return Math.round(pizza.cena * velikost.koeficient) + priplatky + doprava.cena;
}
```

```jsx
import { createContext, useContext, useReducer } from 'react';
import { pizzy, prilohy, velikosti, zpusobyDoruceni } from './menu';
import { spocitejCenu } from './cena';

const VYCHOZI = {
  krok: 1,
  pizza: null,
  velikost: 'stredni',
  prilohy: [],
  doruceni: 'vyzvednuti',
  hotovo: false,
};

function objednavkaReducer(stav, akce) {
  switch (akce.typ) {
    case 'pizza':
      return { ...stav, pizza: akce.id };
    case 'dalsi':
      return { ...stav, krok: stav.krok + 1 };
    case 'zpet':
      return { ...stav, krok: stav.krok - 1 };
    case 'velikost':
      return { ...stav, velikost: akce.id };
    case 'doruceni':
      return { ...stav, doruceni: akce.id };
    case 'odeslano':
      return { ...stav, hotovo: true };
    case 'reset':
      return VYCHOZI;
    case 'priloha': {
      const uz = stav.prilohy.includes(akce.id);
      return {
        ...stav,
        prilohy: uz ? stav.prilohy.filter((id) => id !== akce.id) : [...stav.prilohy, akce.id],
      };
    }
    default:
      throw new Error('Neznámá akce: ' + akce.typ);
  }
}

const StavKontext = createContext(null);
const AkceKontext = createContext(null);

function KrokJedna() {
  const stav = useContext(StavKontext);
  const poslat = useContext(AkceKontext);

  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Vyber si pizzu</h2>
      <ul className="nabidka">
        {pizzy.map((pizza) => (
          <li key={pizza.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.pizza === pizza.id}
              onClick={() => poslat({ typ: 'pizza', id: pizza.id })}
            >
              {pizza.stitek && <span className="polozka__stitek">{pizza.stitek}</span>}
              <p className="polozka__nazev">{pizza.nazev}</p>
              <p className="polozka__popis">{pizza.popis}</p>
              <p className="polozka__cena">{pizza.cena} Kč</p>
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

function KrokDva() {
  const stav = useContext(StavKontext);
  const poslat = useContext(AkceKontext);

  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Velikost a přílohy</h2>
      <ul className="nabidka">
        {velikosti.map((velikost) => (
          <li key={velikost.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.velikost === velikost.id}
              onClick={() => poslat({ typ: 'velikost', id: velikost.id })}
            >
              <p className="polozka__nazev">{velikost.nazev}</p>
              <p className="polozka__popis">{velikost.prumer}</p>
            </button>
          </li>
        ))}
      </ul>
      <h3 className="sekce__nadpis">Přílohy navíc</h3>
      <ul className="nabidka">
        {prilohy.map((priloha) => (
          <li key={priloha.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.prilohy.includes(priloha.id)}
              onClick={() => poslat({ typ: 'priloha', id: priloha.id })}
            >
              <p className="polozka__nazev">{priloha.nazev}</p>
              <p className="polozka__cena">+{priloha.cena} Kč</p>
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

function KrokTri() {
  const stav = useContext(StavKontext);
  const poslat = useContext(AkceKontext);

  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Doručení</h2>
      <ul className="nabidka">
        {zpusobyDoruceni.map((zpusob) => (
          <li key={zpusob.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.doruceni === zpusob.id}
              onClick={() => poslat({ typ: 'doruceni', id: zpusob.id })}
            >
              <p className="polozka__nazev">{zpusob.nazev}</p>
              <p className="polozka__popis">{zpusob.popis}</p>
              <p className="polozka__cena">{zpusob.cena === 0 ? 'zdarma' : `+${zpusob.cena} Kč`}</p>
            </button>
          </li>
        ))}
      </ul>

      <div className="rekapitulace">
        <p className="rekapitulace__radek">
          <span>
            {pizzy.find((p) => p.id === stav.pizza).nazev} (
            {velikosti.find((v) => v.id === stav.velikost).nazev})
          </span>
          <span>
            {Math.round(
              pizzy.find((p) => p.id === stav.pizza).cena *
                velikosti.find((v) => v.id === stav.velikost).koeficient,
            )}{' '}
            Kč
          </span>
        </p>
        {stav.prilohy.map((id) => {
          const priloha = prilohy.find((p) => p.id === id);
          return (
            <p key={id} className="rekapitulace__radek">
              <span>{priloha.nazev}</span>
              <span>{priloha.cena} Kč</span>
            </p>
          );
        })}
        <p className="rekapitulace__radek">
          <span>{zpusobyDoruceni.find((z) => z.id === stav.doruceni).nazev}</span>
          <span>{zpusobyDoruceni.find((z) => z.id === stav.doruceni).cena} Kč</span>
        </p>
        <p className="rekapitulace__radek rekapitulace__radek--soucet">
          <span>Celkem</span>
          <span>{spocitejCenu(stav)} Kč</span>
        </p>
      </div>
    </section>
  );
}

function Kroky() {
  const stav = useContext(StavKontext);
  const nazvy = ['Pizza', 'Velikost', 'Doručení'];

  return (
    <ol className="kroky">
      {nazvy.map((nazev, index) => (
        <li
          key={nazev}
          className={index + 1 === stav.krok ? 'kroky__krok kroky__krok--aktivni' : 'kroky__krok'}
          aria-current={index + 1 === stav.krok ? 'step' : undefined}
        >
          {index + 1}. {nazev}
        </li>
      ))}
    </ol>
  );
}

function Hotovo() {
  const poslat = useContext(AkceKontext);

  return (
    <div className="hotovo">
      <h2 className="hotovo__nadpis">Objednávka odeslaná</h2>
      <p>Pizza bude hotová do 25 minut. Díky!</p>
      <button type="button" className="tlacitko" onClick={() => poslat({ typ: 'reset' })}>
        Objednat další
      </button>
    </div>
  );
}

function Lista() {
  const stav = useContext(StavKontext);
  const poslat = useContext(AkceKontext);

  return (
    <div className="lista">
      <div className="lista__obsah">
        <p className="lista__cena">
          {stav.pizza ? `Celkem ${spocitejCenu(stav)} Kč` : 'Zatím nic nevybráno'}
        </p>
        {stav.krok > 1 && (
          <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'zpet' })}>
            Zpět
          </button>
        )}
        {stav.krok < 3 && (
          <button
            type="button"
            className="tlacitko"
            disabled={stav.pizza === null}
            onClick={() => poslat({ typ: 'dalsi' })}
          >
            Pokračovat
          </button>
        )}
        {stav.krok === 3 && (
          <button type="button" className="tlacitko" onClick={() => poslat({ typ: 'odeslano' })}>
            Objednat
          </button>
        )}
        <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'reset' })}>
          Začít znovu
        </button>
      </div>
    </div>
  );
}

export default function App() {
  const [stav, poslat] = useReducer(objednavkaReducer, VYCHOZI);

  return (
    <StavKontext value={stav}>
      <AkceKontext value={poslat}>
        <div className="objednavka">
          <h1 className="objednavka__nadpis">Pizzeria Na Rohu</h1>
          <p className="objednavka__popis">Objednávka ve třech krocích: pizza, velikost a doručení.</p>

          {stav.hotovo ? (
            <Hotovo />
          ) : (
            <>
              <Kroky />
              {stav.krok === 1 && <KrokJedna />}
              {stav.krok === 2 && <KrokDva />}
              {stav.krok === 3 && <KrokTri />}

              <Lista />
            </>
          )}
        </div>
      </AkceKontext>
    </StavKontext>
  );
}
```
