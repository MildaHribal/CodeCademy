# --description--

Každá komponenta průvodce dostává `stav` a `poslat` jako props a `App` je posílá čtyřikrát dokola. Až přibude pátá komponenta, přibude i pátý pár props. Právě na tohle je kontext.

**Úkol:** Vytvoř v `App.jsx` kontext s výchozí hodnotou `null`, obal jím celý obsah `App` a jako hodnotu mu dej objekt se stavem i s funkcí na posílání akcí. Komponenty `KrokJedna`, `KrokDva`, `KrokTri` a `Lista` ať si obojí berou z kontextu a žádné props už nepřijímají.

# --help--

## --tip--

Kontext se vyrábí mimo komponentu a čte se hookem — viz [Kontext: hodnota pro celý podstrom](see:react-hloubka/reducer-a-context#kontext-hodnota-pro-cely-podstrom).

## --tip--

V React 19 se poskytovatel píše rovnou jako komponenta s prop `value`. Obalit musí všechno, co má hodnotu číst.

## --tip--

Na jiných datech: `const UzivatelKontext = createContext(null);` a v komponentě `const uzivatel = useContext(UzivatelKontext);`.

# --explain--

Vysvětli vlastními slovy, kdy je kontext lepší než props — a kdy je naopak přehnaný.

## --model--

Kontext se vyplatí u hodnot, které potřebuje větší část stromu a které by se jinak musely protahovat komponentami, jež je samy nepoužijí: přihlášený uživatel, téma, jazyk, obsah košíku. Ušetří to props i chyby při přejmenovávání. Přehnaný je tam, kde jde o jednu nebo dvě úrovně — props jsou čitelnější, protože je na první pohled vidět, co komponenta potřebuje. Komponenta, která si všechno tahá z kontextu, se navíc těžko použije podruhé jinde a hůř se testuje, protože ji nejde zavolat bez poskytovatele.

## --checklist--

- Kontext šetří protahování props přes komponenty, které hodnotu nepoužijí.
- Hodí se na věci potřebné v celém podstromu (uživatel, téma, košík).
- Na jednu dvě úrovně jsou props čitelnější.
- Komponenta závislá na kontextu se hůř používá jinde a hůř testuje.

```jsx
import { useReducer } from 'react';
import { pizzy, prilohy, velikosti, zpusobyDoruceni } from './menu';
import { spocitejCenu } from './cena';

const VYCHOZI = {
  krok: 1,
  pizza: null,
  velikost: 'stredni',
  prilohy: [],
  doruceni: 'vyzvednuti',
  hotovo: false,
};

function objednavkaReducer(stav, akce) {
  switch (akce.typ) {
    case 'pizza':
      return { ...stav, pizza: akce.id };
    case 'dalsi':
      return { ...stav, krok: stav.krok + 1 };
    case 'zpet':
      return { ...stav, krok: stav.krok - 1 };
    case 'velikost':
      return { ...stav, velikost: akce.id };
    case 'doruceni':
      return { ...stav, doruceni: akce.id };
    case 'odeslano':
      return { ...stav, hotovo: true };
    case 'reset':
      return VYCHOZI;
    case 'priloha': {
      const uz = stav.prilohy.includes(akce.id);
      return {
        ...stav,
        prilohy: uz ? stav.prilohy.filter((id) => id !== akce.id) : [...stav.prilohy, akce.id],
      };
    }
    default:
      throw new Error('Neznámá akce: ' + akce.typ);
  }
}

function KrokJedna({ stav, poslat }) {
  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Vyber si pizzu</h2>
      <ul className="nabidka">
        {pizzy.map((pizza) => (
          <li key={pizza.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.pizza === pizza.id}
              onClick={() => poslat({ typ: 'pizza', id: pizza.id })}
            >
              {pizza.stitek && <span className="polozka__stitek">{pizza.stitek}</span>}
              <p className="polozka__nazev">{pizza.nazev}</p>
              <p className="polozka__popis">{pizza.popis}</p>
              <p className="polozka__cena">{pizza.cena} Kč</p>
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Velikost a přílohy</h2>

          </li>

        <p className="rekapitulace__radek">

          <span>{velikosti.find((v) => v.id === stav.velikost).nazev}</span>
        </p>

          <span>Přílohy</span>

          <span>Doručení</span>
          <span>{zpusobyDoruceni.find((z) => z.id === stav.doruceni).nazev}</span>

          <span>Celkem</span>
          <span>{spocitejCenu(stav)} Kč</span>
        </p>
      </div>
    </section>
  );
}

function Hotovo() {
  return (
    <div className="hotovo">
      <h2 className="hotovo__nadpis">Objednávka odeslaná</h2>
      <p>Pizza bude hotová do 25 minut. Díky!</p>
    </div>
  );
}

function Lista({ stav, poslat }) {
  return (
    <div className="lista">
      <div className="lista__obsah">
        <p className="lista__cena">
          {stav.pizza ? `Celkem ${spocitejCenu(stav)} Kč` : 'Zatím nic nevybráno'}
        </p>
        {stav.krok > 1 && (
          <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'zpet' })}>
            Zpět
          </button>
        )}
        {stav.krok < 3 && (
          <button
            type="button"
            className="tlacitko"
            disabled={stav.pizza === null}
            onClick={() => poslat({ typ: 'dalsi' })}
          >
            Pokračovat
          </button>
        )}
        {stav.krok === 3 && (
          <button type="button" className="tlacitko" onClick={() => poslat({ typ: 'odeslano' })}>
            Objednat
          </button>
        )}
        <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'reset' })}>
          Začít znovu
        </button>
      </div>
    </div>
  );
}

export default function App() {
  const [stav, poslat] = useReducer(objednavkaReducer, VYCHOZI);

  return (
    <div className="objednavka">
      <h1 className="objednavka__nadpis">Pizzeria Na Rohu</h1>
      <p className="objednavka__popis">Objednávka ve třech krocích: pizza, velikost a doručení.</p>

      {stav.hotovo ? (
        <Hotovo />
      ) : (
        <>
          {stav.krok === 1 && <KrokJedna stav={stav} poslat={poslat} />}
          {stav.krok === 2 && <KrokDva stav={stav} poslat={poslat} />}
          {stav.krok === 3 && <KrokTri stav={stav} poslat={poslat} />}

          <Lista stav={stav} poslat={poslat} />
        </>
      )}
    </div>
  );
}
```

```js
import { pizzy, prilohy, velikosti, zpusobyDoruceni } from './menu';

/**
 * Spočítá celkovou cenu objednávky: pizzu přepočtenou na vybranou velikost,
 * přílohy a dopravu. Dokud není vybraná pizza, vrací nulu.
 */
export function spocitejCenu(stav) {
const pizza = pizzy.find((p) => p.id === stav.pizza);
if (!pizza) return 0;
const velikost = velikosti.find((v) => v.id === stav.velikost);
const doprava = zpusobyDoruceni.find((z) => z.id === stav.doruceni);
const priplatky = stav.prilohy.reduce((soucet, id) => soucet + prilohy.find((p) => p.id === id).cena, 0);
return Math.round(pizza.cena * velikost.koeficient) + priplatky + doprava.cena;
}
```

```jsx
import { createContext, useContext, useReducer } from 'react';
import { pizzy, prilohy, velikosti, zpusobyDoruceni } from './menu';
import { spocitejCenu } from './cena';

const VYCHOZI = {
  krok: 1,
  pizza: null,
  velikost: 'stredni',
  prilohy: [],
  doruceni: 'vyzvednuti',
  hotovo: false,
};

function objednavkaReducer(stav, akce) {
  switch (akce.typ) {
    case 'pizza':
      return { ...stav, pizza: akce.id };
    case 'dalsi':
      return { ...stav, krok: stav.krok + 1 };
    case 'zpet':
      return { ...stav, krok: stav.krok - 1 };
    case 'velikost':
      return { ...stav, velikost: akce.id };
    case 'doruceni':
      return { ...stav, doruceni: akce.id };
    case 'odeslano':
      return { ...stav, hotovo: true };
    case 'reset':
      return VYCHOZI;
    case 'priloha': {
      const uz = stav.prilohy.includes(akce.id);
      return {
        ...stav,
        prilohy: uz ? stav.prilohy.filter((id) => id !== akce.id) : [...stav.prilohy, akce.id],
      };
    }
    default:
      throw new Error('Neznámá akce: ' + akce.typ);
  }
}

const ObjednavkaKontext = createContext(null);

function KrokJedna() {
  const { stav, poslat } = useContext(ObjednavkaKontext);

  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Vyber si pizzu</h2>
      <ul className="nabidka">
        {pizzy.map((pizza) => (
          <li key={pizza.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.pizza === pizza.id}
              onClick={() => poslat({ typ: 'pizza', id: pizza.id })}
            >
              {pizza.stitek && <span className="polozka__stitek">{pizza.stitek}</span>}
              <p className="polozka__nazev">{pizza.nazev}</p>
              <p className="polozka__popis">{pizza.popis}</p>
              <p className="polozka__cena">{pizza.cena} Kč</p>
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

function KrokDva() {
  const { stav, poslat } = useContext(ObjednavkaKontext);

  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Velikost a přílohy</h2>
      <ul className="nabidka">
        {velikosti.map((velikost) => (
          <li key={velikost.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.velikost === velikost.id}
              onClick={() => poslat({ typ: 'velikost', id: velikost.id })}
            >
              <p className="polozka__nazev">{velikost.nazev}</p>
              <p className="polozka__popis">{velikost.prumer}</p>
            </button>
          </li>
        ))}
      </ul>
      <h3 className="sekce__nadpis">Přílohy navíc</h3>
      <ul className="nabidka">
        {prilohy.map((priloha) => (
          <li key={priloha.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.prilohy.includes(priloha.id)}
              onClick={() => poslat({ typ: 'priloha', id: priloha.id })}
            >
              <p className="polozka__nazev">{priloha.nazev}</p>
              <p className="polozka__cena">+{priloha.cena} Kč</p>
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

function KrokTri() {
  const { stav, poslat } = useContext(ObjednavkaKontext);

  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Doručení</h2>
      <ul className="nabidka">
        {zpusobyDoruceni.map((zpusob) => (
          <li key={zpusob.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.doruceni === zpusob.id}
              onClick={() => poslat({ typ: 'doruceni', id: zpusob.id })}
            >
              <p className="polozka__nazev">{zpusob.nazev}</p>
              <p className="polozka__popis">{zpusob.popis}</p>
              <p className="polozka__cena">{zpusob.cena === 0 ? 'zdarma' : `+${zpusob.cena} Kč`}</p>
            </button>
          </li>
        ))}
      </ul>

      <div className="rekapitulace">
        <p className="rekapitulace__radek">
          <span>{pizzy.find((p) => p.id === stav.pizza).nazev}</span>
          <span>{velikosti.find((v) => v.id === stav.velikost).nazev}</span>
        </p>
        <p className="rekapitulace__radek">
          <span>Přílohy</span>
          <span>
            {stav.prilohy.length === 0
              ? 'žádné'
              : stav.prilohy.map((id) => prilohy.find((p) => p.id === id).nazev).join(', ')}
          </span>
        </p>
        <p className="rekapitulace__radek">
          <span>Doručení</span>
          <span>{zpusobyDoruceni.find((z) => z.id === stav.doruceni).nazev}</span>
        </p>
        <p className="rekapitulace__radek rekapitulace__radek--soucet">
          <span>Celkem</span>
          <span>{spocitejCenu(stav)} Kč</span>
        </p>
      </div>
    </section>
  );
}

function Hotovo() {
  return (
    <div className="hotovo">
      <h2 className="hotovo__nadpis">Objednávka odeslaná</h2>
      <p>Pizza bude hotová do 25 minut. Díky!</p>
    </div>
  );
}

function Lista() {
  const { stav, poslat } = useContext(ObjednavkaKontext);

  return (
    <div className="lista">
      <div className="lista__obsah">
        <p className="lista__cena">
          {stav.pizza ? `Celkem ${spocitejCenu(stav)} Kč` : 'Zatím nic nevybráno'}
        </p>
        {stav.krok > 1 && (
          <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'zpet' })}>
            Zpět
          </button>
        )}
        {stav.krok < 3 && (
          <button
            type="button"
            className="tlacitko"
            disabled={stav.pizza === null}
            onClick={() => poslat({ typ: 'dalsi' })}
          >
            Pokračovat
          </button>
        )}
        {stav.krok === 3 && (
          <button type="button" className="tlacitko" onClick={() => poslat({ typ: 'odeslano' })}>
            Objednat
          </button>
        )}
        <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'reset' })}>
          Začít znovu
        </button>
      </div>
    </div>
  );
}

export default function App() {
  const [stav, poslat] = useReducer(objednavkaReducer, VYCHOZI);

  return (
    <ObjednavkaKontext value={{ stav, poslat }}>
      <div className="objednavka">
        <h1 className="objednavka__nadpis">Pizzeria Na Rohu</h1>
        <p className="objednavka__popis">Objednávka ve třech krocích: pizza, velikost a doručení.</p>

        {stav.hotovo ? (
          <Hotovo />
        ) : (
          <>
            {stav.krok === 1 && <KrokJedna />}
            {stav.krok === 2 && <KrokDva />}
            {stav.krok === 3 && <KrokTri />}

            <Lista />
          </>
        )}
      </div>
    </ObjednavkaKontext>
  );
}
```
