# --description--

Objednávka má tři obrazovky. Číslo té aktuální už ve stavu je (`krok`), teď podle něj rozhodneš, co se vykreslí — a dáš uživateli tlačítka dopředu a zpět.

**Úkol:** Rozděl `App.jsx` na komponenty `KrokJedna` (nabídka pizz, kterou už máš), `KrokDva` a `KrokTri` (zatím jen `<section className="sekce">` s nadpisem `Velikost a přílohy` resp. `Doručení`) a `Lista` (spodní lišta). Do reduceru přidej akce `dalsi` a `zpet`, které `krok` zvýší a sníží o jedna. V `App` vykresli podle `stav.krok` tu správnou komponentu a `Lista` ať ukáže tlačítko `Zpět` (jen mimo první krok) a `Pokračovat` (jen mimo třetí). Stav a `poslat` zatím posílej komponentám jako props.

# --help--

## --tip--

Podmíněné vykreslení podle čísla kroku je obyčejné `&&` v JSX, jen se porovnává číslo.

## --tip--

Komponenty zatím dostanou dvě props: aktuální stav a funkci na posílání akcí. Uvnitř se nic jiného nemění.

## --tip--

Na jiných datech: `{stav.zalozka === 'popis' && <Popis text={produkt.popis} />}`.

```jsx
import { useReducer } from 'react';
import { pizzy } from './menu';

const VYCHOZI = {
  krok: 1,
  pizza: null,
  velikost: 'stredni',
  prilohy: [],
  doruceni: 'vyzvednuti',
};

function objednavkaReducer(stav, akce) {
  switch (akce.typ) {
    case 'pizza':
      return { ...stav, pizza: akce.id };
    default:
      throw new Error('Neznámá akce: ' + akce.typ);
  }
}

export default function App() {
  const [stav, poslat] = useReducer(objednavkaReducer, VYCHOZI);

  return (
    <div className="objednavka">
      <h1 className="objednavka__nadpis">Pizzeria Na Rohu</h1>
      <p className="objednavka__popis">Objednávka ve třech krocích: pizza, velikost a doručení.</p>

      <section className="sekce">
        <h2 className="sekce__nadpis">Vyber si pizzu</h2>
        <ul className="nabidka">
          {pizzy.map((pizza) => (
            <li key={pizza.id}>
              <button
                type="button"
                className="polozka"
                aria-pressed={stav.pizza === pizza.id}
                onClick={() => poslat({ typ: 'pizza', id: pizza.id })}
              >
                {pizza.stitek && <span className="polozka__stitek">{pizza.stitek}</span>}
                <p className="polozka__nazev">{pizza.nazev}</p>
                <p className="polozka__popis">{pizza.popis}</p>
                <p className="polozka__cena">{pizza.cena} Kč</p>
              </button>
            </li>
          ))}
        </ul>
      </section>

      <div className="lista">
        <div className="lista__obsah">
          <p className="lista__cena">Zatím nic nevybráno</p>
        </div>
      </div>
    </div>
  );
}
```

## --file-- App.jsx

```jsx
import { useReducer } from 'react';
import { pizzy } from './menu';

const VYCHOZI = {
  krok: 1,
  pizza: null,
  velikost: 'stredni',
  prilohy: [],
  doruceni: 'vyzvednuti',
};

function objednavkaReducer(stav, akce) {
  switch (akce.typ) {
    case 'pizza':
      return { ...stav, pizza: akce.id };
    case 'dalsi':
      return { ...stav, krok: stav.krok + 1 };
    case 'zpet':
      return { ...stav, krok: stav.krok - 1 };
    default:
      throw new Error('Neznámá akce: ' + akce.typ);
  }
}

function KrokJedna({ stav, poslat }) {
  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Vyber si pizzu</h2>
      <ul className="nabidka">
        {pizzy.map((pizza) => (
          <li key={pizza.id}>
            <button
              type="button"
              className="polozka"
              aria-pressed={stav.pizza === pizza.id}
              onClick={() => poslat({ typ: 'pizza', id: pizza.id })}
            >
              {pizza.stitek && <span className="polozka__stitek">{pizza.stitek}</span>}
              <p className="polozka__nazev">{pizza.nazev}</p>
              <p className="polozka__popis">{pizza.popis}</p>
              <p className="polozka__cena">{pizza.cena} Kč</p>
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

function KrokDva() {
  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Velikost a přílohy</h2>
    </section>
  );
}

function KrokTri() {
  return (
    <section className="sekce">
      <h2 className="sekce__nadpis">Doručení</h2>
    </section>
  );
}

function Lista({ stav, poslat }) {
  return (
    <div className="lista">
      <div className="lista__obsah">
        <p className="lista__cena">Zatím nic nevybráno</p>
        {stav.krok > 1 && (
          <button type="button" className="tlacitko tlacitko--tichy" onClick={() => poslat({ typ: 'zpet' })}>
            Zpět
          </button>
        )}
        {stav.krok < 3 && (
          <button type="button" className="tlacitko" onClick={() => poslat({ typ: 'dalsi' })}>
            Pokračovat
          </button>
        )}
      </div>
    </div>
  );
}

export default function App() {
  const [stav, poslat] = useReducer(objednavkaReducer, VYCHOZI);

  return (
    <div className="objednavka">
      <h1 className="objednavka__nadpis">Pizzeria Na Rohu</h1>
      <p className="objednavka__popis">Objednávka ve třech krocích: pizza, velikost a doručení.</p>

      {stav.krok === 1 && <KrokJedna stav={stav} poslat={poslat} />}
      {stav.krok === 2 && <KrokDva stav={stav} poslat={poslat} />}
      {stav.krok === 3 && <KrokTri stav={stav} poslat={poslat} />}

      <Lista stav={stav} poslat={poslat} />
    </div>
  );
}
```
