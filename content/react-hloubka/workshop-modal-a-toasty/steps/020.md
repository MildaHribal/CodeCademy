# --description--

Když uživatel vrátí všechny vstupenky, zůstane po nich prázdná bílá plocha. Prázdný stav je součást návrhu, ne chyba: řekne, co se stalo, a nabídne, co dělat dál.

**Úkol:** Když v seznamu není žádná vstupenka, ukaž místo karet odstavec s třídou `prazdno` a textem, který uživateli řekne, že žádnou vstupenku nemá. Text si zvol sám.

Tím je stránka hotová. Postavil jsi modální okno, jaké se dnes očekává od každé aplikace: vykreslené portálem mimo svůj kout stránky, zavíratelné tlačítkem, Escapem i kliknutím vedle, se správně nastaveným a vráceným fokusem a s rolí, kterou odečítač obrazovky pozná. K tomu oznámení, která se sama uklidí, jdou zavřít dřív a umí nabídnout krok zpět — a celá jejich logika je v jednom vlastním hooku, který příště jen zavoláš.

Z Reactu jsi přitom použil skoro všechno z posledních tří lekcí: efekt jako synchronizaci s vnějším systémem (posluchač, časovač, zámek scrollu), úklid, který ho zase vypne, ref na prvek i ref jako schránku mimo render a stabilní identitu funkce, bez které se efekt spouští pořád dokola.

Příště se podíváš na vlastní hooky pořádně — a pak na `useReducer` a kontext, se kterými zvládneš stav, na který už jednotlivé `useState` nestačí.

## Udělej po svém

- **Vzhled:** dej oznámením ukazatel zbývajícího času — tenký proužek, který se během tří sekund smrskne (CSS animace na šířku, spuštěná při namontování).
- **Chování:** přidej pod seznam součet ceny zbývajících vstupenek a nech ho měnit se s každým vrácením.
- **Chování:** nech okno reagovat na Tab tak, aby fokus nevyskočil ven (najdi v okně zaostřitelné prvky a na posledním přesuň fokus na první).
- **Téma:** přepiš stránku na něco svého — objednávky v e-shopu, rezervace stolů, půjčené knihy. Data, texty i barvy jsou tvoje.

```jsx
export function Nahled({ vstupenka, onZavrit, onVratit }) {
  const oknoRef = useRef(null);

  useEffect(() => {
    oknoRef.current.focus();

  useEffect(() => {
    document.body.style.overflow = 'hidden';
    return () => {

  useEffect(() => {
    const naKlavesu = (udalost) => {
      if (udalost.key === 'Escape') onZavrit();

export default function App() {
  const [vstupenky, setVstupenky] = useState(vychoziVstupenky);
  const [otevrena, setOtevrena] = useState(null);
  const { oznameni, oznam, zavriOznameni } = useOznameni();

  function otevri(udalost, id) {
    spoustecRef.current = udalost.currentTarget;
    setOtevrena(id);

  function zavri() {
    setOtevrena(null);
    spoustecRef.current?.focus();

  function vrat(vstupenka) {
    setVstupenky((stare) => stare.filter((v) => v.id !== vstupenka.id));

    oznam(`Vstupenka na ${vstupenka.kapela} je vrácená.`, () => {
      setVstupenky((stare) => [...stare, vstupenka]);

      <h1 className="stranka__nadpis">Moje vstupenky</h1>

      <ul className="seznam">
        {vstupenky.map((vstupenka) => (
          <li key={vstupenka.id} className={`vstupenka vstupenka--${vstupenka.odstin}`}>

            <div className="vstupenka__telo">
              <h2 className="vstupenka__kapela">{vstupenka.kapela}</h2>
              <p className="vstupenka__radek">{vstupenka.misto}</p>
              <p className="vstupenka__radek">{vstupenka.sektor}</p>
              <p className="vstupenka__cena">{vstupenka.cena} Kč</p>
              <button
                type="button"
                className="tlacitko tlacitko--auto"
                onClick={(udalost) => otevri(udalost, vstupenka.id)}
              >
                Detail
              </button>
            </div>
            {otevrena === vstupenka.id && (
              <Nahled vstupenka={vstupenka} onZavrit={zavri} onVratit={vrat} />
            )}
          </li>
        ))}
      </ul>

      <ul className="oznameni" role="status">
        {oznameni.map((zprava) => (
          <Oznameni key={zprava.id} zprava={zprava} onZavrit={zavriOznameni} />
        ))}
      </ul>
    </div>
  );
}
```

```js
import { useCallback, useState } from 'react';

let dalsiId = 1;

/**
 * Drží seznam oznámení a stará se o jejich přidávání a zavírání.
 * @returns {{ oznameni: Array, oznam: Function, zavriOznameni: Function }}
 */
export function useOznameni() {
  const [oznameni, setOznameni] = useState([]);

  const oznam = useCallback((text, akce = null) => {
    setOznameni((stara) => [...stara, { id: dalsiId++, text, akce }]);
  }, []);

  const zavriOznameni = useCallback((id) => {
    setOznameni((stara) => stara.filter((zprava) => zprava.id !== id));
  }, []);

  return { oznameni, oznam, zavriOznameni };
}
```

```jsx
        <h3 className="nahled__nadpis" id="nahled-nadpis">
          {vstupenka.kapela}
        </h3>
        <p className="nahled__radek">{vstupenka.datum}</p>
        <p className="nahled__radek">{vstupenka.misto}</p>
        <p className="nahled__radek">{vstupenka.sektor}</p>
        <p className="nahled__kod">{vstupenka.kod}</p>
        <div className="nahled__ovladani">
          <button
            type="button"
            className="tlacitko tlacitko--nebezpeci"
            onClick={() => onVratit(vstupenka)}
          >
            Vrátit vstupenku
          </button>
          <button type="button" className="tlacitko tlacitko--tichy" onClick={onZavrit}>
            Zavřít
          </button>
        </div>
      </div>
    </div>,
    document.body,
  );
}

export default function App() {
  const [vstupenky, setVstupenky] = useState(vychoziVstupenky);
  const [otevrena, setOtevrena] = useState(null);
  const { oznameni, oznam, zavriOznameni } = useOznameni();
  const spoustecRef = useRef(null);

  function otevri(udalost, id) {
    spoustecRef.current = udalost.currentTarget;
    setOtevrena(id);
  }

  function zavri() {
    setOtevrena(null);
    spoustecRef.current?.focus();
  }

  function vrat(vstupenka) {
    setVstupenky((stare) => stare.filter((v) => v.id !== vstupenka.id));
    setOtevrena(null);
    oznam(`Vstupenka na ${vstupenka.kapela} je vrácená.`, () => {
      setVstupenky((stare) => [...stare, vstupenka]);
    });
  }

  return (
    <div className="stranka">
      <h1 className="stranka__nadpis">Moje vstupenky</h1>
      <p className="stranka__popis">Koncerty, na které už máš koupeno.</p>

      {vstupenky.length === 0 && (
        <p className="prazdno">Žádná vstupenka. Všechny jsi vrátil.</p>
      )}

      <ul className="seznam">
        {vstupenky.map((vstupenka) => (
          <li key={vstupenka.id} className={`vstupenka vstupenka--${vstupenka.odstin}`}>
            <p className="vstupenka__pas">{vstupenka.datum}</p>
            <div className="vstupenka__telo">
              <h2 className="vstupenka__kapela">{vstupenka.kapela}</h2>
              <p className="vstupenka__radek">{vstupenka.misto}</p>
              <p className="vstupenka__radek">{vstupenka.sektor}</p>
              <p className="vstupenka__cena">{vstupenka.cena} Kč</p>
              <button
                type="button"
                className="tlacitko tlacitko--auto"
                onClick={(udalost) => otevri(udalost, vstupenka.id)}
              >
                Detail
              </button>
            </div>
            {otevrena === vstupenka.id && (
              <Nahled vstupenka={vstupenka} onZavrit={zavri} onVratit={vrat} />
            )}
          </li>
        ))}
      </ul>

      <ul className="oznameni" role="status">
        {oznameni.map((zprava) => (
          <Oznameni key={zprava.id} zprava={zprava} onZavrit={zavriOznameni} />
        ))}
      </ul>
    </div>
  );
}
```
