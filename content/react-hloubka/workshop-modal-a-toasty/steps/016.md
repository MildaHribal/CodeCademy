# --description--

## Hlášení

„Vrátím vstupenku, objeví se oznámení — a když mezitím otevřu a zavřu detail jiného koncertu, oznámení na obrazovce zůstane viset. Když se ničeho nedotknu, zmizí správně po třech sekundách."

Časovač i jeho úklid jsou přitom napsané správně a `clearTimeout` se volá.

## Úkol

Najdi, proč se odpočet u každého překreslení `App` začíná od začátku, a oprav to. Oznámení musí zmizet do tří sekund od svého vzniku, ať uživatel mezitím dělá cokoli. Změň co nejmenší kus kódu.

# --help--

## --tip--

Efekt se spustí znovu vždycky, když se **změní** některá jeho závislost. Podívej se, co je mezi závislostmi kromě identifikátoru zprávy, a přečti si [Identita hodnot](see:react-hloubka/render-a-rerender#identita-kazdy-render-vyrabi-nove-objekty-a-funkce).

## --tip--

Funkce definovaná v tělě komponenty je při každém renderu nová hodnota. Potřebuješ, aby si mezi rendery držela identitu — na to je hook, který si pamatuje funkci, dokud se nezmění závislosti.

## --tip--

Na jiných datech: `const posliFormular = useCallback((data) => odesli(data), []);` — funkce, která nečte nic z renderu, může mít prázdný seznam závislostí.

import { useEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';
import { vstupenky as vychoziVstupenky } from './vstupenky';

let dalsiId = 1;

export function Oznameni({ zprava, onZavrit }) {
  useEffect(() => {
    const id = setTimeout(() => onZavrit(zprava.id), 3000);
    return () => clearTimeout(id);
  }, [zprava.id, onZavrit]);

  return (
    <li className="oznameni__polozka">
      <span className="oznameni__text">{zprava.text}</span>
    </li>
  );
}

export function Nahled({ vstupenka, onZavrit, onVratit }) {
  const oknoRef = useRef(null);

  useEffect(() => {
    oknoRef.current.focus();
  }, []);

  useEffect(() => {
    document.body.style.overflow = 'hidden';
    return () => {
      document.body.style.overflow = '';
    };
  }, []);

  useEffect(() => {
    const naKlavesu = (udalost) => {
      if (udalost.key === 'Escape') onZavrit();
    };
    document.addEventListener('keydown', naKlavesu);
    return () => document.removeEventListener('keydown', naKlavesu);
  }, [onZavrit]);

  return createPortal(
    <div
      className="prekryv"
      onClick={(udalost) => {
        if (udalost.target === udalost.currentTarget) onZavrit();
      }}
    >
      <div
        className="nahled"
        role="dialog"
        aria-modal="true"
        aria-labelledby="nahled-nadpis"
        tabIndex={-1}
        ref={oknoRef}
      >
        <h3 className="nahled__nadpis" id="nahled-nadpis">
          {vstupenka.kapela}
        </h3>
        <p className="nahled__radek">{vstupenka.datum}</p>
        <p className="nahled__radek">{vstupenka.misto}</p>
        <p className="nahled__radek">{vstupenka.sektor}</p>
        <p className="nahled__kod">{vstupenka.kod}</p>
        <div className="nahled__ovladani">
          <button
            type="button"
            className="tlacitko tlacitko--nebezpeci"
            onClick={() => onVratit(vstupenka)}
          >
            Vrátit vstupenku
          </button>
          <button type="button" className="tlacitko tlacitko--tichy" onClick={onZavrit}>
            Zavřít
          </button>
        </div>
      </div>
    </div>,
    document.body,
  );
}

export default function App() {
  const [vstupenky, setVstupenky] = useState(vychoziVstupenky);
  const [otevrena, setOtevrena] = useState(null);
  const [oznameni, setOznameni] = useState([]);
  const spoustecRef = useRef(null);

  function oznam(text) {
    setOznameni((stara) => [...stara, { id: dalsiId++, text }]);
  }

  function zavriOznameni(id) {
    setOznameni((stara) => stara.filter((zprava) => zprava.id !== id));
  }

  function otevri(udalost, id) {
    spoustecRef.current = udalost.currentTarget;
    setOtevrena(id);
  }

  function zavri() {
    setOtevrena(null);
    spoustecRef.current?.focus();
  }

  function vrat(vstupenka) {
    setVstupenky((stare) => stare.filter((v) => v.id !== vstupenka.id));
    setOtevrena(null);
    oznam(`Vstupenka na ${vstupenka.kapela} je vrácená.`);
  }

  return (
    <div className="stranka">
      <h1 className="stranka__nadpis">Moje vstupenky</h1>
      <p className="stranka__popis">Koncerty, na které už máš koupeno.</p>

      <ul className="seznam">
        {vstupenky.map((vstupenka) => (
          <li key={vstupenka.id} className={`vstupenka vstupenka--${vstupenka.odstin}`}>
            <p className="vstupenka__pas">{vstupenka.datum}</p>
            <div className="vstupenka__telo">
              <h2 className="vstupenka__kapela">{vstupenka.kapela}</h2>
              <p className="vstupenka__radek">{vstupenka.misto}</p>
              <p className="vstupenka__radek">{vstupenka.sektor}</p>
              <p className="vstupenka__cena">{vstupenka.cena} Kč</p>
              <button
                type="button"
                className="tlacitko tlacitko--auto"
                onClick={(udalost) => otevri(udalost, vstupenka.id)}
              >
                Detail
              </button>
            </div>
            {otevrena === vstupenka.id && (
              <Nahled vstupenka={vstupenka} onZavrit={zavri} onVratit={vrat} />
            )}
          </li>
        ))}
      </ul>

      <ul className="oznameni" role="status">
        {oznameni.map((zprava) => (
          <Oznameni key={zprava.id} zprava={zprava} onZavrit={zavriOznameni} />
        ))}
      </ul>
    </div>
  );
}
