# --description--

Vstupenka zmizela a uživatel neví, jestli se to povedlo, nebo mu spadl internet. Aplikace to řeší **oznámením** — krátkou zprávou v rohu, která se sama objeví po každé akci.

Oznámení může být na obrazovce víc naráz, takže je nebudeš držet jako jednu hodnotu, ale jako seznam. Každé má vlastní `id` (ať víš, které zavřít) a `text`.

Tohle je opakování bez návodu: pole ve stavu, přidání nové položky bez mutace a výpis seznamu — všechno už umíš.

**Úkol:** V `App` drž stav `oznameni` (na začátku prázdné pole) a napiš funkci `oznam(text)`, která do něj přidá nové oznámení s vlastním `id`. Zavolej ji při vrácení vstupenky s textem, který jmenuje kapelu (třeba `Vstupenka na Zrní je vrácená.`). Oznámení vypiš pod seznam vstupenek do seznamu `ul` s třídou `oznameni` a rolí `status`; každé oznámení je `li` s třídou `oznameni__polozka` a text v něm je zabalený ve `span` s třídou `oznameni__text`.

Jednoznačné `id` si vyrob, jak chceš — počítadlem v modulu, časem, nebo `crypto.randomUUID()`.

> [!PITFALL]
> Atribut `role="status"` patří na seznam, který je ve stránce **pořád** (i prázdný). Kdyby vznikal až s prvním oznámením, odečítač obrazovky ho nestihne zaregistrovat a zprávu nepřečte.

# --help--

## --tip--

Přidání do pole ve stavu se dělá novým polem, ne `push`. Vzor znáš z práce s poli.

## --tip--

Funkce `oznam` patří do `App` vedle ostatních obsluh a volá se ze stejné funkce, která vstupenku odebírá.

## --tip--

Na jiných datech: `setZpravy((stare) => [...stare, { id: dalsiId++, text: 'Objednávka odeslána.' }]);`.

import { useEffect, useRef, useState } from 'react';
import { createPortal } from 'react-dom';
import { vstupenky as vychoziVstupenky } from './vstupenky';

export function Nahled({ vstupenka, onZavrit, onVratit }) {
  const oknoRef = useRef(null);

  useEffect(() => {
    oknoRef.current.focus();
  }, []);

  useEffect(() => {
    document.body.style.overflow = 'hidden';
    return () => {
      document.body.style.overflow = '';
    };
  }, []);

  useEffect(() => {
    const naKlavesu = (udalost) => {
      if (udalost.key === 'Escape') onZavrit();
    };
    document.addEventListener('keydown', naKlavesu);
    return () => document.removeEventListener('keydown', naKlavesu);
  }, [onZavrit]);

  return createPortal(
    <div
      className="prekryv"
      onClick={(udalost) => {
        if (udalost.target === udalost.currentTarget) onZavrit();
      }}
    >
      <div
        className="nahled"
        role="dialog"
        aria-modal="true"
        aria-labelledby="nahled-nadpis"
        tabIndex={-1}
        ref={oknoRef}
      >
        <h3 className="nahled__nadpis" id="nahled-nadpis">
          {vstupenka.kapela}
        </h3>
        <p className="nahled__radek">{vstupenka.datum}</p>
        <p className="nahled__radek">{vstupenka.misto}</p>
        <p className="nahled__radek">{vstupenka.sektor}</p>
        <p className="nahled__kod">{vstupenka.kod}</p>
        <div className="nahled__ovladani">
          <button
            type="button"
            className="tlacitko tlacitko--nebezpeci"
            onClick={() => onVratit(vstupenka)}
          >
            Vrátit vstupenku
          </button>
          <button type="button" className="tlacitko tlacitko--tichy" onClick={onZavrit}>
            Zavřít
          </button>
        </div>
      </div>
    </div>,
    document.body,
  );
}

export default function App() {
  const [vstupenky, setVstupenky] = useState(vychoziVstupenky);
  const [otevrena, setOtevrena] = useState(null);
  const spoustecRef = useRef(null);

  function otevri(udalost, id) {
    spoustecRef.current = udalost.currentTarget;
    setOtevrena(id);
  }

  function zavri() {
    setOtevrena(null);
    spoustecRef.current?.focus();
  }

  function vrat(vstupenka) {
    setVstupenky((stare) => stare.filter((v) => v.id !== vstupenka.id));
    setOtevrena(null);
  }

  return (
    <div className="stranka">
      <h1 className="stranka__nadpis">Moje vstupenky</h1>
      <p className="stranka__popis">Koncerty, na které už máš koupeno.</p>

      <ul className="seznam">
        {vstupenky.map((vstupenka) => (
          <li key={vstupenka.id} className={`vstupenka vstupenka--${vstupenka.odstin}`}>
            <p className="vstupenka__pas">{vstupenka.datum}</p>
            <div className="vstupenka__telo">
              <h2 className="vstupenka__kapela">{vstupenka.kapela}</h2>
              <p className="vstupenka__radek">{vstupenka.misto}</p>
              <p className="vstupenka__radek">{vstupenka.sektor}</p>
              <p className="vstupenka__cena">{vstupenka.cena} Kč</p>
              <button
                type="button"
                className="tlacitko tlacitko--auto"
                onClick={(udalost) => otevri(udalost, vstupenka.id)}
              >
                Detail
              </button>
            </div>
            {otevrena === vstupenka.id && (
              <Nahled vstupenka={vstupenka} onZavrit={zavri} onVratit={vrat} />
            )}
          </li>
        ))}
      </ul>
    </div>
  );
}
