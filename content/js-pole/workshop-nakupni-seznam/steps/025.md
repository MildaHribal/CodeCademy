# --description--

**Úkol:** Napiš funkci `quantityByCategory(items)`, která vrátí objekt, kde klíčem je kategorie a hodnotou **počet kusů** (`quantity`) ze všech jejích položek. Ze seznamu `groceries` vznikne `{ 'pečivo': 7, 'mléčné výrobky': 3, 'ovoce a zelenina': 1, 'drogerie': 1 }`.

Je to stejný vzor jako v minulém kroku — tentokrát ho napiš sám, bez nabídky řádků.

- Kategorie, které v seznamu nejsou, ve výsledku nejsou ani jako klíč.
- Prázdný seznam vrátí prázdný objekt.
- Když funkci zavoláš dvakrát se stejným seznamem, oba výsledky jsou stejné.

# --hints--

`quantityByCategory` sečte kusy v každé kategorii.

```js
const items = [
  { id: 7, name: 'Káva', category: 'nápoje', quantity: 2 },
  { id: 8, name: 'Čaj', category: 'nápoje', quantity: 1 },
  { id: 9, name: 'Rýže', category: 'přílohy', quantity: 3 },
];
assert.deepEqual(quantityByCategory(items), { 'nápoje': 3, 'přílohy': 3 }, 'quantityByCategory([Káva 2 ks (nápoje), Čaj 1 ks (nápoje), Rýže 3 ks (přílohy)]) má vrátit { nápoje: 3, přílohy: 3 }');
```

Pro prázdné pole vrátí `quantityByCategory` prázdný objekt.

```js
const result = quantityByCategory([]);
assert.equal(typeof result, 'object', 'quantityByCategory([]) má vrátit objekt');
assert.ok(result !== null, 'quantityByCategory([]) nemá vrátit null');
assert.deepEqual(Object.keys(result), [], 'quantityByCategory([]) má vrátit objekt bez klíčů');
```

Dvě zavolání se stejným seznamem vrátí stejný výsledek.

```js
const items = [{ id: 7, name: 'Káva', category: 'nápoje', quantity: 2 }];
quantityByCategory(items);
assert.deepEqual(quantityByCategory(items), { 'nápoje': 2 }, 'druhé zavolání quantityByCategory([Káva 2 ks]) má vrátit { nápoje: 2 } — vzniká nový akumulátor při každém zavolání?');
```

# --help--

## --tip--

Stejný vzor jako `countByCategory` — viz [Akumulátor nemusí být číslo](see:js-pole/metody-pole-do-hloubky#akumulator-nemusi-byt-cislo).

## --tip--

Kostra je stejná jako v minulém kroku. Liší se jen to, co k dosavadnímu počtu kategorie přičítáš: ne jedničku, ale kusy položky. Prázdný objekt jako počáteční hodnotu vytvoř uvnitř funkce.

## --tip--

Na jiných datech — počet stran podle autora:

```js
const pagesByAuthor = books.reduce((pages, book) => {
  pages[book.author] = (pages[book.author] ?? 0) + book.pages;
  return pages;
}, {});
```

// Nákupní seznam. Každá položka je objekt se stejnými klíči,
// cena (price) je za jeden kus v korunách.
const groceries = [
  { id: 1, name: 'Chleba', category: 'pečivo', price: 45, quantity: 1, bought: false },
  { id: 2, name: 'Mléko', category: 'mléčné výrobky', price: 24, quantity: 2, bought: true },
  { id: 3, name: 'Jablka', category: 'ovoce a zelenina', price: 39, quantity: 1, bought: false },
  { id: 4, name: 'Máslo', category: 'mléčné výrobky', price: 59, quantity: 1, bought: false },
  { id: 5, name: 'Rohlíky', category: 'pečivo', price: 5, quantity: 6, bought: true },
  { id: 6, name: 'Zubní pasta', category: 'drogerie', price: 89, quantity: 1, bought: false },
];

// Vytvoří novou položku seznamu: jeden kus, zatím nekoupená.
function createItem(id, name, category, price) {
  return { id, name, category, price, quantity: 1, bought: false };
}

const categories = ['pečivo', 'mléčné výrobky', 'ovoce a zelenina', 'drogerie'];

// Vrátí první položku seznamu, u prázdného seznamu undefined.
function firstItem(items) {
  return items[0];
}

// Vrátí poslední položku seznamu, u prázdného seznamu undefined.
function lastItem(items) {
  return items.at(-1);
}

// Vrátí nový seznam s položkou přidanou na konec.
function addItem(items, item) {
  return [...items, item];
}

// Vrátí nový seznam bez položky na daném indexu.
function removeAt(items, index) {
  return items.toSpliced(index, 1);
}

// Kolik kusů je na seznamu celkem.
function totalQuantity(items) {
  let total = 0;
  for (const item of items) {
    total += item.quantity;
  }
  return total;
}

// Názvy položek ve stejném pořadí.
function itemNames(items) {
  return items.map((item) => item.name);
}

// Pro každou položku název a cenu za všechny kusy.
function lineTotals(items) {
  return items.map((item) => ({ name: item.name, total: item.price * item.quantity }));
}

// Popisky ve tvaru „Rohlíky (6 ks)".
function itemLabels(items) {
  return items.map((item) => `${item.name} (${item.quantity} ks)`);
}

// Položky, které ještě nejsou v košíku.
function toBuy(items) {
  return items.filter((item) => !item.bought);
}

// Vrátí nový seznam bez položky s daným id.
function removeById(items, id) {
  return items.filter((item) => item.id !== id);
}

// Položka s daným id, nebo undefined.
function findItem(items, id) {
  return items.find((item) => item.id === id);
}

// Index položky s daným id, nebo -1.
function indexById(items, id) {
  return items.findIndex((item) => item.id === id);
}

// Od kolegy: je položka s tímhle id už v seznamu?
function hasItem(items, id) {
  const position = items.findIndex((item) => item.id === id);
  if (position !== -1) {
    return true;
  }
  return false;
}

// Je v seznamu aspoň jedna nekoupená položka?
function hasUnbought(items) {
  return items.some((item) => !item.bought);
}

// Jsou koupené všechny položky? U prázdného seznamu ano.
function allBought(items) {
  return items.every((item) => item.bought);
}

// Zná obchod tuhle kategorii?
function isKnownCategory(category) {
  return categories.includes(category);
}

// Cena celého nákupu v korunách.
function totalPrice(items) {
  return items.reduce((sum, item) => sum + item.price * item.quantity, 0);
}

// Nový seznam seřazený od nejlevnější položky.
function sortedByPrice(items) {
  return items.toSorted((a, b) => a.price - b.price);
}

// Nový seznam seřazený podle názvu podle české abecedy.
function sortedByName(items) {
  return items.toSorted((a, b) => a.name.localeCompare(b.name, 'cs'));
}

// Od kolegy: jednotkové ceny všech položek od nejnižší.
function pricesAscending(items) {
  return items.map((item) => item.price).sort((a, b) => a - b);
}

// Položka, za kterou v nákupu utratíš nejvíc; při shodě ta dřívější.
function priciestItem(items) {
  let priciest;
  for (const item of items) {
    if (priciest === undefined || item.price * item.quantity > priciest.price * priciest.quantity) {
      priciest = item;
    }
  }
  return priciest;
}

function countByCategory(items) {
  return items.reduce((counts, item) => {
    counts[item.category] = (counts[item.category] ?? 0) + 1;
    return counts;
  }, {});
}
