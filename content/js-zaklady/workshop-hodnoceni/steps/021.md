# --description--

Druhý řádek přehledu ukazuje známku.

**Úkol:** Pod `headline` napiš funkci `gradeLine(earned)`, která vrátí `Známka: 1 (výborný)` — číslo a slovní název známky za body `earned`. Kdo nepsal (`null`), dostane `Známka: bez známky`. Když body známku určit nedovolí (třeba 45 bodů ze 40), vrátí `Známka: chyba v bodech`.

`gradeOf` vrací `null` v obou případech, ale do přehledu patří dvě různé zprávy. Rozliš je tím, jestli chybí body.

# --hints--

`gradeLine` vrátí číslo a slovní název známky: 36 → `Známka: 1 (výborný)`, 21 → `Známka: 3 (dobrý)`, 0 → `Známka: 5 (nedostatečný)`.

```js
assert.equal(gradeLine(36), 'Známka: 1 (výborný)', "gradeLine(36) má vrátit 'Známka: 1 (výborný)'");
assert.equal(gradeLine(21), 'Známka: 3 (dobrý)', "gradeLine(21) má vrátit 'Známka: 3 (dobrý)'");
assert.equal(gradeLine(0), 'Známka: 5 (nedostatečný)', "gradeLine(0) má vrátit 'Známka: 5 (nedostatečný)'");
```

`gradeLine(null)` vrátí `Známka: bez známky`.

```js
assert.equal(gradeLine(null), 'Známka: bez známky', "gradeLine(null) má vrátit 'Známka: bez známky'");
```

`gradeLine(45)` vrátí `Známka: chyba v bodech`.

```js
assert.equal(gradeLine(45), 'Známka: chyba v bodech', "gradeLine(45) má vrátit 'Známka: chyba v bodech' — 45 bodů ze 40 známku neurčí");
```

# --help--

## --tip--

Dvě různé příčiny `null` rozlišíš pořadím kontrol — viz [Rozhodnutí zabalené do funkce](see:js-zaklady/porovnani-a-logika#rozhodnuti-zabalene-do-funkce).

## --tip--

Nejdřív se zeptej, jestli chybí body. Až potom si nech spočítat známku a zeptej se, jestli vyšla `null`. Slovní název vrátí `gradeLabel`.

## --tip--

Na jiných datech: `if (order === null) { return 'Bez objednávky'; }` a teprve pod tím výpočet a druhá kontrola `if (price === null) { return 'Chyba v ceníku'; }`.

```js
// Písemka z fyziky, třída 9.A. Maximum je 40 bodů.
const maxPoints = 40;

// Výsledek Terezy z formuláře. Formulář posílá body vždycky jako text.
const studentName = 'Tereza Malá';
const pointsInput = '36';
const attendance = 82; // docházka za pololetí v procentech

// ---- Od kolegy: funkce pro webový přehled třídy ----

// Návrh na pochvalu: aspoň 90 % v písemce a zároveň aspoň 90 % docházky.
function hasPraise(percent, attendancePercent) {
  return percent >= 90 && attendancePercent >= 90;
}

// Barva štítku se známkou v přehledu (barva pro CSS).
function gradeColor(grade) {
  let color;
  switch (grade) {
    case 1:
      color = 'green';
      break;
    case 2:
      color = 'yellowgreen';
      break;
    case 3:
      color = 'gold';
      break;
    case 4:
      color = 'orange';
      break;
    default:
      color = 'red';
  }
  return color;
}

// ---- Tvoje funkce ----

// Úspěšnost v procentech zaokrouhlená na celé číslo.
function percentOf(earned, max) {
  return Math.round((earned / max) * 100);
}

function gradeFor(percent) {
  if (percent < 0 || percent > 100) {
    return null;
  }
  if (percent >= 90) {
    return 1;
  } else if (percent >= 75) {
    return 2;
  } else if (percent >= 50) {
    return 3;
  } else if (percent >= 30) {
    return 4;
  } else {
    return 5;
  }
}

// Slovní poznámka k docházce.
function attendanceNote(attendancePercent) {
  if (attendancePercent >= 90) {
    return 'výborná docházka';
  } else if (attendancePercent >= 75) {
    return 'docházka v pořádku';
  } else if (attendancePercent >= 60) {
    return 'nízká docházka';
  } else {
    return 'nelze klasifikovat';
  }
}

// Slovní název známky na vysvědčení.
function gradeLabel(grade) {
  switch (grade) {
    case 1:
      return 'výborný';
    case 2:
      return 'chvalitebný';
    case 3:
      return 'dobrý';
    case 4:
      return 'dostatečný';
    case 5:
      return 'nedostatečný';
    default:
      return 'neznámá známka';
  }
}

// Text do přehledu: navrhnout pochvalu, nebo ne.
function praiseText(percent, attendancePercent) {
  return hasPraise(percent, attendancePercent) ? 'navrhnout pochvalu' : 'bez pochvaly';
}

// Má si výsledku všimnout třídní učitelka?
function needsAttention(grade, attendancePercent) {
  return grade === 5 || attendancePercent < 60;
}

// Body do tabulky třídy; kdo nepsal, nemá body.
function pointsCell(earned) {
  return earned ?? 'nepsal(a)';
}

// Upozornění k výsledku, nebo null, když není co hlásit.
function warningFor(grade, previousGrade) {
  if (grade === 5) {
    return 'nedostatečná, nabídni opravný termín';
  }
  if (previousGrade !== null && grade - previousGrade >= 2) {
    return `zhoršení o ${grade - previousGrade} stupně`;
  }
  return null;
}

// Tvar slova „bod" podle počtu.
function pointsWord(count) {
  if (count === 1) {
    return 'bod';
  }
  if (count >= 2 && count <= 4) {
    return 'body';
  }
  return 'bodů';
}

// Známka z bodů; kdo nepsal (null), známku nemá.
function gradeOf(earned) {
  if (earned === null) {
    return null;
  }
  return gradeFor(percentOf(earned, maxPoints));
}

// První řádek přehledu, třeba „Tereza Malá: 36 bodů (90 %)".
function headline(name, earned) {
  if (earned === null) {
    return `${name}: nepsal(a)`;
  }
  return `${name}: ${earned} ${pointsWord(earned)} (${percentOf(earned, maxPoints)} %)`;
}

--edit--

// ---- Výpis pro Terezu ----

const studentPoints = Number(pointsInput);
console.log(`${studentName}: ${studentPoints}/${maxPoints} bodů`);

const studentPercent = percentOf(studentPoints, maxPoints);
console.log(`Úspěšnost: ${studentPercent} %`);

const hasPassed = studentPercent >= 50;
console.log('Splněno:', hasPassed);

if (hasPassed) {
  console.log('Písemka je splněná.');
} else {
  console.log('Písemka není splněná, domluv si opravný termín.');
}

const studentGrade = gradeFor(studentPercent);
console.log(`Známka: ${studentGrade}`);
console.log(`Docházka ${attendance} %: ${attendanceNote(attendance)}`);

console.log(`${studentName}: ${studentPercent} % → ${studentGrade} (${gradeLabel(studentGrade)})`);
```
