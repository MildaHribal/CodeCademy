# --description--

Čeština má u počtů tři tvary: `1 bod`, `2 body` (stejně 3 a 4 body), ale `0 bodů`, `5 bodů` a víc.

**Úkol:** Pod `warningFor` napiš funkci `pointsWord(count)`, která pro celé číslo `count` od 0 do 100 vrátí správný tvar slova: `bod`, `body`, nebo `bodů`.

Nástroj vyber sám, rozumných cest je víc a všechny už znáš. Až bude hotovo, rozmysli si i jinou.

# --hints--

`pointsWord(1)` vrátí `bod`.

```js
assert.equal(pointsWord(1), 'bod', "pointsWord(1) má vrátit 'bod'");
```

`pointsWord` vrátí `body` pro 2, 3 a 4.

```js
assert.equal(pointsWord(2), 'body', "pointsWord(2) má vrátit 'body'");
assert.equal(pointsWord(3), 'body', "pointsWord(3) má vrátit 'body'");
assert.equal(pointsWord(4), 'body', "pointsWord(4) má vrátit 'body'");
```

`pointsWord` vrátí `bodů` pro 0 a od 5 výš.

```js
assert.equal(pointsWord(0), 'bodů', "pointsWord(0) má vrátit 'bodů'");
assert.equal(pointsWord(5), 'bodů', "pointsWord(5) má vrátit 'bodů'");
assert.equal(pointsWord(11), 'bodů', "pointsWord(11) má vrátit 'bodů'");
assert.equal(pointsWord(36), 'bodů', "pointsWord(36) má vrátit 'bodů'");
assert.equal(pointsWord(100), 'bodů', "pointsWord(100) má vrátit 'bodů'");
```

# --help--

## --tip--

Tři tvary jsou tři větve — viz [Rozhodování](see:js-zaklady/porovnani-a-logika#rozhodovani-if-else-if-else) a [Skládání podmínek](see:js-zaklady/porovnani-a-logika#skladani-podminek-a).

## --tip--

Rozmysli, který tvar má nejužší podmínku (jediné číslo), který platí pro rozsah od–do a který je „všechno ostatní". To poslední patří na konec.

## --tip--

Rozepsané na podcíle: `// 1. přesně jeden kus → první tvar` `// 2. rozsah dva až čtyři → druhý tvar` `// 3. cokoli jiného → třetí tvar`

const maxPoints = 40;

// Výsledek Terezy z formuláře. Formulář posílá body vždycky jako text.
const studentName = 'Tereza Malá';
const pointsInput = '36';
const attendance = 82; // docházka za pololetí v procentech

// ---- Od kolegy: funkce pro webový přehled třídy ----

// Návrh na pochvalu: aspoň 90 % v písemce a zároveň aspoň 90 % docházky.
function hasPraise(percent, attendancePercent) {
  return percent >= 90 && attendancePercent >= 90;
}

// Barva štítku se známkou v přehledu (barva pro CSS).
function gradeColor(grade) {
  let color;
  switch (grade) {
    case 1:
      color = 'green';
    case 2:
      color = 'yellowgreen';
      break;
    case 3:
      color = 'gold';
      break;
    case 4:
      color = 'orange';
      break;
    default:
      color = 'red';
  }
  return color;
}

// ---- Tvoje funkce ----

// Úspěšnost v procentech zaokrouhlená na celé číslo.
function percentOf(earned, max) {
  return Math.round((earned / max) * 100);
}

function gradeFor(percent) {
  if (percent < 0 || percent > 100) {
    return null;
  }
  if (percent >= 90) {
    return 1;
  } else if (percent >= 75) {
    return 2;
  } else if (percent >= 50) {
    return 3;
  } else if (percent >= 30) {
    return 4;
  } else {
    return 5;
  }
}

// Slovní poznámka k docházce.
function attendanceNote(attendancePercent) {
  if (attendancePercent >= 90) {
    return 'výborná docházka';
  } else if (attendancePercent >= 75) {
    return 'docházka v pořádku';
  } else if (attendancePercent >= 60) {
    return 'nízká docházka';
  } else {
    return 'nelze klasifikovat';
  }
}

// Slovní název známky na vysvědčení.
function gradeLabel(grade) {
  switch (grade) {
    case 1:
      return 'výborný';
    case 2:
      return 'chvalitebný';
    case 3:
      return 'dobrý';
    case 4:
      return 'dostatečný';
    case 5:
      return 'nedostatečný';
    default:
      return 'neznámá známka';
  }
}

// Text do přehledu: navrhnout pochvalu, nebo ne.
function praiseText(percent, attendancePercent) {
  return hasPraise(percent, attendancePercent) ? 'navrhnout pochvalu' : 'bez pochvaly';
}

// Má si výsledku všimnout třídní učitelka?
function needsAttention(grade, attendancePercent) {
  return grade === 5 || attendancePercent < 60;
}

// Body do tabulky třídy; kdo nepsal, nemá body.
function pointsCell(earned) {
  return earned ?? 'nepsal(a)';
}

// Upozornění k výsledku, nebo null, když není co hlásit.
function warningFor(grade, previousGrade) {
  if (grade === 5) {
    return 'nedostatečná, nabídni opravný termín';
  }
  if (previousGrade !== null && grade - previousGrade >= 2) {
    return `zhoršení o ${grade - previousGrade} stupně`;
  }
  return null;
}
