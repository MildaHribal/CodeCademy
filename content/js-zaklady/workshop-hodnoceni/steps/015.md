# --description--

Učitelka chce u každého studenta jeden řádek, ze kterého pozná všechno podstatné.

**Úkol:** Na konec výpisu pro Terezu vypiš souhrnný řádek `Tereza Malá: 90 % → 1 (výborný)`. Jméno, procenta, známka i její slovní název ať pocházejí z proměnných a z tvých funkcí. Šipku `→` zkopíruj ze zadání.

# --hints--

V konzoli je řádek `Tereza Malá: 90 % → 1 (výborný)`.

```js
assert.ok(logs.some((line) => line.text === 'Tereza Malá: 90 % → 1 (výborný)'), 'V konzoli má být přesně řádek Tereza Malá: 90 % → 1 (výborný)');
```

Souhrnný řádek se skládá z dat: pro Petra Nováka s 20 body je `Petr Novák: 50 % → 3 (dobrý)`.

```js
// Spustí tvůj kód znovu s jinými daty a vrátí vybrané proměnné a výpisy.
const rerun = (changes, names = '') => {
  let src = files['script.js'];
  for (const [from, to] of changes) {
    assert.ok(src.includes(from), `Řádek ${from} nech v souboru beze změny — test ho přepíná na jiná data.`);
    src = src.replace(from, to);
  }
  const lines = [];
  const quietConsole = { log: (...parts) => lines.push(parts.map(String).join(' ')) };
  const values = new Function('console', `${src}\nreturn { ${names} };`)(quietConsole);
  return { ...values, lines };
};
const other = rerun([["const studentName = 'Tereza Malá';", "const studentName = 'Petr Novák';"], ["const pointsInput = '36';", "const pointsInput = '20';"]]);
assert.ok(other.lines.includes('Petr Novák: 50 % → 3 (dobrý)'), 'Pro Petra Nováka s 20 body má souhrn ukázat Petr Novák: 50 % → 3 (dobrý) — skládáš řádek z proměnných a z gradeLabel?');
```

# --help--

## --tip--

Hodnoty i volání funkcí vkládáš do [šablonového řetězce](see:js-zaklady/promenne-a-typy#sablonovy-retezec).

## --tip--

Jméno, procenta a známku už máš v konstantách z dřívějších kroků; slovo ke známce vrátí funkce, kterou jsi psal pro vysvědčení.

## --tip--

Rozepsané na podcíle: `// 1. jméno a procenta z konstant` `// 2. známka z konstanty` `// 3. slovo ke známce voláním funkce` `// 4. celé do jednoho výpisu`

// Písemka z fyziky, třída 9.A. Maximum je 40 bodů.
const maxPoints = 40;

// Výsledek Terezy z formuláře. Formulář posílá body vždycky jako text.
const studentName = 'Tereza Malá';
const pointsInput = '36';
const attendance = 82; // docházka za pololetí v procentech

// ---- Od kolegy: funkce pro webový přehled třídy ----

// Návrh na pochvalu: aspoň 90 % v písemce a zároveň aspoň 90 % docházky.
function hasPraise(percent, attendancePercent) {
  return percent >= 90 && attendancePercent >= 90;
}

// Barva štítku se známkou v přehledu (barva pro CSS).
function gradeColor(grade) {
  let color;
  switch (grade) {
    case 1:
      color = 'green';
    case 2:
      color = 'yellowgreen';
      break;
    case 3:
      color = 'gold';
      break;
    case 4:
      color = 'orange';
      break;
    default:
      color = 'red';
  }
  return color;
}

// ---- Tvoje funkce ----

// Úspěšnost v procentech zaokrouhlená na celé číslo.
function percentOf(earned, max) {
  return Math.round((earned / max) * 100);
}

function gradeFor(percent) {
  if (percent < 0 || percent > 100) {
    return null;
  }
  if (percent >= 90) {
    return 1;
  } else if (percent >= 75) {
    return 2;
  } else if (percent >= 50) {
    return 3;
  } else if (percent >= 30) {
    return 4;
  } else {
    return 5;
  }
}

// Slovní poznámka k docházce.
function attendanceNote(attendancePercent) {
  if (attendancePercent >= 90) {
    return 'výborná docházka';
  } else if (attendancePercent >= 75) {
    return 'docházka v pořádku';
  } else if (attendancePercent >= 60) {
    return 'nízká docházka';
  } else {
    return 'nelze klasifikovat';
  }
}

// Slovní název známky na vysvědčení.
function gradeLabel(grade) {
  switch (grade) {
    case 1:
      return 'výborný';
    case 2:
      return 'chvalitebný';
    case 3:
      return 'dobrý';
    case 4:
      return 'dostatečný';
    case 5:
      return 'nedostatečný';
    default:
      return 'neznámá známka';
  }
}

// Text do přehledu: navrhnout pochvalu, nebo ne.
function praiseText(percent, attendancePercent) {
  return hasPraise(percent, attendancePercent) ? 'navrhnout pochvalu' : 'bez pochvaly';
}

// Má si výsledku všimnout třídní učitelka?
function needsAttention(grade, attendancePercent) {
  return grade === 5 || attendancePercent < 60;
}

// Body do tabulky třídy; kdo nepsal, nemá body.
function pointsCell(earned) {
  return earned ?? 'nepsal(a)';
}

// ---- Výpis pro Terezu ----

const studentPoints = Number(pointsInput);
console.log(`${studentName}: ${studentPoints}/${maxPoints} bodů`);

const studentPercent = percentOf(studentPoints, maxPoints);
console.log(`Úspěšnost: ${studentPercent} %`);

const hasPassed = studentPercent >= 50;
console.log('Splněno:', hasPassed);

if (hasPassed) {
  console.log('Písemka je splněná.');
} else {
  console.log('Písemka není splněná, domluv si opravný termín.');
}

const studentGrade = gradeFor(studentPercent);
console.log(`Známka: ${studentGrade}`);
console.log(`Docházka ${attendance} %: ${attendanceNote(attendance)}`);
