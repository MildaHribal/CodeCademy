# --description--

Vedle známky ukazuje přehled i docházku. Pravidla školy: od 90 % `výborná docházka`, od 75 % `docházka v pořádku`, od 60 % `nízká docházka` a pod 60 % `nelze klasifikovat`.

**Úkol:** Pod `gradeFor` napiš funkci `attendanceNote(attendancePercent)`, která vrátí text podle těchto pravidel. Je to stejný vzor jako v minulém kroku, jen s jinými hranicemi a texty — tentokrát ho píšeš sám.

Na jiných datech, cena dopravy podle hmotnosti balíku:

```js
function shippingFor(weightKg) {
  // 1. nejvyšší hranice první
  if (weightKg >= 20) {
    return 249;
  } else if (weightKg >= 5) {
    return 129;
  } else {
    // 2. všechno pod nejnižší hranicí
    return 79;
  }
}
```

# --hints--

`attendanceNote` vrátí `výborná docházka` od 90 %.

```js
assert.equal(attendanceNote(100), 'výborná docházka', "attendanceNote(100) má vrátit 'výborná docházka'");
assert.equal(attendanceNote(90), 'výborná docházka', "attendanceNote(90) má vrátit 'výborná docházka' — hranice patří do vyšší skupiny");
```

`attendanceNote` vrátí `docházka v pořádku` od 75 % a `nízká docházka` od 60 %.

```js
assert.equal(attendanceNote(89), 'docházka v pořádku', "attendanceNote(89) má vrátit 'docházka v pořádku'");
assert.equal(attendanceNote(75), 'docházka v pořádku', "attendanceNote(75) má vrátit 'docházka v pořádku'");
assert.equal(attendanceNote(74), 'nízká docházka', "attendanceNote(74) má vrátit 'nízká docházka'");
assert.equal(attendanceNote(60), 'nízká docházka', "attendanceNote(60) má vrátit 'nízká docházka'");
```

`attendanceNote` vrátí `nelze klasifikovat` pod 60 %.

```js
assert.equal(attendanceNote(59), 'nelze klasifikovat', "attendanceNote(59) má vrátit 'nelze klasifikovat'");
assert.equal(attendanceNote(0), 'nelze klasifikovat', "attendanceNote(0) má vrátit 'nelze klasifikovat'");
```

# --help--

## --tip--

Stejný řetěz jako `gradeFor` — viz [Rozhodování: if, else if, else](see:js-zaklady/porovnani-a-logika#rozhodovani-if-else-if-else).

## --tip--

Začni podmínkou pro nejvyšší hranici a každou větví vrať text. Poslední `else` zachytí všechno pod 60 %.

## --tip--

Porovnej svou funkci s `gradeFor` z minulého kroku: stejná stavba, jen jiné hranice a místo čísel se vracejí texty.

```js
// Písemka z fyziky, třída 9.A. Maximum je 40 bodů.
const maxPoints = 40;

// Výsledek Terezy z formuláře. Formulář posílá body vždycky jako text.
const studentName = 'Tereza Malá';
const pointsInput = '36';
const attendance = 82; // docházka za pololetí v procentech

// ---- Od kolegy: funkce pro webový přehled třídy ----

// Návrh na pochvalu: aspoň 90 % v písemce a zároveň aspoň 90 % docházky.
function hasPraise(percent, attendancePercent) {
  return percent > 90 && attendancePercent > 90;
}

// Barva štítku se známkou v přehledu (barva pro CSS).
function gradeColor(grade) {
  let color;
  switch (grade) {
    case 1:
      color = 'green';
    case 2:
      color = 'yellowgreen';
      break;
    case 3:
      color = 'gold';
      break;
    case 4:
      color = 'orange';
      break;
```

```js
// ---- Tvoje funkce ----

// Úspěšnost v procentech zaokrouhlená na celé číslo.
function percentOf(earned, max) {
  return Math.round((earned / max) * 100);
}

function gradeFor(percent) {
  if (percent >= 90) {
    return 1;
  } else if (percent >= 75) {
    return 2;
  } else if (percent >= 50) {
    return 3;
  } else if (percent >= 30) {
    return 4;
  } else {
    return 5;
  }
}

// Slovní poznámka k docházce.
function attendanceNote(attendancePercent) {
  if (attendancePercent >= 90) {
    return 'výborná docházka';
  } else if (attendancePercent >= 75) {
    return 'docházka v pořádku';
  } else if (attendancePercent >= 60) {
    return 'nízká docházka';
  } else {
    return 'nelze klasifikovat';
  }
}

// ---- Výpis pro Terezu ----

const studentPoints = Number(pointsInput);
console.log(`${studentName}: ${studentPoints}/${maxPoints} bodů`);

const studentPercent = percentOf(studentPoints, maxPoints);
console.log(`Úspěšnost: ${studentPercent} %`);

const hasPassed = studentPercent >= 50;
console.log('Splněno:', hasPassed);

if (hasPassed) {
  console.log('Písemka je splněná.');
} else {
  console.log('Písemka není splněná, domluv si opravný termín.');
}
```
