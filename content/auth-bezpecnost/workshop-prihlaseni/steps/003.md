# --description--

Při přihlášení se heslo nedešifruje, spočítá se znovu a porovná.

**Úkol:** v `passwords.js` napiš a vyexportuj asynchronní funkci
`verifyPassword(password, stored)`, která vrátí `true`, když heslo odpovídá uloženému
řetězci, jinak `false`:

1. rozděl `stored` podle `$` na algoritmus, sůl a hash,
2. když algoritmus není `scrypt`, vrať `false`,
3. převeď sůl i hash zpátky na bajty: `Buffer.from(hex, 'hex')`,
4. spočítej hash zadaného hesla se **stejnou solí** a s délkou **uloženého** hashe,
5. porovnej oba buffery přes `timingSafeEqual` (přidej ho do importu z `node:crypto`).

Rozklad řetězce na části na jiných datech:

```js
const [provider, userId] = 'github$58213'.split('$');
```

> [!PITFALL]
> Když délku výpočtu napíšeš natvrdo (`64`) a v databázi je starší hash s jinou délkou,
> `timingSafeEqual` spadne na `RangeError: Input buffers must have the same byte length`
> a s ním celý požadavek. Délku ber z uloženého hashe.

# --hints--

`verifyPassword` vrátí `true` pro správné heslo a `false` pro špatné.

```js
const { hashPassword, verifyPassword } = await helpers.importFile('passwords.js');
assert.equal(typeof verifyPassword, 'function', 'passwords.js má exportovat funkci verifyPassword');
const stored = await hashPassword('ranni-beh-2026');
assert.equal(await verifyPassword('ranni-beh-2026', stored), true, 'verifyPassword("ranni-beh-2026", hash téhož hesla) má vrátit true');
assert.equal(await verifyPassword('ranni-beh-2027', stored), false, 'verifyPassword("ranni-beh-2027", hash hesla ranni-beh-2026) má vrátit false');
assert.equal(await verifyPassword('Ranni-beh-2026', stored), false, 'verifyPassword s jiným velkým písmenem má vrátit false');
```

Ověří i řetězec, který nevyrobila tvoje `hashPassword`, a vezme z něj sůl i délku hashe (tady 32 bajtů).

```js
const { scrypt } = await import('node:crypto');
const { verifyPassword } = await helpers.importFile('passwords.js');
const salt = Buffer.from('00112233445566778899aabbccddeeff', 'hex');
const key = await new Promise((resolve, reject) =>
  scrypt('Brno2026', salt, 32, (error, derived) => (error ? reject(error) : resolve(derived))));
const stored = `scrypt$${salt.toString('hex')}$${key.toString('hex')}`;
assert.equal(await verifyPassword('Brno2026', stored), true, 'verifyPassword("Brno2026", scrypt$…$<hash 32 bajtů>) má vrátit true — délku výpočtu ber z uloženého hashe');
assert.equal(await verifyPassword('brno2026', stored), false, 'verifyPassword("brno2026", hash hesla Brno2026) má vrátit false');
```

Řetězec s jiným algoritmem než `scrypt` vrátí `false`.

```js
const { verifyPassword } = await helpers.importFile('passwords.js');
const md5 = 'md5$00112233445566778899aabbccddeeff$9f86d081884c7d659a2feaa0c55ad015';
assert.equal(await verifyPassword('Brno2026', md5), false, 'verifyPassword s uloženým řetězcem md5$… má vrátit false');
```

Porovnání hashů používá `timingSafeEqual`.

```js
const source = helpers.stripComments(files['passwords.js'], 'js');
assert.match(source, /timingSafeEqual\s*\(/, 'verifyPassword má porovnat hashe přes timingSafeEqual(…), ne přes === nebo equals');
```

# --explain--

Vysvětli vlastními slovy, proč `verifyPassword` hash porovnává přes `timingSafeEqual`
a proč musí znát sůl z uloženého řetězce.

## --model--

Porovnání přes `===` skončí u prvního rozdílného bajtu, takže z doby odpovědi jde
postupně odhadnout, kolik bajtů sedí. `timingSafeEqual` projde vždy všechny bajty.
Sůl je potřeba, protože hash hesla se solí se dá jen spočítat znovu se stejnou solí;
jiná sůl by dala jiný hash i pro správné heslo.

## --checklist--

- `===` skončí u prvního rozdílu a doba porovnání prozradí, kolik sedí.
- `timingSafeEqual` projde vždy všechny bajty.
- Hash se nedešifruje, spočítá se znovu a porovná.
- Bez stejné soli vyjde pro správné heslo jiný hash.

```js
import { createServer } from 'node:http';
import { openDb } from './db.js';
import { HttpError, readJson, sendJson } from './http.js';

const db = openDb();
const port = Number(process.env.PORT ?? 3000);

async function handle(req, res) {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && pathname === '/api/trainings') {
    const trainings = db
      .prepare('SELECT id, title, starts_at AS startsAt, place FROM trainings ORDER BY starts_at')
      .all();
    return sendJson(res, 200, trainings);
  }

  if (req.method === 'POST' && pathname === '/api/register') {
    const { email, password } = await readJson(req);
    if (typeof email !== 'string' || !email.includes('@')) {
      return sendJson(res, 400, { error: 'Zadej platný e-mail.' });
    }
    if (typeof password !== 'string' || password.length < 8 || password.length > 128) {
      return sendJson(res, 400, { error: 'Heslo musí mít 8 až 128 znaků.' });
    }
    return sendJson(res, 201, { email });
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
}

createServer(async (req, res) => {
  try {
    await handle(req, res);
  } catch (error) {
    if (error instanceof HttpError) {
      return sendJson(res, error.status, { error: error.message });
    }
    console.error(error);
    sendJson(res, 500, { error: 'Na serveru se něco pokazilo.' });
  }
}).listen(port, () => {
  console.log(`Klub na stezce běží na http://localhost:${port}`);
});
```

```js
// Pomocné funkce pro HTTP, stejné jako v serveru knihovny z node-zaklady.

// Chyba, která už ví, jaký stavový kód má klient dostat.
export class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

export function sendJson(res, status, data, headers = {}) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', ...headers });
  res.end(JSON.stringify(data));
}

// Přečte tělo požadavku jako JSON. Moc velké nebo rozbité tělo skončí HttpError.
export async function readJson(req, maxBytes = 10_000) {
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > maxBytes) throw new HttpError(413, 'Tělo požadavku je moc velké.');
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    throw new HttpError(400, 'Tělo požadavku není platný JSON.');
  }
}
```

```js
import { randomBytes, scrypt } from 'node:crypto';
import { promisify } from 'node:util';

// scrypt v Node hlásí výsledek callbackem, promisify z něj udělá funkci s Promise.
const scryptAsync = promisify(scrypt);

// Hesla členů. Uložený tvar: scrypt$<sůl v hex>$<hash v hex>

export async function hashPassword(password) {
  const salt = randomBytes(16);
  const hash = await scryptAsync(password, salt, 64);
  return `scrypt$${salt.toString('hex')}$${hash.toString('hex')}`;
}
```

## --file-- passwords.js

```js
import { randomBytes, scrypt, timingSafeEqual } from 'node:crypto';
import { promisify } from 'node:util';

// scrypt v Node hlásí výsledek callbackem, promisify z něj udělá funkci s Promise.
const scryptAsync = promisify(scrypt);

// Hesla členů. Uložený tvar: scrypt$<sůl v hex>$<hash v hex>

export async function hashPassword(password) {
  const salt = randomBytes(16);
  const hash = await scryptAsync(password, salt, 64);
  return `scrypt$${salt.toString('hex')}$${hash.toString('hex')}`;
}

export async function verifyPassword(password, stored) {
  const [algorithm, saltHex, hashHex] = stored.split('$');
  if (algorithm !== 'scrypt') return false;
  const expected = Buffer.from(hashHex, 'hex');
  const actual = await scryptAsync(password, Buffer.from(saltHex, 'hex'), expected.length);
  return timingSafeEqual(actual, expected);
}
```
