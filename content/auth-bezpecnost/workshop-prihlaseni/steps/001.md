# --description--

Běžecký klub Na stezce má web s tréninky. Teď na něj přidáš účty: registraci, přihlášení,
session v cookie, odhlášení a routy, které pustí jen přihlášené členy. Server je v čistém
Node nad SQLite, bez knihoven, abys viděl každý kus, který za tebe jinak dělá framework.

V `server.js` už je `GET /api/trainings`, v `http.js` funkce `sendJson` a `readJson`
a v `db.js` tabulky `users`, `sessions`, `trainings` a `signups`.

Začni tím, co znáš ze sekce Node.js základy.

**Úkol:** přidej routu `POST /api/register`. Z těla požadavku (JSON) vezme `email`
a `password` a zkontroluje je:

- `email` je řetězec, který obsahuje `@`,
- `password` je řetězec dlouhý 8 až 128 znaků (horní mez chrání server před hashováním
  megabajtových „hesel").

Když něco nesedí, odpoví `400` a `{ "error": "…" }` s českou zprávou. Když je všechno
v pořádku, odpoví zatím `201` a `{ "email": "…" }`. Ukládat do databáze budeš za chvíli.

Vyzkoušej to: **Spustit** a v HTTP klientovi `POST /api/register` s tělem
`{"email": "eva@klub.cz", "password": "ranni-beh-2026"}`.

# --hints--

`POST /api/register` s platným e-mailem a heslem vrátí `201` a `{ email }`.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(new URL('/api/register', server.url), {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ email: 'eva@klub.cz', password: 'ranni-beh-2026' }),
});
assert.equal(res.status, 201, 'POST /api/register s eva@klub.cz a heslem ranni-beh-2026 má vrátit 201');
const body = await res.json();
assert.equal(body.email, 'eva@klub.cz', 'Odpověď na registraci má obsahovat email eva@klub.cz');
```

Chybějící nebo neplatný e-mail vrátí `400` s klíčem `error`.

```js
const server = await helpers.startServer('server.js');
for (const input of [{ password: 'ranni-beh-2026' }, { email: 'eva.klub.cz', password: 'ranni-beh-2026' }, { email: 42, password: 'ranni-beh-2026' }]) {
  const res = await fetch(new URL('/api/register', server.url), {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  });
  assert.equal(res.status, 400, `POST /api/register s ${JSON.stringify(input)} má vrátit 400`);
  const body = await res.json();
  assert.equal(typeof body.error, 'string', `Odpověď na ${JSON.stringify(input)} má mít klíč error s textem`);
}
```

Heslo kratší než 8 nebo delší než 128 znaků, nebo heslo, které není řetězec, vrátí `400`. Přesně 8 a přesně 128 znaků projde.

```js
const server = await helpers.startServer('server.js');
const register = (password) => fetch(new URL('/api/register', server.url), {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ email: 'petr@klub.cz', password }),
});
for (const password of ['kratke7', 'x'.repeat(129), 12345678, undefined]) {
  const res = await register(password);
  assert.equal(res.status, 400, `Heslo ${JSON.stringify(password)?.slice(0, 20) ?? 'undefined'} (délka ${String(password ?? '').length}) má vrátit 400`);
}
assert.equal((await register('osm-znak')).status, 201, 'Heslo o přesně 8 znacích (osm-znak) má projít s 201');
assert.equal((await register('y'.repeat(128))).status, 201, 'Heslo o přesně 128 znacích má projít s 201');
```

Ostatní routy fungují dál: `GET /api/trainings` vrátí tři tréninky.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(new URL('/api/trainings', server.url));
assert.equal(res.status, 200, 'GET /api/trainings má dál vracet 200');
assert.equal((await res.json()).length, 3, 'GET /api/trainings má dál vracet tři tréninky');
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { openDb } from './db.js';
import { HttpError, readJson, sendJson } from './http.js';

const db = openDb();
const port = Number(process.env.PORT ?? 3000);

async function handle(req, res) {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && pathname === '/api/trainings') {
    const trainings = db
      .prepare('SELECT id, title, starts_at AS startsAt, place FROM trainings ORDER BY starts_at')
      .all();
    return sendJson(res, 200, trainings);
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
}

createServer(async (req, res) => {
  try {
    await handle(req, res);
  } catch (error) {
    if (error instanceof HttpError) {
      return sendJson(res, error.status, { error: error.message });
    }
    console.error(error);
    sendJson(res, 500, { error: 'Na serveru se něco pokazilo.' });
  }
}).listen(port, () => {
  console.log(`Klub na stezce běží na http://localhost:${port}`);
});
```

## --file-- http.js

```js
// Pomocné funkce pro HTTP, stejné jako v serveru knihovny z node-zaklady.

// Chyba, která už ví, jaký stavový kód má klient dostat.
export class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

export function sendJson(res, status, data, headers = {}) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', ...headers });
  res.end(JSON.stringify(data));
}

// Přečte tělo požadavku jako JSON. Moc velké nebo rozbité tělo skončí HttpError.
export async function readJson(req, maxBytes = 10_000) {
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > maxBytes) throw new HttpError(413, 'Tělo požadavku je moc velké.');
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    throw new HttpError(400, 'Tělo požadavku není platný JSON.');
  }
}
```

## --file-- db.js

```js
import { DatabaseSync } from 'node:sqlite';

// Databáze běžeckého klubu. Žije jen v paměti: po restartu serveru jsou účty
// a session pryč a tréninky se založí znovu.
export function openDb() {
  const db = new DatabaseSync(':memory:');
  db.exec(`
    CREATE TABLE users (
      id INTEGER PRIMARY KEY,
      email TEXT NOT NULL UNIQUE,
      password_hash TEXT NOT NULL,
      created_at INTEGER NOT NULL
    ) STRICT;

    CREATE TABLE sessions (
      id TEXT PRIMARY KEY,
      user_id INTEGER NOT NULL REFERENCES users (id) ON DELETE CASCADE,
      expires_at INTEGER NOT NULL
    ) STRICT;

    CREATE TABLE trainings (
      id INTEGER PRIMARY KEY,
      title TEXT NOT NULL,
      starts_at TEXT NOT NULL,
      place TEXT NOT NULL
    ) STRICT;

    CREATE TABLE signups (
      training_id INTEGER NOT NULL REFERENCES trainings (id),
      user_id INTEGER NOT NULL REFERENCES users (id) ON DELETE CASCADE,
      PRIMARY KEY (training_id, user_id)
    ) STRICT;

    INSERT INTO trainings (title, starts_at, place) VALUES
      ('Intervaly na dráze', '2026-10-06T18:00', 'Stadion Za Lužánkami'),
      ('Dlouhý běh kolem přehrady', '2026-10-11T09:00', 'Brněnská přehrada, přístaviště'),
      ('Kopce v Pisárkách', '2026-10-13T17:30', 'Pisárky, konečná tramvaje');
  `);
  return db;
}
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { openDb } from './db.js';
import { HttpError, readJson, sendJson } from './http.js';

const db = openDb();
const port = Number(process.env.PORT ?? 3000);

async function handle(req, res) {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && pathname === '/api/trainings') {
    const trainings = db
      .prepare('SELECT id, title, starts_at AS startsAt, place FROM trainings ORDER BY starts_at')
      .all();
    return sendJson(res, 200, trainings);
  }

  if (req.method === 'POST' && pathname === '/api/register') {
    const { email, password } = await readJson(req);
    if (typeof email !== 'string' || !email.includes('@')) {
      return sendJson(res, 400, { error: 'Zadej platný e-mail.' });
    }
    if (typeof password !== 'string' || password.length < 8 || password.length > 128) {
      return sendJson(res, 400, { error: 'Heslo musí mít 8 až 128 znaků.' });
    }
    return sendJson(res, 201, { email });
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
}

createServer(async (req, res) => {
  try {
    await handle(req, res);
  } catch (error) {
    if (error instanceof HttpError) {
      return sendJson(res, error.status, { error: error.message });
    }
    console.error(error);
    sendJson(res, 500, { error: 'Na serveru se něco pokazilo.' });
  }
}).listen(port, () => {
  console.log(`Klub na stezce běží na http://localhost:${port}`);
});
```
