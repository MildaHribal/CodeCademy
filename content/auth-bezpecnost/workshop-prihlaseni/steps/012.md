# --description--

Prohlížeč teď s každým požadavkem posílá hlavičku `Cookie`. Server ji v Node najde
v `req.headers.cookie` jako jeden řetězec, třeba `theme=dark; session=k9Qe7x`, a když
prohlížeč žádnou cookie nemá, je tam `undefined`.

**Úkol:** v `cookies.js` napiš a vyexportuj funkci `parseCookies(header)`, která vrátí
objekt `{ jméno: hodnota }` se všemi cookie z hlavičky. Postup i nástroje jsou na tobě.
Musí zvládnout:

- chybějící hlavičku (`undefined`) → prázdný objekt,
- mezery kolem středníků, i když chybí,
- hodnotu, která sama obsahuje `=` (base64 s doplněním),
- cookie, jejíž jméno končí na `session`, třeba `oldsession`.

# --hints--

`parseCookies('theme=dark; session=k9Qe7x')` vrátí `{ theme: 'dark', session: 'k9Qe7x' }` a `parseCookies(undefined)` vrátí `{}`.

```js
const { parseCookies } = await helpers.importFile('cookies.js');
assert.equal(typeof parseCookies, 'function', 'cookies.js má exportovat funkci parseCookies');
assert.deepEqual({ ...parseCookies('theme=dark; session=k9Qe7x') }, { theme: 'dark', session: 'k9Qe7x' }, "parseCookies('theme=dark; session=k9Qe7x') má vrátit { theme: 'dark', session: 'k9Qe7x' }");
assert.deepEqual({ ...parseCookies(undefined) }, {}, 'parseCookies(undefined) má vrátit prázdný objekt');
```

Mezery kolem středníku nerozhodují: `'lang=cs;session=abc'` i `'lang=cs ;  session=abc'` dají `session` s hodnotou `abc`.

```js
const { parseCookies } = await helpers.importFile('cookies.js');
assert.equal(parseCookies('lang=cs;session=abc').session, 'abc', "parseCookies('lang=cs;session=abc').session má být abc");
assert.equal(parseCookies('lang=cs ;  session=abc').session, 'abc', "parseCookies('lang=cs ;  session=abc').session má být abc");
```

Hodnota s `=` zůstane celá a cookie `oldsession` se s `session` nesplete.

```js
const { parseCookies } = await helpers.importFile('cookies.js');
assert.equal(parseCookies('session=YWJjZA==; theme=dark').session, 'YWJjZA==', "parseCookies('session=YWJjZA==; theme=dark').session má být YWJjZA== včetně rovnítek");
assert.equal(parseCookies('oldsession=x1; session=k9Qe').session, 'k9Qe', "parseCookies('oldsession=x1; session=k9Qe').session má být k9Qe, ne hodnota oldsession");
assert.equal(parseCookies('session=k9Qe; oldsession=x1').session, 'k9Qe', "parseCookies('session=k9Qe; oldsession=x1').session má být k9Qe");
```

# --help--

## --tip--

Jak vypadá hlavička `Cookie` a proč nestačí hledat `session=` v celém textu, je v části
[Cookie: Set-Cookie a Cookie](see:auth-bezpecnost/session-a-cookies#cookie-set-cookie-a-cookie).

## --tip--

Nejdřív rozděl hlavičku na dvojice, pak každou dvojici na jméno a hodnotu. Hodnotu
neber jako „text za posledním `=`", ale jako všechno za **prvním** `=`.

## --tip--

Stejnou úlohu řeší čtení query stringu ručně, jen s jiným oddělovačem:

```text
// 'q=běh&page=2'
// 1. rozděl podle &
// 2. u každé části najdi první =
// 3. vlevo jméno, vpravo hodnota
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { createSession, registerUser } from './auth.js';
import { sessionCookie } from './cookies.js';
import { openDb } from './db.js';
import { HttpError, readJson, sendJson } from './http.js';
import { verifyPassword } from './passwords.js';

const db = openDb();
const port = Number(process.env.PORT ?? 3000);

async function handle(req, res) {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && pathname === '/api/trainings') {
    const trainings = db
      .prepare('SELECT id, title, starts_at AS startsAt, place FROM trainings ORDER BY starts_at')
      .all();
    return sendJson(res, 200, trainings);
  }

  if (req.method === 'POST' && pathname === '/api/register') {
    const body = await readJson(req);
    const email = typeof body.email === 'string' ? body.email.trim().toLowerCase() : '';
    const { password } = body;
    if (!email.includes('@')) {
      return sendJson(res, 400, { error: 'Zadej platný e-mail.' });
    }
    if (typeof password !== 'string' || password.length < 8 || password.length > 128) {
      return sendJson(res, 400, { error: 'Heslo musí mít 8 až 128 znaků.' });
    }
    const existing = db.prepare('SELECT id FROM users WHERE email = ?').get(email);
    if (existing) {
      return sendJson(res, 409, { error: 'Účet s tímhle e-mailem už existuje.' });
    }
    const user = await registerUser(db, { email, password });
    return sendJson(res, 201, user);
  }

  if (req.method === 'POST' && pathname === '/api/login') {
    const body = await readJson(req);
    const email = String(body.email ?? '').trim().toLowerCase();
    const found = db.prepare('SELECT id, email, password_hash FROM users WHERE email = ?').get(email);
    if (!found) {
      return sendJson(res, 401, { error: 'Špatný e-mail nebo heslo.' });
    }
    const passwordOk = await verifyPassword(String(body.password ?? ''), found.password_hash);
    if (!passwordOk) {
      return sendJson(res, 401, { error: 'Špatný e-mail nebo heslo.' });
    }
    const token = createSession(db, found.id);
    return sendJson(res, 200, { id: found.id, email: found.email }, { 'Set-Cookie': sessionCookie(token) });
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
}

createServer(async (req, res) => {
  try {
    await handle(req, res);
  } catch (error) {
    if (error instanceof HttpError) {
      return sendJson(res, error.status, { error: error.message });
    }
    console.error(error);
    sendJson(res, 500, { error: 'Na serveru se něco pokazilo.' });
  }
}).listen(port, () => {
  console.log(`Klub na stezce běží na http://localhost:${port}`);
});
```

## --file-- http.js

```js
// Pomocné funkce pro HTTP, stejné jako v serveru knihovny z node-zaklady.

// Chyba, která už ví, jaký stavový kód má klient dostat.
export class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

export function sendJson(res, status, data, headers = {}) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', ...headers });
  res.end(JSON.stringify(data));
}

// Přečte tělo požadavku jako JSON. Moc velké nebo rozbité tělo skončí HttpError.
export async function readJson(req, maxBytes = 10_000) {
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > maxBytes) throw new HttpError(413, 'Tělo požadavku je moc velké.');
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    throw new HttpError(400, 'Tělo požadavku není platný JSON.');
  }
}
```

## --file-- passwords.js

```js
import { randomBytes, scrypt, timingSafeEqual } from 'node:crypto';
import { promisify } from 'node:util';

// scrypt v Node hlásí výsledek callbackem, promisify z něj udělá funkci s Promise.
const scryptAsync = promisify(scrypt);

// Hesla členů. Uložený tvar: scrypt$<sůl v hex>$<hash v hex>

export async function hashPassword(password) {
  const salt = randomBytes(16);
  const hash = await scryptAsync(password, salt, 64);
  return `scrypt$${salt.toString('hex')}$${hash.toString('hex')}`;
}

export async function verifyPassword(password, stored) {
  const [algorithm, saltHex, hashHex] = stored.split('$');
  if (algorithm !== 'scrypt') return false;
  const expected = Buffer.from(hashHex, 'hex');
  const actual = await scryptAsync(password, Buffer.from(saltHex, 'hex'), expected.length);
  return timingSafeEqual(actual, expected);
}
```

## --file-- auth.js

```js
import { randomBytes } from 'node:crypto';
import { hashPassword } from './passwords.js';

// Účty a session. Každá funkce dostane otevřenou databázi db.

const SESSION_DAYS = 7;

export async function registerUser(db, { email, password }) {
  const passwordHash = await hashPassword(password);
  const { lastInsertRowid } = db
    .prepare('INSERT INTO users (email, password_hash, created_at) VALUES (?, ?, ?)')
    .run(email, passwordHash, Date.now());
  return { id: Number(lastInsertRowid), email };
}

export function createSession(db, userId) {
  const token = randomBytes(32).toString('base64url');
  const expiresAt = Date.now() + SESSION_DAYS * 24 * 60 * 60 * 1000;
  db.prepare('INSERT INTO sessions (id, user_id, expires_at) VALUES (?, ?, ?)').run(token, userId, expiresAt);
  return token;
}
```

## --file-- cookies.js

```js
// Session cookie: hlavička Set-Cookie pro prohlížeč a čtení hlavičky Cookie.

const MAX_AGE_SECONDS = 7 * 24 * 60 * 60;

export function sessionCookie(token) {
  return `session=${token}; HttpOnly; SameSite=Lax; Path=/; Max-Age=${MAX_AGE_SECONDS}`;
}
```

```js
export function parseCookies(header = '') {
  const cookies = {};
  for (const pair of header.split(';')) {
    const index = pair.indexOf('=');
    if (index === -1) continue;
    cookies[pair.slice(0, index).trim()] = pair.slice(index + 1).trim();
  }
  return cookies;
}
```
