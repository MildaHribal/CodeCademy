# --description--

Tady máš hotové účty: registraci, přihlášení, `GET /api/me` a odhlášení. Klidně si dej
pauzu. Teď přijde to, kvůli čemu účty existují: akce, kterou smí jen přihlášený člen.

**Úkol:** přidej `POST /api/trainings/:id/signups`, kterým se přihlášený člen zapíše
na trénink s daným `id`.

- Nepřihlášený dostane `401` a `{ error }`.
- Přihlášený dostane `201` a `{ trainingId, userId }` a v tabulce `signups` vznikne řádek.
- Člen se zapisuje vždycky **sám za sebe**: kdo to je, určí session, ne tělo požadavku.
  Když klient pošle v těle cizí `userId`, server ho ignoruje.
- Opakovaný zápis na stejný trénink server neshodí a vrátí zase `201`.

Id tréninku vytáhneš z cesty stejně jako id knihy v `node-zaklady`.

# --hints--

Bez přihlášení vrátí `POST /api/trainings/1/signups` stav `401`.

```js
const server = await helpers.startServer('server.js');
const res = await fetch(new URL('/api/trainings/1/signups', server.url), { method: 'POST' });
assert.equal(res.status, 401, 'POST /api/trainings/1/signups bez cookie má vrátit 401');
assert.equal(typeof (await res.json()).error, 'string', 'Odpověď 401 má mít klíč error s textem');
```

Přihlášený člen dostane `201` a `{ trainingId, userId }` se svým id, i když v těle pošle cizí `userId`.

```js
const server = await helpers.startServer('server.js');
const post = (path, body, cookie = '') => fetch(new URL(path, server.url), {
  method: 'POST',
  headers: { 'Content-Type': 'application/json', Cookie: cookie },
  body: JSON.stringify(body ?? {}),
});
await post('/api/register', { email: 'eva@klub.cz', password: 'ranni-beh-2026' });
await post('/api/register', { email: 'petr@klub.cz', password: 'vecerni-beh-2026' });
const login = await post('/api/login', { email: 'petr@klub.cz', password: 'vecerni-beh-2026' });
const cookie = (login.headers.get('set-cookie') ?? '').split(';')[0];
const res = await post('/api/trainings/3/signups', { userId: 1 }, cookie);
assert.equal(res.status, 201, 'POST /api/trainings/3/signups přihlášeného petr@klub.cz má vrátit 201');
assert.deepEqual(await res.json(), { trainingId: 3, userId: 2 }, 'Odpověď má být { trainingId: 3, userId: 2 } — člen se zapisuje sám za sebe, userId z těla se ignoruje');
```

# --help--

## --tip--

Chráněná akce začíná autentizací, viz
[Autentizace a autorizace](see:auth-bezpecnost/session-a-cookies#autentizace-a-autorizace).
Id z cesty znáš z lekce [HTTP v Node](see:node-zaklady/http-v-node#adresa-req-url-cesta-a-query-string).

## --tip--

Cesta obsahuje číslo, takže přesné porovnání `pathname === …` nepomůže; hodí se regulární
výraz nebo rozdělení cesty podle `/`. Druhý zápis stejné dvojice by narazil na primární
klíč tabulky `signups`: SQLite umí `INSERT OR IGNORE`, nebo řádek předem vyhledej.

## --tip--

Stejný tvar má „líbí se mi" u fotky z běhu:

```text
// POST /api/photos/:id/likes
// 1. kdo je přihlášený? nikdo → 401
// 2. id fotky z cesty
// 3. vlož (photo_id, user.id), duplicitu ignoruj
// 4. 201 { photoId, userId }
```

## --file-- server.js

```js
import { createServer } from 'node:http';
import { createSession, registerUser } from './auth.js';
import { parseCookies, sessionCookie } from './cookies.js';
import { getCurrentUser } from './current-user.js';
import { openDb } from './db.js';
import { HttpError, readJson, sendJson } from './http.js';
import { verifyPassword } from './passwords.js';

const db = openDb();
const port = Number(process.env.PORT ?? 3000);

async function handle(req, res) {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && pathname === '/api/trainings') {
    const trainings = db
      .prepare('SELECT id, title, starts_at AS startsAt, place FROM trainings ORDER BY starts_at')
      .all();
    return sendJson(res, 200, trainings);
  }

  if (req.method === 'GET' && pathname === '/api/me') {
    const user = getCurrentUser(db, req);
    if (!user) {
      return sendJson(res, 401, { error: 'Nejdřív se přihlas.' });
    }
    return sendJson(res, 200, user);
  }

  if (req.method === 'POST' && pathname === '/api/logout') {
    // Odhlášení dopsal Ondra.
    const token = parseCookies(req.headers.cookie).session;
    if (token) db.prepare('DELETE FROM sessions WHERE id = ?').run(token);
    return sendJson(res, 200, { ok: true }, { 'Set-Cookie': 'session=; HttpOnly; SameSite=Lax; Path=/; Max-Age=0' });
  }

  if (req.method === 'POST' && pathname === '/api/register') {
    const body = await readJson(req);
    const email = typeof body.email === 'string' ? body.email.trim().toLowerCase() : '';
    const { password } = body;
    if (!email.includes('@')) {
      return sendJson(res, 400, { error: 'Zadej platný e-mail.' });
    }
    if (typeof password !== 'string' || password.length < 8 || password.length > 128) {
      return sendJson(res, 400, { error: 'Heslo musí mít 8 až 128 znaků.' });
    }
    const existing = db.prepare('SELECT id FROM users WHERE email = ?').get(email);
    if (existing) {
      return sendJson(res, 409, { error: 'Účet s tímhle e-mailem už existuje.' });
    }
    const user = await registerUser(db, { email, password });
    return sendJson(res, 201, user);
  }

  if (req.method === 'POST' && pathname === '/api/login') {
    const body = await readJson(req);
    const email = String(body.email ?? '').trim().toLowerCase();
    const found = db.prepare('SELECT id, email, password_hash FROM users WHERE email = ?').get(email);
    if (!found) {
      return sendJson(res, 401, { error: 'Špatný e-mail nebo heslo.' });
    }
    const passwordOk = await verifyPassword(String(body.password ?? ''), found.password_hash);
    if (!passwordOk) {
      return sendJson(res, 401, { error: 'Špatný e-mail nebo heslo.' });
    }
    const token = createSession(db, found.id);
    return sendJson(res, 200, { id: found.id, email: found.email }, { 'Set-Cookie': sessionCookie(token) });
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
}

createServer(async (req, res) => {
  try {
    await handle(req, res);
  } catch (error) {
    if (error instanceof HttpError) {
      return sendJson(res, error.status, { error: error.message });
    }
    console.error(error);
    sendJson(res, 500, { error: 'Na serveru se něco pokazilo.' });
  }
}).listen(port, () => {
  console.log(`Klub na stezce běží na http://localhost:${port}`);
});
```

## --file-- http.js

```js
// Pomocné funkce pro HTTP, stejné jako v serveru knihovny z node-zaklady.

// Chyba, která už ví, jaký stavový kód má klient dostat.
export class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

export function sendJson(res, status, data, headers = {}) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', ...headers });
  res.end(JSON.stringify(data));
}

// Přečte tělo požadavku jako JSON. Moc velké nebo rozbité tělo skončí HttpError.
export async function readJson(req, maxBytes = 10_000) {
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > maxBytes) throw new HttpError(413, 'Tělo požadavku je moc velké.');
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    throw new HttpError(400, 'Tělo požadavku není platný JSON.');
  }
}
```

```js
import { DatabaseSync } from 'node:sqlite';

// Databáze běžeckého klubu. Žije jen v paměti: po restartu serveru jsou účty
// a session pryč a tréninky se založí znovu.
export function openDb() {
  const db = new DatabaseSync(':memory:');
  db.exec(`
    CREATE TABLE users (
      id INTEGER PRIMARY KEY,
      email TEXT NOT NULL UNIQUE,
      password_hash TEXT NOT NULL,
      created_at INTEGER NOT NULL
    ) STRICT;

    CREATE TABLE sessions (
      id TEXT PRIMARY KEY,
      user_id INTEGER NOT NULL REFERENCES users (id) ON DELETE CASCADE,
      expires_at INTEGER NOT NULL
    ) STRICT;

    CREATE TABLE trainings (
      id INTEGER PRIMARY KEY,
      title TEXT NOT NULL,
      starts_at TEXT NOT NULL,
      place TEXT NOT NULL
    ) STRICT;

    CREATE TABLE signups (
      training_id INTEGER NOT NULL REFERENCES trainings (id),
      user_id INTEGER NOT NULL REFERENCES users (id) ON DELETE CASCADE,
      PRIMARY KEY (training_id, user_id)
    ) STRICT;

    INSERT INTO trainings (title, starts_at, place) VALUES
      ('Intervaly na dráze', '2026-10-06T18:00', 'Stadion Za Lužánkami'),

      ('Kopce v Pisárkách', '2026-10-13T17:30', 'Pisárky, konečná tramvaje');
  `);
  return db;
}
```

## --file-- passwords.js

```js
import { randomBytes, scrypt, timingSafeEqual } from 'node:crypto';
import { promisify } from 'node:util';

// scrypt v Node hlásí výsledek callbackem, promisify z něj udělá funkci s Promise.
const scryptAsync = promisify(scrypt);

// Hesla členů. Uložený tvar: scrypt$<sůl v hex>$<hash v hex>

export async function hashPassword(password) {
  const salt = randomBytes(16);
  const hash = await scryptAsync(password, salt, 64);
  return `scrypt$${salt.toString('hex')}$${hash.toString('hex')}`;
}

export async function verifyPassword(password, stored) {
  const [algorithm, saltHex, hashHex] = stored.split('$');
  if (algorithm !== 'scrypt') return false;
  const expected = Buffer.from(hashHex, 'hex');
  const actual = await scryptAsync(password, Buffer.from(saltHex, 'hex'), expected.length);
  return timingSafeEqual(actual, expected);
}
```

## --file-- auth.js

```js
import { randomBytes } from 'node:crypto';
import { hashPassword } from './passwords.js';

// Účty a session. Každá funkce dostane otevřenou databázi db.

const SESSION_DAYS = 7;

export async function registerUser(db, { email, password }) {
  const passwordHash = await hashPassword(password);
  const { lastInsertRowid } = db
    .prepare('INSERT INTO users (email, password_hash, created_at) VALUES (?, ?, ?)')
    .run(email, passwordHash, Date.now());
  return { id: Number(lastInsertRowid), email };
}

export function createSession(db, userId) {
  const token = randomBytes(32).toString('base64url');
  const expiresAt = Date.now() + SESSION_DAYS * 24 * 60 * 60 * 1000;
  db.prepare('INSERT INTO sessions (id, user_id, expires_at) VALUES (?, ?, ?)').run(token, userId, expiresAt);
  return token;
}
```

```js
// Session cookie: hlavička Set-Cookie pro prohlížeč a čtení hlavičky Cookie.

const MAX_AGE_SECONDS = 7 * 24 * 60 * 60;

export function sessionCookie(token) {
  return `session=${token}; HttpOnly; SameSite=Lax; Path=/; Max-Age=${MAX_AGE_SECONDS}`;
}

export function parseCookies(header = '') {
  const cookies = {};
  for (const pair of header.split(';')) {
    const index = pair.indexOf('=');
    if (index === -1) continue;
    cookies[pair.slice(0, index).trim()] = pair.slice(index + 1).trim();
  }
  return cookies;
}
```

## --file-- current-user.js

```js
import { parseCookies } from './cookies.js';

// Najde přihlášeného člena podle cookie session. Když session chybí, neexistuje
// nebo vypršela, vrátí null.
export function getCurrentUser(db, req) {
  const token = parseCookies(req.headers.cookie).session;
  if (!token) return null;
  const user = db
    .prepare('SELECT users.id, users.email FROM sessions JOIN users ON users.id = sessions.user_id WHERE sessions.id = ? AND sessions.expires_at > ?')
    .get(token, Date.now());
  return user ?? null;
}
```

```js
import { createServer } from 'node:http';
import { createSession, registerUser } from './auth.js';
import { parseCookies, sessionCookie } from './cookies.js';
import { getCurrentUser } from './current-user.js';
import { openDb } from './db.js';
import { HttpError, readJson, sendJson } from './http.js';
import { verifyPassword } from './passwords.js';

const db = openDb();
const port = Number(process.env.PORT ?? 3000);

async function handle(req, res) {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && pathname === '/api/trainings') {
    const trainings = db
      .prepare('SELECT id, title, starts_at AS startsAt, place FROM trainings ORDER BY starts_at')
      .all();
    return sendJson(res, 200, trainings);
  }

  if (req.method === 'GET' && pathname === '/api/me') {
    const user = getCurrentUser(db, req);
    if (!user) {
      return sendJson(res, 401, { error: 'Nejdřív se přihlas.' });
    }
    return sendJson(res, 200, user);
  }

  const signupMatch = pathname.match(/^\/api\/trainings\/(\d+)\/signups$/);
  if (req.method === 'POST' && signupMatch) {
    const user = getCurrentUser(db, req);
    if (!user) {
      return sendJson(res, 401, { error: 'Nejdřív se přihlas.' });
    }
    const trainingId = Number(signupMatch[1]);
    db.prepare('INSERT OR IGNORE INTO signups (training_id, user_id) VALUES (?, ?)').run(trainingId, user.id);
    return sendJson(res, 201, { trainingId, userId: user.id });
  }

  if (req.method === 'POST' && pathname === '/api/logout') {
    // Odhlášení dopsal Ondra.
    const token = parseCookies(req.headers.cookie).session;
    if (token) db.prepare('DELETE FROM sessions WHERE id = ?').run(token);
    return sendJson(res, 200, { ok: true }, { 'Set-Cookie': 'session=; HttpOnly; SameSite=Lax; Path=/; Max-Age=0' });
  }

  if (req.method === 'POST' && pathname === '/api/register') {
    const body = await readJson(req);
    const email = typeof body.email === 'string' ? body.email.trim().toLowerCase() : '';
    const { password } = body;
    if (!email.includes('@')) {
      return sendJson(res, 400, { error: 'Zadej platný e-mail.' });
    }
    if (typeof password !== 'string' || password.length < 8 || password.length > 128) {
      return sendJson(res, 400, { error: 'Heslo musí mít 8 až 128 znaků.' });
    }
    const existing = db.prepare('SELECT id FROM users WHERE email = ?').get(email);
    if (existing) {
      return sendJson(res, 409, { error: 'Účet s tímhle e-mailem už existuje.' });
    }
    const user = await registerUser(db, { email, password });
    return sendJson(res, 201, user);
  }

  if (req.method === 'POST' && pathname === '/api/login') {
    const body = await readJson(req);
    const email = String(body.email ?? '').trim().toLowerCase();
    const found = db.prepare('SELECT id, email, password_hash FROM users WHERE email = ?').get(email);
    if (!found) {
      return sendJson(res, 401, { error: 'Špatný e-mail nebo heslo.' });
    }
    const passwordOk = await verifyPassword(String(body.password ?? ''), found.password_hash);
    if (!passwordOk) {
      return sendJson(res, 401, { error: 'Špatný e-mail nebo heslo.' });
    }
    const token = createSession(db, found.id);
    return sendJson(res, 200, { id: found.id, email: found.email }, { 'Set-Cookie': sessionCookie(token) });
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
}

createServer(async (req, res) => {
  try {
    await handle(req, res);
  } catch (error) {
    if (error instanceof HttpError) {
      return sendJson(res, error.status, { error: error.message });
    }
    console.error(error);
    sendJson(res, 500, { error: 'Na serveru se něco pokazilo.' });
  }
}).listen(port, () => {
  console.log(`Klub na stezce běží na http://localhost:${port}`);
});
```
