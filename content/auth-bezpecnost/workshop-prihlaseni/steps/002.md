# --description--

Heslo se do databáze nikdy neuloží, uloží se jeho pomalý [[hash hesla|hash]] se [[sůl|solí]].
V novém souboru `passwords.js` je připravená funkce `scryptAsync`.

**Úkol:** napiš a vyexportuj asynchronní funkci `hashPassword(password)`, která:

1. vygeneruje sůl: 16 náhodných bajtů přes `randomBytes(16)`,
2. spočítá hash: `await scryptAsync(password, salt, 64)` (64 = délka výsledku v bajtech),
3. vrátí jeden řetězec `scrypt$<sůl>$<hash>`, sůl i hash převedené přes `.toString('hex')`.

Stejný postup na jiných datech: jednorázový kód pro obnovu účtu, uložený jako base64
s dvojtečkou:

```js
export async function hashRecoveryCode(code) {
  const salt = randomBytes(8);
  const key = await scryptAsync(code, salt, 32);
  return `${salt.toString('base64')}:${key.toString('base64')}`;
}
```

Výsledek si můžeš vypsat: dočasně dej na konec souboru
`console.log(await hashPassword('ranni-beh-2026'))` a spusť soubor přes `node passwords.js`
v terminálu. Dvě spuštění vypíšou dva různé řetězce, to je sůl.

# --hints--

`hashPassword('ranni-beh-2026')` vrátí řetězec ve tvaru `scrypt$<32 hex znaků>$<128 hex znaků>`.

```js
const { hashPassword } = await helpers.importFile('passwords.js');
assert.equal(typeof hashPassword, 'function', 'passwords.js má exportovat funkci hashPassword');
const stored = await hashPassword('ranni-beh-2026');
assert.equal(typeof stored, 'string', 'hashPassword má vrátit řetězec (nezapomněl jsi await u scryptAsync?)');
assert.match(stored, /^scrypt\$[0-9a-f]{32}\$[0-9a-f]{128}$/, `hashPassword('ranni-beh-2026') má vrátit scrypt$<sůl 16 bajtů hex>$<hash 64 bajtů hex>, vrátila: ${stored.slice(0, 60)}…`);
```

Dvě volání se stejným heslem vrátí různé řetězce (každé má jinou sůl).

```js
const { hashPassword } = await helpers.importFile('passwords.js');
const first = await hashPassword('ranni-beh-2026');
const second = await hashPassword('ranni-beh-2026');
assert.notEqual(first, second, 'Dvě volání hashPassword se stejným heslem mají vrátit různé řetězce — sůl se generuje při každém volání');
```

Hash v řetězci je opravdu `scrypt` zadaného hesla se solí z řetězce.

```js
const { scrypt } = await import('node:crypto');
const { hashPassword } = await helpers.importFile('passwords.js');
const [, saltHex, hashHex] = (await hashPassword('Brno 2026')).split('$');
const expected = await new Promise((resolve, reject) =>
  scrypt('Brno 2026', Buffer.from(saltHex, 'hex'), 64, (error, key) => (error ? reject(error) : resolve(key))));
assert.equal(hashHex, expected.toString('hex'), 'Hash z hashPassword("Brno 2026") má být scrypt hesla se solí z řetězce a délkou 64 bajtů');
```

```js
import { createServer } from 'node:http';
import { openDb } from './db.js';
import { HttpError, readJson, sendJson } from './http.js';

const db = openDb();
const port = Number(process.env.PORT ?? 3000);

async function handle(req, res) {
  const { pathname } = new URL(req.url, 'http://localhost');

  if (req.method === 'GET' && pathname === '/api/trainings') {
    const trainings = db
      .prepare('SELECT id, title, starts_at AS startsAt, place FROM trainings ORDER BY starts_at')
      .all();
    return sendJson(res, 200, trainings);
  }

  if (req.method === 'POST' && pathname === '/api/register') {
    const { email, password } = await readJson(req);
    if (typeof email !== 'string' || !email.includes('@')) {
      return sendJson(res, 400, { error: 'Zadej platný e-mail.' });
    }
    if (typeof password !== 'string' || password.length < 8 || password.length > 128) {
      return sendJson(res, 400, { error: 'Heslo musí mít 8 až 128 znaků.' });
    }
    return sendJson(res, 201, { email });
  }

  sendJson(res, 404, { error: 'Tahle adresa neexistuje.' });
}

createServer(async (req, res) => {
  try {
    await handle(req, res);
  } catch (error) {
    if (error instanceof HttpError) {
      return sendJson(res, error.status, { error: error.message });
    }
    console.error(error);
    sendJson(res, 500, { error: 'Na serveru se něco pokazilo.' });
  }
}).listen(port, () => {
  console.log(`Klub na stezce běží na http://localhost:${port}`);
});
```

```js
// Pomocné funkce pro HTTP, stejné jako v serveru knihovny z node-zaklady.

// Chyba, která už ví, jaký stavový kód má klient dostat.
export class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

export function sendJson(res, status, data, headers = {}) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', ...headers });
  res.end(JSON.stringify(data));
}

// Přečte tělo požadavku jako JSON. Moc velké nebo rozbité tělo skončí HttpError.
export async function readJson(req, maxBytes = 10_000) {
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > maxBytes) throw new HttpError(413, 'Tělo požadavku je moc velké.');
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    throw new HttpError(400, 'Tělo požadavku není platný JSON.');
  }
}
```

## --file-- passwords.js

```js
import { randomBytes, scrypt } from 'node:crypto';
import { promisify } from 'node:util';

// scrypt v Node hlásí výsledek callbackem, promisify z něj udělá funkci s Promise.
const scryptAsync = promisify(scrypt);

// Hesla členů. Uložený tvar: scrypt$<sůl v hex>$<hash v hex>

export async function hashPassword(password) {
  const salt = randomBytes(16);
  const hash = await scryptAsync(password, salt, 64);
  return `scrypt$${salt.toString('hex')}$${hash.toString('hex')}`;
}
```
