# --description--

Když v obsluze požadavku něco spadne, bazar pošle klientovi celou chybu i s výpisem
zásobníku. Vyzkoušej to: přihlas se a založ inzerát s cenou `1999.5` — sloupec `price`
v databázi je celé číslo, takže zápis skončí chybou a odpověď vypadá takhle:

```text
500 {"error":"cannot store REAL value in INTEGER column listings.price",
     "stack":"Error: cannot store REAL value in INTEGER column listings.price
        at obsluz (file:///home/bazar/server.js:164:53) ..."}
```

Útočník se z toho dozví jména tabulek a sloupců, absolutní cesty na serveru i to, na jakém
řádku kód leží. Tyhle drobnosti dohromady dávají mapu k dalšímu útoku.

**Úkol:** uprav zpracování chyb v `server.js` (blok `catch` kolem volání `obsluz`) tak, aby:

- klient dostal dál `500`, ale v těle jen obecnou českou hlášku v klíči `error`,
- v odpovědi nebyl výpis zásobníku, hláška databáze, jména sloupců ani cesty k souborům,
- **podrobnosti se vypsaly do logu serveru**, ať víš, co se stalo — na to je `console.error`.

    CREATE TABLE users (
      id INTEGER PRIMARY KEY,
      email TEXT NOT NULL UNIQUE,
      password_hash TEXT NOT NULL,
      role TEXT NOT NULL
    ) STRICT;

    CREATE TABLE sessions (
      id TEXT PRIMARY KEY,
      user_id INTEGER NOT NULL REFERENCES users (id),
      expires_at INTEGER NOT NULL
    ) STRICT;

    CREATE TABLE listings (
      id INTEGER PRIMARY KEY,
      author_id INTEGER NOT NULL REFERENCES users (id),
      title TEXT NOT NULL,
      description TEXT NOT NULL,
      price INTEGER NOT NULL,
      web TEXT NOT NULL
    ) STRICT;

    CREATE TABLE messages (
      id INTEGER PRIMARY KEY,
      listing_id INTEGER NOT NULL REFERENCES listings (id),
      sender_id INTEGER NOT NULL REFERENCES users (id),
      recipient_id INTEGER NOT NULL REFERENCES users (id),
      body TEXT NOT NULL
    ) STRICT;
  `);

## --file-- hesla.js

```js
import { randomBytes, scryptSync, timingSafeEqual } from 'node:crypto';

// Otisk hesla: náhodná sůl a pomalý hash. Sůl se ukládá vedle otisku,
// tajná být nemusí — jde o to, aby dvě stejná hesla měla různý otisk.
export function zahashujHeslo(heslo) {
  const sul = randomBytes(16).toString('hex');
  const otisk = scryptSync(heslo, sul, 64).toString('hex');
  return `scrypt:${sul}:${otisk}`;
}

// Porovnání v konstantním čase: neskončí u prvního rozdílu.
export function hesloSedi(heslo, ulozenyHash) {
  const [algoritmus, sul, otisk] = String(ulozenyHash ?? '').split(':');
  if (algoritmus !== 'scrypt' || !sul || !otisk) return false;
  const ocekavany = Buffer.from(otisk, 'hex');
  const spocitany = scryptSync(heslo, sul, ocekavany.length);
  return spocitany.length === ocekavany.length && timingSafeEqual(spocitany, ocekavany);
}
```

// Escapování hodnot, které se vkládají do HTML stránky.
export function escapeHtml(hodnota) {
  return String(hodnota)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#39;');
}

// Odkaz od uživatele smí do stránky jen tehdy, když vede přes http(s).
// Schéma javascript: by se po kliknutí spustilo jako kód.
export function bezpecnyOdkaz(adresa) {
  const text = String(adresa ?? '').trim();
  return /^https?:\/\//i.test(text) ? text : '#';
}

## --file-- server.js

```js
import { createServer } from 'node:http';
import { randomBytes } from 'node:crypto';
import { otevriDb } from './db.js';
import { hesloSedi, zahashujHeslo } from './hesla.js';
import { strankaInzeratu } from './sablona.js';

const db = otevriDb();
const port = Number(process.env.PORT ?? 3000);
const DEN = 24 * 60 * 60 * 1000;

// Jméno sloupce parametrem být nemůže. Projde jen to, co je tady.
const SLOUPCE = { id: 'id', cena: 'price', nazev: 'title' };

function vyrobSessionId() {
  return randomBytes(24).toString('base64url');
}

function posliJson(res, status, data, hlavicky = {}) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', ...hlavicky });
  res.end(JSON.stringify(data));
}

function posliHtml(res, status, html) {
  res.writeHead(status, { 'Content-Type': 'text/html; charset=utf-8' });
  res.end(html);
}

async function prectiJson(req) {
  const kusy = [];
  for await (const kus of req) kusy.push(kus);
  if (kusy.length === 0) return {};
  try {
    return JSON.parse(Buffer.concat(kusy).toString('utf8'));
  } catch {
    return {};
  }
}

function prectiCookie(req, jmeno) {
  for (const kus of (req.headers.cookie ?? '').split(';')) {
    const [klic, ...zbytek] = kus.trim().split('=');
    if (klic === jmeno) return zbytek.join('=');
  }
  return null;
}

function prihlasenyUzivatel(req) {
  const sid = prectiCookie(req, 'sid');
  if (!sid) return null;
  const session = db.prepare('SELECT user_id, expires_at FROM sessions WHERE id = ?').get(sid);
  if (!session || session.expires_at < Date.now()) return null;
  return db.prepare('SELECT id, email, role FROM users WHERE id = ?').get(session.user_id);
}

async function obsluz(req, res, url) {
  const { pathname } = url;

  // Hledání v inzerátech.
  if (req.method === 'GET' && pathname === '/api/inzeraty') {
    const hledej = url.searchParams.get('hledej') ?? '';
    const razeni = SLOUPCE[url.searchParams.get('razeni')] ?? 'id';
    const dotaz = `SELECT id, title AS nazev, price AS cena FROM listings
                   WHERE title LIKE '%' || ? || '%' ORDER BY ${razeni}`;
    return posliJson(res, 200, db.prepare(dotaz).all(hledej));
  }

  // Stránka jednoho inzerátu.
  const detail = pathname.match(/^\/inzerat\/(\d+)$/);
  if (req.method === 'GET' && detail) {
    const inzerat = db.prepare('SELECT * FROM listings WHERE id = ?').get(Number(detail[1]));
    if (!inzerat) return posliHtml(res, 404, '<p>Takový inzerát tu není.</p>');
    const autor = db.prepare('SELECT email FROM users WHERE id = ?').get(inzerat.author_id);
    return posliHtml(res, 200, strankaInzeratu(inzerat, autor));
  }

  // Návrat po přihlášení tam, odkud uživatel přišel.
  if (req.method === 'GET' && pathname === '/prihlaseni/zpet') {
    const pozadovana = url.searchParams.get('pokracovat') ?? '/';
    // Projde jen cesta na tomhle webu: začíná jedním lomítkem a hned za ním
    // není další lomítko ani zpětné lomítko (to by byla cizí doména).
    const pokracovat = /^\/[^/\\]/.test(pozadovana) ? pozadovana : '/';
    res.writeHead(303, { Location: pokracovat });
    return res.end();
  }

  // Registrace.
  if (req.method === 'POST' && pathname === '/api/registrace') {
    const { email, heslo } = await prectiJson(req);
    if (typeof email !== 'string' || !email.includes('@')) {
      return posliJson(res, 400, { error: 'Zadej platný e-mail.' });
    }
    if (typeof heslo !== 'string' || heslo.length < 8 || heslo.length > 128) {
      return posliJson(res, 400, { error: 'Heslo musí mít 8 až 128 znaků.' });
    }
    const adresa = email.trim().toLowerCase();
    if (db.prepare('SELECT id FROM users WHERE email = ?').get(adresa)) {
      return posliJson(res, 409, { error: 'Tenhle e-mail už je registrovaný.' });
    }
    const zapis = db
      .prepare('INSERT INTO users (email, password_hash, role) VALUES (?, ?, ?)')
      .run(adresa, zahashujHeslo(heslo), 'user');
    return posliJson(res, 201, { id: Number(zapis.lastInsertRowid), email: adresa });
  }

  // Přihlášení.
  if (req.method === 'POST' && pathname === '/api/prihlaseni') {
    const { email, heslo } = await prectiJson(req);
    const adresa = String(email ?? '').trim().toLowerCase();
    const uzivatel = db
      .prepare('SELECT id, email, role, password_hash FROM users WHERE email = ?')
      .get(adresa);
    if (!uzivatel || !hesloSedi(String(heslo ?? ''), uzivatel.password_hash)) {
      return posliJson(res, 401, { error: 'E-mail nebo heslo nesedí.' });
    }
    const sid = vyrobSessionId();
    db.prepare('INSERT INTO sessions (id, user_id, expires_at) VALUES (?, ?, ?)')
      .run(sid, uzivatel.id, Date.now() + DEN);
    return posliJson(res, 200, { email: uzivatel.email, role: uzivatel.role }, {
      'Set-Cookie': `sid=${sid}; Path=/; HttpOnly; SameSite=Lax; Max-Age=${DEN / 1000}`,
    });
  }

  // Odhlášení.
  if (req.method === 'POST' && pathname === '/api/odhlaseni') {
    const sid = prectiCookie(req, 'sid');
    if (sid) db.prepare('DELETE FROM sessions WHERE id = ?').run(sid);
    return posliJson(res, 200, { odhlaseno: true }, {
      'Set-Cookie': 'sid=; Path=/; HttpOnly; SameSite=Lax; Max-Age=0',
    });
  }

  // Jedna zpráva mezi kupujícím a prodejcem.
  const zprava = pathname.match(/^\/api\/zpravy\/(\d+)$/);
  if (req.method === 'GET' && zprava) {
    const uzivatel = prihlasenyUzivatel(req);
    if (!uzivatel) return posliJson(res, 401, { error: 'Nejdřív se přihlas.' });
    const radek = db
      .prepare(`SELECT id, listing_id AS inzerat, sender_id AS odesilatel,
                       recipient_id AS prijemce, body AS text
                FROM messages WHERE id = ?`)
      .get(Number(zprava[1]));
    // Cizí zprávu ani nepřiznáme — jinak by šlo podle kódu odpovědi zjistit,
    // která id existují.
    const moje = radek && (radek.odesilatel === uzivatel.id || radek.prijemce === uzivatel.id);
    if (!moje) return posliJson(res, 404, { error: 'Taková zpráva tu není.' });
    return posliJson(res, 200, radek);
  }

  // Nový inzerát.
  if (req.method === 'POST' && pathname === '/api/inzeraty') {
    const uzivatel = prihlasenyUzivatel(req);
    if (!uzivatel) return posliJson(res, 401, { error: 'Nejdřív se přihlas.' });
    const { nazev, popis, cena, web } = await prectiJson(req);
    if (typeof nazev !== 'string' || nazev.trim() === '') {
      return posliJson(res, 400, { error: 'Inzerát musí mít název.' });
    }
    const zapis = db
      .prepare('INSERT INTO listings (author_id, title, description, price, web) VALUES (?, ?, ?, ?, ?)')
      .run(uzivatel.id, nazev, String(popis ?? ''), Number(cena ?? 0), String(web ?? ''));
    return posliJson(res, 201, { id: Number(zapis.lastInsertRowid), nazev });
  }

  // Smazání inzerátu.
  const smazat = pathname.match(/^\/api\/inzeraty\/(\d+)$/);
  if (req.method === 'DELETE' && smazat) {
    const uzivatel = prihlasenyUzivatel(req);
    if (!uzivatel) return posliJson(res, 401, { error: 'Nejdřív se přihlas.' });
    const inzerat = db.prepare('SELECT id, author_id FROM listings WHERE id = ?').get(Number(smazat[1]));
    if (!inzerat) return posliJson(res, 404, { error: 'Takový inzerát tu není.' });
    if (uzivatel.role !== 'admin' && inzerat.author_id !== uzivatel.id) {
      return posliJson(res, 403, { error: 'Cizí inzerát smazat nemůžeš.' });
    }
    db.prepare('DELETE FROM listings WHERE id = ?').run(inzerat.id);
    return posliJson(res, 200, { smazano: inzerat.id });
  }

  posliJson(res, 404, { error: 'Tahle adresa tu není.' });
}

createServer(async (req, res) => {
  const url = new URL(req.url, 'http://localhost');
  try {
    await obsluz(req, res, url);
  } catch (chyba) {
    // Ať je hned vidět, co se pokazilo.
    posliJson(res, 500, { error: chyba.message, stack: chyba.stack });
  }
}).listen(port, () => {
  console.log(`Bazar Šuplík běží na http://localhost:${port}`);
});
```

  const detail = pathname.match(/^\/inzerat\/(\d+)$/);
  if (req.method === 'GET' && detail) {
    const inzerat = db.prepare('SELECT * FROM listings WHERE id = ?').get(Number(detail[1]));
    if (!inzerat) return posliHtml(res, 404, '<p>Takový inzerát tu není.</p>');
    const autor = db.prepare('SELECT email FROM users WHERE id = ?').get(inzerat.author_id);
    return posliHtml(res, 200, strankaInzeratu(inzerat, autor));
  }

  // Návrat po přihlášení tam, odkud uživatel přišel.
  if (req.method === 'GET' && pathname === '/prihlaseni/zpet') {
    const pozadovana = url.searchParams.get('pokracovat') ?? '/';
    // Projde jen cesta na tomhle webu: začíná jedním lomítkem a hned za ním
    // není další lomítko ani zpětné lomítko (to by byla cizí doména).
    const pokracovat = /^\/[^/\\]/.test(pozadovana) ? pozadovana : '/';
    res.writeHead(303, { Location: pokracovat });
    return res.end();
  }

  // Registrace.
  if (req.method === 'POST' && pathname === '/api/registrace') {
    const { email, heslo } = await prectiJson(req);
    if (typeof email !== 'string' || !email.includes('@')) {
      return posliJson(res, 400, { error: 'Zadej platný e-mail.' });
    }
    if (typeof heslo !== 'string' || heslo.length < 8 || heslo.length > 128) {
      return posliJson(res, 400, { error: 'Heslo musí mít 8 až 128 znaků.' });
    }
    const adresa = email.trim().toLowerCase();
    if (db.prepare('SELECT id FROM users WHERE email = ?').get(adresa)) {
      return posliJson(res, 409, { error: 'Tenhle e-mail už je registrovaný.' });
    }
    const zapis = db
      .prepare('INSERT INTO users (email, password_hash, role) VALUES (?, ?, ?)')
      .run(adresa, zahashujHeslo(heslo), 'user');
    return posliJson(res, 201, { id: Number(zapis.lastInsertRowid), email: adresa });
  }

  // Přihlášení.
  if (req.method === 'POST' && pathname === '/api/prihlaseni') {
    const { email, heslo } = await prectiJson(req);
    const adresa = String(email ?? '').trim().toLowerCase();
    const uzivatel = db
      .prepare('SELECT id, email, role, password_hash FROM users WHERE email = ?')
      .get(adresa);
    if (!uzivatel || !hesloSedi(String(heslo ?? ''), uzivatel.password_hash)) {
      return posliJson(res, 401, { error: 'E-mail nebo heslo nesedí.' });
    }
    const sid = vyrobSessionId();
    db.prepare('INSERT INTO sessions (id, user_id, expires_at) VALUES (?, ?, ?)')
      .run(sid, uzivatel.id, Date.now() + DEN);
    return posliJson(res, 200, { email: uzivatel.email, role: uzivatel.role }, {
      'Set-Cookie': `sid=${sid}; Path=/; HttpOnly; SameSite=Lax; Max-Age=${DEN / 1000}`,
    });
  }

  // Odhlášení.
  if (req.method === 'POST' && pathname === '/api/odhlaseni') {
    const sid = prectiCookie(req, 'sid');
    if (sid) db.prepare('DELETE FROM sessions WHERE id = ?').run(sid);
    return posliJson(res, 200, { odhlaseno: true }, {
      'Set-Cookie': 'sid=; Path=/; HttpOnly; SameSite=Lax; Max-Age=0',
    });
  }

  // Jedna zpráva mezi kupujícím a prodejcem.
  const zprava = pathname.match(/^\/api\/zpravy\/(\d+)$/);
  if (req.method === 'GET' && zprava) {
    const uzivatel = prihlasenyUzivatel(req);
    if (!uzivatel) return posliJson(res, 401, { error: 'Nejdřív se přihlas.' });
    const radek = db
      .prepare(`SELECT id, listing_id AS inzerat, sender_id AS odesilatel,
                       recipient_id AS prijemce, body AS text
                FROM messages WHERE id = ?`)
      .get(Number(zprava[1]));
    // Cizí zprávu ani nepřiznáme — jinak by šlo podle kódu odpovědi zjistit,
    // která id existují.
    const moje = radek && (radek.odesilatel === uzivatel.id || radek.prijemce === uzivatel.id);
    if (!moje) return posliJson(res, 404, { error: 'Taková zpráva tu není.' });
    return posliJson(res, 200, radek);
  }

  // Nový inzerát.
  if (req.method === 'POST' && pathname === '/api/inzeraty') {
    const uzivatel = prihlasenyUzivatel(req);
    if (!uzivatel) return posliJson(res, 401, { error: 'Nejdřív se přihlas.' });
    const { nazev, popis, cena, web } = await prectiJson(req);
    if (typeof nazev !== 'string' || nazev.trim() === '') {
      return posliJson(res, 400, { error: 'Inzerát musí mít název.' });
    }
    const zapis = db
      .prepare('INSERT INTO listings (author_id, title, description, price, web) VALUES (?, ?, ?, ?, ?)')
      .run(uzivatel.id, nazev, String(popis ?? ''), Number(cena ?? 0), String(web ?? ''));
    return posliJson(res, 201, { id: Number(zapis.lastInsertRowid), nazev });
  }

  // Smazání inzerátu.
  const smazat = pathname.match(/^\/api\/inzeraty\/(\d+)$/);
  if (req.method === 'DELETE' && smazat) {
    const uzivatel = prihlasenyUzivatel(req);
    if (!uzivatel) return posliJson(res, 401, { error: 'Nejdřív se přihlas.' });
    const inzerat = db.prepare('SELECT id, author_id FROM listings WHERE id = ?').get(Number(smazat[1]));
    if (!inzerat) return posliJson(res, 404, { error: 'Takový inzerát tu není.' });
    if (uzivatel.role !== 'admin' && inzerat.author_id !== uzivatel.id) {
      return posliJson(res, 403, { error: 'Cizí inzerát smazat nemůžeš.' });
    }
    db.prepare('DELETE FROM listings WHERE id = ?').run(inzerat.id);
    return posliJson(res, 200, { smazano: inzerat.id });
  }

  posliJson(res, 404, { error: 'Tahle adresa tu není.' });
}

createServer(async (req, res) => {
  const url = new URL(req.url, 'http://localhost');
  try {
    await obsluz(req, res, url);
  } catch (chyba) {
    // Podrobnosti patří do logu serveru. Klient se dozví jen to, že se to nepovedlo.
    console.error('[500]', req.method, url.pathname, chyba);
    posliJson(res, 500, { error: 'Na serveru se něco pokazilo.' });
  }
}).listen(port, () => {
  console.log(`Bazar Šuplík běží na http://localhost:${port}`);
});
