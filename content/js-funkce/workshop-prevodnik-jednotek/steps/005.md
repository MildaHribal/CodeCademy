# --description--

Dvě desetinná místa se hodí na kilogramy, ale teplotu trouby chceš na celé stupně a vzdálenost na jedno místo. Počet míst proto bude parametr. Aby dosavadní volání `formatNumber(value)` fungovala dál, dostane parametr **výchozí hodnotu**.

**Úkol:** Uprav `formatNumber` na `formatNumber(value, decimals = 2)` a zaokrouhluj na `decimals` míst.

Výchozí hodnota se píše za rovnítko přímo do závorek. Na jiných datech:

```js
function greet(name, greeting = 'Dobrý den') {
  return `${greeting}, ${name}`;
}
```

Výchozí hodnota se použije jen tehdy, když argument chybí (je `undefined`). Nula je platný argument, takže `formatNumber(x, 0)` musí zaokrouhlit na celé číslo.

# --hints--

`formatNumber(42.1648128)` bez druhého argumentu vrátí pořád `'42,16'`.

```js
assert.equal(formatNumber(42.1648128), '42,16', "formatNumber(42.1648128) má vrátit '42,16' — výchozí jsou 2 desetinná místa");
```

`formatNumber(42.1648128, 1)` vrátí `'42,2'`.

```js
assert.equal(formatNumber(42.1648128, 1), '42,2', "formatNumber(42.1648128, 1) má vrátit '42,2'");
```

`formatNumber(176.666, 0)` vrátí `'177'` — nula desetinných míst se nepřepíše výchozí hodnotou.

```js
assert.equal(formatNumber(176.666, 0), '177', "formatNumber(176.666, 0) má vrátit '177'; když dostaneš '176,67', nula se nahradila výchozí hodnotou");
```

# --help--

## --tip--

Výchozí parametry jsou v části [Výchozí a zbytkové parametry](see:js-funkce/funkce#vychozi-a-zbytkove-parametry).

## --tip--

Výchozí hodnotu napiš do hlavičky funkce a v těle místo pevné dvojky použij parametr. Nepoužívej `||` — ten přepíše i nulu.

// Převodník jednotek pro nákupy, recepty a závody ze zahraničí.
// Americké weby uvádějí míle, libry, palce a stupně Fahrenheita.
const KM_PER_MILE = 1.609344;
const KG_PER_POUND = 0.45359237;
const CM_PER_INCH = 2.54;

// Převody liber už napsal kolega.
function poundsToKg(pounds) {
  return pounds * KG_PER_POUND;
}

function kgToPounds(kg) {
  return kg / KG_PER_POUND;
}

console.log(poundsToKg(5));

function roundTo(value, decimals) {
  const factor = 10 ** decimals;
  return Math.round(value * factor) / factor;
}

console.log(roundTo(poundsToKg(5), 2));

function milesToKm(miles) {
  return miles * KM_PER_MILE;
}

console.log(milesToKm(26.2));

const kmToMiles = (km) => km / KM_PER_MILE;

console.log(kmToMiles(42.195));

--edit--
function formatNumber(value) {
  return String(roundTo(value, 2)).replace('.', ',');
}

console.log(formatNumber(milesToKm(26.2)));
--edit--

```js
// Převodník jednotek pro nákupy, recepty a závody ze zahraničí.
// Americké weby uvádějí míle, libry, palce a stupně Fahrenheita.
const KM_PER_MILE = 1.609344;
const KG_PER_POUND = 0.45359237;
const CM_PER_INCH = 2.54;

// Převody liber už napsal kolega.
function poundsToKg(pounds) {
  return pounds * KG_PER_POUND;
}

function kgToPounds(kg) {
  return kg / KG_PER_POUND;
}

console.log(poundsToKg(5));

function roundTo(value, decimals) {
  const factor = 10 ** decimals;
  return Math.round(value * factor) / factor;
}

console.log(roundTo(poundsToKg(5), 2));

function milesToKm(miles) {
  return miles * KM_PER_MILE;
}

console.log(milesToKm(26.2));

const kmToMiles = (km) => km / KM_PER_MILE;

console.log(kmToMiles(42.195));

function formatNumber(value, decimals = 2) {
  return String(roundTo(value, decimals)).replace('.', ',');
}

console.log(formatNumber(milesToKm(26.2)));
console.log(formatNumber(milesToKm(26.2), 0));
```
