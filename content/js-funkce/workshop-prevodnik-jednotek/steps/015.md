# --description--

Tohle je funkce, kterou bude volat stránka: dostane text z políčka a hodnotu z výběru a vrátí to, co uživatel uvidí.

**Úkol:** Napiš funkci `describeConversion(text, conversion)`, která vrátí:

- `'Zadej číslo'`, když text není číslo,
- `'Neznámý převod'`, když převodník `conversion` nezná,
- jinak popis ve tvaru `'26,2 → 42,2 km'`: vstupní číslo na 1 desetinné místo, šipka `→` s mezerami a výsledek s cílovou jednotkou.

Skoro nic nového nepíšeš — skládáš `parseAmount`, `convert`, `targetUnit`, `formatNumber` a `formatResult`. Kontroly napiš jako guard clauses, nebo jako zanořené `if` / `else`; porovnej, co se čte líp.

> [!TIP]
> Šipku `→` nemusíš hledat na klávesnici, zkopíruj ji ze zadání.

# --hints--

`describeConversion('26,2', 'mi-km')` vrátí `'26,2 → 42,2 km'`.

```js
assert.equal(describeConversion('26,2', 'mi-km'), '26,2 → 42,2 km', "describeConversion('26,2', 'mi-km') má vrátit '26,2 → 42,2 km'");
```

`describeConversion('350', 'f-c')` vrátí `'350 → 176,7 °C'` a `describeConversion('5', 'lb-kg')` vrátí `'5 → 2,3 kg'`.

```js
assert.equal(describeConversion('350', 'f-c'), '350 → 176,7 °C', "describeConversion('350', 'f-c') má vrátit '350 → 176,7 °C'");
assert.equal(describeConversion('5', 'lb-kg'), '5 → 2,3 kg', "describeConversion('5', 'lb-kg') má vrátit '5 → 2,3 kg'");
```

Vstupní číslo je zaokrouhlené na jedno desetinné místo: `describeConversion('0,75', 'lb-kg')` vrátí `'0,8 → 0,3 kg'`.

```js
assert.equal(describeConversion('0,75', 'lb-kg'), '0,8 → 0,3 kg', "describeConversion('0,75', 'lb-kg') má vrátit '0,8 → 0,3 kg' — vstup na 1 desetinné místo, ne na 2");
```

`describeConversion` vrátí `'Zadej číslo'` pro prázdný text i nesmysl.

```js
assert.equal(describeConversion('', 'mi-km'), 'Zadej číslo', "describeConversion('', 'mi-km') má vrátit 'Zadej číslo'");
assert.equal(describeConversion('pět', 'mi-km'), 'Zadej číslo', "describeConversion('pět', 'mi-km') má vrátit 'Zadej číslo'");
```

`describeConversion('5', 'yd-m')` vrátí `'Neznámý převod'`.

```js
assert.equal(describeConversion('5', 'yd-m'), 'Neznámý převod', "describeConversion('5', 'yd-m') má vrátit 'Neznámý převod'");
```

# --help--

## --tip--

Funkce, která jen skládá výsledky jiných funkcí, stojí na návratových hodnotách — viz [return: jak funkce vrací výsledek](see:js-funkce/funkce#return-jak-funkce-vraci-vysledek).

## --tip--

Pořadí kroků: text převeď na číslo a zkontroluj ho, pak převeď a zkontroluj `null`, nakonec slož popis. Na kontrolu `NaN` nestačí `=== NaN`, použij `Number.isNaN`.

## --tip--

Stejná stavba na jiných datech, rozepsaná na podcíle:

```js
// 1. vstup → hodnota (a guard clause pro nesmysl)
// 2. hodnota → výsledek (a guard clause pro neznámou operaci)
// 3. vrať text poskládaný z hodnoty a výsledku
```

# --explain--

Vysvětli vlastními slovy, co získáš tím, že `describeConversion` jen skládá malé funkce, místo aby všechno počítala sama.

## --model--

Každá malá funkce dělá jednu věc a má jasný vstup a výstup, takže ji jde otestovat a opravit zvlášť. `describeConversion` pak čte jako popis postupu: přečti číslo, převeď, zformátuj. Když se změní formát čísla nebo přibude převod, upravím jedno místo a projeví se všude, kde se funkce používá. Chyba se taky snáz hledá, protože vím, která funkce za kterou část výsledku odpovídá.

## --checklist--

- Každá malá funkce dělá jednu věc s jasným vstupem a výstupem.
- Malé funkce jde otestovat a opravit zvlášť.
- Skládající funkce se čte jako popis postupu.
- Změna na jednom místě se projeví všude, kde se funkce používá.

// Převodník jednotek pro nákupy, recepty a závody ze zahraničí.
// Americké weby uvádějí míle, libry, palce a stupně Fahrenheita.
const KM_PER_MILE = 1.609344;
const KG_PER_POUND = 0.45359237;
const CM_PER_INCH = 2.54;

// Převody liber už napsal kolega.
function poundsToKg(pounds) {
  return pounds * KG_PER_POUND;
}

function kgToPounds(kg) {
  return kg / KG_PER_POUND;
}

console.log(poundsToKg(5));

function roundTo(value, decimals) {
  const factor = 10 ** decimals;
  return Math.round(value * factor) / factor;
}

console.log(roundTo(poundsToKg(5), 2));

function milesToKm(miles) {
  return miles * KM_PER_MILE;
}

console.log(milesToKm(26.2));

const kmToMiles = (km) => km / KM_PER_MILE;

console.log(kmToMiles(42.195));

function formatNumber(value, decimals = 2) {
  return String(roundTo(value, decimals)).replace('.', ',');
}

console.log(formatNumber(milesToKm(26.2)));
console.log(formatNumber(milesToKm(26.2), 0));

function formatResult(value, unit, decimals = 1) {
  return `${formatNumber(value, decimals)} ${unit}`;
}

console.log(formatResult(milesToKm(26.2), 'km'));

// Teploty poslal kolega: recepty z USA uvádějí teplotu trouby ve °F.
const ABSOLUTE_ZERO_F = -459.67;
const ABSOLUTE_ZERO_C = -273.15;

function degreesAboveFreezing(fahrenheit) {
  return fahrenheit - 32;
}

function fahrenheitToCelsius(fahrenheit) {
  return degreesAboveFreezing(fahrenheit) * 5 / 9;
}

function celsiusToFahrenheit(celsius) {
  return celsius * 9 / 5 + 32;
}

console.log(formatResult(fahrenheitToCelsius(350), '°C'));

function temperatureLabel(fahrenheit) {
  if (!Number.isFinite(fahrenheit)) {
    return 'Zadej číslo';
  }
  if (fahrenheit < ABSOLUTE_ZERO_F) {
    return 'Pod absolutní nulou';
  }
  return formatResult(fahrenheitToCelsius(fahrenheit), '°C');
}

function distanceLabel(miles) {
  if (!Number.isFinite(miles)) {
    return 'Zadej číslo';
  }
  if (miles < 0) {
    return 'Vzdálenost nemůže být záporná';
  }
  return formatResult(milesToKm(miles), 'km');
}

console.log(distanceLabel(26.2));

function convert(value, conversion) {
  if (conversion === 'mi-km') {
    return milesToKm(value);
  }
  if (conversion === 'km-mi') {
    return kmToMiles(value);
  }
  if (conversion === 'lb-kg') {
    return poundsToKg(value);
  }
  if (conversion === 'kg-lb') {
    return kgToPounds(value);
  }
  if (conversion === 'f-c') {
    return fahrenheitToCelsius(value);
  }
  if (conversion === 'c-f') {
    return celsiusToFahrenheit(value);
  }
  return null;
}

console.log(convert(5, 'lb-kg'));

function targetUnit(conversion) {
  const unit = conversion.slice(conversion.indexOf('-') + 1);
  if (unit === 'c') {
    return '°C';
  }
  if (unit === 'f') {
    return '°F';
  }
  return unit;
}

console.log(targetUnit('f-c'));

function parseAmount(text) {
  const trimmed = text.trim();
  if (trimmed === '') {
    return NaN;
  }
  return Number(trimmed.replace(',', '.'));
}

console.log(parseAmount(' 26,2 '));

// Kolegův modul pro běžce: tempo z amerických závodů v minutách na kilometr.
function paceLabel(miles, minutes, decimals) {
  const km = milesToKm(miles);
  const pace = minutes / km;
  const places = decimals ?? 2;
  return `${formatNumber(pace, places)} min/km`;
}

console.log(paceLabel(26.2, 210, 0));

function totalKm(...milesLegs) {
  let total = 0;
  for (const miles of milesLegs) {
    total += milesToKm(miles);
  }
  return total;
}

console.log(formatResult(totalKm(297, 396, 370), 'km'));

function describeConversion(text, conversion) {
  const amount = parseAmount(text);
  if (Number.isNaN(amount)) {
    return 'Zadej číslo';

  const result = convert(amount, conversion);
  if (result === null) {
    return 'Neznámý převod';
  }
  return `${formatNumber(amount, 1)} → ${formatResult(result, targetUnit(conversion))}`;
}
