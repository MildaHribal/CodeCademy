# --description--

`describeConversion` jen vrací text a nic nevypisuje, takže ji může použít stránka, test i konzole. To je čistá funkce. Výpis do konzole je vedlejší efekt a patří do jiné funkce.

**Úkol:** Napiš funkci `printConversion(text, conversion)`, která popis převodu **vypíše** do konzole a nic nevrací. Popis nepočítej znovu, použij `describeConversion`. Pod funkci ji zavolej, třeba pro 5 liber.

Kandidáti na to, kam dát `console.log`: dovnitř `describeConversion`, nebo do nové funkce, která `describeConversion` zavolá. Jen jedna cesta nechá `describeConversion` čistou.

# --hints--

`printConversion('5', 'lb-kg')` vypíše do konzole `5 → 2,3 kg`.

```js
const printed = [];
const original = console.log;
console.log = (...args) => printed.push(args.join(' '));
try {
  printConversion('5', 'lb-kg');
} finally {
  console.log = original;
}
assert.deepEqual(printed, ['5 → 2,3 kg'], "printConversion('5', 'lb-kg') má vypsat jediný řádek '5 → 2,3 kg'");
```

`printConversion` nic nevrací.

```js
const original = console.log;
console.log = () => {};
let result;
try {
  result = printConversion('26,2', 'mi-km');
} finally {
  console.log = original;
}
assert.equal(result, undefined, "printConversion('26,2', 'mi-km') má vracet undefined — jen vypisuje");
```

`describeConversion` zůstane čistá: vrací text a nic nevypisuje.

```js
const printed = [];
const original = console.log;
console.log = (...args) => printed.push(args.join(' '));
let result;
try {
  result = describeConversion('10', 'km-mi');
} finally {
  console.log = original;
}
assert.equal(printed.length, 0, "describeConversion('10', 'km-mi') nemá nic vypisovat");
assert.equal(result, '10 → 6,2 mi', "describeConversion('10', 'km-mi') má vrátit '10 → 6,2 mi'");
```

# --help--

## --tip--

Rozdíl mezi čistou funkcí a vedlejším efektem je v části [Čistá funkce](see:js-funkce/funkce#cista-funkce).

## --tip--

Nová funkce má v těle jediný řádek: výpis výsledku `describeConversion` se stejnými argumenty. `return` nepotřebuje.

## --tip--

Stejné rozdělení znáš z běžného webu: funkce, která spočítá text potvrzení objednávky, ho jen vrátí, a o tom, jestli se ukáže v konzoli, v e-mailu nebo na stránce, rozhoduje až jiná funkce, která ji zavolá. Tvoje nová funkce je ta druhá.

# --explain--

Vysvětli vlastními slovy, proč `console.log` nepatří dovnitř `describeConversion`.

## --model--

`describeConversion` je čistá funkce: pro stejné argumenty vrátí vždy stejný text a nic kolem sebe nemění. Díky tomu ji použije stránka, test i konzole a otestuje se jedním porovnáním. Kdyby uvnitř vypisovala, každé použití by navíc psalo do konzole, i tam, kde to nikdo nechce, a test by musel výpis zachytávat. Vedlejší efekty jako výpis proto dávám do samostatné tenké funkce na okraji programu.

## --checklist--

- Čistá funkce vrací výsledek a nic kolem sebe nemění.
- Výpis do konzole je vedlejší efekt.
- Čistou funkci může použít stránka, test i konzole.
- Vedlejší efekt patří do samostatné funkce, která čistou funkci zavolá.

// Převodník jednotek pro nákupy, recepty a závody ze zahraničí.
// Americké weby uvádějí míle, libry, palce a stupně Fahrenheita.
const KM_PER_MILE = 1.609344;
const KG_PER_POUND = 0.45359237;
const CM_PER_INCH = 2.54;

// Převody liber už napsal kolega.
function poundsToKg(pounds) {
  return pounds * KG_PER_POUND;
}

function kgToPounds(kg) {
  return kg / KG_PER_POUND;
}

console.log(poundsToKg(5));

function roundTo(value, decimals) {
  const factor = 10 ** decimals;
  return Math.round(value * factor) / factor;
}

console.log(roundTo(poundsToKg(5), 2));

function milesToKm(miles) {
  return miles * KM_PER_MILE;
}

console.log(milesToKm(26.2));

const kmToMiles = (km) => km / KM_PER_MILE;

console.log(kmToMiles(42.195));

function formatNumber(value, decimals = 2) {
  return String(roundTo(value, decimals)).replace('.', ',');
}

console.log(formatNumber(milesToKm(26.2)));
console.log(formatNumber(milesToKm(26.2), 0));

function formatResult(value, unit, decimals = 1) {
  return `${formatNumber(value, decimals)} ${unit}`;
}

console.log(formatResult(milesToKm(26.2), 'km'));

// Teploty poslal kolega: recepty z USA uvádějí teplotu trouby ve °F.
const ABSOLUTE_ZERO_F = -459.67;
const ABSOLUTE_ZERO_C = -273.15;

function degreesAboveFreezing(fahrenheit) {
  return fahrenheit - 32;
}

function fahrenheitToCelsius(fahrenheit) {
  return degreesAboveFreezing(fahrenheit) * 5 / 9;
}

function celsiusToFahrenheit(celsius) {
  return celsius * 9 / 5 + 32;
}

console.log(formatResult(fahrenheitToCelsius(350), '°C'));

function temperatureLabel(fahrenheit) {
  if (!Number.isFinite(fahrenheit)) {
    return 'Zadej číslo';
  }
  if (fahrenheit < ABSOLUTE_ZERO_F) {
    return 'Pod absolutní nulou';
  }
  return formatResult(fahrenheitToCelsius(fahrenheit), '°C');
}

function distanceLabel(miles) {
  if (!Number.isFinite(miles)) {
    return 'Zadej číslo';
  }
  if (miles < 0) {
    return 'Vzdálenost nemůže být záporná';
  }
  return formatResult(milesToKm(miles), 'km');
}

console.log(distanceLabel(26.2));

function convert(value, conversion) {
  if (conversion === 'mi-km') {
    return milesToKm(value);
  }
  if (conversion === 'km-mi') {
    return kmToMiles(value);
  }
  if (conversion === 'lb-kg') {
    return poundsToKg(value);
  }
  if (conversion === 'kg-lb') {
    return kgToPounds(value);
  }
  if (conversion === 'f-c') {
    return fahrenheitToCelsius(value);
  }
  if (conversion === 'c-f') {
    return celsiusToFahrenheit(value);
  }
  return null;
}

console.log(convert(5, 'lb-kg'));

function targetUnit(conversion) {
  const unit = conversion.slice(conversion.indexOf('-') + 1);
  if (unit === 'c') {
    return '°C';
  }
  if (unit === 'f') {
    return '°F';
  }
  return unit;
}

console.log(targetUnit('f-c'));

function parseAmount(text) {
  const trimmed = text.trim();
  if (trimmed === '') {
    return NaN;
  }
  return Number(trimmed.replace(',', '.'));
}

console.log(parseAmount(' 26,2 '));

// Kolegův modul pro běžce: tempo z amerických závodů v minutách na kilometr.
function paceLabel(miles, minutes, decimals) {
  const km = milesToKm(miles);
  const pace = minutes / km;
  const places = decimals ?? 2;
  return `${formatNumber(pace, places)} min/km`;
}

console.log(paceLabel(26.2, 210, 0));

function totalKm(...milesLegs) {
  let total = 0;
  for (const miles of milesLegs) {
    total += milesToKm(miles);
  }
  return total;
}

console.log(formatResult(totalKm(297, 396, 370), 'km'));

function describeConversion(text, conversion) {
  const amount = parseAmount(text);
  if (Number.isNaN(amount)) {
    return 'Zadej číslo';
  }
  const result = convert(amount, conversion);
  if (result === null) {
    return 'Neznámý převod';
  }
  return `${formatNumber(amount, 1)} → ${formatResult(result, targetUnit(conversion))}`;
}

console.log(describeConversion('26,2', 'mi-km'));
