# --description--

Než převodník teplotu převede, musí odmítnout nesmysly: text místo čísla a teplotu pod absolutní nulou (−459,67 °F), protože nic chladnějšího neexistuje.

**Úkol:** Seřaď řádky funkce `temperatureLabel(fahrenheit)` tak, aby vrátila:

- `'Zadej číslo'`, když `fahrenheit` není běžné číslo,
- `'Pod absolutní nulou'`, když je teplota nižší než `ABSOLUTE_ZERO_F`,
- jinak převedenou teplotu s jednotkou, třeba `'176,7 °C'`.

Kontroly stojí na začátku funkce a každá hned skončí `return` — to jsou [[guard clause|guard clauses]]. Hlavní výpočet je až na konci, bez `else`. `Number.isFinite(x)` vrátí `true` jen pro běžné číslo; pro `NaN`, `Infinity` i text vrátí `false`.

Odsazení nastavíš klávesou Tab, do mezery doplň porovnávací operátor. Dva řádky jsou navíc.

# --hints--

`temperatureLabel(350)` vrátí `'176,7 °C'`.

```js
assert.equal(temperatureLabel(350), '176,7 °C', "temperatureLabel(350) má vrátit '176,7 °C'");
```

`temperatureLabel(-500)` vrátí `'Pod absolutní nulou'` a `temperatureLabel(-40)` vrátí `'-40 °C'`.

```js
assert.equal(temperatureLabel(-500), 'Pod absolutní nulou', "temperatureLabel(-500) má vrátit 'Pod absolutní nulou'");
assert.equal(temperatureLabel(-40), '-40 °C', "temperatureLabel(-40) má vrátit '-40 °C' — −40 °F je nad absolutní nulou");
```

`temperatureLabel(NaN)` vrátí `'Zadej číslo'`.

```js
assert.equal(temperatureLabel(NaN), 'Zadej číslo', "temperatureLabel(NaN) má vrátit 'Zadej číslo'");
assert.equal(temperatureLabel(Infinity), 'Zadej číslo', "temperatureLabel(Infinity) má vrátit 'Zadej číslo'");
```

## --tip--

Vzor je v části [Předčasný `return` (guard clause)](see:js-funkce/funkce#predcasny-return-guard-clause).

## --tip--

Každá kontrola má vlastní `if` s `return` uvnitř. Poslední `return` s převodem stojí mimo všechny `if`.

function temperatureLabel(fahrenheit) {
  if (!Number.isFinite(fahrenheit)) {
    return 'Zadej číslo';
  }
  if (fahrenheit __1__ ABSOLUTE_ZERO_F) {
    return 'Pod absolutní nulou';
  }
  return formatResult(fahrenheitToCelsius(fahrenheit), '°C');
}

// Převodník jednotek pro nákupy, recepty a závody ze zahraničí.
// Americké weby uvádějí míle, libry, palce a stupně Fahrenheita.
const KM_PER_MILE = 1.609344;
const KG_PER_POUND = 0.45359237;
const CM_PER_INCH = 2.54;

// Převody liber už napsal kolega.
function poundsToKg(pounds) {
  return pounds * KG_PER_POUND;
}

function kgToPounds(kg) {
  return kg / KG_PER_POUND;
}

console.log(poundsToKg(5));

function roundTo(value, decimals) {
  const factor = 10 ** decimals;
  return Math.round(value * factor) / factor;
}

console.log(roundTo(poundsToKg(5), 2));

function milesToKm(miles) {
  return miles * KM_PER_MILE;
}

console.log(milesToKm(26.2));

const kmToMiles = (km) => km / KM_PER_MILE;

console.log(kmToMiles(42.195));

function formatNumber(value, decimals = 2) {
  return String(roundTo(value, decimals)).replace('.', ',');
}

console.log(formatNumber(milesToKm(26.2)));
console.log(formatNumber(milesToKm(26.2), 0));

function formatResult(value, unit, decimals = 1) {
  return `${formatNumber(value, decimals)} ${unit}`;
}

console.log(formatResult(milesToKm(26.2), 'km'));

// Teploty poslal kolega: recepty z USA uvádějí teplotu trouby ve °F.
const ABSOLUTE_ZERO_F = -459.67;
const ABSOLUTE_ZERO_C = -273.15;

function degreesAboveFreezing(fahrenheit) {
  return fahrenheit - 32;
}

function fahrenheitToCelsius(fahrenheit) {
  return degreesAboveFreezing(fahrenheit) * 5 / 9;
}

function celsiusToFahrenheit(celsius) {
  return celsius * 9 / 5 + 32;
}

console.log(formatResult(fahrenheitToCelsius(350), '°C'));
