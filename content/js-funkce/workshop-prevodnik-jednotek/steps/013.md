# --description--

## Hlášení

Kolega přidal na konec souboru funkci `paceLabel(miles, minutes, decimals)` pro běžce, kteří se chystají na závod v USA. Vrací tempo v minutách na kilometr s `decimals` desetinnými místy, bez třetího argumentu se dvěma.

Uživatel zadal maraton (26,2 mile) za 210 minut a chtěl tempo zaokrouhlené na celé minuty: `paceLabel(26.2, 210, 0)`. Čekal `'5 min/km'`, ale dostal `'4,98 min/km'` — poslední řádek v konzoli. S jedním desetinným místem (`paceLabel(26.2, 210, 1)`) vše funguje.

## Úkol

Oprav `paceLabel` tak, aby nula desetinných míst fungovala a bez třetího argumentu zůstala dvě. Změň co nejmenší kus kódu.

# --hints--

`paceLabel(26.2, 210, 0)` vrátí `'5 min/km'`.

```js
assert.equal(paceLabel(26.2, 210, 0), '5 min/km', "paceLabel(26.2, 210, 0) má vrátit '5 min/km' — nula desetinných míst se nesmí nahradit výchozí hodnotou");
```

`paceLabel(26.2, 210)` bez třetího argumentu vrátí `'4,98 min/km'`.

```js
assert.equal(paceLabel(26.2, 210), '4,98 min/km', "paceLabel(26.2, 210) má vrátit '4,98 min/km' — bez argumentu platí 2 desetinná místa");
```

`paceLabel(13.1, 95, 1)` vrátí `'4,5 min/km'`.

```js
assert.equal(paceLabel(13.1, 95, 1), '4,5 min/km', "paceLabel(13.1, 95, 1) má vrátit '4,5 min/km'");
```

`formatNumber` a `formatResult` fungují dál.

```js
assert.equal(formatNumber(4.98, 0), '5', "formatNumber(4.98, 0) má vrátit '5'");
assert.equal(formatResult(42.1648128, 'km'), '42,2 km', "formatResult(42.1648128, 'km') má vrátit '42,2 km'");
```

# --help--

## --tip--

Nula se ztrácí cestou k `formatNumber` — viz past [Výchozí hodnota přes `||`](see:js-funkce/funkce#vychozi-hodnota-pres).

## --tip--

Podívej se na řádek s `places`. Co vrátí `0 || 2`? Který zápis výchozí hodnoty reaguje jen na chybějící argument?

## --tip--

Stejná chyba na jiných datech: `const volume = level || 50;` nastaví hlasitost na 50 i tehdy, když uživatel schválně zvolil 0.

# --explain--

Vysvětli vlastními slovy, proč `decimals || 2` pokazilo zaokrouhlení na celé minuty a čím se od toho liší výchozí parametr `decimals = 2`.

## --model--

`||` vrátí pravou stranu, kdykoli je levá strana nepravdivá, a nula je nepravdivá hodnota. `0 || 2` proto dá `2` a funkce zaokrouhlila na dvě místa, přestože uživatel chtěl nulu. Výchozí parametr i operátor `??` reagují jen na chybějící hodnotu (`undefined`, u `??` i `null`), takže nula projde beze změny. Na výchozí hodnoty čísel a textů, kde je nula nebo prázdný text platný vstup, proto `||` nepoužívám.

## --checklist--

- `||` vrátí pravou stranu pro každou nepravdivou hodnotu, tedy i pro `0`.
- Nula je tu platný počet desetinných míst.
- Výchozí parametr se použije jen pro `undefined`.
- `??` nahradí jen `undefined` a `null`, nulu nechá být.

```js
// Převodník jednotek pro nákupy, recepty a závody ze zahraničí.
// Americké weby uvádějí míle, libry, palce a stupně Fahrenheita.
const KM_PER_MILE = 1.609344;
const KG_PER_POUND = 0.45359237;
const CM_PER_INCH = 2.54;

// Převody liber už napsal kolega.
function poundsToKg(pounds) {
  return pounds * KG_PER_POUND;
}

function kgToPounds(kg) {
  return kg / KG_PER_POUND;
}

console.log(poundsToKg(5));

function roundTo(value, decimals) {
  const factor = 10 ** decimals;
  return Math.round(value * factor) / factor;
}

console.log(roundTo(poundsToKg(5), 2));

function milesToKm(miles) {
  return miles * KM_PER_MILE;
}

console.log(milesToKm(26.2));

const kmToMiles = (km) => km / KM_PER_MILE;

console.log(kmToMiles(42.195));

function formatNumber(value, decimals = 2) {
  return String(roundTo(value, decimals)).replace('.', ',');
}

console.log(formatNumber(milesToKm(26.2)));
console.log(formatNumber(milesToKm(26.2), 0));

function formatResult(value, unit, decimals = 1) {
  return `${formatNumber(value, decimals)} ${unit}`;
}

console.log(formatResult(milesToKm(26.2), 'km'));

// Teploty poslal kolega: recepty z USA uvádějí teplotu trouby ve °F.
const ABSOLUTE_ZERO_F = -459.67;
const ABSOLUTE_ZERO_C = -273.15;

function degreesAboveFreezing(fahrenheit) {
  return fahrenheit - 32;
}

function fahrenheitToCelsius(fahrenheit) {
  return degreesAboveFreezing(fahrenheit) * 5 / 9;
}

function celsiusToFahrenheit(celsius) {
  return celsius * 9 / 5 + 32;
}

console.log(formatResult(fahrenheitToCelsius(350), '°C'));

function temperatureLabel(fahrenheit) {
  if (!Number.isFinite(fahrenheit)) {
    return 'Zadej číslo';
  }
  if (fahrenheit < ABSOLUTE_ZERO_F) {
    return 'Pod absolutní nulou';
  }
  return formatResult(fahrenheitToCelsius(fahrenheit), '°C');
}

function distanceLabel(miles) {
  if (!Number.isFinite(miles)) {
    return 'Zadej číslo';
  }
  if (miles < 0) {
    return 'Vzdálenost nemůže být záporná';
  }
  return formatResult(milesToKm(miles), 'km');
}

console.log(distanceLabel(26.2));

function convert(value, conversion) {
  if (conversion === 'mi-km') {
    return milesToKm(value);
  }
  if (conversion === 'km-mi') {
    return kmToMiles(value);
  }
  if (conversion === 'lb-kg') {
    return poundsToKg(value);
  }
  if (conversion === 'kg-lb') {
    return kgToPounds(value);
  }
  if (conversion === 'f-c') {
    return fahrenheitToCelsius(value);
  }
  if (conversion === 'c-f') {
    return celsiusToFahrenheit(value);
  }
  return null;
}

console.log(convert(5, 'lb-kg'));

function targetUnit(conversion) {
  const unit = conversion.slice(conversion.indexOf('-') + 1);
  if (unit === 'c') {
    return '°C';
  }
  if (unit === 'f') {
    return '°F';
  }
  return unit;
}

console.log(targetUnit('f-c'));

function parseAmount(text) {
  const trimmed = text.trim();
  if (trimmed === '') {
    return NaN;
  }
  return Number(trimmed.replace(',', '.'));
}

console.log(parseAmount(' 26,2 '));

--edit--
// Kolegův modul pro běžce: tempo z amerických závodů v minutách na kilometr.
function paceLabel(miles, minutes, decimals) {
  const km = milesToKm(miles);
  const pace = minutes / km;
  const places = decimals || 2;
  return `${formatNumber(pace, places)} min/km`;
}

console.log(paceLabel(26.2, 210, 0));
--edit--
```
