# --description--

Tvoje funkce teď pohánějí skutečnou stránku. Přibyly soubory `index.html`, `styles.css` a `page.js`. V `page.js` je kód, který při každé změně políčka zavolá `describeConversion` a `conversionTable` a výsledek zapíše do stránky — napiš do políčka jiné číslo a sleduj náhled. Jak `page.js` funguje uvnitř, se naučíš v sekci *DOM, události a prohlížeč*. Všimni si jen, že funkci `updatePage` předává bez závorek: stránka si ji zavolá sama, až uživatel něco změní. Proč to tak jde, vysvětlí lekce *Funkce jako hodnoty* hned za lekcí o rozsahu platnosti.

Výběr převodu nabízí i palce a centimetry, ale zatím ukazuje „Neznámý převod". Tlačítko **Prohodit směr** rozchodíš v dalším kroku.

**Úkol:** Doplň převodník o převody `'in-cm'` (palce na centimetry) a `'cm-in'` (centimetry na palce) tak, aby fungovaly ve stránce. Konstantu `CM_PER_INCH` máš v souboru od začátku. Ostatní převody musí fungovat dál.

# --hints--

`convert` převede palce na centimetry a zpět.

```js
assert.ok(Math.abs(convert(10, 'in-cm') - 25.4) < 1e-9, "convert(10, 'in-cm') má vrátit 25.4");
assert.ok(Math.abs(convert(25.4, 'cm-in') - 10) < 1e-9, "convert(25.4, 'cm-in') má vrátit 10");
```

`targetUnit` vrátí `'cm'` a `'in'`.

```js
assert.equal(targetUnit('in-cm'), 'cm', "targetUnit('in-cm') má vrátit 'cm'");
assert.equal(targetUnit('cm-in'), 'in', "targetUnit('cm-in') má vrátit 'in'");
```

Když ve stránce zadáš `10` a vybereš palce → centimetry, výsledek ukáže `10 → 25,4 cm`.

```js
const select = document.querySelector('#conversion');
select.value = 'in-cm';
select.dispatchEvent(new Event('change'));
await helpers.type(document.querySelector('#amount'), '10');
assert.equal(document.querySelector('#result').textContent, '10 → 25,4 cm', 'po zadání 10 a výběru palce → centimetry má #result ukazovat „10 → 25,4 cm"');
```

Původní převody fungují dál.

```js
assert.ok(Math.abs(convert(26.2, 'mi-km') - 42.1648128) < 1e-9, "convert(26.2, 'mi-km') má dál vracet 42.1648128");
assert.equal(convert(212, 'f-c'), 100, "convert(212, 'f-c') má dál vracet 100");
assert.equal(convert(1, 'yd-m'), null, "convert(1, 'yd-m') má dál vracet null");
```

# --help--

## --tip--

Stránka volá tvoje funkce jako callbacky — viz [Callback: funkce, kterou zavolá někdo jiný](see:js-funkce/funkce-jako-hodnoty#callback-funkce-kterou-zavola-nekdo-jiny). Stačí upravit funkce, stránka se přizpůsobí sama.

## --tip--

Nový převod potřebuje místo v `convert` (výpočet s `CM_PER_INCH`) a správnou jednotku z `targetUnit`. Když máš `targetUnit` jako řadu podmínek, přibudou dvě i tam.

## --tip--

Na jiných datech: převod `'oz-g'` (unce na gramy) by v `convert` násobil 28.3495 a `targetUnit` by pro něj vracela `'g'`.

// Americké weby uvádějí míle, libry, palce a stupně Fahrenheita.
const KM_PER_MILE = 1.609344;
const KG_PER_POUND = 0.45359237;
const CM_PER_INCH = 2.54;

// Převody liber už napsal kolega.
function poundsToKg(pounds) {
  return pounds * KG_PER_POUND;
}

function kgToPounds(kg) {
  return kg / KG_PER_POUND;
}

console.log(poundsToKg(5));

function roundTo(value, decimals) {
  const factor = 10 ** decimals;
  return Math.round(value * factor) / factor;
}

console.log(roundTo(poundsToKg(5), 2));

function milesToKm(miles) {
  return miles * KM_PER_MILE;
}

console.log(milesToKm(26.2));

const kmToMiles = (km) => km / KM_PER_MILE;

console.log(kmToMiles(42.195));

function formatNumber(value, decimals = 2) {
  return String(roundTo(value, decimals)).replace('.', ',');
}

console.log(formatNumber(milesToKm(26.2)));
console.log(formatNumber(milesToKm(26.2), 0));

function formatResult(value, unit, decimals = 1) {
  return `${formatNumber(value, decimals)} ${unit}`;
}

console.log(formatResult(milesToKm(26.2), 'km'));

// Teploty poslal kolega: recepty z USA uvádějí teplotu trouby ve °F.
const ABSOLUTE_ZERO_F = -459.67;
const ABSOLUTE_ZERO_C = -273.15;

function degreesAboveFreezing(fahrenheit) {
  return fahrenheit - 32;
}

function fahrenheitToCelsius(fahrenheit) {
  return degreesAboveFreezing(fahrenheit) * 5 / 9;
}

function celsiusToFahrenheit(celsius) {
  return celsius * 9 / 5 + 32;
}

console.log(formatResult(fahrenheitToCelsius(350), '°C'));

function temperatureLabel(fahrenheit) {
  if (!Number.isFinite(fahrenheit)) {
    return 'Zadej číslo';
  }
  if (fahrenheit < ABSOLUTE_ZERO_F) {
    return 'Pod absolutní nulou';
  }
  return formatResult(fahrenheitToCelsius(fahrenheit), '°C');
}

function distanceLabel(miles) {
  if (!Number.isFinite(miles)) {
    return 'Zadej číslo';
  }
  if (miles < 0) {
    return 'Vzdálenost nemůže být záporná';
  }
  return formatResult(milesToKm(miles), 'km');
}

console.log(distanceLabel(26.2));

function targetUnit(conversion) {
  const unit = conversion.slice(conversion.indexOf('-') + 1);
  if (unit === 'c') {
    return '°C';
  }
  if (unit === 'f') {
    return '°F';
  }
  return unit;
}

console.log(targetUnit('f-c'));

function parseAmount(text) {
  const trimmed = text.trim();
  if (trimmed === '') {
    return NaN;
  }
  return Number(trimmed.replace(',', '.'));
}

console.log(parseAmount(' 26,2 '));

// Kolegův modul pro běžce: tempo z amerických závodů v minutách na kilometr.
function paceLabel(miles, minutes, decimals) {
  const km = milesToKm(miles);
  const pace = minutes / km;
  const places = decimals ?? 2;
  return `${formatNumber(pace, places)} min/km`;
}

console.log(paceLabel(26.2, 210, 0));

function totalKm(...milesLegs) {
  let total = 0;
  for (const miles of milesLegs) {
    total += milesToKm(miles);
  }
  return total;
}

console.log(formatResult(totalKm(297, 396, 370), 'km'));

function describeConversion(text, conversion) {
  const amount = parseAmount(text);
  if (Number.isNaN(amount)) {
    return 'Zadej číslo';
  }
  const result = convert(amount, conversion);
  if (result === null) {
    return 'Neznámý převod';
  }
  return `${formatNumber(amount, 1)} → ${formatResult(result, targetUnit(conversion))}`;
}

console.log(describeConversion('26,2', 'mi-km'));

function printConversion(text, conversion) {
  console.log(describeConversion(text, conversion));
}

printConversion('5', 'lb-kg');

function conversionTable(conversion, from, to) {
  let table = '';
  for (let amount = from; amount <= to; amount++) {
    table += `${describeConversion(String(amount), conversion)}\n`;
  }
  return table.trim();
}

console.log(conversionTable('mi-km', 1, 5));

## --file-- page.js

```js
// Tahle část propojuje tvoje funkce se stránkou. Hledání prvků (querySelector),
// posluchače událostí (addEventListener) a textContent se naučíš v sekci
// DOM, události a prohlížeč. Všimni si ale, že updatePage se předává bez závorek.
const form = document.querySelector('#converter');
const amountInput = document.querySelector('#amount');
const conversionSelect = document.querySelector('#conversion');
const swapButton = document.querySelector('#swap');
const resultOutput = document.querySelector('#result');
const tableOutput = document.querySelector('#table');

function updatePage() {
  resultOutput.textContent = describeConversion(amountInput.value, conversionSelect.value);
  tableOutput.textContent = conversionTable(conversionSelect.value, 1, 5);
}

function swapDirection() {
  conversionSelect.value = swapConversion(conversionSelect.value);
  updatePage();
}

form.addEventListener('submit', (event) => event.preventDefault());
amountInput.addEventListener('input', updatePage);
conversionSelect.addEventListener('change', updatePage);
swapButton.addEventListener('click', swapDirection);
updatePage();
```
