# --description--

Sada je hotová. Aby ji kamarád (nebo kdokoli, kdo projekt otevře) pustil bez
přemýšlení, patří do `package.json` skript `test`. Příkaz `npm test` je dohoda:
v každém projektu v JavaScriptu spustí testy, ať jsou napsané v čemkoli.

**Úkol:** přidej do `package.json` skript `test`, který spustí `node --test`.

Kontrola pak spustí `npm test` a nakonec tvoji sadu pustí proti osmi rozbitým
verzím `utils.js` — po jedné chybě v každé funkci, kterou jsi testoval. Sada musí
chytit všechny.

## Co ses naučil

- Test je kód, který volá tvůj kód: `test`, `assert.equal`, `assert.deepEqual`.
- Chyby bydlí na krajích — nula, hranice podmínky, desetiny, prázdný týden.
- Tabulka případů s testem v cyklu ušetří opisování a ve výpisu ukáže vstup.
- Chybu, kterou funkce slibuje, hlídá `assert.throws` s funkcí a `await assert.rejects`.
- Nahlášenou chybu nejdřív zachytíš červeným testem, pak opravíš.
- Každý test si připraví vlastní data, jinak prochází jen v určitém pořadí.

Stejně otestuješ utility v jakémkoli projektu: formátování cen, validaci formuláře,
výpočty v košíku. Příště z toho uděláš **refaktoring pod ochranou testů** — přepíšeš
přerostlou funkci a testy ti řeknou, že se nic nerozbilo.

## Udělej po svém

- **Spouštění při ukládání:** zkus na svém počítači `node --test --watch` a sleduj,
  jak se sada pouští sama po každém uložení souboru.
- **Nová funkce test-first:** vymysli `formatPace` pro tempo na míli a napiš na ni
  nejdřív červené testy, teprve pak kód.
- **Skupiny testů:** v dokumentaci `node:test` najdi `describe` a seskup testy podle
  funkcí, ať je výpis přehlednější.
- **Tvoje téma:** přepiš deník na to, co měříš ty — plavání, kolo, hodiny u klavíru —
  a sadu testů uprav tak, aby pořád chytila každou podstrčenou chybu.

```js
// Pomocné funkce Běžeckého deníku. Napsal je kolega Ondra, testy zatím nemají.

// Délka běhu v sekundách (z hodinek i s desetinami) → text zaokrouhlený na celé sekundy, třeba 2525 → „42:05".
export function formatDuration(totalSeconds) {
  const rounded = Math.round(totalSeconds);
  const hours = Math.floor(rounded / 3600);
  const minutes = Math.floor((rounded % 3600) / 60);
  const seconds = String(rounded % 60).padStart(2, '0');
  if (hours > 0) {
    return `${hours}:${String(minutes).padStart(2, '0')}:${seconds}`;
  }
  return `${minutes}:${seconds}`;
}

// Text z formuláře „42:05" nebo „1:05:30" → počet sekund. Neplatný čas vyhodí RangeError.
export function parseDuration(text) {
  const parts = text.split(':').map(Number);
  const [seconds, minutes, hours = 0] = [...parts].reverse();
  const valid =
    parts.length >= 2 &&
    parts.length <= 3 &&
    parts.every((part) => Number.isInteger(part) && part >= 0) &&
    seconds <= 59 &&
    (parts.length === 2 || minutes <= 59);
  if (!valid) {
    throw new RangeError(`Neplatný čas „${text}", čekám m:ss nebo h:mm:ss`);
  }
  return hours * 3600 + minutes * 60 + seconds;
}

// Tempo na kilometr zaokrouhlené na celé sekundy, třeba „5:03 /km". Vzdálenost musí být kladná.
export function pace(totalSeconds, km) {
  if (!(km > 0)) {
    throw new RangeError('Vzdálenost musí být větší než 0 km');
  }
  return `${formatDuration(Math.round(totalSeconds / km))} /km`;
}

// Název běhu → kousek adresy: malá písmena bez diakritiky, slova oddělená jednou pomlčkou.
export function slugify(title) {
  return title
    .normalize('NFD')
    .replace(/\p{Diacritic}/gu, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');
}

// Souhrn týdne: počet běhů, kilometry zaokrouhlené na desetiny a celkový čas v sekundách.
export function weekSummary(runs) {
  const km = runs.reduce((sum, run) => sum + run.km, 0);
  const seconds = runs.reduce((sum, run) => sum + run.seconds, 0);
  return { count: runs.length, km: Math.round(km * 10) / 10, seconds };
}

// Načte běh z úložiště. Neznámé id skončí chybou „Běh … nenalezen".
export async function loadRun(store, id) {
  const run = store.get(id);
  if (!run) {
    throw new Error(`Běh ${id} nenalezen`);
  }
  return run;
}

// Uloží běh a vrátí jeho nové id. Běh bez kladné vzdálenosti odmítne chybou RangeError.
export async function saveRun(store, run) {
  if (!(run.km > 0)) {
    throw new RangeError('Běh musí mít kladnou vzdálenost');
  }
  const id = store.size + 1;
  store.set(id, { ...run, id });
  return id;
}
```

## --file-- utils.test.js

```js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { formatDuration, parseDuration, pace, slugify, weekSummary, loadRun, saveRun } from './utils.js';

test('formatDuration píše minuty a sekundy', () => {
  assert.equal(formatDuration(2525), '42:05');
});

test('formatDuration píše u dlouhého běhu hodiny', () => {
  assert.equal(formatDuration(3725), '1:02:05');
});

test('formatDuration u běhu pod minutu začíná nulou', () => {
  assert.equal(formatDuration(59), '0:59');
});

test('formatDuration píše přesně hodinu už s hodinami', () => {
  assert.equal(formatDuration(3600), '1:00:00');
});

test('parseDuration odmítne 60 sekund', () => {
  assert.throws(() => parseDuration('7:60'), RangeError);
});

const validTimes = [
  { text: '42:05', seconds: 2525 },
  { text: '0:59', seconds: 59 },
  { text: '1:05:30', seconds: 3930 },
];
for (const { text, seconds } of validTimes) {
  test(`parseDuration('${text}') vrátí ${seconds} s`, () => {
    assert.equal(parseDuration(text), seconds);
  });
}

const slugCases = [
  { title: 'Běh kolem Máchova jezera', slug: 'beh-kolem-machova-jezera' },
  { title: '  Půlmaraton Praha 2026! ', slug: 'pulmaraton-praha-2026' },
  { title: 'Ranní   výklus -- Stromovka', slug: 'ranni-vyklus-stromovka' },
];
for (const { title, slug } of slugCases) {
  test(`slugify('${title}') vrátí ${slug}`, () => {
    assert.equal(slugify(title), slug);
  });
}

test('loadRun odmítne neznámé id', async () => {
  const store = new Map();
  await assert.rejects(loadRun(store, 99), { message: /nenalezen/ });
});

test('saveRun odmítne běh bez vzdálenosti', async () => {
  const store = new Map();
  await assert.rejects(saveRun(store, { title: 'Rozcvička', km: 0, seconds: 300 }), RangeError);
});

test('saveRun uloží běh a loadRun ho vrátí', async () => {
  const store = new Map();
  const run = { title: 'Divoká Šárka', km: 9.8, seconds: 3300 };

  const id = await saveRun(store, run);
  const loaded = await loadRun(store, id);

  assert.equal(id, 1);
  assert.deepEqual(loaded, { ...run, id: 1 });
});

test('weekSummary sečte týden a zaokrouhlí kilometry na desetiny', () => {
  const runs = [
    { title: 'Stromovka', km: 5.2, seconds: 1680 },
    { title: 'Letná', km: 10.1, seconds: 3060 },
    { title: 'Pražský půlmaraton', km: 21.1, seconds: 6900 },
  ];
  assert.deepEqual(weekSummary(runs), { count: 3, km: 36.4, seconds: 11640 });
});

test('weekSummary vrátí pro týden bez běhu nuly', () => {
  assert.deepEqual(weekSummary([]), { count: 0, km: 0, seconds: 0 });
});

test('pace zaokrouhlí tempo na celé sekundy dolů i nahoru', () => {
  assert.equal(pace(1517, 5), '5:03 /km');
  assert.equal(pace(1518, 5), '5:04 /km');
});

test('pace odmítne nulovou i zápornou vzdálenost', () => {
  assert.throws(() => pace(1500, 0), RangeError);
  assert.throws(() => pace(1500, -3), RangeError);
});

const invalidTimes = ['-5:00', '1:75:00', '4:30.5', '1:2:3:4', 'čtyřicet minut'];
for (const text of invalidTimes) {
  test(`parseDuration('${text}') vyhodí RangeError`, () => {
    assert.throws(() => parseDuration(text), RangeError);
  });
}

test('formatDuration zaokrouhlí desetiny z hodinek na celé sekundy', () => {
  assert.equal(formatDuration(2525.6), '42:06');
});

test('saveRun dá prvnímu běhu id 1', async () => {
  const store = new Map();
  const id = await saveRun(store, { title: 'Stromovka', km: 6.2, seconds: 2010 });
  assert.equal(id, 1);
});

test('loadRun najde běh podle id', async () => {
  const store = new Map();
  await saveRun(store, { title: 'Hvězda', km: 7.5, seconds: 2400 });
  const run = await loadRun(store, 1);
  assert.equal(run.title, 'Hvězda');
});
```
