# --description--

Kamarád běží půlmaratony a v přehledu mu svítí `67:00` místo `1:07:00`. Klikni na
**Spustit** — `ukazka.js` vypíše přesně to. Chce hodiny.

Než sáhneš do `utils.js`, napiš test na nové chování. Teď **musí selhat**, protože
`formatDuration` hodiny ještě neumí. Tím si ověříš dvě věci: že test opravdu hlídá
hodiny (kdyby prošel, hlídá něco jiného) a že po opravě v dalším kroku zezelená
právě kvůli tvé změně.

Formát s hodinami je `h:mm:ss`: minuty i sekundy mají dvě číslice. Například
běh 1 h 2 min 5 s trvá 3 725 sekund a má se zobrazit jako `1:02:05`.

**Úkol:** přidej do `utils.test.js` druhý test, který pro dlouhý běh čeká formát
s hodinami. `utils.js` neměň. Výsledek `node --test` bude `pass 1`, `fail 1` — a to
je správně.

# --hints--

`node --test utils.test.js` skončí přesně jedním selhaným testem a původní test dál prochází.

```js
const result = await helpers.run('node --test utils.test.js');
const pass = Number(/pass (\d+)/.exec(result.stdout)?.[1] ?? 0);
const fail = Number(/fail (\d+)/.exec(result.stdout)?.[1] ?? 0);
assert.equal(fail, 1, `nový test má teď selhat (formatDuration hodiny neumí), node --test hlásí fail ${fail}:\n${result.stdout}${result.stderr}`);
assert.ok(pass >= 1, `původní test na 42:05 má dál projít, node --test hlásí pass ${pass}`);
```

Nový test čeká formát `h:mm:ss`: nad `formatDuration`, která hodiny umí, projdou všechny tvoje testy.

```js
const fs = await import('node:fs/promises');
const source = files['utils.js'];
const from = `export function formatDuration(totalSeconds) {
  const minutes = Math.floor(totalSeconds / 60);`;
assert.ok(source.includes(from), 'utils.js v tomhle kroku neměň — hodiny doplníš až v dalším kroku');
const withHours = source.replace(from, `export function formatDuration(totalSeconds) {
  const hours = Math.floor(totalSeconds / 3600);
  if (hours > 0) {
    const mm = String(Math.floor((totalSeconds % 3600) / 60)).padStart(2, '0');
    const ss = String(totalSeconds % 60).padStart(2, '0');
    return \`\${hours}:\${mm}:\${ss}\`;
  }
  const minutes = Math.floor(totalSeconds / 60);`);
await fs.mkdir(`${helpers.dir}/mutant-hours`, { recursive: true });
await fs.writeFile(`${helpers.dir}/mutant-hours/utils.js`, withHours);
await fs.writeFile(`${helpers.dir}/mutant-hours/utils.test.js`, files['utils.test.js']);
const fixed = await helpers.run('node --test mutant-hours/utils.test.js');
assert.equal(fixed.code, 0, `se správně doplněnými hodinami (3725 → „1:02:05") mají tvoje testy projít — čeká nový test opravdu h:mm:ss?\n${fixed.stdout}`);
```

```js
// Pomocné funkce Běžeckého deníku. Napsal je kolega Ondra, testy zatím nemají.

// Délka běhu v sekundách → text pro přehled, třeba 2525 → „42:05".
export function formatDuration(totalSeconds) {
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = String(totalSeconds % 60).padStart(2, '0');
  return `${minutes}:${seconds}`;
}

// Text z formuláře „42:05" nebo „1:05:30" → počet sekund. Neplatný čas vyhodí RangeError.
export function parseDuration(text) {
  const parts = text.split(':').map(Number);
  const [seconds, minutes, hours = 0] = [...parts].reverse();
  const valid =
    parts.length >= 2 &&
    parts.length <= 3 &&
    parts.every((part) => Number.isInteger(part) && part >= 0) &&
    seconds <= 59 &&
    (parts.length === 2 || minutes <= 59);
  if (!valid) {
    throw new RangeError(`Neplatný čas „${text}", čekám m:ss nebo h:mm:ss`);
  }
  return hours * 3600 + minutes * 60 + seconds;
}

// Tempo na kilometr zaokrouhlené na celé sekundy, třeba „5:03 /km". Vzdálenost musí být kladná.
export function pace(totalSeconds, km) {
  if (!(km > 0)) {
    throw new RangeError('Vzdálenost musí být větší než 0 km');
  }
  return `${formatDuration(Math.round(totalSeconds / km))} /km`;
}

// Název běhu → kousek adresy: malá písmena bez diakritiky, slova oddělená jednou pomlčkou.
export function slugify(title) {
  return title
    .normalize('NFD')
    .replace(/\p{Diacritic}/gu, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '');
}

// Souhrn týdne: počet běhů, kilometry zaokrouhlené na desetiny a celkový čas v sekundách.
export function weekSummary(runs) {
  const km = runs.reduce((sum, run) => sum + run.km, 0);
  const seconds = runs.reduce((sum, run) => sum + run.seconds, 0);
  return { count: runs.length, km: Math.round(km * 10) / 10, seconds };
}

// Načte běh z úložiště. Neznámé id skončí chybou „Běh … nenalezen".
export async function loadRun(store, id) {
  const run = store.get(id);
  if (!run) {
    throw new Error(`Běh ${id} nenalezen`);
  }
  return run;
}

// Uloží běh a vrátí jeho nové id. Běh bez kladné vzdálenosti odmítne chybou RangeError.
export async function saveRun(store, run) {
  if (!(run.km > 0)) {
    throw new RangeError('Běh musí mít kladnou vzdálenost');
  }
  const id = store.size + 1;
  store.set(id, { ...run, id });
  return id;
}
```
