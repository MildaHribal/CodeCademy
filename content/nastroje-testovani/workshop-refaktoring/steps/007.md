# --description--

Slevy se v knihovně mění nejčastěji ze všeho: student, senior, brigádník, nová akce
na jaro. Dneska jsou schované ve dvou skoro stejných podmínkách na konci `vypisUpominky`.
Po tomhle kroku budou v samostatném souboru, kde je najde i člověk, který o zbytku kódu
nic neví.

Modul `slevy.js` už v projektu je, zatím prázdný. Bude držet tabulku slev a jednu
funkci, která pro kategorii čtenáře vrátí **koeficient**, kterým se násobí celková
částka: dnes platí student i senior polovinu, ostatní plnou cenu.

**Úkol:** seřaď řádky v `slevy.js` a doplň hodnotu do mezery. Odsazení nastavíš
klávesou Tab. Dva řádky jsou navíc — oba odpovídají chybě, která se tu nabízí.

Zapojíš ho do upomínky až v dalším kroku, teď stačí, že funkce sama počítá správně.

```js
// Upomínky Městské knihovny Kutná Hora.
// První verzi napsal Ivan Sýkora v roce 2019, od té doby sem každý přidal svoje pravidlo.

export const SAZEBNIK = {
  kniha: { zaDen: 5, odklad: 3, strop: 300, body: 1 },
  'časopis': { zaDen: 8, odklad: 0, strop: 200, body: 1 },
  audiokniha: { zaDen: 12, odklad: 1, strop: 400, body: 2 },
};

// Co v sazebníku není (deskové hry, mapy, stolní hry), počítá se takhle.
export const VYCHOZI_SAZBA = { zaDen: 10, odklad: 0, strop: 250, body: 1 };

export function vypisUpominky(ctenar) {
  let text = 'Upomínka — Městská knihovna Kutná Hora\n';
  text += `Čtenář: ${ctenar.jmeno} (${ctenar.kategorie})\n`;
  let celkem = 0;
  let body = 0;
  for (const vypujcka of ctenar.vypujcky) {
    const poplatek = poplatekZaVypujcku(vypujcka);
    body += bodyZaVypujcku(vypujcka);
    text += `  ${vypujcka.nazev} (${vypujcka.druh}): ${poplatek} Kč, po termínu ${tvarDnu(vypujcka.dniPoTerminu)}\n`;
    celkem += poplatek;
  }
  if (ctenar.kategorie === 'student') {
    celkem = Math.round(celkem * 0.5);
  }
  if (ctenar.kategorie === 'senior') {
    celkem = Math.round(celkem * 0.5);
  }
  text += `Celkem k zaplacení: ${celkem} Kč\n`;
  text += `Body do čtenářského klubu: ${body}\n`;
  return text;
}

export function poplatekZaVypujcku(vypujcka) {
  const sazba = SAZEBNIK[vypujcka.druh] ?? VYCHOZI_SAZBA;
  const dniNavic = vypujcka.dniPoTerminu - sazba.odklad;
  if (dniNavic <= 0) {
    return 0;
  }
  return Math.min(dniNavic * sazba.zaDen, sazba.strop);
}

export function bodyZaVypujcku(vypujcka) {
  const sazba = SAZEBNIK[vypujcka.druh] ?? VYCHOZI_SAZBA;
  return sazba.body;
}

function tvarDnu(pocet) {
  if (pocet === 1) {
    return '1 den';
  }
  if (pocet >= 2 && pocet <= 4) {
    return `${pocet} dny`;
  }
  return `${pocet} dní`;
}
```

## --file-- poplatky.test.js

```js
import { test } from 'node:test';
import assert from 'node:assert/strict';

import { vypisUpominky } from './poplatky.js';
import { jana, petr, marie } from './ctenari.js';

test('upomínka pro Janu Novákovou (student) vypadá přesně takhle', () => {
  const ocekavano = [
    'Upomínka — Městská knihovna Kutná Hora',
    'Čtenář: Jana Nováková (student)',
    '  Alchymista (kniha): 55 Kč, po termínu 14 dní',
    '  Respekt 12/2026 (časopis): 24 Kč, po termínu 3 dny',
    '  Šifra mistra Leonarda (audiokniha): 0 Kč, po termínu 0 dní',
    'Celkem k zaplacení: 40 Kč',
    'Body do čtenářského klubu: 4',
    '',
  ].join('\n');

  assert.equal(vypisUpominky(jana), ocekavano, 'Upomínka pro Janu se změnila proti dnešnímu stavu');
});

test('Petr Doležal nic nedluží, ale body dostane', () => {
  const ocekavano = [
    'Upomínka — Městská knihovna Kutná Hora',
    'Čtenář: Petr Doležal (dospělý)',
    '  Bílá velryba (kniha): 0 Kč, po termínu 2 dny',
    '  Reflex 09/2026 (časopis): 0 Kč, po termínu 0 dní',
    '  Saturnin (audiokniha): 0 Kč, po termínu 1 den',
    'Celkem k zaplacení: 0 Kč',
    'Body do čtenářského klubu: 4',
    '',
  ].join('\n');

  assert.equal(vypisUpominky(petr), ocekavano, 'Upomínka pro Petra se změnila proti dnešnímu stavu');
});

test('Marie Kratochvílová (senior) narazí na stropy a slevu', () => {
  const ocekavano = [
    'Upomínka — Městská knihovna Kutná Hora',
    'Čtenář: Marie Kratochvílová (senior)',
    '  Osadníci z Katanu (desková hra): 250 Kč, po termínu 40 dní',
    '  Kroniky Narnie (kniha): 300 Kč, po termínu 95 dní',
    'Celkem k zaplacení: 275 Kč',
    'Body do čtenářského klubu: 2',
    '',
  ].join('\n');

  assert.equal(vypisUpominky(marie), ocekavano, 'Upomínka pro Marii se změnila proti dnešnímu stavu');
});
```
