# --description--

Tohle je řez, kvůli kterému celý refaktoring začal. `vypisUpominky` dnes v jednom
cyklu počítá částky a zároveň skládá text. Web ale žádný text nechce — chce čísla.
Po tomhle kroku půjde upomínku spočítat, aniž by vznikl jediný řádek textu.

Rozdělíš ji na dvě funkce:

- `spoctiUpominku(ctenar)` spočítá a vrátí data, nic nevypisuje,
- `formatujUpominku(souhrn)` dostane ta data a poskládá z nich text. **Nic nepočítá**
  — ani součty, ani slevu.

`spoctiUpominku` vrací objekt v tomhle tvaru (tady pro vymyšlenou čtenářku
s jednou výpůjčkou):

```json
{
  "jmeno": "Eva Marková",
  "kategorie": "dospělý",
  "polozky": [
    { "nazev": "Hobit", "druh": "kniha", "dniPoTerminu": 6, "poplatek": 15, "body": 1 }
  ],
  "celkem": 15,
  "body": 1
}
```

**Úkol:** vyveze obě funkce a z `vypisUpominky` nech jen předání výsledku jedné
do druhé, aby ji mohl `ukazka.js` volat dál beze změny.

# --help--

## --tip--

Řez vede tam, kde se v cyklu potkávají dvě různé práce: přičítání čísel a přilepování řádků k textu.

## --tip--

Funkce, která skládá text, nesmí sáhnout na sazebník ani na slevu. Všechno, co potřebuje, má dostat v objektu.

## --tip--

Původní `vypisUpominky` nemaž ani nepřejmenovávej — `ukazka.js` i tvoje testy ji volají dál a díky tomu uvidíš, že se chování nezměnilo.

# --explain--

Pojmenuj vlastními slovy kroky, kterými jsi z `vypisUpominky` udělal dvě funkce. Podle čeho jsi poznal, kudy vede řez?

## --model--

Nejdřív jsem si všiml, že cyklus dělá dvě různé práce najednou: počítá čísla
a skládá text. To je hranice řezu — jedna funkce má mít jeden účel. Pak jsem vymyslel
tvar dat, které si obě části předají, protože ten určuje, co bude muset formátování
vědět. Původní kód jsem rozkopíroval do dvou funkcí a starý název nechal jako spojku,
aby volající nic nepoznali. Nakonec jsem spustil testy: kdyby zčervenaly, vracím se
o krok zpátky, ne opravuju dopředu.

## --checklist--

- Hranice řezu byla tam, kde jedna funkce dělala dvě různé práce.
- Tvar předávaných dat jsem vymyslel dřív, než jsem začal přesouvat kód.
- Původní název zůstal jako spojka, takže volající kód se nemusel měnit.
- Testy po každém kroku říkají, jestli se chování nezměnilo.

```js
// Upomínky Městské knihovny Kutná Hora.
// První verzi napsal Ivan Sýkora v roce 2019, od té doby sem každý přidal svoje pravidlo.

import { slevaProKategorii } from './slevy.js';

export const SAZEBNIK = {
  kniha: { zaDen: 5, odklad: 3, strop: 300, body: 1 },
  'časopis': { zaDen: 8, odklad: 0, strop: 200, body: 1 },
  audiokniha: { zaDen: 12, odklad: 1, strop: 400, body: 2 },
  'desková hra': { zaDen: 10, odklad: 0, strop: 250, body: 1 },
};

// Co v sazebníku není (mapy, stolní hry), počítá se takhle.
export const VYCHOZI_SAZBA = { zaDen: 10, odklad: 0, strop: 250, body: 1 };

export function vypisUpominky(ctenar) {
  let text = 'Upomínka — Městská knihovna Kutná Hora\n';
  text += `Čtenář: ${ctenar.jmeno} (${ctenar.kategorie})\n`;
  let celkem = 0;
  let body = 0;
  for (const vypujcka of ctenar.vypujcky) {
    const poplatek = poplatekZaVypujcku(vypujcka);
    body += bodyZaVypujcku(vypujcka);
    text += `  ${vypujcka.nazev} (${vypujcka.druh}): ${poplatek} Kč, po termínu ${tvarDnu(vypujcka.dniPoTerminu)}\n`;
    celkem += poplatek;
  }
  celkem = Math.round(celkem * slevaProKategorii(ctenar.kategorie));
  text += `Celkem k zaplacení: ${celkem} Kč\n`;
  text += `Body do čtenářského klubu: ${body}\n`;
  return text;
}

export function poplatekZaVypujcku(vypujcka) {
  const sazba = SAZEBNIK[vypujcka.druh] ?? VYCHOZI_SAZBA;
  const dniNavic = vypujcka.dniPoTerminu - sazba.odklad;
  if (dniNavic <= 0) {
    return 0;
  }
  return Math.min(dniNavic * sazba.zaDen, sazba.strop);
}

export function bodyZaVypujcku(vypujcka) {
  const sazba = SAZEBNIK[vypujcka.druh] ?? VYCHOZI_SAZBA;
  return sazba.body;
}

function tvarDnu(pocet) {
  if (pocet === 1) {
    return '1 den';
  }
  if (pocet >= 2 && pocet <= 4) {
    return `${pocet} dny`;
  }
  return `${pocet} dní`;
}
```

## --file-- poplatky.js

```js
// Upomínky Městské knihovny Kutná Hora.
// První verzi napsal Ivan Sýkora v roce 2019, od té doby sem každý přidal svoje pravidlo.

import { slevaProKategorii } from './slevy.js';

export const SAZEBNIK = {
  kniha: { zaDen: 5, odklad: 3, strop: 300, body: 1 },
  'časopis': { zaDen: 8, odklad: 0, strop: 200, body: 1 },
  audiokniha: { zaDen: 12, odklad: 1, strop: 400, body: 2 },
  'desková hra': { zaDen: 10, odklad: 0, strop: 250, body: 1 },
};

// Co v sazebníku není (mapy, stolní hry), počítá se takhle.
export const VYCHOZI_SAZBA = { zaDen: 10, odklad: 0, strop: 250, body: 1 };

export function spoctiUpominku(ctenar) {
  const polozky = ctenar.vypujcky.map((vypujcka) => ({
    nazev: vypujcka.nazev,
    druh: vypujcka.druh,
    dniPoTerminu: vypujcka.dniPoTerminu,
    poplatek: poplatekZaVypujcku(vypujcka),
    body: bodyZaVypujcku(vypujcka),
  }));
  const poplatky = polozky.reduce((soucet, polozka) => soucet + polozka.poplatek, 0);
  return {
    jmeno: ctenar.jmeno,
    kategorie: ctenar.kategorie,
    polozky,
    celkem: Math.round(poplatky * slevaProKategorii(ctenar.kategorie)),
    body: polozky.reduce((soucet, polozka) => soucet + polozka.body, 0),
  };
}

export function formatujUpominku(souhrn) {
  const radky = [
    'Upomínka — Městská knihovna Kutná Hora',
    `Čtenář: ${souhrn.jmeno} (${souhrn.kategorie})`,
    ...souhrn.polozky.map(
      (polozka) =>
        `  ${polozka.nazev} (${polozka.druh}): ${polozka.poplatek} Kč, po termínu ${tvarDnu(polozka.dniPoTerminu)}`,
    ),
    `Celkem k zaplacení: ${souhrn.celkem} Kč`,
    `Body do čtenářského klubu: ${souhrn.body}`,
    '',
  ];
  return radky.join('\n');
}

export function vypisUpominky(ctenar) {
  return formatujUpominku(spoctiUpominku(ctenar));
}

export function poplatekZaVypujcku(vypujcka) {
  const sazba = SAZEBNIK[vypujcka.druh] ?? VYCHOZI_SAZBA;
  const dniNavic = vypujcka.dniPoTerminu - sazba.odklad;
  if (dniNavic <= 0) {
    return 0;
  }
  return Math.min(dniNavic * sazba.zaDen, sazba.strop);
}

export function bodyZaVypujcku(vypujcka) {
  const sazba = SAZEBNIK[vypujcka.druh] ?? VYCHOZI_SAZBA;
  return sazba.body;
}

function tvarDnu(pocet) {
  if (pocet === 1) {
    return '1 den';
  }
  if (pocet >= 2 && pocet <= 4) {
    return `${pocet} dny`;
  }
  return `${pocet} dní`;
}
```
