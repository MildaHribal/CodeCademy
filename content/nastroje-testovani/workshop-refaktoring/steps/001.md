# --description--

Městská knihovna v Kutné Hoře ti dala do ruky jeden soubor. `poplatky.js` počítá
upomínky za pozdě vrácené knihy: první verzi napsal v roce 2019 Ivan Sýkora a od té
doby do ní každý přidal svoje pravidlo. Teď chce knihovna posílat upomínky i e-mailem
a ukazovat je ve čtenářském účtu na webu — a do téhle jediné funkce se to nevejde.

Za celý workshop z ní uděláš několik malých funkcí, které jdou použít samostatně.
Knihovna má jedinou podmínku: **částka na upomínce musí vyjít na korunu stejně jako
dnes.** Proto ještě nic nepřepisuješ a nejdřív se podíváš, co kód vlastně umí.

**Úkol:** v `ukazka.js` naimportuj funkci `vypisUpominky` ze souboru `poplatky.js`
a čtenářku `jana` ze souboru `ctenari.js`, zavolej funkci se čtenářkou a výsledek
vypiš do konzole. Pak zmáčkni **Spustit**.

Tohle je opakování bez návodu — pojmenované exporty a jejich import znáš z lekce
[ES moduly](see:nastroje-moduly-vite/es-moduly).

## --file-- poplatky.js

```js
// Upomínky Městské knihovny Kutná Hora.
// První verzi napsal Ivan Sýkora v roce 2019, od té doby sem každý přidal svoje pravidlo.

export function vypisUpominky(ctenar) {
  let text = 'Upomínka — Městská knihovna Kutná Hora\n';
  text += `Čtenář: ${ctenar.jmeno} (${ctenar.kategorie})\n`;
  let celkem = 0;
  let body = 0;
  for (const vypujcka of ctenar.vypujcky) {
    let poplatek = 0;
    if (vypujcka.druh === 'kniha') {
      if (vypujcka.dniPoTerminu > 3) {
        poplatek = (vypujcka.dniPoTerminu - 3) * 5;
      }
      if (poplatek > 300) {
        poplatek = 300;
      }
      body += 1;
    } else if (vypujcka.druh === 'časopis') {
      if (vypujcka.dniPoTerminu > 0) {
        poplatek = vypujcka.dniPoTerminu * 8;
      }
      if (poplatek > 200) {
        poplatek = 200;
      }
      body += 1;
    } else if (vypujcka.druh === 'audiokniha') {
      if (vypujcka.dniPoTerminu > 1) {
        poplatek = (vypujcka.dniPoTerminu - 1) * 12;
      }
      if (poplatek > 400) {
        poplatek = 400;
      }
      body += 2;
    } else {
      if (vypujcka.dniPoTerminu > 0) {
        poplatek = vypujcka.dniPoTerminu * 10;
      }
      if (poplatek > 250) {
        poplatek = 250;
      }
      body += 1;
    }
    text += `  ${vypujcka.nazev} (${vypujcka.druh}): ${poplatek} Kč, po termínu ${tvarDnu(vypujcka.dniPoTerminu)}\n`;
    celkem += poplatek;
  }
  if (ctenar.kategorie === 'student') {
    celkem = Math.round(celkem * 0.5);
  }
  if (ctenar.kategorie === 'senior') {
    celkem = Math.round(celkem * 0.5);
  }
  text += `Celkem k zaplacení: ${celkem} Kč\n`;
  text += `Body do čtenářského klubu: ${body}\n`;
  return text;
}

function tvarDnu(pocet) {
  if (pocet === 1) {
    return '1 den';
  }
  if (pocet >= 2 && pocet <= 4) {
    return `${pocet} dny`;
  }
  return `${pocet} dní`;
}
```
