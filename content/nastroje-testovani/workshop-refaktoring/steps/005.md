# --description--

Sazby jsou dnes rozsypané v podmínkách: každý druh výpůjčky má svoje tři čísla
schovaná v jiné větvi. Po tomhle kroku bude ceník na jednom místě jako data, takže
změna sazby přestane být zásahem do kódu.

Z kaskády vytáhni tuhle tabulku:

| druh | za den | odklad (dny zdarma) | strop |
|---|---|---|---|
| kniha | 5 Kč | 3 | 300 Kč |
| časopis | 8 Kč | 0 | 200 Kč |
| audiokniha | 12 Kč | 1 | 400 Kč |
| cokoli dalšího | 10 Kč | 0 | 250 Kč |

Pozor na poslední řádek: dnešní `else` větev platí pro **všechno ostatní**, tedy
i pro deskové hry, mapy nebo společenské hry. Kdybys ji zahodil, upomínky se změní.

**Úkol:** v `poplatky.js` vyveze objekt `SAZEBNIK` s klíči `kniha`, `časopis`
a `audiokniha` a objekt `VYCHOZI_SAZBA` pro všechno ostatní. Každá položka má
vlastnosti `zaDen`, `odklad` a `strop`. Funkci `poplatekZaVypujcku` pak přepiš tak,
aby si z tabulky vzala sazbu a spočítala poplatek jediným vzorcem, bez podmínek
na druh výpůjčky.

## --tip--

Tabulka je obyčejný objekt, ve kterém je klíčem druh výpůjčky a hodnotou objekt se třemi čísly. Vyhledání pak nahradí celou kaskádu.

## --tip--

Když v tabulce klíč není, vrátí vyhledání `undefined` — na tohle místo patří výchozí sazba.

## --tip--

Vzorec má tři kroky na jiných datech: od počtu dní odečti dny zdarma, výsledek porovnej s nulou a teprve pak vynásob a ořízni stropem.

```js
export function vypisUpominky(ctenar) {
  let text = 'Upomínka — Městská knihovna Kutná Hora\n';
  text += `Čtenář: ${ctenar.jmeno} (${ctenar.kategorie})\n`;
  let celkem = 0;
  let body = 0;
  for (const vypujcka of ctenar.vypujcky) {
    const poplatek = poplatekZaVypujcku(vypujcka);
    if (vypujcka.druh === 'kniha') {
      body += 1;
    } else if (vypujcka.druh === 'časopis') {
      body += 1;
    } else if (vypujcka.druh === 'audiokniha') {
      body += 2;
    } else {
      body += 1;
    }
    text += `  ${vypujcka.nazev} (${vypujcka.druh}): ${poplatek} Kč, po termínu ${tvarDnu(vypujcka.dniPoTerminu)}\n`;
    celkem += poplatek;
  }
  if (ctenar.kategorie === 'student') {
    celkem = Math.round(celkem * 0.5);
  }
  if (ctenar.kategorie === 'senior') {
    celkem = Math.round(celkem * 0.5);
  }
  text += `Celkem k zaplacení: ${celkem} Kč\n`;
  text += `Body do čtenářského klubu: ${body}\n`;
  return text;
}

export function poplatekZaVypujcku(vypujcka) {
  let poplatek = 0;
  if (vypujcka.druh === 'kniha') {
    if (vypujcka.dniPoTerminu > 3) {
      poplatek = (vypujcka.dniPoTerminu - 3) * 5;
    }
    if (poplatek > 300) {
      poplatek = 300;
    }
  } else if (vypujcka.druh === 'časopis') {
    if (vypujcka.dniPoTerminu > 0) {
      poplatek = vypujcka.dniPoTerminu * 8;
    }
    if (poplatek > 200) {
      poplatek = 200;
    }
  } else if (vypujcka.druh === 'audiokniha') {
    if (vypujcka.dniPoTerminu > 1) {
      poplatek = (vypujcka.dniPoTerminu - 1) * 12;
    }
    if (poplatek > 400) {
      poplatek = 400;
    }
  } else {
    if (vypujcka.dniPoTerminu > 0) {
      poplatek = vypujcka.dniPoTerminu * 10;
    }
    if (poplatek > 250) {
      poplatek = 250;
    }
  }
  return poplatek;
}

function tvarDnu(pocet) {
  if (pocet === 1) {
    return '1 den';
  }
  if (pocet >= 2 && pocet <= 4) {
    return `${pocet} dny`;
  }
  return `${pocet} dní`;
}
```

## --file-- poplatky.js

```js
// Upomínky Městské knihovny Kutná Hora.
// První verzi napsal Ivan Sýkora v roce 2019, od té doby sem každý přidal svoje pravidlo.

export const SAZEBNIK = {
  kniha: { zaDen: 5, odklad: 3, strop: 300 },
  'časopis': { zaDen: 8, odklad: 0, strop: 200 },
  audiokniha: { zaDen: 12, odklad: 1, strop: 400 },
};

// Co v sazebníku není (deskové hry, mapy, stolní hry), počítá se takhle.
export const VYCHOZI_SAZBA = { zaDen: 10, odklad: 0, strop: 250 };

export function vypisUpominky(ctenar) {
  let text = 'Upomínka — Městská knihovna Kutná Hora\n';
  text += `Čtenář: ${ctenar.jmeno} (${ctenar.kategorie})\n`;
  let celkem = 0;
  let body = 0;
  for (const vypujcka of ctenar.vypujcky) {
    const poplatek = poplatekZaVypujcku(vypujcka);
    if (vypujcka.druh === 'kniha') {
      body += 1;
    } else if (vypujcka.druh === 'časopis') {
      body += 1;
    } else if (vypujcka.druh === 'audiokniha') {
      body += 2;
    } else {
      body += 1;
    }
    text += `  ${vypujcka.nazev} (${vypujcka.druh}): ${poplatek} Kč, po termínu ${tvarDnu(vypujcka.dniPoTerminu)}\n`;
    celkem += poplatek;
  }
  if (ctenar.kategorie === 'student') {
    celkem = Math.round(celkem * 0.5);
  }
  if (ctenar.kategorie === 'senior') {
    celkem = Math.round(celkem * 0.5);
  }
  text += `Celkem k zaplacení: ${celkem} Kč\n`;
  text += `Body do čtenářského klubu: ${body}\n`;
  return text;
}

export function poplatekZaVypujcku(vypujcka) {
  const sazba = SAZEBNIK[vypujcka.druh] ?? VYCHOZI_SAZBA;
  const dniNavic = vypujcka.dniPoTerminu - sazba.odklad;
  if (dniNavic <= 0) {
    return 0;
  }
  return Math.min(dniNavic * sazba.zaDen, sazba.strop);
}

function tvarDnu(pocet) {
  if (pocet === 1) {
    return '1 den';
  }
  if (pocet >= 2 && pocet <= 4) {
    return `${pocet} dny`;
  }
  return `${pocet} dní`;
}
```
