# --description--

`data ?? []` z minulého kroku je berlička: v první chvíli ukáže uživateli prázdný
seznam a souhrn „Hotovo 0 z 0", jako by na směně nebylo nic. To je horší než
poctivé „načítám".

Dotaz má vlastnost `status`, která nabývá tří hodnot: `'pending'` (data ještě
nejsou), `'error'` (poslední pokus selhal) a `'success'` (data jsou).

**Úkol:** V `Ukoly.jsx` si z dotazu vezmi kromě `data` i `status`. Když je
`status` roven `'pending'`, vrať z komponenty rovnou
`<p className="hlaska">Načítám úkoly směny…</p>`. Ve zbytku komponenty pak už
můžeš s `data` pracovat přímo — berličku s `?? []` smaž.

Brzký `return` je tady na místě: zbytek komponenty se o „ještě nemáme data"
nemusí starat vůbec.

Jakmile data dorazí, hláška zmizí a objeví se seznam.

```js
await helpers.waitFor(() => document.querySelectorAll('.ukoly li').length === 5);
assert.doesNotMatch(document.body.textContent, /Načítám úkoly směny/, 'po načtení už hláška být vidět nemá');
assert.match(document.querySelector('.souhrn').textContent, /Hotovo\s*2 z 5/, 'po načtení má souhrn ukazovat skutečná čísla');
```

Souhrn nikdy neukáže „0 z 0" — berlička s prázdným polem je pryč.

```js
const zdroj = helpers.stripComments(files['Ukoly.jsx'], 'js');

assert.match(zdroj, /['"]pending['"]/, 'komponenta má rozpoznat stav pending');
```

```jsx
import { useQuery } from '@tanstack/react-query';
import { nactiUkoly } from './api';
import { souhrn } from './pomocne';

export default function Ukoly() {
  const { data } = useQuery({
    queryKey: ['ukoly'],
    queryFn: () => nactiUkoly(),
  });

  const ukoly = data ?? [];
  const prehled = souhrn(ukoly);

  return (
    <>
      <p className="souhrn">
        Hotovo <strong>{prehled.hotovo} z {prehled.celkem}</strong> · důležité a nehotové: {prehled.dulezite}
      </p>
      <ul className="ukoly">
        {ukoly.map((ukol) => (
          <li key={ukol.id} className={ukol.hotovo ? 'hotovo' : ''}>
            <span className="text">{ukol.text}</span>
            {ukol.dulezite && <span className="stitek">důležité</span>}
          </li>
        ))}
      </ul>
    </>
  );
}
```

## --file-- pomocne.js

```js
/**
 * Souhrn směny: kolik úkolů je hotových, kolik jich je celkem
 * a kolik nehotových je označených jako důležité.
 * @param {{ hotovo: boolean, dulezite: boolean }[]} ukoly
 * @returns {{ hotovo: number, celkem: number, dulezite: number }}
 */
export function souhrn(ukoly) {
  return {
    hotovo: ukoly.filter((ukol) => ukol.hotovo).length,
    celkem: ukoly.length,
    dulezite: ukoly.filter((ukol) => ukol.dulezite && !ukol.hotovo).length,
  };
}
```

## --file-- Ukoly.jsx

```jsx
import { useQuery } from '@tanstack/react-query';
import { nactiUkoly } from './api';
import { souhrn } from './pomocne';

export default function Ukoly() {
  const { data, status } = useQuery({
    queryKey: ['ukoly'],
    queryFn: () => nactiUkoly(),
  });

  if (status === 'pending') {
    return <p className="hlaska">Načítám úkoly směny…</p>;
  }

  const prehled = souhrn(data);

  return (
    <>
      <p className="souhrn">
        Hotovo <strong>{prehled.hotovo} z {prehled.celkem}</strong> · důležité a nehotové: {prehled.dulezite}
      </p>
      <ul className="ukoly">
        {data.map((ukol) => (
          <li key={ukol.id} className={ukol.hotovo ? 'hotovo' : ''}>
            <span className="text">{ukol.text}</span>
            {ukol.dulezite && <span className="stitek">důležité</span>}
          </li>
        ))}
      </ul>
    </>
  );
}
```
