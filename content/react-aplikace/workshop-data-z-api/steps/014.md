# --description--

Poslední chybějící operace: úkol, který se na směnu omylem dostal, musí jít
smazat.

**Úkol:** V `Ukoly.jsx` přidej na konec každé položky tlačítko s třídou `smazat`,
textem `×` a přístupným názvem složeným z textu „Smazat: " a textu úkolu.
Kliknutí spustí mutaci nad `smazUkol` z `api.js` a po úspěchu se seznam srovná.

Tady **nechceme** optimistickou úpravu. Mazání je nevratné: když je řádek pryč
dřív, než server odpoví, a server pak odmítne, uživatel už mezitím zavřel kartu
v přesvědčení, že je hotovo. U nevratných věcí se poctivě počká.

# --hints--

Každý úkol má tlačítko na smazání.

```js
await helpers.waitFor(() => document.querySelectorAll('.ukoly li').length === 5);
const tlacitka = document.querySelectorAll('.ukoly .smazat');
assert.equal(tlacitka.length, 5, 'každý úkol má mít tlačítko s třídou smazat');
assert.match(tlacitka[0].getAttribute('aria-label'), /Smazat: Umýt pákový kávovar/, 'tlačítko má mít aria-label „Smazat: " a text úkolu');
```

Kliknutí úkol smaže a seznam se srovná.

```js
await helpers.waitFor(() => document.querySelectorAll('.ukoly li').length === 5);
await helpers.click(document.querySelectorAll('.ukoly .smazat')[0]);
await helpers.waitFor(() => document.querySelectorAll('.ukoly li').length === 4, 3000);
const texty = [...document.querySelectorAll('.ukoly .text')].map((prvek) => prvek.textContent.trim());
assert.ok(!texty.includes('Umýt pákový kávovar'), 'smazaný úkol má ze seznamu zmizet');
assert.match(document.querySelector('.souhrn').textContent, /Hotovo\s*2 z 4/, 'souhrn se má po smazání přepočítat');
```

Mazání se neprojeví dřív, než ho server potvrdí.

```js
const { rezim } = await helpers.importFile('api.js');
await helpers.waitFor(() => document.querySelectorAll('.ukoly li').length === 5);
rezim.selhatZapis = true;
await helpers.click(document.querySelectorAll('.ukoly .smazat')[0]);
await helpers.flush();
assert.equal(document.querySelectorAll('.ukoly li').length, 5, 'u mazání se čeká na server — řádek nemá zmizet hned');
await helpers.wait(500);
await helpers.flush();
rezim.selhatZapis = false;
assert.equal(document.querySelectorAll('.ukoly li').length, 5, 'když server mazání odmítne, úkol má v seznamu zůstat');
```

Odškrtávání i hvězdičky fungují dál.

```js
await helpers.waitFor(() => document.querySelectorAll('.ukoly li').length === 5);
await helpers.click(document.querySelectorAll('.ukoly .odskrtnout')[0]);
await helpers.flush();
assert.match(document.querySelector('.souhrn').textContent, /Hotovo\s*3 z 5/, 'odškrtávání má fungovat dál');
await helpers.click(document.querySelectorAll('.ukoly .hvezda')[3]);
await helpers.flush();
assert.match(document.querySelector('.souhrn').textContent, /důležité a nehotové:\s*2/, 'hvězdičky mají fungovat dál');
```

# --help--

## --tip--

Jde o třetí mutaci ve stejné komponentě. Tentokrát bez `onMutate` a bez zálohy —
stačí zneplatnit po úspěchu, jako u přidávání.

## --tip--

Tlačítko patří dovnitř položky seznamu, za hvězdičku. Mutaci pošli `id` úkolu.

## --tip--

Přístupný název se skládá stejně jako u dvou tlačítek nad ním: pevný text plus
text úkolu.

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { nactiUkoly, prepniDulezite, prepniHotovo } from './api';
import { souhrn } from './pomocne';

export default function Ukoly({ filtr }) {
  const klient = useQueryClient();
  const { data, status, error, refetch } = useQuery({
    queryKey: ['ukoly', filtr],
    queryFn: () => nactiUkoly(filtr),
  });

  const odskrtnuti = useMutation({
    mutationFn: prepniHotovo,
    onMutate: async (id) => {
      await klient.cancelQueries({ queryKey: ['ukoly', filtr] });
      const zaloha = klient.getQueryData(['ukoly', filtr]);
      klient.setQueryData(['ukoly', filtr], (stare) =>
        stare.map((ukol) => (ukol.id === id ? { ...ukol, hotovo: !ukol.hotovo } : ukol)));
      return { zaloha };
    },
    onError: (chyba, id, kontext) => klient.setQueryData(['ukoly', filtr], kontext.zaloha),
    onSettled: () => klient.invalidateQueries({ queryKey: ['ukoly'] }),
  });

  const oznaceni = useMutation({
    mutationFn: prepniDulezite,
    onMutate: async (id) => {
      await klient.cancelQueries({ queryKey: ['ukoly', filtr] });
      const zaloha = klient.getQueryData(['ukoly', filtr]);
      klient.setQueryData(['ukoly', filtr], (stare) =>
        stare.map((ukol) => (ukol.id === id ? { ...ukol, dulezite: !ukol.dulezite } : ukol)));
      return { zaloha };
    },
    onError: (chyba, id, kontext) => klient.setQueryData(['ukoly', filtr], kontext.zaloha),
    onSettled: () => klient.invalidateQueries({ queryKey: ['ukoly'] }),
  });

  if (status === 'pending') {
    return <p className="hlaska">Načítám úkoly směny…</p>;
  }

  if (status === 'error') {
    return (
      <div className="chyba" role="alert">
        <span>Úkoly se nepodařilo načíst: {error.message}</span>
        <button type="button" onClick={() => refetch()}>Zkusit znovu</button>
      </div>
    );
  }

  if (data.length === 0) {
    return <p className="prazdno">Tady nic není. Zkus jiný filtr, nebo si přidej úkol.</p>;
  }

  const prehled = souhrn(data);

  return (
    <>
      <p className="souhrn">
        Hotovo <strong>{prehled.hotovo} z {prehled.celkem}</strong> · důležité a nehotové: {prehled.dulezite}
      </p>
      <ul className="ukoly">
        {data.map((ukol) => (
          <li key={ukol.id} className={ukol.hotovo ? 'hotovo' : ''}>
            <button
              type="button"
              className="odskrtnout"
              aria-pressed={ukol.hotovo}
              aria-label={'Hotovo: ' + ukol.text}
              onClick={() => odskrtnuti.mutate(ukol.id)}
            >
              ✓
            </button>
            <span className="text">{ukol.text}</span>
            <button
              type="button"
              className="hvezda"
              aria-pressed={ukol.dulezite}
              aria-label={'Důležité: ' + ukol.text}
              onClick={() => oznaceni.mutate(ukol.id)}
            >
              ★
            </button>
          </li>
        ))}
      </ul>
    </>
  );
}

// Náhrada skutečného API: data žijí v paměti stránky a odpovědi mají zpoždění,
// aby byly vidět stavy načítání. Ve svém projektu tyhle funkce nahradíš fetchem.
// Objekt rezim je tu jen pro tenhle workshop — dá se jím vynutit chyba serveru.
export const rezim = { selhatNacteni: false, selhatZapis: false, zpozdeni: 120 };

let ukoly = [
  { id: 1, text: 'Umýt pákový kávovar', hotovo: false, dulezite: true },
  { id: 2, text: 'Doplnit zrnka do mlýnku', hotovo: true, dulezite: false },
  { id: 3, text: 'Objednat ovesné mléko', hotovo: false, dulezite: true },
  { id: 4, text: 'Přepsat nabídku na tabuli', hotovo: false, dulezite: false },
  { id: 5, text: 'Spočítat kasu', hotovo: true, dulezite: false },
];
let dalsiId = 6;

function odpoved(vyrob, selhat, zprava) {
  return new Promise((splnit, odmitnout) => {
    setTimeout(() => {
      if (selhat) odmitnout(new Error(zprava));
      else splnit(vyrob());
    }, rezim.zpozdeni);
  });
}

/** Úkoly směny. Filtr je 'vse', 'zbyva' nebo 'hotove'. */
export function nactiUkoly(filtr = 'vse') {
  return odpoved(
    () => ukoly
      .filter((ukol) => filtr === 'vse' || (filtr === 'hotove' ? ukol.hotovo : !ukol.hotovo))
      .map((ukol) => ({ ...ukol })),
    rezim.selhatNacteni,
    'Server neodpovídá',
  );
}

export function pridejUkol(text) {
  return odpoved(() => {
    const novy = { id: dalsiId, text, hotovo: false, dulezite: false };
    dalsiId += 1;
    ukoly = [...ukoly, novy];
    return { ...novy };
  }, rezim.selhatZapis, 'Úkol se nepodařilo uložit');
}

export function prepniHotovo(id) {
  return odpoved(() => {
    ukoly = ukoly.map((ukol) => (ukol.id === id ? { ...ukol, hotovo: !ukol.hotovo } : ukol));
    return { ...ukoly.find((ukol) => ukol.id === id) };
  }, rezim.selhatZapis, 'Změnu se nepodařilo uložit');
}

export function prepniDulezite(id) {
  return odpoved(() => {
    ukoly = ukoly.map((ukol) => (ukol.id === id ? { ...ukol, dulezite: !ukol.dulezite } : ukol));
    return { ...ukoly.find((ukol) => ukol.id === id) };
  }, rezim.selhatZapis, 'Změnu se nepodařilo uložit');
}

export function smazUkol(id) {
  return odpoved(() => {
    ukoly = ukoly.filter((ukol) => ukol.id !== id);
    return { id };
  }, rezim.selhatZapis, 'Úkol se nepodařilo smazat');
}

const dodavatele = [
  { id: 1, nazev: 'Pražírna Kolonial', stav: 'objednano' },
  { id: 2, nazev: 'Mlékárna Valašsko', stav: 'ceka' },
  { id: 3, nazev: 'Pekárna U Nádraží', stav: 'objednano' },
  { id: 4, nazev: 'Ovesná farma Brdy', stav: 'ceka' },
];

/** Dodavatelé směny. Stav je 'vse', 'objednano' nebo 'ceka'. */
export function nactiDodavatele(stav = 'vse') {
  return odpoved(
    () => dodavatele.filter((polozka) => stav === 'vse' || polozka.stav === stav).map((polozka) => ({ ...polozka })),
    rezim.selhatNacteni,
    'Server neodpovídá',
  );
}

/**
 * Souhrn směny: kolik úkolů je hotových, kolik jich je celkem
 * a kolik nehotových je označených jako důležité.
 * @param {{ hotovo: boolean, dulezite: boolean }[]} ukoly
 * @returns {{ hotovo: number, celkem: number, dulezite: number }}
 */
export function souhrn(ukoly) {
  return {
    hotovo: ukoly.filter((ukol) => ukol.hotovo).length,
    celkem: ukoly.length,
    dulezite: ukoly.filter((ukol) => ukol.dulezite && !ukol.hotovo).length,
  };
}

import { useState } from 'react';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { pridejUkol } from './api';

export default function PridatUkol() {
  const klient = useQueryClient();
  const [text, setText] = useState('');
  const pridani = useMutation({
    mutationFn: pridejUkol,
    onSuccess: () => {
      setText('');
      klient.invalidateQueries({ queryKey: ['ukoly'] });
    },
  });

  function odesli(udalost) {
    udalost.preventDefault();
    if (text.trim() === '') return;
    pridani.mutate(text.trim());
  }

  return (
    <>
      <form className="pridat" onSubmit={odesli}>
        <input
          value={text}
          onChange={(udalost) => setText(udalost.target.value)}
          placeholder="nový úkol na směnu"
          aria-label="Nový úkol"
          disabled={pridani.isPending}
        />
        <button type="submit" disabled={pridani.isPending}>
          {pridani.isPending ? 'Ukládám…' : 'Přidat'}
        </button>
      </form>
      {pridani.isError && (
        <p className="chyba" role="alert">
          {pridani.error.message}. Text máš pořád v poli, zkus odeslat znovu.
        </p>
      )}
    </>
  );
}

    return (

        <span>Úkoly se nepodařilo načíst: {error.message}</span>
        <button type="button" onClick={() => refetch()}>Zkusit znovu</button>
      </div>
    );
  }

  if (data.length === 0) {
    return <p className="prazdno">Tady nic není. Zkus jiný filtr, nebo si přidej úkol.</p>;
  }

  const prehled = souhrn(data);

  return (
    <>
      <p className="souhrn">
        Hotovo <strong>{prehled.hotovo} z {prehled.celkem}</strong> · důležité a nehotové: {prehled.dulezite}
      </p>
      <ul className="ukoly">
        {data.map((ukol) => (
          <li key={ukol.id} className={ukol.hotovo ? 'hotovo' : ''}>
            <button
              type="button"
              className="odskrtnout"
              aria-pressed={ukol.hotovo}
              aria-label={'Hotovo: ' + ukol.text}
              onClick={() => odskrtnuti.mutate(ukol.id)}
            >
              ✓
            </button>
            <span className="text">{ukol.text}</span>
            <button
              type="button"
              className="hvezda"
              aria-pressed={ukol.dulezite}
              aria-label={'Důležité: ' + ukol.text}
              onClick={() => oznaceni.mutate(ukol.id)}
            >
              ★
            </button>
            <button
              type="button"
              className="smazat"
              aria-label={'Smazat: ' + ukol.text}
              onClick={() => mazani.mutate(ukol.id)}
            >
              ×
            </button>
          </li>
        ))}
      </ul>
    </>
  );
}
