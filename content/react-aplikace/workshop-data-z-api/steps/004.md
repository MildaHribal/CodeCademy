# --description--

Wi-Fi v kavárně vypadává. Když se seznam nenačte, teď se nestane nic — komponenta
zůstane viset na hlášce „Načítám", protože stav `'error'` nikdo neošetřil.

V `api.js` je objekt `rezim`. Když mu nastavíš `selhatNacteni` na `true`, začne
`nactiUkoly` odmítat Promise. Zkusit si to můžeš tak, že v `api.js` dočasně
změníš výchozí hodnotu.

**Úkol:** V `Ukoly.jsx` si z dotazu vezmi navíc `error` a `refetch`. Když je
`status` roven `'error'`, vrať `<div className="chyba" role="alert">` s textem
`Úkoly se nepodařilo načíst:` a hláškou z `error.message`, a vedle toho tlačítko
`Zkusit znovu`, které zavolá `refetch()`.

`role="alert"` říká čtečce obrazovky, že se objevilo něco, co má uživatel slyšet
hned — u chyb je to povinnost, ne ozdoba.

```jsx
// tvar, který hledáš, jinde
if (status === 'error') {
  return <p role="alert">Fotky se nenačetly: {error.message}</p>;
}
```

Když načtení selže, je vidět chybová hláška se zprávou ze serveru.


Tlačítko „Zkusit znovu" načtení zopakuje — a když už server odpovídá, seznam se objeví.

```js
const { rezim } = await helpers.importFile('api.js');
const { default: App } = await helpers.importFile('App.jsx');
const { createElement } = await import('react');
const { createRoot } = await import('react-dom/client');

document.body.append(misto);
const koren = createRoot(misto);
koren.render(createElement(App));
await helpers.waitFor(() => misto.querySelector('.chyba button'));
rezim.selhatNacteni = false;
await helpers.click(misto.querySelector('.chyba button'));
await helpers.waitFor(() => misto.querySelectorAll('.ukoly li').length === 5, 3000);
assert.equal(misto.querySelector('.chyba'), null, 'po úspěšném zopakování má chybová hláška zmizet');
koren.unmount();
misto.remove();
```

Když server odpovídá, běží všechno jako dřív.

```js
await helpers.waitFor(() => document.querySelectorAll('.ukoly li').length === 5);
assert.equal(document.querySelector('.chyba'), null, 'bez chyby se chybová hláška ukazovat nemá');
```

## --tip--

Stavy dotazu jsou popsané v lekci [Serverová data](see:react-aplikace/tanstack-query#problem-tri-stavy-dva-efekty-a-jeden-zavod).

## --tip--

Větev pro chybu patří hned za větev s načítáním, ještě než komponenta sáhne na
`data`. Funkci na zopakování dotazu dává `useQuery` vedle `data` a `status`.

## --tip--

Text chyby je v `error.message`. Vypisuje se stejně jako každá jiná hodnota,
tedy ve složených závorkách uvnitř JSX.

```jsx
import { useQuery } from '@tanstack/react-query';
import { nactiUkoly } from './api';
import { souhrn } from './pomocne';

export default function Ukoly() {
  const { data, status } = useQuery({
    queryKey: ['ukoly'],
    queryFn: () => nactiUkoly(),
  });

  if (status === 'pending') {
    return <p className="hlaska">Načítám úkoly směny…</p>;
  }

  const prehled = souhrn(data);

  return (
    <>
      <p className="souhrn">
        Hotovo <strong>{prehled.hotovo} z {prehled.celkem}</strong> · důležité a nehotové: {prehled.dulezite}
      </p>
      <ul className="ukoly">
        {data.map((ukol) => (
          <li key={ukol.id} className={ukol.hotovo ? 'hotovo' : ''}>
            <span className="text">{ukol.text}</span>
            {ukol.dulezite && <span className="stitek">důležité</span>}
          </li>
        ))}
      </ul>
    </>
  );
}
```

```js
// Náhrada skutečného API: data žijí v paměti stránky a odpovědi mají zpoždění,
// aby byly vidět stavy načítání. Ve svém projektu tyhle funkce nahradíš fetchem.
// Objekt rezim je tu jen pro tenhle workshop — dá se jím vynutit chyba serveru.
export const rezim = { selhatNacteni: false, selhatZapis: false, zpozdeni: 120 };

let ukoly = [
  { id: 1, text: 'Umýt pákový kávovar', hotovo: false, dulezite: true },
  { id: 2, text: 'Doplnit zrnka do mlýnku', hotovo: true, dulezite: false },
  { id: 3, text: 'Objednat ovesné mléko', hotovo: false, dulezite: true },
  { id: 4, text: 'Přepsat nabídku na tabuli', hotovo: false, dulezite: false },
  { id: 5, text: 'Spočítat kasu', hotovo: true, dulezite: false },
];
let dalsiId = 6;

function odpoved(vyrob, selhat, zprava) {
  return new Promise((splnit, odmitnout) => {
    setTimeout(() => {
      if (selhat) odmitnout(new Error(zprava));
      else splnit(vyrob());
    }, rezim.zpozdeni);
  });
}

/** Úkoly směny. Filtr je 'vse', 'zbyva' nebo 'hotove'. */
export function nactiUkoly(filtr = 'vse') {
  return odpoved(
    () => ukoly
      .filter((ukol) => filtr === 'vse' || (filtr === 'hotove' ? ukol.hotovo : !ukol.hotovo))
      .map((ukol) => ({ ...ukol })),
    rezim.selhatNacteni,
    'Server neodpovídá',
  );
}

export function pridejUkol(text) {
  return odpoved(() => {
    const novy = { id: dalsiId, text, hotovo: false, dulezite: false };
    dalsiId += 1;
    ukoly = [...ukoly, novy];
    return { ...novy };
  }, rezim.selhatZapis, 'Úkol se nepodařilo uložit');
}

export function prepniHotovo(id) {
  return odpoved(() => {
    ukoly = ukoly.map((ukol) => (ukol.id === id ? { ...ukol, hotovo: !ukol.hotovo } : ukol));
    return { ...ukoly.find((ukol) => ukol.id === id) };
  }, rezim.selhatZapis, 'Změnu se nepodařilo uložit');
}

export function prepniDulezite(id) {
  return odpoved(() => {
    ukoly = ukoly.map((ukol) => (ukol.id === id ? { ...ukol, dulezite: !ukol.dulezite } : ukol));
    return { ...ukoly.find((ukol) => ukol.id === id) };
  }, rezim.selhatZapis, 'Změnu se nepodařilo uložit');
}

export function smazUkol(id) {
  return odpoved(() => {
    ukoly = ukoly.filter((ukol) => ukol.id !== id);
    return { id };
  }, rezim.selhatZapis, 'Úkol se nepodařilo smazat');
}

const dodavatele = [
  { id: 1, nazev: 'Pražírna Kolonial', stav: 'objednano' },
  { id: 2, nazev: 'Mlékárna Valašsko', stav: 'ceka' },
  { id: 3, nazev: 'Pekárna U Nádraží', stav: 'objednano' },
  { id: 4, nazev: 'Ovesná farma Brdy', stav: 'ceka' },
];

/** Dodavatelé směny. Stav je 'vse', 'objednano' nebo 'ceka'. */
export function nactiDodavatele(stav = 'vse') {
  return odpoved(
    () => dodavatele.filter((polozka) => stav === 'vse' || polozka.stav === stav).map((polozka) => ({ ...polozka })),
    rezim.selhatNacteni,
    'Server neodpovídá',
  );
}
```

## --file-- pomocne.js

```js
/**
 * Souhrn směny: kolik úkolů je hotových, kolik jich je celkem
 * a kolik nehotových je označených jako důležité.
 * @param {{ hotovo: boolean, dulezite: boolean }[]} ukoly
 * @returns {{ hotovo: number, celkem: number, dulezite: number }}
 */
export function souhrn(ukoly) {
  return {
    hotovo: ukoly.filter((ukol) => ukol.hotovo).length,
    celkem: ukoly.length,
    dulezite: ukoly.filter((ukol) => ukol.dulezite && !ukol.hotovo).length,
  };
}
```

## --file-- Ukoly.jsx

```jsx
import { useQuery } from '@tanstack/react-query';
import { nactiUkoly } from './api';
import { souhrn } from './pomocne';

export default function Ukoly() {
  const { data, status, error, refetch } = useQuery({
    queryKey: ['ukoly'],
    queryFn: () => nactiUkoly(),
  });

  if (status === 'pending') {
    return <p className="hlaska">Načítám úkoly směny…</p>;
  }

  if (status === 'error') {
    return (
      <div className="chyba" role="alert">
        <span>Úkoly se nepodařilo načíst: {error.message}</span>
        <button type="button" onClick={() => refetch()}>Zkusit znovu</button>
      </div>
    );
  }

  const prehled = souhrn(data);

  return (
    <>
      <p className="souhrn">
        Hotovo <strong>{prehled.hotovo} z {prehled.celkem}</strong> · důležité a nehotové: {prehled.dulezite}
      </p>
      <ul className="ukoly">
        {data.map((ukol) => (
          <li key={ukol.id} className={ukol.hotovo ? 'hotovo' : ''}>
            <span className="text">{ukol.text}</span>
            {ukol.dulezite && <span className="stitek">důležité</span>}
          </li>
        ))}
      </ul>
    </>
  );
}
```
