# --description--

Napiš do hledání `xyz`. Filtr sedí, adresa sedí — a uživatel se dívá na prázdné
místo, kde nic není. Neví, jestli se stránka rozbila, nebo jestli půjčovna nemá
kola.

**Úkol:** V `Katalog.jsx` zařiď, aby se v případě, že žádné kolo neprojde
filtrem, místo prázdné mřížky vykreslil `div` s třídou `prazdno`, v něm větu
„Žádné kolo neodpovídá tomu, co hledáš." a tlačítko s popiskem „Zrušit filtry"
(`button` typu `button` s třídou `zpet-tlacitko`), které vyčistí adresu od všech
parametrů.

Jak to zařídíš, je na tobě — v JSX se to dá udělat nejmíň třemi způsoby
a všechny jsou v pořádku.

Prázdný stav (*empty state*) je jedna z věcí, které dělí cvičnou stránku od
hotové aplikace. Dobrý prázdný stav řekne, **co se stalo** a **co s tím**.

# --hints--

Když filtru nic neodpovídá, ukáže se hláška místo prázdné mřížky.

```js
  for (let i = 1; i <= text.length; i += 1) {

    pole.dispatchEvent(new InputEvent('input', { bubbles: true, data: text[i - 1], inputType: 'insertText' }));
    await helpers.flush();
  }
}

assert.equal(document.querySelectorAll('.katalog li').length, 0, 'při hledání „xyz" nemá projít žádné kolo');
```

# --help--

## --tip--

Jde o podmíněné vykreslení v JSX. V Reactu se to dělá výrazem, ne příkazem `if`
uvnitř návratového JSX — ale `if` s brzkým `return` nad ním je taky v pořádku.

## --tip--

Rozhodnout se musíš podle toho, kolik kol zbylo po filtrování. Tlačítko dělá
totéž co tlačítko **Vše** ve filtru.

## --tip--

Podobnou křižovatku už jsi psal v detailu kola, když se kolo nenašlo: nejdřív
odbav výjimečný případ, pak běžný.

```jsx
import { Link, useSearchParams } from 'react-router';
import { kola, popisekTypu, typy } from './data';

export default function Katalog() {
  const [parametry, setParametry] = useSearchParams();
  const typ = parametry.get('typ') ?? 'vse';
  const hledani = parametry.get('hledani') ?? '';
  const razeni = parametry.get('razeni') ?? 'nejlevnejsi';

  const videt = kola
    .filter((kolo) => typ === 'vse' || kolo.typ === typ)
    .filter((kolo) => kolo.nazev.toLowerCase().includes(hledani.toLowerCase()))
    .toSorted((prvni, druhe) =>
      razeni === 'nejdrazsi' ? druhe.cenaDen - prvni.cenaDen : prvni.cenaDen - druhe.cenaDen,
    );

  function zmenHledani(text) {
    const dalsi = new URLSearchParams(parametry);
    if (text === '') dalsi.delete('hledani');
    else dalsi.set('hledani', text);
    setParametry(dalsi, { replace: true });
  }

  function zmenRazeni(hodnota) {
    const dalsi = new URLSearchParams(parametry);
    dalsi.set('razeni', hodnota);
    setParametry(dalsi);
  }

  return (
    <>
      <div className="filtr">
        <button
          type="button"
          className={typ === 'vse' ? 'zvolene' : ''}
          onClick={() => setParametry({})}
        >
          Vše
        </button>
        {typy.map((moznost) => (
          <button
            key={moznost.hodnota}
            type="button"
            className={typ === moznost.hodnota ? 'zvolene' : ''}
            onClick={() => setParametry({ typ: moznost.hodnota })}
          >
            {moznost.popisek}
          </button>
        ))}
      </div>
      <label className="hledani">
        <span>Hledat kolo</span>
        <input
          type="search"
          value={hledani}
          placeholder="název kola"
          onChange={(udalost) => zmenHledani(udalost.target.value)}
        />
      </label>
      <label className="razeni">
        <span>Řadit</span>
        <select value={razeni} onChange={(udalost) => zmenRazeni(udalost.target.value)}>
          <option value="nejlevnejsi">od nejlevnějšího</option>
          <option value="nejdrazsi">od nejdražšího</option>
        </select>
      </label>
      <ul className="katalog">
        {videt.map((kolo) => (
          <li key={kolo.id}>
            <Link className="karta" to={'/kola/' + kolo.id}>
              <h2>{kolo.nazev}</h2>
              <p className="typ">{popisekTypu(kolo.typ)}</p>
              <p className="cena">{kolo.cenaDen} Kč / den</p>
            </Link>
          </li>
        ))}
      </ul>
    </>
  );
}
```

```jsx
import { Link, useParams } from 'react-router';

export default function Detail() {
  const { id } = useParams();
  const kolo = kola.find((polozka) => polozka.id === Number(id));

  if (!kolo) {
    return (
      <article className="detail">
```

```jsx
            {moznost.popisek}
          </button>
        ))}
      </div>
      <label className="hledani">
        <span>Hledat kolo</span>
        <input
          type="search"
          value={hledani}
          placeholder="název kola"
          onChange={(udalost) => zmenHledani(udalost.target.value)}
        />
      </label>
      <label className="razeni">
        <span>Řadit</span>
        <select value={razeni} onChange={(udalost) => zmenRazeni(udalost.target.value)}>
          <option value="nejlevnejsi">od nejlevnějšího</option>
          <option value="nejdrazsi">od nejdražšího</option>
        </select>
      </label>
      {videt.length === 0 ? (
        <div className="prazdno">
          <p>Žádné kolo neodpovídá tomu, co hledáš.</p>
          <button type="button" className="zpet-tlacitko" onClick={() => setParametry({})}>
            Zrušit filtry
          </button>
        </div>
      ) : (
        <ul className="katalog">
          {videt.map((kolo) => (
            <li key={kolo.id}>
              <Link className="karta" to={'/kola/' + kolo.id}>
                <h2>{kolo.nazev}</h2>
                <p className="typ">{popisekTypu(kolo.typ)}</p>
                <p className="cena">{kolo.cenaDen} Kč / den</p>
              </Link>
            </li>
          ))}
        </ul>
      )}
    </>
  );
}
```
